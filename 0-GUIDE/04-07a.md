## Key Terms and Concepts

- **Data Quality**: The measure of how well data meets the requirements of its intended use.
- **Missing Values**: Data points absent from the dataset, represented as NaN, NULL, or special placeholders.
- **Missing Data Mechanisms**: MCAR (Missing Completely At Random), MAR (Missing At Random), and MNAR (Missing Not At Random).
- **Outliers**: Data points that significantly deviate from the general distribution of the dataset.
- **Imputation**: The process of replacing missing values with substituted values.
- **Feature Scaling**: Transforming numeric variables to a standard scale (e.g., normalization, standardization).
- **Data Validation**: Systematically checking data for inconsistencies and errors.
- **Data Drift**: Changes in the statistical properties of data over time.
- **Skewness**: Asymmetry in the probability distribution of data.
- **Robust Statistics**: Statistical methods resistant to outliers.
- **Data Leakage**: When information from outside the training dataset is used to create the model.

def identify_data_problems(df):
    """
    Perform a comprehensive analysis to identify various data quality issues.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        The dataset to analyze
    
    Returns:
    --------
    data_report : dict
        Dictionary containing various data quality metrics
    """
    data_report = {}
    
    # 1. Basic dataset information
    data_report['shape'] = df.shape
    data_report['dtypes'] = df.dtypes
    
    # 2. Missing values analysis
    missing_values = df.isnull().sum()
    missing_percentage = (missing_values / len(df)) * 100
    data_report['missing_values'] = pd.DataFrame({
        'Count': missing_values,
        'Percentage': missing_percentage
    }).sort_values('Percentage', ascending=False)
    
    # 3. Duplicate rows
    data_report['duplicate_rows'] = df.duplicated().sum()
    
    # 4. Descriptive statistics for numerical features
    numeric_cols = df.select_dtypes(include=['number']).columns
    data_report['numeric_stats'] = df[numeric_cols].describe(percentiles=[.01, .05, .25, .5, .75, .95, .99])
    
    # 5. Skewness and kurtosis (for detecting distribution anomalies)
    skewness = df[numeric_cols].skew()
    kurtosis = df[numeric_cols].kurtosis()
    data_report['distribution_metrics'] = pd.DataFrame({
        'Skewness': skewness,
        'Kurtosis': kurtosis
    })
    
    # 6. Outlier detection using Z-score
    z_scores = pd.DataFrame()
    for col in numeric_cols:
        if not np.issubdtype(df[col].dtype, np.datetime64):
            z_scores[col] = np.abs(stats.zscore(df[col], nan_policy='omit'))
    
    potential_outliers = (z_scores > 3).sum()
    data_report['potential_outliers_zscore'] = potential_outliers
    
    # 7. Outlier detection using IQR
    q1 = df[numeric_cols].quantile(0.25)
    q3 = df[numeric_cols].quantile(0.75)
    iqr = q3 - q1
    outliers_iqr = ((df[numeric_cols] < (q1 - 1.5 * iqr)) | 
                     (df[numeric_cols] > (q3 + 1.5 * iqr))).sum()
    data_report['potential_outliers_iqr'] = outliers_iqr
    
    # 8. Categorical features analysis
    cat_cols = df.select_dtypes(include=['object', 'category']).columns
    
    if not cat_cols.empty:
        category_counts = {}
        for col in cat_cols:
            category_counts[col] = df[col].value_counts(dropna=False)
        data_report['category_counts'] = category_counts
    
    # 9. Check for constant columns (low variance)
    data_report['constant_columns'] = [col for col in df.columns 
                                     if df[col].nunique() == 1]
    
    # 10. Check for highly correlated features
    if len(numeric_cols) > 1:
        correlation_matrix = df[numeric_cols].corr()
        high_correlation = (correlation_matrix.abs() > 0.8) & (correlation_matrix.abs() < 1.0)
        high_corr_pairs = []
        
        for i in range(len(correlation_matrix.columns)):
            for j in range(i):
                if high_correlation.iloc[i, j]:
                    high_corr_pairs.append(
                        (correlation_matrix.columns[i], 
                         correlation_matrix.columns[j], 
                         correlation_matrix.iloc[i, j])
                    )
        
        data_report['high_correlation_pairs'] = high_corr_pairs
    
    # 11. Check for columns with many unique values (potential IDs or noise)
    unique_counts = df.nunique()
    unique_percentage = (unique_counts / len(df)) * 100
    high_cardinality = unique_percentage > 80
    data_report['high_cardinality_columns'] = list(df.columns[high_cardinality])
    
    return data_report

# Detecting outliers with multiple methods
def detect_outliers(df, column, methods=['zscore', 'iqr', 'isolation_forest']):
    """
    Detect outliers in a specific column using multiple methods.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        The dataset
    column : str
        The column to analyze for outliers
    methods : list
        List of methods to use for outlier detection
        
    Returns:
    --------
    outliers : dict
        Dictionary with outlier indices for each method
    """
    outliers = {}
    data = df[column].dropna()
    
    if 'zscore' in methods:
        z_scores = np.abs(stats.zscore(data))
        outliers['zscore'] = data.index[z_scores > 3].tolist()
    
    if 'iqr' in methods:
        q1 = data.quantile(0.25)
        q3 = data.quantile(0.75)
        iqr = q3 - q1
        outliers['iqr'] = data.index[(data < (q1 - 1.5 * iqr)) | 
                                    (data > (q3 + 1.5 * iqr))].tolist()
    
    if 'isolation_forest' in methods:
        isolation_forest = IsolationForest(contamination=0.05, random_state=42)
        predictions = isolation_forest.fit_predict(data.values.reshape(-1, 1))
        outliers['isolation_forest'] = data.index[predictions == -1].tolist()
    
    return outliers

def fix_data_problems(df, strategies=None):
    """
    Apply various strategies to fix data quality issues.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        The dataset to clean
    strategies : dict
        Dictionary with strategies for different issue types
        
    Returns:
    --------
    df_cleaned : pandas.DataFrame
        The cleaned dataset
    transformers : dict
        Dictionary with fitted transformers for future use
    """
    df_cleaned = df.copy()
    transformers = {}
    
    if strategies is None:
        strategies = {
            'missing_values': 'knn',  # Options: 'mean', 'median', 'most_frequent', 'knn', 'iterative'
            'outliers': 'winsorize',  # Options: 'remove', 'winsorize', 'mean', 'median'
            'scaling': 'standard',    # Options: 'standard', 'minmax', 'robust', 'power'
            'skewness': 'yeo-johnson' # Options: 'log', 'sqrt', 'box-cox', 'yeo-johnson'
        }
    
    # 1. Handle missing values
    numeric_cols = df_cleaned.select_dtypes(include=['number']).columns
    categorical_cols = df_cleaned.select_dtypes(include=['object', 'category']).columns
    
    # Separate handling for numeric and categorical
    if strategies['missing_values'] == 'mean':
        num_imputer = SimpleImputer(strategy='mean')
    elif strategies['missing_values'] == 'median':
        num_imputer = SimpleImputer(strategy='median')
    elif strategies['missing_values'] == 'knn':
        num_imputer = KNNImputer(n_neighbors=5)
    elif strategies['missing_values'] == 'iterative':
        num_imputer = IterativeImputer(max_iter=10, random_state=42)
    else:
        num_imputer = SimpleImputer(strategy='median')
    
    if len(numeric_cols) > 0:
        df_cleaned[numeric_cols] = pd.DataFrame(
            num_imputer.fit_transform(df_cleaned[numeric_cols]),
            columns=numeric_cols,
            index=df_cleaned.index
        )
        transformers['numeric_imputer'] = num_imputer
    
    if len(categorical_cols) > 0:
        cat_imputer = SimpleImputer(strategy='most_frequent')
        df_cleaned[categorical_cols] = pd.DataFrame(
            cat_imputer.fit_transform(df_cleaned[categorical_cols]),
            columns=categorical_cols,
            index=df_cleaned.index
        )
        transformers['categorical_imputer'] = cat_imputer
    
    # 2. Handle outliers in numeric columns
    for col in numeric_cols:
        if col == 'target':  # Skip target variable
            continue
            
        if strategies['outliers'] == 'remove':
            # Identify outliers using Z-score
            z_scores = np.abs(stats.zscore(df_cleaned[col]))
            df_cleaned = df_cleaned[z_scores <= 3]
        
        elif strategies['outliers'] == 'winsorize':
            # Cap outliers at 1st and 99th percentiles
            lower_bound = np.percentile(df_cleaned[col], 1)
            upper_bound = np.percentile(df_cleaned[col], 99)
            df_cleaned[col] = df_cleaned[col].clip(lower=lower_bound, upper=upper_bound)
        
        elif strategies['outliers'] == 'mean':
            # Replace outliers with mean
            z_scores = np.abs(stats.zscore(df_cleaned[col]))
            col_mean = df_cleaned[col].mean()
            df_cleaned.loc[z_scores > 3, col] = col_mean
        
        elif strategies['outliers'] == 'median':
            # Replace outliers with median
            z_scores = np.abs(stats.zscore(df_cleaned[col]))
            col_median = df_cleaned[col].median()
            df_cleaned.loc[z_scores > 3, col] = col_median
    
    # 3. Handle scaling
    if strategies['scaling'] == 'standard':
        scaler = StandardScaler()
    elif strategies['scaling'] == 'minmax':
        scaler = MinMaxScaler()
    elif strategies['scaling'] == 'robust':
        scaler = RobustScaler()
    else:
        scaler = StandardScaler()
    
    # Apply scaling to numeric features (excluding target)
    numeric_features = [col for col in numeric_cols if col != 'target']
    if numeric_features:
        df_cleaned[numeric_features] = pd.DataFrame(
            scaler.fit_transform(df_cleaned[numeric_features]),
            columns=numeric_features,
            index=df_cleaned.index
        )
        transformers['scaler'] = scaler
    
    # 4. Handle skewness
    if strategies['skewness'] in ['box-cox', 'yeo-johnson']:
        for col in numeric_features:
            # Check if data is severely skewed
            skewness = df_cleaned[col].skew()
            if abs(skewness) > 1:  # Arbitrary threshold for skewness
                if strategies['skewness'] == 'box-cox':
                    # Box-Cox requires positive values
                    if (df_cleaned[col] <= 0).any():
                        continue
                    
                    # Apply Box-Cox transformation
                    df_cleaned[col], _ = stats.boxcox(df_cleaned[col])
                
                elif strategies['skewness'] == 'yeo-johnson':
                    # Yeo-Johnson works with any values
                    pt = PowerTransformer(method='yeo-johnson')
                    df_cleaned[col] = pt.fit_transform(df_cleaned[col].values.reshape(-1, 1)).flatten()
                    transformers[f'power_transformer_{col}'] = pt
    
    elif strategies['skewness'] == 'log':
        for col in numeric_features:
            skewness = df_cleaned[col].skew()
            if abs(skewness) > 1:
                # Ensure all values are positive for log transform
                if (df_cleaned[col] <= 0).any():
                    min_val = df_cleaned[col].min()
                    df_cleaned[col] = df_cleaned[col] - min_val + 1
                
                df_cleaned[col] = np.log1p(df_cleaned[col])
    
    elif strategies['skewness'] == 'sqrt':
        for col in numeric_features:
            skewness = df_cleaned[col].skew()
            if abs(skewness) > 1:
                # Ensure all values are positive for sqrt transform
                if (df_cleaned[col] < 0).any():
                    min_val = df_cleaned[col].min()
                    df_cleaned[col] = df_cleaned[col] - min_val
                
                df_cleaned[col] = np.sqrt(df_cleaned[col])
    
    return df_cleaned, transformers

def build_data_cleaning_pipeline(categorical_features=None, numerical_features=None):
    """
    Build a complete data cleaning pipeline using scikit-learn.
    
    Parameters:
    -----------
    categorical_features : list
        List of categorical feature names
    numerical_features : list
        List of numerical feature names
        
    Returns:
    --------
    pipeline : sklearn.pipeline.Pipeline
        The data cleaning pipeline
    """
    from sklearn.compose import ColumnTransformer
    from sklearn.pipeline import Pipeline
    
    # Define default transformers
    num_transformer = Pipeline(steps=[
        ('imputer', KNNImputer(n_neighbors=5)),
        ('outlier_remover', 'passthrough'),  # We'll implement a custom transformer later
        ('scaler', RobustScaler())
    ])
    
    cat_transformer = Pipeline(steps=[
        ('imputer', SimpleImputer(strategy='most_frequent')),
        ('encoder', pd.get_dummies)  # For simplicity, we're using pandas' get_dummies here
    ])
    
    # Combine transformers using ColumnTransformer
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', num_transformer, numerical_features),
            ('cat', cat_transformer, categorical_features)
        ]
    )
    
    # Create the full pipeline
    pipeline = Pipeline(steps=[
        ('preprocessor', preprocessor)
    ])
    
    return pipeline

# Custom transformer for outlier handling
from sklearn.base import BaseEstimator, TransformerMixin

class OutlierRemover(BaseEstimator, TransformerMixin):
    """Custom transformer for handling outliers in numeric features."""
    
    def __init__(self, method='winsorize', threshold=3):
        self.method = method
        self.threshold = threshold
        self.bounds_ = {}
    
    def fit(self, X, y=None):
        """Learn the bounds for outlier removal."""
        if isinstance(X, pd.DataFrame):
            for col in X.columns:
                if self.method == 'winsorize':
                    self.bounds_[col] = {
                        'lower': np.percentile(X[col], 1),
                        'upper': np.percentile(X[col], 99)
                    }
                elif self.method == 'zscore':
                    mean = X[col].mean()
                    std = X[col].std()
                    self.bounds_[col] = {
                        'mean': mean,
                        'std': std
                    }
        else:
            # If X is a numpy array, treat as a single feature
            if self.method == 'winsorize':
                self.bounds_['feature'] = {
                    'lower': np.percentile(X, 1),
                    'upper': np.percentile(X, 99)
                }
            elif self.method == 'zscore':
                mean = np.mean(X)
                std = np.std(X)
                self.bounds_['feature'] = {
                    'mean': mean,
                    'std': std
                }
        
        return self
    
    def transform(self, X):
        """Apply outlier removal or capping."""
        X_copy = X.copy()
        
        if isinstance(X_copy, pd.DataFrame):
            for col in X_copy.columns:
                if col in self.bounds_:
                    if self.method == 'winsorize':
                        X_copy[col] = X_copy[col].clip(
                            lower=self.bounds_[col]['lower'],
                            upper=self.bounds_[col]['upper']
                        )
                    elif self.method == 'zscore':
                        mean = self.bounds_[col]['mean']
                        std = self.bounds_[col]['std']
                        z_scores = np.abs((X_copy[col] - mean) / std)
                        X_copy.loc[z_scores > self.threshold, col] = mean
        else:
            # If X is a numpy array
            if self.method == 'winsorize':
                X_copy = np.clip(
                    X_copy,
                    self.bounds_['feature']['lower'],
                    self.bounds_['feature']['upper']
                )
            elif self.method == 'zscore':
                mean = self.bounds_['feature']['mean']
                std = self.bounds_['feature']['std']
                z_scores = np.abs((X_copy - mean) / std)
                X_copy[z_scores > self.threshold] = mean
        
        return X_copy

def build_advanced_pipeline(categorical_features=None, numerical_features=None):

            ('num', num_transformer, numerical_features),
            ('cat', cat_transformer, categorical_features)
        ]
    )
    
    # Create the full pipeline
    pipeline = Pipeline(steps=[
        ('preprocessor', preprocessor)
    ])
    
    return pipeline
