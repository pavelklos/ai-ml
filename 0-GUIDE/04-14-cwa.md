class MLProjectManager:
    """Complete ML project management class"""
    
    def __init__(self, project_name: str):
        self.project_name = project_name
        self.models = {}
        self.pipelines = {}
        self.results = {}
        
        # Setup logging
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s - %(levelname)s - %(message)s',
            handlers=[
                logging.FileHandler(f'{project_name}_ml_log.log'),
                logging.StreamHandler()
            ]
        )
        self.logger = logging.getLogger(__name__)
        
    def load_and_preprocess_data(self, data_path: str, target_column: str):
        """Load and perform initial data preprocessing"""
        try:
            # Load data
            self.data = pd.read_csv(data_path)
            self.logger.info(f"Data loaded: {self.data.shape}")
            
            # Separate features and target
            self.X = self.data.drop(columns=[target_column])
            self.y = self.data[target_column]
            
            # Handle different data types
            self.numeric_features = self.X.select_dtypes(include=[np.number]).columns.tolist()
            self.categorical_features = self.X.select_dtypes(include=['object']).columns.tolist()
            
            self.logger.info(f"Numeric features: {len(self.numeric_features)}")
            self.logger.info(f"Categorical features: {len(self.categorical_features)}")
            
            return True
            
        except Exception as e:
            self.logger.error(f"Data loading failed: {str(e)}")
            return False
    
    def create_advanced_pipeline(self, model_type: str = 'classification'):
        """Create comprehensive ML pipeline"""
        from sklearn.compose import ColumnTransformer
        from sklearn.preprocessing import StandardScaler, OneHotEncoder
        from sklearn.impute import SimpleImputer
        
        # Numeric pipeline
        numeric_pipeline = Pipeline([
            ('imputer', SimpleImputer(strategy='median')),
            ('scaler', StandardScaler())
        ])
        
        # Categorical pipeline
        categorical_pipeline = Pipeline([
            ('imputer', SimpleImputer(strategy='constant', fill_value='missing')),
            ('encoder', OneHotEncoder(drop='first', sparse_output=False))
        ])
        
        # Combine preprocessing
        preprocessor = ColumnTransformer([
            ('num', numeric_pipeline, self.numeric_features),
            ('cat', categorical_pipeline, self.categorical_features)
        ])
        
        # Select model based on type
        if model_type == 'classification':
            model = RandomForestClassifier(
                n_estimators=100,
                max_depth=10,
                random_state=42,
                n_jobs=-1
            )
        else:
            model = GradientBoostingRegressor(
                n_estimators=100,
                learning_rate=0.1,
                max_depth=6,
                random_state=42
            )
        
        # Complete pipeline
        pipeline = Pipeline([
            ('preprocessor', preprocessor),
            ('model', model)
        ])
        
        self.pipelines[model_type] = pipeline
        return pipeline
    
    def train_and_evaluate(self, model_type: str = 'classification'):
        """Train model with cross-validation and evaluation"""
        if model_type not in self.pipelines:
            self.create_advanced_pipeline(model_type)
        
        pipeline = self.pipelines[model_type]
        
        # Split data
        X_train, X_test, y_train, y_test = train_test_split(
            self.X, self.y, test_size=0.2, random_state=42, stratify=self.y if model_type == 'classification' else None
        )
        
        # Cross-validation
        cv_scores = cross_val_score(pipeline, X_train, y_train, cv=5, 
                                   scoring='accuracy' if model_type == 'classification' else 'neg_mean_squared_error')
        
        # Train final model
        pipeline.fit(X_train, y_train)
        
        # Predictions
        y_pred = pipeline.predict(X_test)
        
        # Evaluation
        if model_type == 'classification':
            from sklearn.metrics import accuracy_score, classification_report
            accuracy = accuracy_score(y_test, y_pred)
            report = classification_report(y_test, y_pred)
            
            self.results[model_type] = {
                'cv_scores': cv_scores,
                'test_accuracy': accuracy,
                'classification_report': report
            }
            
            self.logger.info(f"Cross-validation accuracy: {cv_scores.mean():.4f} (+/- {cv_scores.std() * 2:.4f})")
            self.logger.info(f"Test accuracy: {accuracy:.4f}")
            
        else:
            mse = mean_squared_error(y_test, y_pred)
            rmse = np.sqrt(mse)
            
            self.results[model_type] = {
                'cv_scores': cv_scores,
                'test_mse': mse,
                'test_rmse': rmse
            }
            
            self.logger.info(f"Cross-validation RMSE: {np.sqrt(-cv_scores.mean()):.4f}")
            self.logger.info(f"Test RMSE: {rmse:.4f}")
        
        # Save model
        model_path = f"{self.project_name}_{model_type}_model.joblib"
        joblib.dump(pipeline, model_path)
        self.logger.info(f"Model saved to {model_path}")
        
        return pipeline, self.results[model_type]

class FeatureEngineer:
    """Advanced feature engineering techniques"""
    
    @staticmethod
    def create_polynomial_features(X, degree=2):
        """Create polynomial features"""
        from sklearn.preprocessing import PolynomialFeatures
        poly = PolynomialFeatures(degree=degree, include_bias=False)
        return poly.fit_transform(X)
    
    @staticmethod
    def create_interaction_features(X, feature_pairs):
        """Create interaction features for specified pairs"""
        X_interactions = X.copy()
        for feat1, feat2 in feature_pairs:
            if feat1 in X.columns and feat2 in X.columns:
                X_interactions[f'{feat1}_{feat2}_interaction'] = X[feat1] * X[feat2]
        return X_interactions
    
    @staticmethod
    def create_time_features(datetime_col):
        """Extract comprehensive time-based features"""
        features = pd.DataFrame()
        features['year'] = datetime_col.dt.year
        features['month'] = datetime_col.dt.month
        features['day'] = datetime_col.dt.day
        features['dayofweek'] = datetime_col.dt.dayofweek
        features['hour'] = datetime_col.dt.hour
        features['quarter'] = datetime_col.dt.quarter
        features['is_weekend'] = datetime_col.dt.dayofweek.isin([5, 6]).astype(int)
        features['is_month_start'] = datetime_col.dt.is_month_start.astype(int)
        features['is_month_end'] = datetime_col.dt.is_month_end.astype(int)
        return features

# Model deployment utilities
class ModelDeployment:
    """Model deployment and monitoring utilities"""
    
    def __init__(self, model_path: str):
        self.model = joblib.load(model_path)
        self.prediction_log = []
    
    def predict_single(self, input_data: dict):
        """Make prediction for single instance"""
        try:
            # Convert input to DataFrame
            input_df = pd.DataFrame([input_data])
            
            # Make prediction
            prediction = self.model.predict(input_df)[0]
            probability = None
            
            # Get probability for classification
            if hasattr(self.model.named_steps['model'], 'predict_proba'):
                probability = self.model.predict_proba(input_df)[0].max()
            
            # Log prediction
            self.prediction_log.append({
                'timestamp': pd.Timestamp.now(),
                'input': input_data,
                'prediction': prediction,
                'probability': probability
            })
            
            return {
                'prediction': prediction,
                'probability': probability,
                'timestamp': pd.Timestamp.now().isoformat()
            }
            
        except Exception as e:
            return {'error': str(e)}
    
    def batch_predict(self, input_data: pd.DataFrame):
        """Make batch predictions"""
        try:
            predictions = self.model.predict(input_data)
            probabilities = None
            
            if hasattr(self.model.named_steps['model'], 'predict_proba'):
                probabilities = self.model.predict_proba(input_data)
            
            return {
                'predictions': predictions.tolist(),
                'probabilities': probabilities.tolist() if probabilities is not None else None
            }
            
        except Exception as e:
            return {'error': str(e)}

# Common ML interview coding problems
class MLInterviewPrep:
    """Common ML coding interview questions and solutions"""
    
    @staticmethod
    def implement_linear_regression():
        """Implement linear regression from scratch"""
        class LinearRegression:
            def __init__(self):
                self.weights = None
                self.bias = None
            
            def fit(self, X, y, learning_rate=0.01, epochs=1000):
                n_samples, n_features = X.shape
                self.weights = np.zeros(n_features)
                self.bias = 0
                
                for _ in range(epochs):
                    # Forward pass
                    y_pred = X.dot(self.weights) + self.bias
                    
                    # Compute cost
                    cost = (1/n_samples) * np.sum((y_pred - y)**2)
                    
                    # Compute gradients
                    dw = (1/n_samples) * X.T.dot(y_pred - y)
                    db = (1/n_samples) * np.sum(y_pred - y)
                    
                    # Update parameters
                    self.weights -= learning_rate * dw
                    self.bias -= learning_rate * db
            
            def predict(self, X):
                return X.dot(self.weights) + self.bias
        
        return LinearRegression
    
    @staticmethod
    def implement_k_means():
        """Implement K-means clustering from scratch"""
        class KMeans:
            def __init__(self, k=3, max_iters=100):
                self.k = k
                self.max_iters = max_iters
            
            def fit(self, X):
                n_samples, n_features = X.shape
                
                # Initialize centroids randomly
                self.centroids = np.random.uniform(
                    X.min(axis=0), X.max(axis=0), (self.k, n_features)
                )
                
                for _ in range(self.max_iters):
                    # Assign points to clusters
                    distances = np.sqrt(((X - self.centroids[:, np.newaxis])**2).sum(axis=2))
                    labels = np.argmin(distances, axis=0)
                    
                    # Update centroids
                    new_centroids = np.array([
                        X[labels == i].mean(axis=0) for i in range(self.k)
                    ])
                    
                    # Check convergence
                    if np.allclose(self.centroids, new_centroids):
                        break
                    
                    self.centroids = new_centroids
                
                return labels
        
        return KMeans
