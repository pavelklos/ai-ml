# 08. Cross-Validation and Model Selection Art

## Key Terms and Concepts

**Cross-Validation (CV)**: A resampling technique used to evaluate machine learning models by partitioning data into subsets, training on some subsets, and validating on others. This process is repeated multiple times to obtain a more robust estimate of model performance.

**Hyperparameters**: Configuration settings for machine learning algorithms that are set before training begins (e.g., learning rate, regularization strength, tree depth). Unlike model parameters, hyperparameters are not learned from data.

**Train-Test Split**: The fundamental division of dataset into training data (used to fit the model) and testing data (used to evaluate final model performance on unseen data).

**Validation Set**: A subset of training data used during model development to tune hyperparameters and make model selection decisions without touching the test set.

**Overfitting**: When a model learns the training data too well, including noise and random fluctuations, resulting in poor generalization to new data.

**Underfitting**: When a model is too simple to capture the underlying patterns in the data, resulting in poor performance on both training and test data.

**Bias-Variance Tradeoff**: The fundamental tradeoff in machine learning between a model's ability to minimize bias (error from oversimplifying assumptions) and variance (error from sensitivity to small fluctuations in training set).

## The Importance of Cross-Validation

Cross-validation serves as a critical bridge between model training and real-world deployment. Without proper validation techniques, we risk deploying models that appear successful in development but fail catastrophically in production. The primary purposes include:

### Model Performance Estimation
Cross-validation provides a more reliable estimate of how a model will perform on unseen data by testing it on multiple different subsets of the available data. This approach reduces the variance in performance estimates that can occur when using a single train-test split.

### Hyperparameter Optimization
Modern machine learning algorithms contain numerous hyperparameters that significantly impact model performance. Cross-validation enables systematic hyperparameter tuning by providing stable performance metrics across different parameter configurations.

### Model Selection
When comparing different algorithms or architectures, cross-validation ensures that the selection is based on robust performance estimates rather than lucky splits that favor particular models.

```python
# For regression with stratified sampling based on target distribution
def stratified_regression_split(X, y, test_size=0.2, n_bins=5, random_state=42):
    """
    Create stratified split for regression by binning target variable
    """
    # Discretize target for stratification
    discretizer = KBinsDiscretizer(n_bins=n_bins, encode='ordinal', strategy='quantile')
    y_binned = discretizer.fit_transform(y.reshape(-1, 1)).ravel()
    
    # Perform stratified split
    sss = StratifiedShuffleSplit(n_splits=1, test_size=test_size, random_state=random_state)
    train_idx, test_idx = next(sss.split(X, y_binned))
    
    return train_idx, test_idx
```

```python
def monte_carlo_cv(X, y, model, n_iterations=100, test_size=0.2):
    """
    Monte Carlo cross-validation with random sampling
    """
    scores = []
    
    for i in range(n_iterations):
        X_train_mc, X_test_mc, y_train_mc, y_test_mc = train_test_split(
            X, y, test_size=test_size, random_state=i
        )
        
        model.fit(X_train_mc, y_train_mc)
        score = model.score(X_test_mc, y_test_mc)
        scores.append(score)
    
    return np.array(scores)
```

```python
from sklearn.preprocessing import PowerTransformer, QuantileTransformer
from sklearn.feature_selection import SelectKBest, f_regression, RFE

class AdvancedDataPreprocessor:
    """
    Advanced data preprocessing pipeline with outlier detection and feature engineering
    """
    
    def __init__(self, outlier_method='iqr', missing_strategy='knn', scaling_method='standard'):
        self.outlier_method = outlier_method
        self.missing_strategy = missing_strategy
        self.scaling_method = scaling_method
        self.outlier_indices_ = None
        self.feature_stats_ = None
        
    def detect_outliers_iqr(self, X, multiplier=1.5):
        """Detect outliers using IQR method"""
        Q1 = np.percentile(X, 25, axis=0)
        Q3 = np.percentile(X, 75, axis=0)
        IQR = Q3 - Q1
        
        lower_bound = Q1 - multiplier * IQR
        upper_bound = Q3 + multiplier * IQR
        
        outlier_mask = (X < lower_bound) | (X > upper_bound)
        return outlier_mask.any(axis=1)
    
    def detect_outliers_zscore(self, X, threshold=3):
        """Detect outliers using Z-score method"""
        z_scores = np.abs(stats.zscore(X, nan_policy='omit'))
        return (z_scores > threshold).any(axis=1)
    
    def create_preprocessing_pipeline(self, X, y=None):
        """Create comprehensive preprocessing pipeline"""
        
        # Identify numeric columns (assuming all are numeric for this example)
        numeric_features = list(range(X.shape[1]))
        
        # Missing value imputation
        if self.missing_strategy == 'simple':
            imputer = SimpleImputer(strategy='median')
        elif self.missing_strategy == 'knn':
            imputer = KNNImputer(n_neighbors=5)
        else:
            imputer = SimpleImputer(strategy='median')
        
        # Scaling
        if self.scaling_method == 'standard':
            scaler = StandardScaler()
        elif self.scaling_method == 'robust':
            scaler = RobustScaler()
        elif self.scaling_method == 'power':
            scaler = PowerTransformer(method='yeo-johnson')
        else:
            scaler = StandardScaler()
        
        # Create pipeline
        preprocessing_steps = [
            ('imputer', imputer),
            ('scaler', scaler)
        ]
        
        numeric_transformer = Pipeline(steps=preprocessing_steps)
        
        # Column transformer
        preprocessor = ColumnTransformer(
            transformers=[
                ('num', numeric_transformer, numeric_features)
            ],
            remainder='passthrough'
        )
        
        return preprocessor
```

### Bayesian Optimization (Conceptual Implementation)

```python
from scipy.optimize import minimize
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import Matern

class BayesianOptimizer:
    """
    Simplified Bayesian optimization for hyperparameter tuning
    """
    
    def __init__(self, objective_function, bounds, n_initial=5, n_iterations=20):
        self.objective_function = objective_function
        self.bounds = bounds
        self.n_initial = n_initial
        self.n_iterations = n_iterations
        self.X_observed = []
        self.y_observed = []
        
    def acquisition_function(self, X_test, gp, xi=0.01):
        """Expected Improvement acquisition function"""
        mu, sigma = gp.predict(X_test.reshape(-1, len(self.bounds)), return_std=True)
        
        if len(self.y_observed) == 0:
            return mu
        
        mu_best = max(self.y_observed)
        
        with np.errstate(divide='warn'):
            imp = mu - mu_best - xi
            Z = imp / sigma
            ei = imp * stats.norm.cdf(Z) + sigma * stats.norm.pdf(Z)
            ei[sigma == 0.0] = 0.0
        
        return ei
    
    def optimize(self):
        """Perform Bayesian optimization"""
        # Initial random sampling
        for _ in range(self.n_initial):
            x = [np.random.uniform(bound[0], bound[1]) for bound in self.bounds]
            y = self.objective_function(x)
            self.X_observed.append(x)
            self.y_observed.append(y)
        
        # Bayesian optimization iterations
        for iteration in range(self.n_iterations):
            # Fit Gaussian Process
            gp = GaussianProcessRegressor(
                kernel=Matern(length_scale=1.0, nu=2.5),
                alpha=1e-6,
                normalize_y=True,
                n_restarts_optimizer=5,
                random_state=42
            )
            
            gp.fit(self.X_observed, self.y_observed)
            
            # Find next point to evaluate
            def neg_acquisition(x):
                return -self.acquisition_function(np.array(x), gp)
            
            # Multiple random starts for optimization
            best_x = None
            best_acq = float('inf')
            
            for _ in range(10):
                x0 = [np.random.uniform(bound[0], bound[1]) for bound in self.bounds]
                result = minimize(neg_acquisition, x0, bounds=self.bounds, method='L-BFGS-B')
                
                if result.fun < best_acq:
                    best_acq = result.fun
                    best_x = result.x
            
            # Evaluate objective at new point
            y_new = self.objective_function(best_x)
            self.X_observed.append(best_x.tolist())
            self.y_observed.append(y_new)
            
            print(f"Iteration {iteration + 1}: Best so far = {max(self.y_observed):.4f}")
        
        # Return best result
        best_idx = np.argmax(self.y_observed)
        return self.X_observed[best_idx], self.y_observed[best_idx]
```

```python
def nested_cross_validation(X, y, models_with_params, outer_cv, inner_cv, scoring='r2'):
```

### Custom Cross-Validation for Specific Use Cases

```python
from sklearn.model_selection import BaseCrossValidator

class GroupTimeSeriesSplit(BaseCrossValidator):
    """
    Custom cross-validation for grouped time series data
    """
    
    def __init__(self, n_splits=5, max_train_size=None):
        self.n_splits = n_splits
        self.max_train_size = max_train_size
    
    def split(self, X, y=None, groups=None):
        if groups is None:
            raise ValueError("Groups must be provided")
        
        unique_groups = np.unique(groups)
        n_groups = len(unique_groups)
        
        if n_groups < self.n_splits:
            raise ValueError(f"Cannot have {self.n_splits} splits with only {n_groups} groups")
        
        group_size = n_groups // self.n_splits
        
        for i in range(self.n_splits):
            start_idx = i * group_size
            if i == self.n_splits - 1:
                end_idx = n_groups
            else:
                end_idx = (i + 1) * group_size
            
            test_groups = unique_groups[start_idx:end_idx]
            train_groups = np.setdiff1d(unique_groups, test_groups)
            
            if self.max_train_size is not None and len(train_groups) > self.max_train_size:
                train_groups = train_groups[-self.max_train_size:]
            
            train_indices = np.isin(groups, train_groups)
            test_indices = np.isin(groups, test_groups)
            
            yield np.where(train_indices)[0], np.where(test_indices)[0]
    
    def get_n_splits(self, X=None, y=None, groups=None):
        return self.n_splits
```

```python
def bootstrap_validation(X, y, model, n_bootstrap=1000, confidence_level=0.95):
    """
    Bootstrap validation for confidence intervals of model performance
    """
    
    bootstrap_scores = []
    n_samples = len(X)
    
    for i in range(n_bootstrap):
        # Bootstrap sample
        bootstrap_indices = np.random.choice(n_samples, size=n_samples, replace=True)
        X_bootstrap = X.iloc[bootstrap_indices]
        y_bootstrap = y.iloc[bootstrap_indices]
        
        # Out-of-bag samples
        oob_indices = np.setdiff1d(np.arange(n_samples), bootstrap_indices)
        if len(oob_indices) == 0:
            continue
            
        X_oob = X.iloc[oob_indices]
        y_oob = y.iloc[oob_indices]
        
        # Fit and evaluate
        model.fit(X_bootstrap, y_bootstrap)
        score = model.score(X_oob, y_oob)
        bootstrap_scores.append(score)
    
    bootstrap_scores = np.array(bootstrap_scores)
    
    # Calculate confidence intervals
    alpha = 1 - confidence_level
    lower_percentile = (alpha / 2) * 100
    upper_percentile = (1 - alpha / 2) * 100
    
    confidence_interval = np.percentile(bootstrap_scores, [lower_percentile, upper_percentile])
    
    return {
        'mean_score': bootstrap_scores.mean(),
        'std_score': bootstrap_scores.std(),
        'confidence_interval': confidence_interval,
        'all_scores': bootstrap_scores
    }

# Perform bootstrap validation
```

### Progressive Validation for Large Datasets

```python
def progressive_validation(X, y, model, validation_sizes=None, cv_strategy=None):
    """
    Progressive validation to understand model performance with increasing data size
    """
    
    if validation_sizes is None:
        validation_sizes = np.logspace(2, np.log10(len(X)), 10).astype(int)
        validation_sizes = validation_sizes[validation_sizes <= len(X)]
    
    if cv_strategy is None:
        cv_strategy = KFold(n_splits=5, shuffle=True, random_state=42)
    
    results = {
        'train_sizes': [],
        'train_scores': [],
        'val_scores': [],
        'train_times': [],
        'val_times': []
    }
    
    for size in validation_sizes:
        print(f"Evaluating with {size} samples...")
        
        # Sample data
        indices = np.random.choice(len(X), size=size, replace=False)
        X_subset = X.iloc[indices]
        y_subset = y.iloc[indices]
        
        # Perform cross-validation
        start_time = time.time()
        
        train_scores = []
        val_scores = []
        train_times = []
        val_times = []
        
        for train_idx, val_idx in cv_strategy.split(X_subset):
            X_train_fold = X_subset.iloc[train_idx]
            X_val_fold = X_subset.iloc[val_idx]
            y_train_fold = y_subset.iloc[train_idx]
            y_val_fold = y_subset.iloc[val_idx]
            
            # Training
            train_start = time.time()
            model.fit(X_train_fold, y_train_fold)
            train_time = time.time() - train_start
            
            # Validation
            val_start = time.time()
            train_score = model.score(X_train_fold, y_train_fold)
            val_score = model.score(X_val_fold, y_val_fold)
            val_time = time.time() - val_start
            
            train_scores.append(train_score)
            val_scores.append(val_score)
            train_times.append(train_time)
            val_times.append(val_time)
        
        results['train_sizes'].append(size)
        results['train_scores'].append(np.mean(train_scores))
        results['val_scores'].append(np.mean(val_scores))
        results['train_times'].append(np.mean(train_times))
        results['val_times'].append(np.mean(val_times))
    
    return results
```
