# 03. Classification: Teaching Computers to Recognize

## Key Terms and Concepts

**Classification** is a supervised machine learning task where the goal is to predict discrete categories or classes for input data. Unlike regression which predicts continuous values, classification assigns labels to data points based on learned patterns from training examples.

**Binary Classification** involves predicting one of two possible outcomes (e.g., spam/not spam, positive/negative sentiment). **Multi-class Classification** extends this to predict among three or more categories (e.g., image recognition with multiple object types).

**Logistic Regression** uses the logistic function (sigmoid) to model the probability of class membership, despite its name being a classification algorithm. **K-Nearest Neighbors (KNN)** classifies data points based on the majority class among the k nearest neighbors in feature space.

**Class Imbalance** occurs when some classes have significantly fewer samples than others, which can bias model performance. **Evaluation Metrics** like accuracy, precision, recall, and F1-score help assess model quality beyond simple correctness rates.

## Logistic Regression Deep Dive

Logistic regression models the probability of class membership using the sigmoid function:

```python
class LogisticRegressionDetailed:
    def __init__(self, learning_rate=0.01, max_iterations=1000):
        self.learning_rate = learning_rate
        self.max_iterations = max_iterations
        
    def sigmoid(self, z):
        """Sigmoid activation function with numerical stability"""
        z = np.clip(z, -500, 500)  # Prevent overflow
        return 1 / (1 + np.exp(-z))
    
    def fit(self, X, y):
        """Train logistic regression using gradient descent"""
        # Add bias term
        X_bias = np.c_[np.ones(X.shape[0]), X]
        self.weights = np.random.normal(0, 0.01, X_bias.shape[1])
        
        self.cost_history = []
        
        for i in range(self.max_iterations):
            # Forward pass
            z = X_bias @ self.weights
            predictions = self.sigmoid(z)
            
            # Cost function (log-likelihood)
            cost = -np.mean(y * np.log(predictions + 1e-15) + 
                           (1 - y) * np.log(1 - predictions + 1e-15))
            self.cost_history.append(cost)
            
            # Gradient computation
            gradient = X_bias.T @ (predictions - y) / len(y)
            
            # Weight update
            self.weights -= self.learning_rate * gradient
            
            if i % 100 == 0:
                print(f"Iteration {i}, Cost: {cost:.4f}")
    
    def predict_proba(self, X):
        """Predict class probabilities"""
        X_bias = np.c_[np.ones(X.shape[0]), X]
        return self.sigmoid(X_bias @ self.weights)
    
    def predict(self, X):
        """Make binary predictions"""
        return (self.predict_proba(X) >= 0.5).astype(int)

# Train custom logistic regression
```

## K-Nearest Neighbors Implementation

KNN is a lazy learning algorithm that makes predictions based on local neighborhood patterns:

```python
class KNearestNeighborsDetailed:
    def __init__(self, k=5, distance_metric='euclidean'):
        self.k = k
        self.distance_metric = distance_metric
        
    def euclidean_distance(self, x1, x2):
        """Calculate Euclidean distance between two points"""
        return np.sqrt(np.sum((x1 - x2) ** 2))
    
    def manhattan_distance(self, x1, x2):
        """Calculate Manhattan distance between two points"""
        return np.sum(np.abs(x1 - x2))
    
    def fit(self, X, y):
        """Store training data (lazy learning)"""
        self.X_train = X
        self.y_train = y
        
    def predict(self, X):
        """Make predictions for test data"""
        predictions = []
        
        for test_point in X:
            distances = []
            
            # Calculate distances to all training points
            for i, train_point in enumerate(self.X_train):
                if self.distance_metric == 'euclidean':
                    dist = self.euclidean_distance(test_point, train_point)
                else:
                    dist = self.manhattan_distance(test_point, train_point)
                distances.append((dist, self.y_train[i]))
            
            # Sort by distance and get k nearest neighbors
            distances.sort(key=lambda x: x[0])
            k_nearest = distances[:self.k]
            
            # Majority vote
            k_nearest_labels = [label for _, label in k_nearest]
            prediction = max(set(k_nearest_labels), key=k_nearest_labels.count)
            predictions.append(prediction)
            
        return np.array(predictions)

# Compare different k values
k_values = [1, 3, 5, 7, 9, 15, 21]
knn_scores = []

for k in k_values:
    knn = KNeighborsClassifier(n_neighbors=k)

    knn_scores.append(scores.mean())
    print(f"K={k}: Cross-validation accuracy = {scores.mean():.4f} (+/- {scores.std() * 2:.4f})")

# Find optimal k
optimal_k = k_values[np.argmax(knn_scores)]
print(f"\nOptimal k: {optimal_k}")
```

## Comprehensive Model Evaluation

Proper evaluation requires multiple metrics beyond accuracy:

```python
def comprehensive_evaluation(y_true, y_pred, y_pred_proba=None, class_names=None):
    """Comprehensive model evaluation with multiple metrics"""
    
    print("=== Classification Report ===")
    print(classification_report(y_true, y_pred, target_names=class_names))
    
    # Confusion Matrix
    cm = confusion_matrix(y_true, y_pred)
    plt.figure(figsize=(8, 6))
    sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
                xticklabels=class_names, yticklabels=class_names)
    plt.title('Confusion Matrix')
    plt.ylabel('True Label')
    plt.xlabel('Predicted Label')
    plt.show()
    
    # Individual metrics
    accuracy = accuracy_score(y_true, y_pred)
    precision = precision_score(y_true, y_pred, average='weighted')
    recall = recall_score(y_true, y_pred, average='weighted')
    f1 = f1_score(y_true, y_pred, average='weighted')
    
    print(f"\n=== Individual Metrics ===")
    print(f"Accuracy:  {accuracy:.4f}")
    print(f"Precision: {precision:.4f}")
    print(f"Recall:    {recall:.4f}")
    print(f"F1-Score:  {f1:.4f}")
    
    # ROC curve for binary classification
    if len(np.unique(y_true)) == 2 and y_pred_proba is not None:
        auc_score = roc_auc_score(y_true, y_pred_proba)
        fpr, tpr, _ = roc_curve(y_true, y_pred_proba)
        
        plt.figure(figsize=(8, 6))
        plt.plot(fpr, tpr, label=f'ROC Curve (AUC = {auc_score:.4f})')
        plt.plot([0, 1], [0, 1], 'k--', label='Random Classifier')
        plt.xlabel('False Positive Rate')
        plt.ylabel('True Positive Rate')
        plt.title('ROC Curve')
        plt.legend()
        plt.grid(True)
        plt.show()
        
        print(f"AUC Score: {auc_score:.4f}")

# Evaluate logistic regression
lr_pred_proba = sklearn_lr.predict_proba(X_test_scaled)[:, 1]
comprehensive_evaluation(y_test, sklearn_pred, lr_pred_proba, target_names)
```

## Handling Imbalanced Datasets

Class imbalance is a common challenge requiring specialized techniques:

```python
from imblearn.over_sampling import SMOTE, RandomOverSampler
from imblearn.under_sampling import RandomUnderSampler
from imblearn.combine import SMOTETomek
from sklearn.utils.class_weight import compute_class_weight
from collections import Counter

# Create imbalanced dataset
X_imb, y_imb = make_classification(
    n_samples=1000, n_features=20, n_informative=10,
    n_redundant=10, n_clusters_per_class=1, weights=[0.9, 0.1],
    random_state=42
)

print(f"Original dataset distribution: {Counter(y_imb)}")

# Split imbalanced data
X_train_imb, X_test_imb, y_train_imb, y_test_imb = train_test_split(
    X_imb, y_imb, test_size=0.2, random_state=42, stratify=y_imb
)

# Resampling techniques
resampling_methods = {
    'Original': (X_train_imb, y_train_imb),
    'Random Oversampling': RandomOverSampler(random_state=42).fit_resample(X_train_imb, y_train_imb),
    'SMOTE': SMOTE(random_state=42).fit_resample(X_train_imb, y_train_imb),
    'Random Undersampling': RandomUnderSampler(random_state=42).fit_resample(X_train_imb, y_train_imb),
    'SMOTE + Tomek': SMOTETomek(random_state=42).fit_resample(X_train_imb, y_train_imb)
}

# Evaluate each method
results = {}
for method_name, (X_resampled, y_resampled) in resampling_methods.items():
    print(f"\n=== {method_name} ===")
    print(f"Distribution: {Counter(y_resampled)}")
    
    # Train model
    lr_imb = LogisticRegression(random_state=42, max_iter=1000)
    lr_imb.fit(X_resampled, y_resampled)
    
    # Predictions
    pred_imb = lr_imb.predict(X_test_imb)
    
    # Evaluate
    accuracy = accuracy_score(y_test_imb, pred_imb)
    precision = precision_score(y_test_imb, pred_imb)
    recall = recall_score(y_test_imb, pred_imb)
    f1 = f1_score(y_test_imb, pred_imb)
    
    results[method_name] = {
        'Accuracy': accuracy,
        'Precision': precision,
        'Recall': recall,
        'F1-Score': f1
    }
    
    print(f"Accuracy: {accuracy:.4f}, Precision: {precision:.4f}")
    print(f"Recall: {recall:.4f}, F1-Score: {f1:.4f}")

# Class weight balancing
class_weights = compute_class_weight('balanced', classes=np.unique(y_train_imb), y=y_train_imb)
class_weight_dict = dict(zip(np.unique(y_train_imb), class_weights))

lr_weighted = LogisticRegression(
    class_weight=class_weight_dict, 
    random_state=42, 
    max_iter=1000
)
lr_weighted.fit(X_train_imb, y_train_imb)

pred_weighted = lr_weighted.predict(X_test_imb)
print(f"\n=== Class Weight Balancing ===")

print(f"F1-Score: {f1_score(y_test_imb, pred_weighted):.4f}")
```

## Advanced Classification Pipeline

Modern ML workflows require robust pipelines that handle preprocessing, model selection, and evaluation:

```python
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.preprocessing import StandardScaler, RobustScaler
from sklearn.model_selection import GridSearchCV, RandomizedSearchCV
from sklearn.ensemble import VotingClassifier
from sklearn.svm import SVC

class AdvancedClassificationPipeline:
    def __init__(self, random_state=42):
        self.random_state = random_state
        self.best_model = None
        self.results = {}
        
    def create_models(self):
        """Create diverse set of classification models"""
        models = {
            'Logistic Regression': LogisticRegression(
                random_state=self.random_state, 
                max_iter=1000
            ),
            'K-Nearest Neighbors': KNeighborsClassifier(),
            'Support Vector Machine': SVC(
                random_state=self.random_state, 
                probability=True
            ),
            'Random Forest': RandomForestClassifier(
                random_state=self.random_state
            )
        }
        return models
    
    def create_preprocessing_pipeline(self):
        """Create preprocessing pipeline"""
        return Pipeline([
            ('scaler', StandardScaler()),
        ])
    
    def hyperparameter_tuning(self, X, y, cv=5):
        """Perform hyperparameter tuning for each model"""
        models = self.create_models()
        preprocessing = self.create_preprocessing_pipeline()
        
        param_grids = {
            'Logistic Regression': {
                'model__C': [0.1, 1, 10, 100],
                'model__penalty': ['l1', 'l2'],
                'model__solver': ['liblinear']
            },
            'K-Nearest Neighbors': {
                'model__n_neighbors': [3, 5, 7, 9, 11],
                'model__weights': ['uniform', 'distance'],
                'model__metric': ['euclidean', 'manhattan']
            },
            'Support Vector Machine': {
                'model__C': [0.1, 1, 10],
                'model__gamma': ['scale', 'auto', 0.1, 1],
                'model__kernel': ['rbf', 'linear']
            },
            'Random Forest': {
                'model__n_estimators': [50, 100, 200],
                'model__max_depth': [3, 5, 7, None],
                'model__min_samples_split': [2, 5, 10]
            }
        }
        
        best_models = {}
        
        for name, model in models.items():
            print(f"\nTuning {name}...")
            
            # Create pipeline
            pipeline = Pipeline([
                ('preprocessing', preprocessing),
                ('model', model)
            ])
            
            # Grid search
            grid_search = GridSearchCV(
                pipeline, 
                param_grids[name], 
                cv=cv, 
                scoring='f1_weighted',
                n_jobs=-1
            )
            
            grid_search.fit(X, y)
            best_models[name] = grid_search.best_estimator_
            
            print(f"Best parameters: {grid_search.best_params_}")
            print(f"Best cross-validation score: {grid_search.best_score_:.4f}")
            
        return best_models
    
    def create_ensemble(self, models):
        """Create voting ensemble from best models"""
        estimators = [(name, model) for name, model in models.items()]
        
        ensemble = VotingClassifier(
            estimators=estimators,
            voting='soft'  # Use probability voting
        )
        
        return ensemble
    
    def fit(self, X, y):
        """Train the complete pipeline"""
        # Hyperparameter tuning
        best_models = self.hyperparameter_tuning(X, y)
        
        # Create ensemble
        ensemble = self.create_ensemble(best_models)
        
        # Cross-validation evaluation
        cv_scores = cross_val_score(ensemble, X, y, cv=5, scoring='f1_weighted')
        
        print(f"\n=== Ensemble Model ===")
        print(f"Cross-validation F1-score: {cv_scores.mean():.4f} (+/- {cv_scores.std() * 2:.4f})")
        
        # Fit final model
        ensemble.fit(X, y)
        self.best_model = ensemble
        
        return self

# Train advanced pipeline
advanced_pipeline = AdvancedClassificationPipeline()
advanced_pipeline.fit(X_train_scaled, y_train)

# Final evaluation
if advanced_pipeline.best_model:
    final_pred = advanced_pipeline.best_model.predict(X_test_scaled)
    final_pred_proba = advanced_pipeline.best_model.predict_proba(X_test_scaled)[:, 1]
    
    print(f"\n=== Final Model Performance ===")
    comprehensive_evaluation(y_test, final_pred, final_pred_proba, target_names)
```
