# 06. Data Transformers: Creating Effective Models

## Key Terms and Concepts

**Data Transformers** are preprocessing components that modify raw data into formats suitable for machine learning algorithms. They implement the transformer interface with `fit()`, `transform()`, and `fit_transform()` methods.

**Feature Engineering** is the process of selecting, modifying, or creating features from raw data to improve model performance. It's often the most critical step in the machine learning pipeline.

**Categorical Encoding** converts non-numerical categorical variables into numerical representations that algorithms can process. Common methods include one-hot encoding, label encoding, and target encoding.

**Polynomial Features** create interaction terms and higher-order features from existing numerical variables, allowing linear models to capture non-linear relationships.

**Column Transformer** enables applying different transformations to different subsets of features in a single pipeline step.

**Custom Transformers** are user-defined preprocessing components that inherit from scikit-learn's base classes and implement domain-specific transformations.

### Column Transformer for Heterogeneous Data

```python
# Create comprehensive preprocessing pipeline
def create_advanced_pipeline():
    # Define different preprocessing for different column types
    
    # Numerical preprocessing
    numerical_transformer = Pipeline(steps=[
        ('scaler', StandardScaler()),
        ('poly', PolynomialFeatures(degree=2, interaction_only=True, include_bias=False))
    ])
    
    # Categorical preprocessing - different strategies for different columns
    low_cardinality_transformer = Pipeline(steps=[
        ('onehot', OneHotEncoder(drop='first', handle_unknown='ignore', sparse_output=False))
    ])
    
    high_cardinality_transformer = Pipeline(steps=[
        ('target_enc', TargetEncoder(smooth='auto', cv=5, random_state=42))
    ])
    
    ordinal_transformer = Pipeline(steps=[
        ('ordinal', OrdinalEncoder(
            categories=[['high_school', 'bachelor', 'master', 'phd']], 
            handle_unknown='use_encoded_value', 
            unknown_value=-1
        ))
    ])
    
    # Combine all transformers
    preprocessor = ColumnTransformer(
        transformers=[
            ('num', numerical_transformer, numerical_cols),
            ('cat_low', low_cardinality_transformer, ['job_category']),
            ('cat_high', high_cardinality_transformer, ['city']),
            ('cat_ord', ordinal_transformer, ['education'])
        ],
        remainder='drop',  # Drop columns not specified
        n_jobs=-1
    )
    
    # Complete pipeline with model
    pipeline = Pipeline([
        ('preprocessor', preprocessor),
        ('classifier', RandomForestClassifier(n_estimators=100, random_state=42))
    ])
    
    return pipeline

# Train and evaluate the advanced pipeline
```

## Creating Custom Transformers

Custom transformers allow you to implement domain-specific preprocessing logic while maintaining compatibility with scikit-learn pipelines.

```python
class AdvancedFeatureEngineer(BaseEstimator, TransformerMixin):
    """
    Custom transformer for advanced feature engineering
    """
    def __init__(self, create_ratios=True, create_binning=True, create_interactions=True):
        self.create_ratios = create_ratios
        self.create_binning = create_binning
        self.create_interactions = create_interactions
        self.binning_thresholds_ = None
        self.feature_names_out_ = None
    
    def fit(self, X, y=None):
        X = pd.DataFrame(X) if not isinstance(X, pd.DataFrame) else X
        
        # Calculate binning thresholds for numerical features
        if self.create_binning:
            self.binning_thresholds_ = {}
            numerical_features = X.select_dtypes(include=[np.number]).columns
            for col in numerical_features:
                self.binning_thresholds_[col] = [
                    X[col].quantile(0.25),
                    X[col].quantile(0.50),
                    X[col].quantile(0.75)
                ]
        
        return self
    
    def transform(self, X):
        X = pd.DataFrame(X) if not isinstance(X, pd.DataFrame) else X.copy()
        X_transformed = X.copy()
        
        numerical_features = X.select_dtypes(include=[np.number]).columns
        
        # Create ratio features
        if self.create_ratios and len(numerical_features) >= 2:
            for i, col1 in enumerate(numerical_features):
                for col2 in numerical_features[i+1:]:
                    # Avoid division by zero
                    ratio_name = f"{col1}_to_{col2}_ratio"
                    X_transformed[ratio_name] = np.where(
                        X[col2] != 0, 
                        X[col1] / X[col2], 
                        0
                    )
        
        # Create binned features
        if self.create_binning and self.binning_thresholds_:
            for col, thresholds in self.binning_thresholds_.items():
                if col in X_transformed.columns:
                    binned_name = f"{col}_binned"
                    X_transformed[binned_name] = pd.cut(
                        X_transformed[col], 
                        bins=[-np.inf] + thresholds + [np.inf], 
                        labels=['low', 'medium_low', 'medium_high', 'high']
                    ).astype(str)
        
        # Create interaction features
        if self.create_interactions and len(numerical_features) >= 2:
            for i, col1 in enumerate(numerical_features[:3]):  # Limit to avoid explosion
                for col2 in numerical_features[i+1:4]:
                    interaction_name = f"{col1}_x_{col2}"
                    X_transformed[interaction_name] = X[col1] * X[col2]
        
        return X_transformed
    
    def get_feature_names_out(self, input_features=None):
        return self.feature_names_out_

class DomainSpecificTransformer(BaseEstimator, TransformerMixin):
    """
    Domain-specific transformer for business logic
    """
    def __init__(self, income_categories=True, age_groups=True):
        self.income_categories = income_categories
        self.age_groups = age_groups
    
    def fit(self, X, y=None):
        return self
    
    def transform(self, X):
        X = pd.DataFrame(X) if not isinstance(X, pd.DataFrame) else X.copy()
        
        # Create income categories
        if self.income_categories and 'income' in X.columns:
            X['income_category'] = pd.cut(
                X['income'],
                bins=[0, 30000, 60000, 100000, np.inf],
                labels=['low', 'medium', 'high', 'very_high']
            ).astype(str)
        
        # Create age groups
        if self.age_groups and 'age' in X.columns:
            X['age_group'] = pd.cut(
                X['age'],
                bins=[0, 25, 35, 50, 65, np.inf],
                labels=['young', 'young_adult', 'middle_aged', 'senior', 'elderly']
            ).astype(str)
            
            # Create experience-to-age ratio
            if 'experience_years' in X.columns:
                X['experience_ratio'] = X['experience_years'] / (X['age'] - 18).clip(lower=1)
        
        return X

# Test custom transformers
```

## Best Practices and Pipeline Management

```python
class PipelineManager:
    """
    Utility class for managing complex pipelines
    """
    def __init__(self):
        self.pipelines = {}
        self.results = {}
    
    def add_pipeline(self, name, pipeline):
        """Add a pipeline to the manager"""
        self.pipelines[name] = pipeline
    
    def compare_pipelines(self, X_train, y_train, X_test, y_test, cv=5):
        """Compare all registered pipelines"""
        results = {}
        
        for name, pipeline in self.pipelines.items():
            print(f"\nEvaluating {name}...")
            
            # Cross-validation
            cv_scores = cross_val_score(pipeline, X_train, y_train, cv=cv, scoring='f1')
            
            # Test performance
            pipeline.fit(X_train, y_train)
            y_pred = pipeline.predict(X_test)
            
            from sklearn.metrics import f1_score, accuracy_score, precision_score, recall_score
            
            results[name] = {
                'cv_f1_mean': cv_scores.mean(),
                'cv_f1_std': cv_scores.std(),
                'test_f1': f1_score(y_test, y_pred),
                'test_accuracy': accuracy_score(y_test, y_pred),
                'test_precision': precision_score(y_test, y_pred),
                'test_recall': recall_score(y_test, y_pred)
            }
        
        self.results = results
        return pd.DataFrame(results).T
    
    def get_best_pipeline(self, metric='test_f1'):
        """Get the best performing pipeline"""
        if not self.results:
            raise ValueError("No results available. Run compare_pipelines first.")
        
        best_name = max(self.results.keys(), key=lambda x: self.results[x][metric])
        return best_name, self.pipelines[best_name]
```
