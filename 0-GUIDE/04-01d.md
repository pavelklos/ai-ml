## Key Terms and Concepts

- **Artificial Intelligence (AI)**: A field of computer science dedicated to creating systems capable of performing tasks that typically require human intelligence.
- **Machine Learning (ML)**: A subset of AI focusing on algorithms that improve through experience without explicit programming.
- **Supervised Learning**: Learning from labeled training data to make predictions on unseen data.
- **Unsupervised Learning**: Discovering patterns in unlabeled data without predetermined outputs.
- **Optimization Problem**: Mathematical formulation seeking to find parameters that minimize or maximize an objective function.
- **Objective Function**: Function to be optimized (minimized or maximized) in an optimization problem.
- **Loss/Cost Function**: Specific type of objective function measuring prediction error.
- **Gradient Descent**: First-order iterative optimization algorithm for finding local minima.
- **Convex Optimization**: Special case with guaranteed global minimum.
- **NumPy**: Fundamental Python library for numerical computing with powerful N-dimensional array objects.
- **SciPy**: Scientific computing library built on NumPy with modules for optimization, linear algebra, integration, and more.
- **Learning Rate**: Hyperparameter controlling step size in gradient descent.
- **Local vs. Global Minima**: Distinction between points that are minimal within a neighborhood versus minimal over the entire domain.
- **Gradient**: Vector of partial derivatives indicating the direction of steepest ascent.

## Theoretical Foundation of Optimization Problems

Optimization is the mathematical backbone of machine learning. The general form of an optimization problem is:

$$\min_{\theta} f(\theta) \text{ subject to } g_i(\theta) \leq 0, h_j(\theta) = 0$$

Where:
- $\theta$ represents the parameters to optimize
- $f(\theta)$ is the objective function
- $g_i(\theta)$ represents inequality constraints
- $h_j(\theta)$ represents equality constraints

In machine learning, we typically minimize a loss function that quantifies the discrepancy between model predictions and actual values:

```python
# Define a complex multivariate function for optimization demonstration
def rosenbrock(x):
    """
    Rosenbrock function: f(x,y) = (1-x)² + 100(y-x²)²
    Has a global minimum at (1,1) with f(1,1) = 0
    """
    return sum(100.0 * (x[1:] - x[:-1]**2.0)**2.0 + (1 - x[:-1])**2.0)

# Gradient of the Rosenbrock function for more efficient optimization
def rosenbrock_gradient(x):
    grad = np.zeros_like(x)
    grad[0] = -400 * x[0] * (x[1] - x[0]**2) - 2 * (1 - x[0])
    grad[-1] = 200 * (x[-1] - x[-2]**2)
    for i in range(1, len(x) - 1):
        grad[i] = 200 * (x[i] - x[i-1]**2) - 400 * x[i] * (x[i+1] - x[i]**2) - 2 * (1 - x[i])
    return grad

# Define constraints for constrained optimization
def sum_constraint(x):
    """Equality constraint: sum of parameters equals 1"""
    return np.sum(x) - 1.0

def sum_constraint_jac(x):
    """Jacobian of the equality constraint"""
    return np.ones_like(x)
```

```python
def trust_region_optimizer(func, x0, max_iter=100, initial_radius=1.0, tol=1e-6):
    """
    Custom implementation of trust-region optimization method
    """
    x = x0.copy()
    n = len(x)
    radius = initial_radius
    I = np.eye(n)
    
    history = []
    
    for iteration in range(max_iter):
        # Evaluate function and approximate gradient using finite differences
        f_current = func(x)
        history.append(f_current)
        
        # Compute gradient using finite differences
        grad = np.zeros(n)
        h = 1e-8
        for i in range(n):
            x_plus_h = x.copy()
            x_plus_h[i] += h
            grad[i] = (func(x_plus_h) - f_current) / h
        
        # Approximate Hessian using finite differences
        hess = np.zeros((n, n))
        for i in range(n):
            for j in range(n):
                x_plus_h_i = x.copy()
                x_plus_h_i[i] += h
                
                x_plus_h_j = x.copy()
                x_plus_h_j[j] += h
                
                x_plus_h_ij = x.copy()
                x_plus_h_ij[i] += h
                x_plus_h_ij[j] += h
                
                hess[i, j] = (func(x_plus_h_ij) - func(x_plus_h_i) - func(x_plus_h_j) + f_current) / (h * h)
        
        # Ensure the Hessian is positive definite
        eig_vals = np.linalg.eigvals(hess)
        if np.any(eig_vals <= 0):
            hess = hess + (abs(np.min(eig_vals)) + 0.1) * I
        
        # Solve the trust-region subproblem
        # min m(p) = f(x) + g^T p + 0.5 p^T B p  subject to ||p|| <= radius
        # Simplified approach: use a constrained optimizer
        def subproblem(p):
            return f_current + np.dot(grad, p) + 0.5 * np.dot(p, np.dot(hess, p))
        
        def radius_constraint(p):
            return radius - np.linalg.norm(p)
        
        subproblem_constraints = [{'type': 'ineq', 'fun': radius_constraint}]
        p0 = np.zeros(n)
        
        subproblem_result = optimize.minimize(
            subproblem, 
            p0, 
            method='SLSQP',
            constraints=subproblem_constraints
        )
        
        p = subproblem_result.x
        
        # Evaluate the actual reduction vs predicted reduction
        actual_reduction = f_current - func(x + p)
        predicted_reduction = -(np.dot(grad, p) + 0.5 * np.dot(p, np.dot(hess, p)))
        
        # Compute ratio
        ratio = actual_reduction / (predicted_reduction + 1e-10)
        
        # Update trust region radius
        if ratio < 0.25:
            radius = radius * 0.5
        elif ratio > 0.75 and np.linalg.norm(p) >= 0.9 * radius:
            radius = min(2.0 * radius, 10.0)
        
        # Update current point
        if ratio > 0.1:
            x = x + p
        
        # Check for convergence
        if np.linalg.norm(grad) < tol:
            break
    
    return {
        'x': x,
        'fun': func(x),
        'nit': iteration + 1,
        'success': iteration < max_iter - 1,
        'history': history
    }
```

```python
# Advanced optimization for polynomial regression
def polynomial_regression_optimization():
    # Generate synthetic data with non-linear relationship
    np.random.seed(42)
    n_samples = 200
    X = np.sort(np.random.uniform(0, 10, n_samples)).reshape(-1, 1)
    true_function = lambda x: 2 + 3*x - 0.5*x**2 + 0.1*x**3
    y_true = true_function(X)
    y = y_true + np.random.normal(0, 3, size=y_true.shape)  # Add noise
    
    # Split data into train and test sets
    X_train, X_test = X[:150], X[150:]
    y_train, y_test = y[:150], y[150:]
    
    # Create polynomial features
    def create_polynomial_features(X, degree):
        features = [np.ones(X.shape[0])]  # Bias term
        for i in range(1, degree + 1):
            features.append(X.flatten() ** i)
        return np.column_stack(features)
    
    # Create design matrices for different polynomial degrees
    degrees = [1, 3, 5, 10]
    design_matrices = {degree: create_polynomial_features(X_train, degree) for degree in degrees}
    test_matrices = {degree: create_polynomial_features(X_test, degree) for degree in degrees}
    
    # Define MSE loss function and its gradient for direct optimization
    def mse_loss(theta, X, y):
        predictions = X.dot(theta)
        error = predictions - y
        return np.mean(error ** 2)
    
    def mse_gradient(theta, X, y):
        m = X.shape[0]
        predictions = X.dot(theta)
        error = predictions - y
        return (2/m) * X.T.dot(error)
    
    # Ridge regression loss (L2 regularization)
    def ridge_loss(theta, X, y, alpha=1.0):
        predictions = X.dot(theta)
        error = predictions - y
        l2_penalty = alpha * np.sum(theta[1:] ** 2)  # Don't penalize bias term
        return np.mean(error ** 2) + l2_penalty
    
    def ridge_gradient(theta, X, y, alpha=1.0):
        m = X.shape[0]
        predictions = X.dot(theta)
        error = predictions - y
        grad = (2/m) * X.T.dot(error)
        grad[1:] += 2 * alpha * theta[1:]  # Add regularization gradient
        return grad
    
    # Optimize using different methods and compare
    results = {}
    
    for degree in degrees:
        X_poly = design_matrices[degree]
        X_test_poly = test_matrices[degree]
        
        # Initial parameters
        initial_theta = np.zeros(degree + 1)
        
        # Direct solution using normal equations
        direct_theta = np.linalg.inv(X_poly.T.dot(X_poly)).dot(X_poly.T).dot(y_train)
        
        # Regularized solution (Ridge regression)
        alpha = 0.1
        reg_theta = np.linalg.inv(X_poly.T.dot(X_poly) + alpha * np.eye(degree + 1)).dot(X_poly.T).dot(y_train)
        
        # Gradient Descent optimization
        def gd_optimizer(X, y, initial_theta, learning_rate=0.01, iterations=1000, tol=1e-6):
            theta = initial_theta.copy()
            cost_history = np.zeros(iterations)
            
            for i in range(iterations):
                gradient = mse_gradient(theta, X, y)
                theta = theta - learning_rate * gradient
                cost_history[i] = mse_loss(theta, X, y)
                
                if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
                    return theta, cost_history[:i+1]
            
            return theta, cost_history
        
        # L-BFGS optimization
        lbfgs_result = optimize.minimize(
            lambda theta: mse_loss(theta, X_poly, y_train),
            initial_theta,
            method='L-BFGS-B',
            jac=lambda theta: mse_gradient(theta, X_poly, y_train),
            options={'maxiter': 1000, 'disp': False}
        )
        
        # Ridge regression with L-BFGS
        ridge_result = optimize.minimize(
            lambda theta: ridge_loss(theta, X_poly, y_train, alpha),
            initial_theta,
            method='L-BFGS-B',
            jac=lambda theta: ridge_gradient(theta, X_poly, y_train, alpha),
            options={'maxiter': 1000, 'disp': False}
        )
        
        # Gradient descent optimization
        gd_theta, gd_costs = gd_optimizer(X_poly, y_train, initial_theta)
        
        # Calculate training and test errors
        direct_train_mse = mse_loss(direct_theta, X_poly, y_train)
        direct_test_mse = mse_loss(direct_theta, X_test_poly, y_test)
        
        reg_train_mse = mse_loss(reg_theta, X_poly, y_train)
        reg_test_mse = mse_loss(reg_theta, X_test_poly, y_test)
        
        lbfgs_train_mse = mse_loss(lbfgs_result.x, X_poly, y_train)
        lbfgs_test_mse = mse_loss(lbfgs_result.x, X_test_poly, y_test)
        
        ridge_train_mse = mse_loss(ridge_result.x, X_poly, y_train)
        ridge_test_mse = mse_loss(ridge_result.x, X_test_poly, y_test)
        
        gd_train_mse = mse_loss(gd_theta, X_poly, y_train)
        gd_test_mse = mse_loss(gd_theta, X_test_poly, y_test)
        
        results[degree] = {
            'direct': (direct_theta, direct_train_mse, direct_test_mse),
            'ridge': (reg_theta, reg_train_mse, reg_test_mse),
            'lbfgs': (lbfgs_result.x, lbfgs_train_mse, lbfgs_test_mse),
            'ridge_lbfgs': (ridge_result.x, ridge_train_mse, ridge_test_mse),
            'gradient_descent': (gd_theta, gd_train_mse, gd_test_mse)
        }
    
    # Print results
    print("\nPolynomial Regression Results:")
    print(f"{'Degree':<6} {'Method':<15} {'Train MSE':<15} {'Test MSE':<15} {'Overfitting Ratio':<20}")
    print("-" * 75)
    
    for degree in degrees:
        for method, (_, train_mse, test_mse) in results[degree].items():
            overfitting_ratio = test_mse / train_mse if train_mse > 0 else float('inf')
            print(f"{degree:<6} {method:<15} {train_mse:<15.4f} {test_mse:<15.4f} {overfitting_ratio:<20.4f}")
    
    # Create predictions for visualization
    X_range = np.linspace(0, 10, 1000).reshape(-1, 1)
    best_method = results[3]['ridge_lbfgs']  # Typically degree 3 with ridge works well
    best_theta = best_method[0]
    X_range_poly = create_polynomial_features(X_range, 3)
    y_pred = X_range_poly.dot(best_theta)
    
    print("\nBest model prediction characteristics:")
    print(f"- Polynomial degree: 3")
    print(f"- Method: Ridge regression with L-BFGS optimizer")
    print(f"- Parameters: {best_theta}")
    print(f"- Training MSE: {best_method[1]:.4f}")
    print(f"- Test MSE: {best_method[2]:.4f}")
    
    return results
```

```python
                yi = y_shuffled[j:j+1]

                theta = theta - learning_rate * gradient

            theta_history[i] = theta
            cost_history[i] = compute_cost(X, y, theta, lambda_)

            gradient = compute_gradient(X, y, theta, lambda_)
            s = beta * s + (1 - beta) * gradient**2
            theta = theta - (learning_rate / (np.sqrt(s) + epsilon)) * gradient
            
            theta_history[i] = theta
            cost_history[i] = compute_cost(X, y, theta, lambda_)
            
            # Early stopping
            if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
                return theta, cost_history[:i+1], theta_history[:i+1], i
                
        return theta, cost_history, theta_history, iterations
    
    # 6. Adam Optimizer
    def adam_optimizer(X, y, theta, learning_rate, iterations, beta1=0.9, beta2=0.999, epsilon=1e-8, lambda_=0, tol=1e-8):
        m = len(y)
        cost_history = np.zeros(iterations)
        theta_history = np.zeros((iterations, len(theta)))
        v = np.zeros_like(theta)  # First moment
        s = np.zeros_like(theta)  # Second moment
        
        for i in range(iterations):
            gradient = compute_gradient(X, y, theta, lambda_)
            
            # Update biased first moment
            v = beta1 * v + (1 - beta1) * gradient
            
            # Update biased second moment
            s = beta2 * s + (1 - beta2) * gradient**2
            
            # Bias correction
            v_corrected = v / (1 - beta1**(i+1))
            s_corrected = s / (1 - beta2**(i+1))
            
            # Update parameters
            theta = theta - (learning_rate / (np.sqrt(s_corrected) + epsilon)) * v_corrected
            
            theta_history[i] = theta
            cost_history[i] = compute_cost(X, y, theta, lambda_)
            
            # Early stopping
            if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
                return theta, cost_history[:i+1], theta_history[:i+1], i
                
        return theta, cost_history, theta_history, iterations
    
    # Run all optimizers
    print("\nTraining with different gradient descent variants:")
    
    start_time = time.time()
    bgd_theta, bgd_cost, _, bgd_iters = batch_gradient_descent(
        X_train_b, y_train, theta_init.copy(), learning_rate, iterations, lambda_
    )
    bgd_time = time.time() - start_time
    
    start_time = time.time()
    sgd_theta, sgd_cost, _, sgd_iters = stochastic_gradient_descent(
        X_train_b, y_train, theta_init.copy(), learning_rate * 0.1, iterations, lambda_
    )
    sgd_time = time.time() - start_time
    
    start_time = time.time()
    mbgd_theta, mbgd_cost, _, mbgd_iters = mini_batch_gradient_descent(
        X_train_b, y_train, theta_init.copy(), learning_rate, iterations, 32, lambda_
    )
    mbgd_time = time.time() - start_time
    
    start_time = time.time()
    momentum_theta, momentum_cost, _, momentum_iters = momentum_gradient_descent(

    momentum_time = time.time() - start_time

    rmsprop_time = time.time() - start_time
    
    start_time = time.time()
    adam_theta, adam_cost, _, adam_iters = adam_optimizer(

    adam_time = time.time() - start_time

    bgd_test_error = compute_cost(X_test_b, y_test, bgd_theta)
    sgd_test_error = compute_cost(X_test_b, y_test, sgd_theta)
    mbgd_test_error = compute_cost(X_test_b, y_test, mbgd_theta)
    momentum_test_error = compute_cost(X_test_b, y_test, momentum_theta)
    rmsprop_test_error = compute_cost(X_test_b, y_test, rmsprop_theta)
    adam_test_error = compute_cost(X_test_b, y_test, adam_theta)
    
    # Print results
    print("\nGradient Descent Optimization Results:")
    print(f"{'Algorithm':<20} {'Final Cost':<15} {'Test Error':<15} {'Iterations':<15} {'Time (s)':<10}")
    print("-" * 80)
    print(f"{'Batch GD':<20} {bgd_cost[-1]:<15.6f} {bgd_test_error:<15.6f} {bgd_iters:<15} {bgd_time:<10.4f}")
    print(f"{'Stochastic GD':<20} {sgd_cost[-1]:<15.6f} {sgd_test_error:<15.6f} {sgd_iters:<15} {sgd_time:<10.4f}")
    print(f"{'Mini-batch GD':<20} {mbgd_cost[-1]:<15.6f} {mbgd_test_error:<15.6f} {mbgd_iters:<15} {mbgd_time:<10.4f}")
    print(f"{'Momentum':<20} {momentum_cost[-1]:<15.6f} {momentum_test_error:<15.6f} {momentum_iters:<15} {momentum_time:<10.4f}")
    print(f"{'RMSprop':<20} {rmsprop_cost[-1]:<15.6f} {rmsprop_test_error:<15.6f} {rmsprop_iters:<15} {rmsprop_time:<10.4f}")
    print(f"{'Adam':<20} {adam_cost[-1]:<15.6f} {adam_test_error:<15.6f} {adam_iters:<15} {adam_time:<10.4f}")
```

## Implementing a Neural Network with Gradient Descent

To demonstrate the power of gradient descent in deep learning, let's implement a simple neural network from scratch:

```python
def neural_network_with_gradient_descent():
    # Generate classification data
    np.random.seed(42)
    n_samples = 1000
    X = np.random.randn(n_samples, 2)
    y = (X[:, 0]**2 + X[:, 1]**2 < 1).astype(int)  # Circle classification
    
    # Split data
    indices = np.random.permutation(n_samples)
    training_idx, test_idx = indices[:800], indices[800:]
    X_train, X_test = X[training_idx], X[test_idx]
    y_train, y_test = y[training_idx], y[test_idx]
    
    # Convert to one-hot encoding
    def to_one_hot(y, num_classes=2):
        return np.eye(num_classes)[y]
    
    y_train_oh = to_one_hot(y_train)
    y_test_oh = to_one_hot(y_test)
    
    # Neural network activation functions and their derivatives
    def sigmoid(z):
        return 1 / (1 + np.exp(-np.clip(z, -500, 500)))  # Clip to prevent overflow
    
    def sigmoid_derivative(z):
        s = sigmoid(z)
        return s * (1 - s)
    
    def relu(z):
        return np.maximum(0, z)
    
    def relu_derivative(z):
        return (z > 0).astype(float)
    
    def softmax(z):
        exp_z = np.exp(z - np.max(z, axis=1, keepdims=True))
        return exp_z / np.sum(exp_z, axis=1, keepdims=True)
    
    # Define neural network architecture
    def initialize_parameters(layer_dims):
        """
        Initialize parameters for a neural network with specific architecture
        layer_dims: list containing the dimensions of each layer
        """
        np.random.seed(42)
        parameters = {}
        L = len(layer_dims)
        
        for l in range(1, L):
            # He initialization for ReLU
            parameters[f'W{l}'] = np.random.randn(layer_dims[l-1], layer_dims[l]) * np.sqrt(2 / layer_dims[l-1])
            parameters[f'b{l}'] = np.zeros((1, layer_dims[l]))
            
        return parameters
    
    # Forward propagation
    def forward_propagation(X, parameters, activation_fn="relu"):
        """
        Implement forward propagation for neural network
        """
        caches = []
        A = X
        L = len(parameters) // 2
        
        # Hidden layers
        for l in range(1, L):
            A_prev = A
            W = parameters[f'W{l}']
            b = parameters[f'b{l}']
            Z = np.dot(A_prev, W) + b
            
            if activation_fn == "relu":
                A = relu(Z)
            else:
                A = sigmoid(Z)
                
            cache = {"Z": Z, "A": A_prev}
            caches.append(cache)
        
        # Output layer with softmax
        W = parameters[f'W{L}']
        b = parameters[f'b{L}']
        Z = np.dot(A, W) + b
        A = softmax(Z)
        
        cache = {"Z": Z, "A": A}
        caches.append(cache)
        
        return A, caches
    
    # Compute cost
    def compute_cost(A, Y, parameters, lambda_=0.01):
        """
        Compute cross-entropy cost with L2 regularization
        """
        m = Y.shape[0]
        
        # Cross-entropy loss
        log_probs = np.multiply(np.log(A + 1e-8), Y) + np.multiply(np.log(1 - A + 1e-8), 1 - Y)
        cost = -np.sum(log_probs) / m
        
        # Add L2 regularization
        if lambda_ > 0:
            L = len(parameters) // 2
            L2_regularization_cost = 0
            
            for l in range(1, L + 1):
                L2_regularization_cost += np.sum(np.square(parameters[f'W{l}']))
                
            L2_regularization_cost *= (lambda_ / (2 * m))
            cost += L2_regularization_cost
            
        return cost
    
    # Backward propagation
    def backward_propagation(X, Y, parameters, caches, activation_fn="relu", lambda_=0.01):
        """
        Implement backward propagation for neural network
        """
        grads = {}
        L = len(caches)
        m = Y.shape[0]
        
        # Output layer
        AL = caches[-1]["A"]
        dZ = AL - Y
        
        grads[f"dW{L}"] = (1/m) * np.dot(caches[-1]["A"].T, dZ)
        grads[f"db{L}"] = (1/m) * np.sum(dZ, axis=0, keepdims=True)
        
        # Add L2 regularization to gradients
        if lambda_ > 0:
            grads[f"dW{L}"] += (lambda_/m) * parameters[f"W{L}"]
        
        # Hidden layers
        for l in reversed(range(L-1)):
            dA = np.dot(dZ, parameters[f"W{l+2}"].T)
            
            if activation_fn == "relu":
                dZ = dA * relu_derivative(caches[l]["Z"])
            else:
                dZ = dA * sigmoid_derivative(caches[l]["Z"])
                
            grads[f"dW{l+1}"] = (1/m) * np.dot(caches[l]["A"].T, dZ)
            grads[f"db{l+1}"] = (1/m) * np.sum(dZ, axis=0, keepdims=True)
            
            # Add L2 regularization to gradients
            if lambda_ > 0:
                grads[f"dW{l+1}"] += (lambda_/m) * parameters[f"W{l+1}"]
        
        return grads
    
    # Update parameters with gradient descent
    def update_parameters(parameters, grads, learning_rate):
        """
        Update parameters using gradient descent
        """
        L = len(parameters) // 2
        
        for l in range(1, L + 1):
            parameters[f"W{l}"] -= learning_rate * grads[f"dW{l}"]
            parameters[f"b{l}"] -= learning_rate * grads[f"db{l}"]
            
        return parameters
    
    # Predict using trained neural network
    def predict(X, parameters, activation_fn="relu"):
        """
        Make predictions using trained neural network
        """
        A, _ = forward_propagation(X, parameters, activation_fn)
        return np.argmax(A, axis=1)
    
    # Train neural network
    def train_neural_network(X, Y, layer_dims, activation_fn="relu", learning_rate=0.01, 
                            num_iterations=3000, batch_size=64, lambda_=0.01, print_cost=True):
        """
        Train a neural network with gradient descent
        """
        np.random.seed(42)
        costs = []
        accuracies = []
        
        # Initialize parameters
        parameters = initialize_parameters(layer_dims)
        
        # Gradient descent with mini-batches
        num_complete_batches = X.shape[0] // batch_size
        
        for i in range(num_iterations):
            # Shuffle data
            permutation = list(np.random.permutation(X.shape[0]))
            X_shuffled = X[permutation, :]
            Y_shuffled = Y[permutation, :]
            
            cost_batch = 0
            
            # Process mini-batches
            for k in range(num_complete_batches):
                # Select mini-batch
                start_idx = k * batch_size
                end_idx = min((k + 1) * batch_size, X.shape[0])
                X_batch = X_shuffled[start_idx:end_idx, :]
                Y_batch = Y_shuffled[start_idx:end_idx, :]
                
                # Forward propagation
                A, caches = forward_propagation(X_batch, parameters, activation_fn)
                
                # Compute cost
                cost_batch += compute_cost(A, Y_batch, parameters, lambda_)
                
                # Backward propagation
                grads = backward_propagation(X_batch, Y_batch, parameters, caches, activation_fn, lambda_)
                
                # Update parameters
                parameters = update_parameters(parameters, grads, learning_rate)
            
            cost_avg = cost_batch / num_complete_batches
            
            # Print cost every 100 iterations
            if print_cost and i % 100 == 0:
                # Compute training and validation accuracy
                train_predictions = predict(X, parameters, activation_fn)
                train_accuracy = np.mean(train_predictions == np.argmax(Y, axis=1))
                
                test_predictions = predict(X_test, parameters, activation_fn)
                test_accuracy = np.mean(test_predictions == y_test)
                
                print(f"Cost after iteration {i}: {cost_avg}")
                print(f"Training accuracy: {train_accuracy:.4f}")
                print(f"Test accuracy: {test_accuracy:.4f}")
                
                costs.append(cost_avg)
                accuracies.append((train_accuracy, test_accuracy))
                
        return parameters, costs, accuracies
    
    # Define network architecture
    layer_dims = [2, 16, 8, 2]  # 2 inputs, 16 hidden, 8 hidden, 2 outputs
    
    # Train neural network
    print("\nTraining neural network with gradient descent:")
    parameters, costs, accuracies = train_neural_network(
        X_train, y_train_oh, layer_dims, 
        activation_fn="relu", 
        learning_rate=0.01, 
        num_iterations=1000, 
        batch_size=64,
        lambda_=0.01
    )
    
    # Final evaluation
    train_predictions = predict(X_train, parameters, "relu")
    train_accuracy = np.mean(train_predictions == y_train)
    
    test_predictions = predict(X_test, parameters, "relu")
    test_accuracy = np.mean(test_predictions == y_test)
    
    print("\nFinal Neural Network Results:")
    print(f"Training accuracy: {train_accuracy:.4f}")
    print(f"Test accuracy: {test_accuracy:.4f}")
    
    # Model complexity
    total_params = 0
    for l in range(1, len(layer_dims)):
        layer_params = layer_dims[l-1] * layer_dims[l] + layer_dims[l]  # Weights + biases
        total_params += layer_params
        print(f"Layer {l}: {layer_dims[l-1]} x {layer_dims[l]} + {layer_dims[l]} = {layer_params} parameters")
    
    print(f"Total parameters: {total_params}")
    
    return parameters, costs, accuracies

# Train a neural network using gradient descent
```
