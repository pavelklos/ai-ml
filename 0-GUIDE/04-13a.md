## Key Terms and Concepts

- **Large Language Models (LLMs)**: Neural network-based systems trained on vast text corpora to understand and generate human-like text.
- **Transformer Architecture**: The neural network architecture that powers modern LLMs, utilizing self-attention mechanisms to process text.
- **Prompt Engineering**: The practice of crafting inputs to elicit desired outputs from language models.
- **API (Application Programming Interface)**: A set of protocols allowing different software applications to communicate with each other.
- **Tokens**: The basic units that LLMs process, which can be words, subwords, or characters.
- **Context Window**: The maximum amount of text (tokens) that an LLM can consider at once.
- **Temperature**: A parameter controlling randomness in LLM outputs; higher values produce more diverse responses.
- **Fine-tuning**: The process of adapting a pre-trained model to specific tasks or domains.
- **Embedding**: Vector representations of text that capture semantic meaning.
- **LangChain**: A framework for developing applications powered by language models.
- **Retrieval-Augmented Generation (RAG)**: A technique combining information retrieval with text generation to produce factual, grounded responses.
- **Vector Database**: A specialized database optimized for storing and querying vector embeddings.

## LangChain Library

LangChain provides a powerful framework for building applications with LLMs, offering abstractions for working with models, prompts, memory, and data sources:

```python

import os
from langchain.llms import OpenAI
from langchain_openai import ChatOpenAI
from langchain.prompts import PromptTemplate
from langchain.chains import LLMChain, SimpleSequentialChain, SequentialChain
from langchain.memory import ConversationBufferMemory
from langchain.schema import HumanMessage, SystemMessage, AIMessage
from langchain.output_parsers import PydanticOutputParser
from pydantic import BaseModel, Field
from typing import List

def langchain_basics():
    """Demonstrate core LangChain components and patterns"""
    
    # 1. Models - interface with LLMs
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0.7)
    
    # 2. Prompts - structure inputs to LLMs
    prompt_template = PromptTemplate(
        input_variables=["product"],
        template="Generate 5 creative marketing slogans for {product}."
    )
    
    # 3. Chains - combine components for specific tasks
    slogan_chain = LLMChain(
        llm=llm,
        prompt=prompt_template,
        verbose=True
    )
    
    # Execute the chain
    slogans = slogan_chain.run(product="an AI-powered smart water bottle")
    print("\nGenerated Slogans:")
    print(slogans)
    
    # 4. Memory - store conversation history
    memory = ConversationBufferMemory()
    memory.chat_memory.add_message(HumanMessage(content="Hello, I'm interested in AI."))
    memory.chat_memory.add_message(AIMessage(content="That's great! What aspects of AI interest you most?"))
    
    conversation_chain = LLMChain(
        llm=llm,
        prompt=PromptTemplate(
            input_variables=["history", "human_input"],
            template="Conversation history: {history}\nHuman: {human_input}\nAI: "
        ),
        memory=memory,
        verbose=True
    )
    
    # 5. Output Parsers - structure LLM outputs
    class Product(BaseModel):
        name: str = Field(description="Name of the product")
        description: str = Field(description="Short description of the product")
        key_features: List[str] = Field(description="List of key product features")
        target_audience: str = Field(description="Description of the target customer")
        
    parser = PydanticOutputParser(pydantic_object=Product)
    
    structured_prompt = PromptTemplate(
        template="Generate a product concept for a new tech gadget.\n{format_instructions}\n",
        input_variables=[],
        partial_variables={"format_instructions": parser.get_format_instructions()}
    )
    
    # 6. Agents - dynamic chains that use tools based on inputs
    print("\nLangChain also supports Agents, which can:")
    print("- Make decisions about which tools to use")
    print("- Execute multi-step reasoning")
    print("- Adapt to user requests dynamically")
    
    # 7. Sequential chains - combining multiple chains
    analysis_prompt = PromptTemplate(
        input_variables=["slogans"],
        template="Analyze these marketing slogans and explain why they might be effective:\n{slogans}"
    )
    
    analysis_chain = LLMChain(
        llm=llm,
        prompt=analysis_prompt
    )
    
    # Connect chains sequentially
    marketing_chain = SimpleSequentialChain(
        chains=[slogan_chain, analysis_chain],
        verbose=True
    )
    
    print("\nSequential Chain Example:")
    print("This will generate slogans and then analyze them in sequence.")
    
    return {
        "llm": llm,
        "prompt_template": prompt_template,
        "slogan_chain": slogan_chain,
        "memory": memory,
        "parser": parser
    }

# Run LangChain basics
langchain_components = langchain_basics()
```

RAG systems enhance LLM responses by retrieving relevant information from external knowledge sources:

```python

def build_rag_system():
    """Create a complete RAG system from scratch"""
    
    # 1. Data Loading
    print("Step 1: Loading Documents")
    
    # Create a temporary text file for demo purposes
    os.makedirs("data", exist_ok=True)
    
    with open("data/ai_info.txt", "w") as f:
        f.write("""
        Artificial Intelligence (AI) refers to systems or machines that mimic human intelligence.
        Machine Learning is a subset of AI that uses statistical methods to enable machines to improve with experience.
        Deep Learning is a type of machine learning based on artificial neural networks.
        Natural Language Processing (NLP) is a field of AI that gives machines the ability to read and understand human language.
        Computer Vision is an AI field that enables computers to derive meaningful information from digital images and videos.
        Reinforcement Learning is about taking suitable actions to maximize reward in a particular situation.
        Generative AI models can create new content, including text, images, audio, and synthetic data.
        ChatGPT is a language model developed by OpenAI based on the GPT architecture.
        Large Language Models (LLMs) are AI systems trained on vast amounts of text data to understand and generate human-like text.
        Neural networks are computing systems inspired by the biological neural networks that constitute animal brains.
        Transformer architecture revolutionized NLP by introducing attention mechanisms.
        Transfer learning enables leveraging knowledge from one task to improve performance on another related task.
        Fine-tuning adapts a pre-trained model to a specific task or domain with additional training.
        Prompt engineering is the process of designing effective prompts to guide AI models toward desired outputs.
        """)
    
    # Load documents
    loader = DirectoryLoader("data", glob="**/*.txt", loader_cls=TextLoader)
    documents = loader.load()
    print(f"Loaded {len(documents)} documents")
    
    # 2. Text Splitting
    print("\nStep 2: Splitting Documents into Chunks")
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=500,
        chunk_overlap=50,
        separators=["\n\n", "\n", ".", " ", ""]
    )
    
    chunks = text_splitter.split_documents(documents)
    print(f"Split into {len(chunks)} chunks")
    
    # 3. Create Embeddings and Vector Store
    print("\nStep 3: Creating Embeddings and Vector Store")
    embeddings = OpenAIEmbeddings()
    
    # Create vector database
    vector_store = FAISS.from_documents(chunks, embeddings)
    print(f"Created vector store with {vector_store.index.ntotal} vectors")
    
    # 4. Set up Retriever
    print("\nStep 4: Setting up Retriever")
    retriever = vector_store.as_retriever(
        search_type="similarity",
        search_kwargs={"k": 3}  # Return top 3 most relevant chunks
    )
    
    # 5. Create RAG Chain
    print("\nStep 5: Creating RAG Chain")
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
    
    rag_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="stuff",  # Other options: map_reduce, refine
        retriever=retriever,
        verbose=True,
        return_source_documents=True
    )
    
    # 6. Query the System
    print("\nStep 6: Querying the RAG System")
    query = "What is the difference between machine learning and deep learning?"
    print(f"Query: {query}")
    
    result = rag_chain({"query": query})
    print("\nRAG Response:")
    print(result["result"])
    
    print("\nSource Chunks Used:")
    for i, doc in enumerate(result["source_documents"]):
        print(f"\nSource {i+1}:")
        print(doc.page_content)
    
    # 7. Advanced RAG Techniques
    print("\nAdvanced RAG Techniques:")
    print("1. Query Transformation: Rewrite queries to improve retrieval")
    print("2. Ensemble Retrieval: Combine multiple retrieval methods")
    print("3. Multi-stage Retrieval: Retrieve, then filter with LLM")
    print("4. Self-querying: Let the LLM determine query structure")
    print("5. Hypothetical Document Embeddings (HyDE): Generate then retrieve")
    
    # Clean up the demo file
    # os.remove("data/ai_info.txt")
    
    return {
        "vector_store": vector_store,
        "rag_chain": rag_chain
    }
```

```python
def advanced_rag_with_web_content():

    print("\nStep 2: Advanced Text Splitting")
    text_splitter = RecursiveCharacterTextSplitter(
        chunk_size=800,
        chunk_overlap=150,
        separators=["\n\n", "\n", ". ", " ", ""],
        length_function=len
    )
    
    chunks = text_splitter.split_documents(documents)
    print(f"Split into {len(chunks)} chunks")
    
    # 3. Create embeddings and vector store
    print("\nStep 3: Creating Embeddings with Metadata")
    embeddings = OpenAIEmbeddings()
    
    # Enhance chunks with metadata
    for i, chunk in enumerate(chunks):
        if not hasattr(chunk, 'metadata'):
            chunk.metadata = {}
        chunk.metadata["chunk_id"] = i
        chunk.metadata["source"] = chunk.metadata.get("source", "unknown")
    
    # Create vector database
    vector_store = FAISS.from_documents(chunks, embeddings)
    print(f"Created vector store with {vector_store.index.ntotal} vectors")
    
    # 4. Create enhanced prompt for RAG
    print("\nStep 4: Creating Enhanced RAG Prompt")
    
    from langchain.prompts import PromptTemplate
    
    template = """
    You are an AI assistant that provides accurate and helpful information based on the provided context.
    
    Context:
    {context}
    
    Human Question: {question}
    
    Instructions:
    1. Answer the question based only on the provided context
    2. If the context doesn't contain relevant information, say "I don't have enough information to answer that question"
    3. Cite the sources of your information when possible
    4. Keep your answer concise and to the point
    
    Your Answer:
    """
    
    QA_CHAIN_PROMPT = PromptTemplate(
        input_variables=["context", "question"],
        template=template
    )
    
    # 5. Create enhanced RAG chain
    print("\nStep 5: Creating Enhanced RAG Chain")
    llm = ChatOpenAI(model="gpt-3.5-turbo", temperature=0)
    
    rag_chain = RetrievalQA.from_chain_type(
        llm=llm,
        chain_type="stuff",
        retriever=vector_store.as_retriever(search_kwargs={"k": 5}),
        return_source_documents=True,
        chain_type_kwargs={"prompt": QA_CHAIN_PROMPT}

        memory_key="chat_history",
        return_messages=True
    )
    
    conversational_rag = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=vector_store.as_retriever(),
        memory=memory,
        verbose=True
    )
    
    # Example conversation
    print("\nExample Conversation:")
    queries = [
        "What are the main types of machine learning?",
        "What are the challenges in this field?",
        "How do large language models relate to this?"
    ]

    print("1. Chunk documents semantically (paragraphs, sections) when possible")
    print("2. Experiment with chunk size and overlap for optimal retrieval")
    print("3. Use hybrid retrieval combining keyword and semantic search")
    print("4. Implement re-ranking to improve retrieval precision")
    print("5. Add source attribution and citations")
    print("6. Monitor and evaluate RAG system performance")

        "vector_store": vector_store,
        "rag_chain": rag_chain,
```

```python
        - K-Fold Cross-Validation: splits data into k subsets
        - Stratified K-Fold: maintains class distribution in each fold
        - Leave-One-Out Cross-Validation: uses one observation for testing
        - Time Series Cross-Validation: respects temporal order of data
        """
    }
    
    # Write the sample documents
    for filename, content in sample_docs.items():
        with open(f"knowledge_base/{filename}", "w") as f:
            f.write(content)
    
    # 2. Load documents
    print("\nStep 1: Loading Documents")
    loader = DirectoryLoader("knowledge_base/", glob="**/*.txt", loader_cls=TextLoader)
    documents = loader.load()
    print(f"Loaded {len(documents)} documents from knowledge base")
    
    # 3. Split documents into chunks
    print("\nStep 2: Splitting Documents")
    text_splitter = CharacterTextSplitter(
        separator="\n\n",
        chunk_size=1000,
        chunk_overlap=200,
        length_function=len
    )
    
    chunks = text_splitter.split_documents(documents)
    print(f"Split into {len(chunks)} chunks")
    
    # 4. Create embeddings and vector store
    print("\nStep 3: Creating Vector Store")
    embeddings = OpenAIEmbeddings()
    vector_store = FAISS.from_documents(chunks, embeddings)
    print("Vector store created successfully")
    
    # 5. Create a conversation memory
    memory = ConversationBufferMemory(
        memory_key="chat_history",
        return_messages=True
    )
    
    # 6. Create the conversational RAG chain
    print("\nStep 4: Creating Conversational RAG Chain")
    llm = ChatOpenAI(temperature=0, model="gpt-3.5-turbo")
    
    qa_chain = ConversationalRetrievalChain.from_llm(
        llm=llm,
        retriever=vector_store.as_retriever(search_kwargs={"k": 3}),
        memory=memory,
        verbose=True
    )
    
    # 7. Interact with the RAG system
    print("\nStep 5: Interacting with the RAG System")
    
    # Predefined questions for the demo
    questions = [
        "What are the main types of machine learning algorithms?",
        "How do we handle missing values in data preprocessing?",
        "Which metrics should I use to evaluate a classification model?",
        "Can you compare supervised and unsupervised learning approaches?",
        "What is the difference between normalization and standardization?"
    ]
    
    # Query the system with each question
    for i, question in enumerate(questions):
        print(f"\nQuestion {i+1}: {question}")
        
        # Get the answer
        response = qa_chain({"question": question})
        
        print(f"Answer: {response['answer']}")
    
    # 8. Save the vector store for future use
    print("\nStep 6: Saving the Vector Store")
    vector_store.save_local("knowledge_base_index")
    print("Vector store saved to 'knowledge_base_index'")
    
    # 9. Demonstrate loading a saved vector store
    print("\nStep 7: Loading a Saved Vector Store")
    loaded_vector_store = FAISS.load_local("knowledge_base_index", embeddings)
    print("Successfully loaded the vector store")

    print("This system demonstrates how to create a domain-specific AI assistant")
    print("that uses a custom knowledge base to provide accurate information")

    return {

        "qa_chain": qa_chain
```
