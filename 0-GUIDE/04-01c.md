## Key Terms and Concepts

- **Artificial Intelligence (AI)**: The broader field encompassing systems that can perform tasks typically requiring human intelligence.
- **Machine Learning (ML)**: A subset of AI focused on algorithms that improve through experience with data.
- **Supervised Learning**: Learning from labeled data pairs (input → desired output).
- **Unsupervised Learning**: Finding patterns in unlabeled data.
- **Reinforcement Learning**: Learning optimal actions through trial-and-error interactions with an environment.
- **Optimization Problem**: Mathematical formulation seeking to find parameters that minimize or maximize an objective function.
- **Loss/Cost Function**: Function measuring discrepancy between predicted and actual values.
- **Gradient Descent**: First-order iterative optimization algorithm for finding local minima.
- **Convex Optimization**: Special case with guaranteed global minimum.
- **Hessian Matrix**: Matrix of second-order partial derivatives, indicating curvature.
- **Regularization**: Techniques to prevent overfitting by penalizing model complexity.

## Mathematical Foundations of Optimization Problems

Optimization lies at the heart of machine learning, providing the mathematical framework for learning from data. In essence, we define a loss function and find parameters that minimize this loss.

### General Form of Optimization Problems

$$\min_{\theta} f(\theta) \text{ subject to } g_i(\theta) \leq 0, h_j(\theta) = 0$$

Where:
- $\theta$ are the parameters we're optimizing
- $f(\theta)$ is the objective function
- $g_i(\theta)$ are inequality constraints
- $h_j(\theta)$ are equality constraints

Let's implement a complex optimization problem with constraints:

# Define a challenging optimization problem: Rosenbrock function with constraints
def rosenbrock(x):
    """
    Rosenbrock function: f(x,y) = (1-x)² + 100(y-x²)²
    Has a global minimum at (1,1) with f(1,1) = 0
    """
    return sum(100.0 * (x[1:] - x[:-1]**2.0)**2.0 + (1 - x[:-1])**2.0)

# Gradient of Rosenbrock function (for efficiency)
def rosenbrock_gradient(x):
    grad = np.zeros_like(x)
    grad[0] = -400 * x[0] * (x[1] - x[0]**2) - 2 * (1 - x[0])
    grad[-1] = 200 * (x[-1] - x[-2]**2)
    for i in range(1, len(x) - 1):
        grad[i] = 200 * (x[i] - x[i-1]**2) - 400 * x[i] * (x[i+1] - x[i]**2) - 2 * (1 - x[i])
    return grad

# Define constraints
def sum_constraint(x):
    """Equality constraint: sum of parameters equals 1"""
    return np.sum(x) - 1.0

def sum_constraint_jac(x):
    """Jacobian of the equality constraint"""
    return np.ones_like(x)

def bound_constraint(x):
    """Inequality constraint: all parameters must be non-negative"""
    return x

def bound_constraint_jac(x):
    """Jacobian of the inequality constraint"""
    return np.eye(len(x))

# Implement our own custom optimization method: Sequential Quadratic Programming
def custom_sqp_optimizer(obj_func, obj_grad, initial_guess, constraints, max_iter=100, tol=1e-6):
    """
    Custom implementation of Sequential Quadratic Programming (SQP)
    for constrained optimization
    """
    x = initial_guess.copy()
    n = len(x)
    
    # Initialize Hessian approximation using identity matrix
    B = np.eye(n)
    
    history = []
    
    for iteration in range(max_iter):
        # Current gradient and constraint values
        grad = obj_grad(x)
        c_val = sum_constraint(x)
        c_jac = sum_constraint_jac(x)
        
        # Build KKT system (first order optimality conditions)
        # [B   c_jac^T] [p  ] = [-grad]
        # [c_jac  0   ] [λ_k] = [-c   ]
        
        KKT_matrix = np.zeros((n+1, n+1))
        KKT_matrix[:n, :n] = B
        KKT_matrix[:n, n] = c_jac
        KKT_matrix[n, :n] = c_jac
        
        KKT_rhs = np.zeros(n+1)
        KKT_rhs[:n] = -grad
        KKT_rhs[n] = -c_val
        
        # Solve KKT system
        solution = np.linalg.solve(KKT_matrix, KKT_rhs)
        p = solution[:n]  # Search direction
        
        # Line search (simplistic)
        alpha = 1.0
        while alpha > tol:
            x_new = x + alpha * p
            
            # Check if new point satisfies constraints
            if np.all(x_new >= 0) and abs(sum_constraint(x_new)) < 0.01:
                if obj_func(x_new) < obj_func(x):
                    break
            
            alpha *= 0.5
        
        # Check for convergence
        if alpha < tol or np.linalg.norm(p) < tol:
            break
        
        # Update solution
        x_new = x + alpha * p
        
        # Update Hessian approximation using BFGS formula
        s = alpha * p
        y = obj_grad(x_new) - grad
        
        if np.dot(s, y) > 0:  # Ensure curvature condition
            rho = 1.0 / np.dot(s, y)
            v = np.dot(B, s)
            B = B - np.outer(v, v) / np.dot(v, s) + np.outer(y, y) / np.dot(s, y)
        
        x = x_new
        history.append(obj_func(x))
    
    return {
        'x': x,
        'fun': obj_func(x),
        'success': True if iteration < max_iter-1 else False,
        'nit': iteration+1,
        'history': history
    }

# Run our custom SQP implementation
custom_result = custom_sqp_optimizer(
    rosenbrock, 
    rosenbrock_gradient, 
    initial_guess, 
    constraints,
    max_iter=100
)

# Define cost function for polynomial regression with regularization
def poly_regression_cost(theta, X, y, lambda_=0.0):
    """Cost function for polynomial regression with L2 regularization"""
    m = len(y)
    h = X.dot(theta)
    error = h - y
    J = (1/(2*m)) * np.sum(error**2)
    
    # Add regularization, excluding bias term
    if lambda_ > 0:
        J += (lambda_/(2*m)) * np.sum(theta[1:]**2)
    
    return J

# Define gradient of cost function
def poly_regression_gradient(theta, X, y, lambda_=0.0):
    """Gradient of cost function for polynomial regression with L2 regularization"""
    m = len(y)
    h = X.dot(theta)
    error = h - y
    
    grad = (1/m) * X.T.dot(error)
    
    # Add regularization, excluding bias term
    if lambda_ > 0:
        grad_reg = grad.copy()
        grad_reg[1:] += (lambda_/m) * theta[1:]
        return grad_reg
    
    return grad

# 1. Batch Gradient Descent
def batch_gradient_descent(X, y, theta, learning_rate, iterations, lambda_=0.0, tol=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, len(theta)))
    
    for i in range(iterations):
        grad = poly_regression_gradient(theta, X, y, lambda_)
        theta = theta - learning_rate * grad
        
        theta_history[i] = theta.flatten()
        cost_history[i] = poly_regression_cost(theta, X, y, lambda_)
        
        # Early stopping if gradient is small enough
        if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
            return theta, cost_history[:i+1], theta_history[:i+1]
    
    return theta, cost_history, theta_history

# 2. Stochastic Gradient Descent
def stochastic_gradient_descent(X, y, theta, learning_rate, iterations, lambda_=0.0, tol=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, len(theta)))
    
    for i in range(iterations):
        cost = 0
        indices = np.random.permutation(m)
        X_shuffled = X[indices]
        y_shuffled = y[indices]
        
        for j in range(m):
            xi = X_shuffled[j:j+1]
            yi = y_shuffled[j:j+1]
            
            grad = poly_regression_gradient(theta, xi, yi, lambda_)
            theta = theta - learning_rate * grad
            
            cost += poly_regression_cost(theta, xi, yi, lambda_)
        
        theta_history[i] = theta.flatten()
        cost_history[i] = cost/m
        
        # Early stopping
        if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
            return theta, cost_history[:i+1], theta_history[:i+1]
    
    return theta, cost_history, theta_history

# 3. Mini-Batch Gradient Descent
def mini_batch_gradient_descent(X, y, theta, learning_rate, iterations, batch_size=32, lambda_=0.0, tol=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, len(theta)))
    
    for i in range(iterations):
        cost = 0
        indices = np.random.permutation(m)
        X_shuffled = X[indices]
        y_shuffled = y[indices]
        
        for j in range(0, m, batch_size):
            end = min(j + batch_size, m)
            X_batch = X_shuffled[j:end]
            y_batch = y_shuffled[j:end]
            
            grad = poly_regression_gradient(theta, X_batch, y_batch, lambda_)
            theta = theta - learning_rate * grad
            
            cost += poly_regression_cost(theta, X_batch, y_batch, lambda_) * (end - j)
        
        theta_history[i] = theta.flatten()
        cost_history[i] = cost/m
        
        # Early stopping
        if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
            return theta, cost_history[:i+1], theta_history[:i+1]
    
    return theta, cost_history, theta_history

# 4. Adam Optimizer
def adam_optimizer(X, y, theta, learning_rate, iterations, lambda_=0.0, 
                  beta1=0.9, beta2=0.999, epsilon=1e-8, tol=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, len(theta)))
    
    # Initialize moment vectors
    v = np.zeros_like(theta)  # First moment estimate
    s = np.zeros_like(theta)  # Second moment estimate
    
    for i in range(iterations):
        grad = poly_regression_gradient(theta, X, y, lambda_)
        
        # Update biased first moment estimate
        v = beta1 * v + (1 - beta1) * grad
        
        # Update biased second moment estimate
        s = beta2 * s + (1 - beta2) * np.square(grad)
        
        # Bias correction
        v_corrected = v / (1 - beta1**(i+1))
        s_corrected = s / (1 - beta2**(i+1))
        
        # Update parameters
        theta = theta - learning_rate * v_corrected / (np.sqrt(s_corrected) + epsilon)
        
        theta_history[i] = theta.flatten()
        cost_history[i] = poly_regression_cost(theta, X, y, lambda_)
        
        # Early stopping
        if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
            return theta, cost_history[:i+1], theta_history[:i+1]
    
    return theta, cost_history, theta_history

# 5. L-BFGS optimizer using SciPy
def lbfgs_optimizer(X, y, theta_init, lambda_=0.0):

# Implement a suite of advanced gradient descent algorithms with visualizations
def visualize_optimization_2d():
    """
    Visualize different optimization algorithms on a 2D function
    """
    # Define a challenging 2D function: Himmelblau's function
    def himmelblau(x):
        return (x[0]**2 + x[1] - 11)**2 + (x[0] + x[1]**2 - 7)**2
    
    def himmelblau_gradient(x):

        return np.array([dx, dy])

    X, Y = np.meshgrid(x, y)
    Z = np.zeros_like(X)

        for j in range(len(y)):

    # Define optimization algorithms
    def gradient_descent(init_x, learning_rate=0.01, max_iter=100):
        x = init_x.copy()
        path = [x.copy()]

            grad = himmelblau_gradient(x)

            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    def momentum(init_x, learning_rate=0.01, beta=0.9, max_iter=100):
        x = init_x.copy()
        v = np.zeros_like(x)
        path = [x.copy()]
        
        for i in range(max_iter):
            grad = himmelblau_gradient(x)
            v = beta * v - learning_rate * grad
            x = x + v
            path.append(x.copy())
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    def nesterov(init_x, learning_rate=0.01, beta=0.9, max_iter=100):
        x = init_x.copy()
        v = np.zeros_like(x)
        path = [x.copy()]
        
        for i in range(max_iter):
            x_ahead = x + beta * v
            grad = himmelblau_gradient(x_ahead)
            v = beta * v - learning_rate * grad
            x = x + v
            path.append(x.copy())
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    def adagrad(init_x, learning_rate=0.1, epsilon=1e-8, max_iter=100):
        x = init_x.copy()
        cache = np.zeros_like(x)
        path = [x.copy()]
        
        for i in range(max_iter):
            grad = himmelblau_gradient(x)
            cache += grad**2
            x = x - learning_rate * grad / (np.sqrt(cache) + epsilon)
            path.append(x.copy())
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    def rmsprop(init_x, learning_rate=0.01, beta=0.9, epsilon=1e-8, max_iter=100):
        x = init_x.copy()
        cache = np.zeros_like(x)
        path = [x.copy()]
        
        for i in range(max_iter):
            grad = himmelblau_gradient(x)
            cache = beta * cache + (1 - beta) * grad**2
            x = x - learning_rate * grad / (np.sqrt(cache) + epsilon)
            path.append(x.copy())
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    def adam(init_x, learning_rate=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8, max_iter=100):
        x = init_x.copy()
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        path = [x.copy()]
        
        for i in range(max_iter):
            grad = himmelblau_gradient(x)
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            
            m_corrected = m / (1 - beta1**(i+1))
            v_corrected = v / (1 - beta2**(i+1))
            
            x = x - learning_rate * m_corrected / (np.sqrt(v_corrected) + epsilon)
            path.append(x.copy())
            
            if np.linalg.norm(grad) < 1e-6:
                break
                
        return np.array(path)
    
    # Initialize with different starting points
    starting_points = [
        np.array([-4, 4]),
        np.array([4, 4]),
        np.array([4, -4]),
        np.array([-4, -4])
    ]
    
    # Run optimizers
    optimizers = {
        'Gradient Descent': gradient_descent,
        'Momentum': momentum,
        'Nesterov Accelerated Gradient': nesterov,
        'AdaGrad': adagrad,
        'RMSprop': rmsprop,
        'Adam': adam
    }
    
    # Visualize all optimizers from all starting points
    plt.figure(figsize=(20, 15))
    for i, (name, optimizer) in enumerate(optimizers.items()):
        plt.subplot(2, 3, i+1)
        
        # Plot contour
        contours = plt.contour(X, Y, Z, 50, cmap='viridis')
        
        # Plot paths from different starting points
        for j, start in enumerate(starting_points):
            path = optimizer(start)
            plt.plot(path[:, 0], path[:, 1], '-o', linewidth=1.5, markersize=3, 
                     label=f'Start {j+1}' if i == 0 else "")
            plt.plot(path[0, 0], path[0, 1], 'g*', markersize=10)
            plt.plot(path[-1, 0], path[-1, 1], 'r*', markersize=10)
        
        plt.title(name)
        plt.xlabel('x')
        plt.ylabel('y')
        
        if i == 0:
            plt.legend()
    
    plt.tight_layout()
    plt.show()
    
    # Compare convergence rates
    plt.figure(figsize=(12, 8))
    
    for name, optimizer in optimizers.items():
        path = optimizer(starting_points[0])
        values = [himmelblau(point) for point in path]
        plt.semilogy(values, label=name)
    
    plt.title('Convergence Comparison (from starting point 1)')

def visualize_learning_rate_effects():
    """
    Visualize the effect of different learning rates on gradient descent
    """
    # Define a simple 1D function
    def f(x):
        return x**4 - 3*x**3 + 2
    
    def df(x):
        return 4*x**3 - 9*x**2
    
    # Create a range of x values

    y = f(x)

    def gradient_descent(init_x, learning_rate, iterations=50):
        x = init_x
        path = [x]

            x = x - learning_rate * df(x)
            path.append(x)
            
        return np.array(path)

    for lr in learning_rates:

def neural_network_example():
    """
    Demonstrate gradient descent optimization for a simple neural network
    """
    from sklearn.datasets import make_moons
    from sklearn.preprocessing import StandardScaler
    from sklearn.model_selection import train_test_split
    
    # Generate a dataset
    X, y = make_moons(n_samples=1000, noise=0.2, random_state=42)
    X = StandardScaler().fit_transform(X)
    
    # Split the data
    X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)
    
    # Convert to one-hot encoding
    def to_one_hot(y, num_classes=None):
        y = np.array(y, dtype='int')
        input_shape = y.shape
        if input_shape and input_shape[-1] == 1 and len(input_shape) > 1:
            input_shape = tuple(input_shape[:-1])
        y = y.ravel()
        if not num_classes:
            num_classes = np.max(y) + 1
        n = y.shape[0]
        categorical = np.zeros((n, num_classes))
        categorical[np.arange(n), y] = 1
        return categorical
    
    y_train_oh = to_one_hot(y_train)
    y_test_oh = to_one_hot(y_test)
    
    # Define neural network functions
    def sigmoid(x):
        return 1 / (1 + np.exp(-np.clip(x, -15, 15)))
    
    def sigmoid_derivative(x):
        s = sigmoid(x)
        return s * (1 - s)
    
    def relu(x):
        return np.maximum(0, x)
    
    def relu_derivative(x):
        return np.where(x > 0, 1, 0)
    
    def softmax(x):
        e_x = np.exp(x - np.max(x, axis=1, keepdims=True))
        return e_x / np.sum(e_x, axis=1, keepdims=True)
    
    def cross_entropy_loss(y_true, y_pred):
        m = y_true.shape[0]
        log_likelihood = -np.log(y_pred[range(m), np.argmax(y_true, axis=1)])
        loss = np.sum(log_likelihood) / m
        return loss
    
    # Define neural network architecture
    def initialize_parameters(layer_dims):
        """
        Initialize parameters for a neural network with ReLU activations
        """
        np.random.seed(42)
        parameters = {}
        L = len(layer_dims)
        
        for l in range(1, L):
            parameters[f'W{l}'] = np.random.randn(layer_dims[l-1], layer_dims[l]) * np.sqrt(2 / layer_dims[l-1])
            parameters[f'b{l}'] = np.zeros((1, layer_dims[l]))
        
        return parameters
    
    def forward_propagation(X, parameters):
        """
        Forward propagation for a neural network with ReLU activations
        """
        cache = {}
        A = X
        L = len(parameters) // 2
        
        # Hidden layers with ReLU
        for l in range(1, L):
            A_prev = A
            W = parameters[f'W{l}']
            b = parameters[f'b{l}']
            Z = np.dot(A_prev, W) + b
            A = relu(Z)
            
            cache[f'A{l-1}'] = A_prev
            cache[f'Z{l}'] = Z
        
        # Output layer with softmax
        W = parameters[f'W{L}']
        b = parameters[f'b{L}']
        Z = np.dot(A, W) + b
        A = softmax(Z)
        
        cache[f'A{L-1}'] = A
        cache[f'Z{L}'] = Z
        
        return A, cache
    
    def backward_propagation(X, y, parameters, cache):
        """
        Backward propagation for a neural network with ReLU activations
        """
        gradients = {}
        m = y.shape[0]
        L = len(parameters) // 2
        
        # Get output of the last layer
        AL = cache[f'A{L-1}']
        
        # Output layer gradients (assuming softmax + cross-entropy)
        dZ = AL.copy()
        dZ[range(m), np.argmax(y, axis=1)] -= 1
        dZ /= m
        
        gradients[f'dW{L}'] = np.dot(cache[f'A{L-2}'].T, dZ)
        gradients[f'db{L}'] = np.sum(dZ, axis=0, keepdims=True)
        
        # Hidden layers
        dA = np.dot(dZ, parameters[f'W{L}'].T)
        
        for l in reversed(range(1, L)):
            dZ = dA * relu_derivative(cache[f'Z{l}'])
            
            if l == 1:
                A_prev = X
            else:
                A_prev = cache[f'A{l-2}']
            
            gradients[f'dW{l}'] = np.dot(A_prev.T, dZ)
            gradients[f'db{l}'] = np.sum(dZ, axis=0, keepdims=True)
            
            if l > 1:
                dA = np.dot(dZ, parameters[f'W{l}'].T)
        
        return gradients
    
    def update_parameters(parameters, gradients, learning_rate):
        """
        Update parameters using gradient descent
        """
        L = len(parameters) // 2
        
        for l in range(1, L+1):
            parameters[f'W{l}'] -= learning_rate * gradients[f'dW{l}']
            parameters[f'b{l}'] -= learning_rate * gradients[f'db{l}']
        
        return parameters
    
    def update_parameters_momentum(parameters, gradients, v, beta, learning_rate):
        """
        Update parameters using momentum
        """
        L = len(parameters) // 2
        
        for l in range(1, L+1):
            v[f'dW{l}'] = beta * v[f'dW{l}'] + (1 - beta) * gradients[f'dW{l}']
            v[f'db{l}'] = beta * v[f'db{l}'] + (1 - beta) * gradients[f'db{l}']
            
            parameters[f'W{l}'] -= learning_rate * v[f'dW{l}']
            parameters[f'b{l}'] -= learning_rate * v[f'db{l}']
        
        return parameters, v
    
    def update_parameters_adam(parameters, gradients, v, s, t, beta1=0.9, beta2=0.999, learning_rate=0.01, epsilon=1e-8):
        """
        Update parameters using Adam
        """
        L = len(parameters) // 2
        v_corrected = {}
        s_corrected = {}
        
        for l in range(1, L+1):
            # Momentum
            v[f'dW{l}'] = beta1 * v[f'dW{l}'] + (1 - beta1) * gradients[f'dW{l}']
            v[f'db{l}'] = beta1 * v[f'db{l}'] + (1 - beta1) * gradients[f'db{l}']
            
            # RMSprop
            s[f'dW{l}'] = beta2 * s[f'dW{l}'] + (1 - beta2) * np.square(gradients[f'dW{l}'])
            s[f'db{l}'] = beta2 * s[f'db{l}'] + (1 - beta2) * np.square(gradients[f'db{l}'])
            
            # Bias correction
            v_corrected[f'dW{l}'] = v[f'dW{l}'] / (1 - beta1**t)
            v_corrected[f'db{l}'] = v[f'db{l}'] / (1 - beta1**t)
            s_corrected[f'dW{l}'] = s[f'dW{l}'] / (1 - beta2**t)
            s_corrected[f'db{l}'] = s[f'db{l}'] / (1 - beta2**t)
            
            # Update parameters
            parameters[f'W{l}'] -= learning_rate * v_corrected[f'dW{l}'] / (np.sqrt(s_corrected[f'dW{l}']) + epsilon)
            parameters[f'b{l}'] -= learning_rate * v_corrected[f'db{l}'] / (np.sqrt(s_corrected[f'db{l}']) + epsilon)
        
        return parameters, v, s
    
    def predict(X, parameters):
        """
        Predict using trained neural network
        """
        A, _ = forward_propagation(X, parameters)
        return np.argmax(A, axis=1)
    
    def compute_accuracy(predictions, labels):
        """
        Compute classification accuracy
        """
        if len(labels.shape) > 1:
            labels = np.argmax(labels, axis=1)
        return np.mean(predictions == labels)
    
    # Train neural networks with different optimizers
    def train_nn(X_train, y_train, X_test, y_test, layer_dims, optimizer, 
                learning_rate=0.01, epochs=200, batch_size=32, print_every=10):
        """
        Train a neural network with specified optimizer
        """
        parameters = initialize_parameters(layer_dims)
        costs = []
        train_acc = []
        test_acc = []
        m = X_train.shape[0]
        
        # Initialize optimizer-specific variables
        if optimizer == 'momentum':
            v = {}
            for l in range(1, len(layer_dims)):
                v[f'dW{l}'] = np.zeros((layer_dims[l-1], layer_dims[l]))
                v[f'db{l}'] = np.zeros((1, layer_dims[l]))
        elif optimizer == 'adam':
            v = {}
            s = {}
            for l in range(1, len(layer_dims)):
                v[f'dW{l}'] = np.zeros((layer_dims[l-1], layer_dims[l]))
                v[f'db{l}'] = np.zeros((1, layer_dims[l]))
                s[f'dW{l}'] = np.zeros((layer_dims[l-1], layer_dims[l]))
                s[f'db{l}'] = np.zeros((1, layer_dims[l]))
            t = 0
        
        for epoch in range(epochs):
            # Shuffle the data
            indices = np.random.permutation(m)
            X_shuffled = X_train[indices]
            y_shuffled = y_train[indices]
            
            # Mini-batch gradient descent
            for i in range(0, m, batch_size):
                end = min(i + batch_size, m)
                X_batch = X_shuffled[i:end]
                y_batch = y_shuffled[i:end]
                
                # Forward and backward propagation
                AL, cache = forward_propagation(X_batch, parameters)
                gradients = backward_propagation(X_batch, y_batch, parameters, cache)
                
                # Update parameters based on optimizer
                if optimizer == 'sgd':
                    parameters = update_parameters(parameters, gradients, learning_rate)
                elif optimizer == 'momentum':
                    parameters, v = update_parameters_momentum(parameters, gradients, v, 0.9, learning_rate)
                elif optimizer == 'adam':
                    t += 1
                    parameters, v, s = update_parameters_adam(parameters, gradients, v, s, t, 0.9, 0.999, learning_rate)
            
            # Compute cost and accuracy at the end of epoch
            AL, _ = forward_propagation(X_train, parameters)
            cost = cross_entropy_loss(y_train, AL)
            costs.append(cost)
            
            # Compute training and testing accuracy
            y_train_pred = predict(X_train, parameters)
            y_test_pred = predict(X_test, parameters)
            
            train_accuracy = compute_accuracy(y_train_pred, y_train)
            test_accuracy = compute_accuracy(y_test_pred, y_test)
            
            train_acc.append(train_accuracy)
            test_acc.append(test_accuracy)
            
            if epoch % print_every == 0:
                print(f"Epoch {epoch}, Cost: {cost:.4f}, Train Acc: {train_accuracy:.4f}, Test Acc: {test_accuracy:.4f}")
        
        return parameters, costs, train_acc, test_acc
    
    # Define network architecture
    layer_dims = [X_train.shape[1], 16, 8, 2]  # 2 classes
    
    # Train with different optimizers
    optimizers = ['sgd', 'momentum', 'adam']
    results = {}
    
    for opt in optimizers:
        print(f"\nTraining with {opt} optimizer...")
        parameters, costs, train_acc, test_acc = train_nn(
            X_train, y_train_oh, X_test, y_test_oh, 
            layer_dims, opt, learning_rate=0.01 if opt == 'adam' else 0.1, 
            epochs=100, print_every=20
        )
        
        results[opt] = {
            'parameters': parameters,
            'costs': costs,
            'train_acc': train_acc,
            'test_acc': test_acc
        }
    
    # Plot learning curves
    plt.figure(figsize=(18, 6))
    
    # Plot costs
    plt.subplot(1, 3, 1)
    for opt, res in results.items():
        plt.plot(res['costs'], label=opt)
    plt.title('Cost vs. Epochs')
    plt.xlabel('Epochs')
    plt.ylabel('Cost')
    plt.legend()
    plt.grid(True)
    
    # Plot training accuracy
    plt.subplot(1, 3, 2)
    for opt, res in results.items():
        plt.plot(res['train_acc'], label=opt)
    plt.title('Training Accuracy vs. Epochs')
    plt.xlabel('Epochs')
    plt.ylabel('Accuracy')
    plt.legend()
    plt.grid(True)
    
    # Plot testing accuracy
    plt.subplot(1, 3, 3)
    for opt, res in results.items():
        plt.plot(res['test_acc'], label=opt)
    plt.title('Testing Accuracy vs. Epochs')
    plt.xlabel('Epochs')
    plt.ylabel('Accuracy')
    plt.legend()
    plt.grid(True)
    
    plt.tight_layout()
    plt.show()
    
    # Visualize decision boundaries
    def plot_decision_boundary(X, y, parameters):
        h = 0.01
        x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
        y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
        xx, yy = np.meshgrid(np.arange(x_min, x_max, h), np.arange(y_min, y_max, h))

    plt.figure(figsize=(15, 5))
    for i, opt in enumerate(optimizers):
        plt.subplot(1, 3, i+1)
        plot_decision_boundary(X_test, y_test, results[opt]['parameters'])
