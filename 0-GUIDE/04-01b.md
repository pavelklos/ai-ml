## Key Terms

- **Artificial Intelligence (AI)**: Computing systems that can perform tasks requiring human-like intelligence.
- **Machine Learning (ML)**: A subset of AI that allows systems to learn patterns from data without explicit programming.
- **Supervised Learning**: Learning from labeled training data to make predictions.
- **Unsupervised Learning**: Discovering hidden patterns in unlabeled data.
- **Optimization Problem**: Mathematical problem of finding parameter values that minimize or maximize an objective function.
- **Loss/Cost Function**: A function measuring the error between predicted and actual values.
- **Gradient Descent**: First-order iterative optimization algorithm for finding local minima.
- **Convex Optimization**: Optimization where the objective function has a single global minimum.
- **Hessian Matrix**: Square matrix of second-order partial derivatives of a function.

## Advanced Optimization in Machine Learning

Machine learning fundamentally relies on optimization to find parameter values that make models perform well. The general form of an optimization problem in ML is:

```
minimize f(θ) subject to g_i(θ) ≤ 0, h_j(θ) = 0
```

Where:
- θ represents model parameters
- f(θ) is the objective function (typically a loss function)
- g_i(θ) represents inequality constraints
- h_j(θ) represents equality constraints

Most ML problems can be expressed as minimizing a loss function that quantifies the discrepancy between model predictions and ground truth:

```
L(θ) = 1/n ∑ l(f(x_i; θ), y_i)
```

Where l(·) is a specific loss function like squared error, cross-entropy, etc.

## Advanced NumPy and SciPy Optimization

Let's explore a complex optimization problem with constraints using SciPy's optimization toolbox:

```python
import numpy as np
import matplotlib.pyplot as plt
from scipy import optimize
from mpl_toolkits.mplot3d import Axes3D
from matplotlib import cm

# Define a complex objective function (Rosenbrock function)
def rosenbrock(x):
    return sum(100.0 * (x[1:] - x[:-1]**2.0)**2.0 + (1 - x[:-1])**2.0)

# Define gradient of the Rosenbrock function for efficiency
def rosenbrock_grad(x):
    xm = x[1:-1]
    xm_m1 = x[:-2]
    xm_p1 = x[2:]
    grad = np.zeros_like(x)
    grad[0] = -400 * x[0] * (x[1] - x[0]**2) - 2 * (1 - x[0])
    grad[-1] = 200 * (x[-1] - x[-2]**2)
    grad[1:-1] = 200 * (xm - xm_m1**2) - 400 * xm * (xm_p1 - xm**2) - 2 * (1 - xm)
    return grad

# Define constraints (sum of variables equals 1)
def equality_constraint(x):
    return np.sum(x) - 1.0

# Define Jacobian of constraints for efficiency
def equality_constraint_jacobian(x):
    return np.ones_like(x)

# Create constraints dictionary
constraints = [{'type': 'eq', 'fun': equality_constraint, 'jac': equality_constraint_jacobian}]

# Initial guess
x0 = np.array([0.25, 0.25, 0.25, 0.25])

# Bounds for variables
bounds = [(0, 1) for _ in range(4)]

# Perform optimization with different methods and compare
methods = ['SLSQP', 'trust-constr']
results = {}

for method in methods:
    result = optimize.minimize(
        rosenbrock, 
        x0, 
        method=method, 
        jac=rosenbrock_grad,
        constraints=constraints,
        bounds=bounds,
        options={'disp': True, 'maxiter': 1000}
    )
    results[method] = result

# Print comparison of results
print("\nOptimization Results Comparison:")
for method, result in results.items():
    print(f"\n{method}:")
    print(f"  Success: {result.success}")
    print(f"  Message: {result.message}")
    print(f"  Iterations: {result.nit}")
    print(f"  Function evaluations: {result.nfev}")
    print(f"  Optimal solution: {result.x}")
    print(f"  Function value: {result.fun}")
    print(f"  Constraint violation: {equality_constraint(result.x)}")

# Visualize 2D slice of the function
def plot_rosenbrock_contour():
    x = np.linspace(-1.5, 1.5, 300)
    y = np.linspace(-1.5, 1.5, 300)
    X, Y = np.meshgrid(x, y)
    Z = np.zeros_like(X)
    
    for i in range(len(x)):
        for j in range(len(y)):
            Z[j, i] = rosenbrock(np.array([X[j, i], Y[j, i], 0, 0]))
    
    plt.figure(figsize=(10, 8))
    contours = plt.contour(X, Y, Z, 50, cmap='viridis')
    plt.colorbar(contours)
    plt.plot(results['SLSQP'].x[0], results['SLSQP'].x[1], 'r*', markersize=10)
    plt.title('Rosenbrock Function Contour (2D slice)')
    plt.xlabel('x[0]')
    plt.ylabel('x[1]')
    plt.grid(True)
    plt.show()

plot_rosenbrock_contour()
```

## Optimization for Advanced Prediction Tasks

Let's implement a more complex regression task using optimization:

```python
import numpy as np
import matplotlib.pyplot as plt
from sklearn.preprocessing import PolynomialFeatures
from sklearn.metrics import mean_squared_error, r2_score

# Generate synthetic non-linear data with noise
np.random.seed(42)
n_samples = 200
X = np.sort(np.random.uniform(0, 10, n_samples)).reshape(-1, 1)
y = 2 + 3*X - 0.5*X**2 + 0.1*X**3 + np.random.normal(0, 5, size=(n_samples, 1))

# Split data into training and testing sets
train_size = int(0.8 * n_samples)
X_train, X_test = X[:train_size], X[train_size:]
y_train, y_test = y[:train_size], y[train_size:]

# Define polynomial features
poly_degree = 5
poly = PolynomialFeatures(degree=poly_degree, include_bias=False)
X_poly_train = poly.fit_transform(X_train)
X_poly_test = poly.transform(X_test)

# Add bias term
X_poly_train_b = np.c_[np.ones((X_poly_train.shape[0], 1)), X_poly_train]
X_poly_test_b = np.c_[np.ones((X_poly_test.shape[0], 1)), X_poly_test]

# Define the cost function (MSE)
def compute_cost(X, y, theta, lambda_=0):
    m = len(y)
    predictions = X.dot(theta)
    error = predictions - y
    cost = (1/(2*m)) * np.sum(error**2)
    # Add regularization if lambda_ > 0
    if lambda_ > 0:
        cost += (lambda_/(2*m)) * np.sum(theta[1:]**2)  # Exclude bias term
    return cost

# Define gradient of the cost function
def compute_gradient(X, y, theta, lambda_=0):
    m = len(y)
    predictions = X.dot(theta)
    error = predictions - y
    gradient = (1/m) * X.T.dot(error)
    # Add regularization if lambda_ > 0
    if lambda_ > 0:
        gradient_reg = np.copy(gradient)
        gradient_reg[1:] += (lambda_/m) * theta[1:]  # Exclude bias term
        return gradient_reg
    return gradient

# Implement advanced gradient descent with momentum and adaptive learning rate
def gradient_descent_advanced(X, y, theta, learning_rate=0.01, iterations=1000, 
                             lambda_=0, momentum=0.9, lr_decay=0.99, 
                             min_lr=0.0001, tol=1e-6):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, theta.shape[0]))
    velocity = np.zeros_like(theta)
    
    for i in range(iterations):
        gradient = compute_gradient(X, y, theta, lambda_)
        
        # Apply momentum
        velocity = momentum * velocity - learning_rate * gradient
        theta += velocity
        
        # Store history
        theta_history[i] = theta.flatten()
        cost_history[i] = compute_cost(X, y, theta, lambda_)
        
        # Adaptive learning rate
        learning_rate = max(learning_rate * lr_decay, min_lr)
        
        # Early stopping
        if i > 0 and abs(cost_history[i] - cost_history[i-1]) < tol:
            return theta, cost_history[:i+1], theta_history[:i+1]
    
    return theta, cost_history, theta_history

# Initialize parameters and hyperparameters
initial_theta = np.random.randn(X_poly_train_b.shape[1], 1) * 0.01
iterations = 10000
learning_rate = 0.01
lambda_ = 1.0  # Regularization parameter

# Run advanced gradient descent
theta_final, cost_history, _ = gradient_descent_advanced(
    X_poly_train_b, y_train, initial_theta, 
    learning_rate=learning_rate, 
    iterations=iterations,
    lambda_=lambda_,
    momentum=0.9,
    lr_decay=0.999,
    min_lr=0.0001,
    tol=1e-8
)

# Make predictions
y_pred_train = X_poly_train_b.dot(theta_final)
y_pred_test = X_poly_test_b.dot(theta_final)

# Calculate metrics
train_mse = mean_squared_error(y_train, y_pred_train)
test_mse = mean_squared_error(y_test, y_pred_test)
train_r2 = r2_score(y_train, y_pred_train)
test_r2 = r2_score(y_test, y_pred_test)

print(f"Training MSE: {train_mse:.4f}, R²: {train_r2:.4f}")
print(f"Testing MSE: {test_mse:.4f}, R²: {test_r2:.4f}")

# Plot the results
plt.figure(figsize=(15, 10))

# Plot training data and predictions
plt.subplot(2, 2, 1)
plt.scatter(X_train, y_train, color='blue', alpha=0.5, label='Training data')
plt.plot(X_train, y_pred_train, color='red', linewidth=2, label='Model predictions')
plt.title('Training Data and Predictions')
plt.xlabel('X')
plt.ylabel('y')
plt.legend()

# Plot testing data and predictions
plt.subplot(2, 2, 2)
plt.scatter(X_test, y_test, color='green', alpha=0.5, label='Testing data')
plt.plot(X_test, y_pred_test, color='red', linewidth=2, label='Model predictions')
plt.title('Testing Data and Predictions')
plt.xlabel('X')
plt.ylabel('y')
plt.legend()

# Plot cost history
plt.subplot(2, 2, 3)
plt.plot(cost_history)
plt.title('Cost History')
plt.xlabel('Iterations')
plt.ylabel('Cost')
plt.grid(True)

# Plot the learned parameters
plt.subplot(2, 2, 4)
plt.bar(range(len(theta_final)), theta_final.flatten())
plt.title('Model Parameters')
plt.xlabel('Parameter Index')
plt.ylabel('Value')
plt.grid(True)

plt.tight_layout()
plt.show()
```

## Advanced Gradient Descent Algorithms

Gradient descent has numerous variants, each with specific advantages for different scenarios. Let's implement several advanced algorithms:

```python
import numpy as np
import matplotlib.pyplot as plt
from tqdm import tqdm

# Generate more complex data for demonstration
np.random.seed(42)
X = 2 * np.random.rand(1000, 5) - 1
true_theta = np.array([3.5, -2.0, 1.2, -0.5, 2.8]).reshape(-1, 1)
y = X.dot(true_theta) + np.random.normal(0, 1, size=(1000, 1))

# Add bias term
X_b = np.c_[np.ones((X.shape[0], 1)), X]

# Define the cost function for linear regression
def compute_cost(X, y, theta):
    m = len(y)
    predictions = X.dot(theta)
    error = predictions - y
    return (1/(2*m)) * np.sum(error**2)

# Define the gradient of the cost function
def compute_gradient(X, y, theta):
    m = len(y)
    predictions = X.dot(theta)
    error = predictions - y
    return (1/m) * X.T.dot(error)

# Standard Gradient Descent
def batch_gradient_descent(X, y, theta, learning_rate=0.01, iterations=1000):
    m = len(y)
    cost_history = np.zeros(iterations)
    
    for i in range(iterations):
        gradient = compute_gradient(X, y, theta)
        theta = theta - learning_rate * gradient
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# Stochastic Gradient Descent
def stochastic_gradient_descent(X, y, theta, learning_rate=0.01, iterations=1000):
    m = len(y)
    cost_history = np.zeros(iterations)
    
    for i in range(iterations):
        for j in range(m):
            random_index = np.random.randint(m)
            xi = X[random_index:random_index+1]
            yi = y[random_index:random_index+1]
            gradient = compute_gradient(xi, yi, theta)
            theta = theta - learning_rate * gradient
        
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# Mini-Batch Gradient Descent
def mini_batch_gradient_descent(X, y, theta, learning_rate=0.01, iterations=1000, batch_size=32):
    m = len(y)
    cost_history = np.zeros(iterations)
    
    for i in range(iterations):
        indices = np.random.permutation(m)
        X_shuffled = X[indices]
        y_shuffled = y[indices]
        
        for j in range(0, m, batch_size):
            X_batch = X_shuffled[j:j+batch_size]
            y_batch = y_shuffled[j:j+batch_size]
            gradient = compute_gradient(X_batch, y_batch, theta)
            theta = theta - learning_rate * gradient
        
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# Momentum Gradient Descent
def momentum_gradient_descent(X, y, theta, learning_rate=0.01, iterations=1000, beta=0.9):
    m = len(y)
    cost_history = np.zeros(iterations)
    velocity = np.zeros_like(theta)
    
    for i in range(iterations):
        gradient = compute_gradient(X, y, theta)
        velocity = beta * velocity - learning_rate * gradient
        theta = theta + velocity
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# RMSprop
def rmsprop(X, y, theta, learning_rate=0.01, iterations=1000, beta=0.9, epsilon=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    s = np.zeros_like(theta)
    
    for i in range(iterations):
        gradient = compute_gradient(X, y, theta)
        s = beta * s + (1 - beta) * gradient**2
        theta = theta - (learning_rate / (np.sqrt(s) + epsilon)) * gradient
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# Adam Optimizer
def adam_optimizer(X, y, theta, learning_rate=0.01, iterations=1000, beta1=0.9, beta2=0.999, epsilon=1e-8):
    m = len(y)
    cost_history = np.zeros(iterations)
    v = np.zeros_like(theta)  # First moment estimate
    s = np.zeros_like(theta)  # Second moment estimate
    
    for i in range(iterations):
        gradient = compute_gradient(X, y, theta)
        
        # Update biased first moment estimate
        v = beta1 * v + (1 - beta1) * gradient
        
        # Update biased second moment estimate
        s = beta2 * s + (1 - beta2) * gradient**2
        
        # Correct bias in first moment
        v_corrected = v / (1 - beta1**(i+1))
        
        # Correct bias in second moment
        s_corrected = s / (1 - beta2**(i+1))
        
        # Update parameters
        theta = theta - learning_rate * v_corrected / (np.sqrt(s_corrected) + epsilon)
        
        cost_history[i] = compute_cost(X, y, theta)
    
    return theta, cost_history

# Initialize parameters
initial_theta = np.random.randn(X_b.shape[1], 1) * 0.01
iterations = 200

# Run different optimizers
print("Running optimization algorithms...")
theta_bgd, cost_bgd = batch_gradient_descent(X_b, y, initial_theta.copy(), learning_rate=0.01, iterations=iterations)
theta_sgd, cost_sgd = stochastic_gradient_descent(X_b, y, initial_theta.copy(), learning_rate=0.001, iterations=iterations)
theta_mbgd, cost_mbgd = mini_batch_gradient_descent(X_b, y, initial_theta.copy(), learning_rate=0.01, iterations=iterations)
theta_mgd, cost_mgd = momentum_gradient_descent(X_b, y, initial_theta.copy(), learning_rate=0.01, iterations=iterations)
theta_rmsprop, cost_rmsprop = rmsprop(X_b, y, initial_theta.copy(), learning_rate=0.01, iterations=iterations)
theta_adam, cost_adam = adam_optimizer(X_b, y, initial_theta.copy(), learning_rate=0.01, iterations=iterations)

# Print final costs
print(f"Final costs after {iterations} iterations:")
print(f"Batch GD: {cost_bgd[-1]:.6f}")
print(f"SGD: {cost_sgd[-1]:.6f}")
print(f"Mini-batch GD: {cost_mbgd[-1]:.6f}")
print(f"Momentum GD: {cost_mgd[-1]:.6f}")
print(f"RMSprop: {cost_rmsprop[-1]:.6f}")
print(f"Adam: {cost_adam[-1]:.6f}")

# Plot convergence
plt.figure(figsize=(12, 8))
plt.plot(cost_bgd, label='Batch GD')
plt.plot(cost_sgd, label='SGD')
plt.plot(cost_mbgd, label='Mini-batch GD')
plt.plot(cost_mgd, label='Momentum GD')
plt.plot(cost_rmsprop, label='RMSprop')
plt.plot(cost_adam, label='Adam')
plt.title('Convergence of Different Optimization Algorithms')
plt.xlabel('Iterations')
plt.ylabel('Cost')
plt.legend()
plt.grid(True)
plt.yscale('log')
plt.show()

# Print parameters compared to true parameters
print("\nTrue parameters vs. optimized parameters:")
print("True parameters:", true_theta.flatten())
print("Batch GD:", theta_bgd.flatten())
print("SGD:", theta_sgd.flatten())
print("Mini-batch GD:", theta_mbgd.flatten())
print("Momentum GD:", theta_mgd.flatten())
print("RMSprop:", theta_rmsprop.flatten())
print("Adam:", theta_adam.flatten())
```
