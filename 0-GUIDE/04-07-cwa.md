# 07. Data Quality: Foundation of Successful AI

## Key Terms and Concepts

**Data Quality** refers to the condition of data based on factors like accuracy, completeness, reliability, and relevance. High-quality data is essential for building robust machine learning models.

**Missing Values (NaN)** are data points that are absent from the dataset, which can occur due to data collection errors, system failures, or intentional omissions.

**Outliers** are data points that significantly deviate from the majority of observations in a dataset. They can represent genuine extreme cases or data collection errors.

**Data Imputation** is the process of replacing missing data with substituted values using statistical methods or predictive models.

**Feature Scaling** is the process of normalizing or standardizing features to ensure they have similar scales, preventing features with larger magnitudes from dominating the model training process.

**Data Preprocessing Pipeline** is a systematic approach to cleaning, transforming, and preparing raw data for machine learning algorithms.

class DataQualityAnalyzer:
    """
    Comprehensive data quality analysis and reporting tool
    """
    
    def __init__(self, df):
        self.df = df.copy()
        self.quality_report = {}
    
    def analyze_missing_values(self):
        """Analyze patterns and distribution of missing values"""
        missing_stats = {}
        
        # Basic missing value statistics
        missing_counts = self.df.isnull().sum()
        missing_percentages = (missing_counts / len(self.df)) * 100
        
        missing_stats['total_missing'] = missing_counts.sum()
        missing_stats['columns_with_missing'] = (missing_counts > 0).sum()
        missing_stats['missing_by_column'] = pd.DataFrame({
            'missing_count': missing_counts,
            'missing_percentage': missing_percentages
        }).sort_values('missing_percentage', ascending=False)
        
        # Missing value patterns
        missing_patterns = self.df.isnull().value_counts()
        missing_stats['missing_patterns'] = missing_patterns.head(10)
        
        # Completely empty rows
        empty_rows = self.df.isnull().all(axis=1).sum()
        missing_stats['empty_rows'] = empty_rows
        
        self.quality_report['missing_values'] = missing_stats
        return missing_stats
    
    def detect_outliers_multiple_methods(self):
        """Detect outliers using multiple statistical methods"""
        numerical_cols = self.df.select_dtypes(include=[np.number]).columns
        outlier_results = {}
        
        for col in numerical_cols:
            if self.df[col].notna().sum() > 0:
                data = self.df[col].dropna()
                col_outliers = {}
                
                # Z-Score method
                z_scores = np.abs(stats.zscore(data))
                z_outliers = np.where(z_scores > 3)[0]
                col_outliers['zscore'] = len(z_outliers)
                
                # IQR method
                Q1 = data.quantile(0.25)
                Q3 = data.quantile(0.75)
                IQR = Q3 - Q1
                lower_bound = Q1 - 1.5 * IQR
                upper_bound = Q3 + 1.5 * IQR
                iqr_outliers = data[(data < lower_bound) | (data > upper_bound)]
                col_outliers['iqr'] = len(iqr_outliers)
                
                # Modified Z-Score (using median)
                median = data.median()
                mad = np.median(np.abs(data - median))
                modified_z_scores = 0.6745 * (data - median) / mad
                modified_z_outliers = np.abs(modified_z_scores) > 3.5
                col_outliers['modified_zscore'] = modified_z_outliers.sum()
                
                outlier_results[col] = col_outliers
        
        self.quality_report['outliers'] = outlier_results
        return outlier_results
    
    def analyze_data_distribution(self):
        """Analyze distribution characteristics of numerical features"""
        numerical_cols = self.df.select_dtypes(include=[np.number]).columns
        distribution_stats = {}
        
        for col in numerical_cols:
            if self.df[col].notna().sum() > 0:
                data = self.df[col].dropna()
                
                stats_dict = {
                    'mean': data.mean(),
                    'median': data.median(),
                    'std': data.std(),
                    'skewness': stats.skew(data),
                    'kurtosis': stats.kurtosis(data),
                    'min': data.min(),
                    'max': data.max(),
                    'range': data.max() - data.min(),
                    'coefficient_of_variation': data.std() / data.mean() if data.mean() != 0 else np.inf
                }
                
                distribution_stats[col] = stats_dict
        
        self.quality_report['distributions'] = distribution_stats
        return distribution_stats
    
    def generate_quality_report(self):
        """Generate comprehensive data quality report"""
        print("=== DATA QUALITY ANALYSIS REPORT ===\n")
        
        # Dataset overview
        print(f"Dataset Shape: {self.df.shape}")
        print(f"Memory Usage: {self.df.memory_usage(deep=True).sum() / 1024**2:.2f} MB\n")
        
        # Missing values analysis
        missing_stats = self.analyze_missing_values()
        print("=== MISSING VALUES ANALYSIS ===")
        print(f"Total missing values: {missing_stats['total_missing']}")
        print(f"Columns with missing values: {missing_stats['columns_with_missing']}")
        print("\nTop columns with missing values:")
        print(missing_stats['missing_by_column'].head())
        
        # Outliers analysis
        print("\n=== OUTLIERS ANALYSIS ===")
        outlier_stats = self.detect_outliers_multiple_methods()
        for col, methods in outlier_stats.items():
            print(f"{col}: Z-Score: {methods['zscore']}, IQR: {methods['iqr']}, Modified Z-Score: {methods['modified_zscore']}")
        
        return self.quality_report

class AdvancedImputer:
    """
    Advanced imputation strategies for different types of missing data
    """
    
    def __init__(self, strategy='adaptive'):
        self.strategy = strategy
        self.imputers = {}
        self.missing_patterns = {}
    
    def analyze_missing_mechanism(self, df):
        """
        Analyze missing data mechanism: MCAR, MAR, or MNAR
        """
        missing_matrix = df.isnull()
        
        # Calculate correlations between missing indicators
        missing_corr = missing_matrix.corr()
        
        # Test for MCAR using Little's MCAR test (simplified version)
        patterns = {}
        for col in df.columns:
            if missing_matrix[col].any():
                # Analyze correlation with other variables
                col_missing = missing_matrix[col]
                correlations = {}
                
                for other_col in df.columns:
                    if other_col != col and df[other_col].dtype in ['int64', 'float64']:
                        # Point-biserial correlation
                        valid_data = df[other_col].dropna()
                        valid_missing = col_missing[valid_data.index]
                        
                        if len(valid_data) > 0 and valid_missing.var() > 0:
                            corr_coef = np.corrcoef(valid_data, valid_missing)[0, 1]
                            correlations[other_col] = corr_coef
                
                patterns[col] = correlations
        
        return patterns
    
    def impute_numerical_advanced(self, df, column):
        """Advanced imputation for numerical variables"""
        if self.strategy == 'knn':
            imputer = KNNImputer(n_neighbors=5)
        elif self.strategy == 'iterative':
            imputer = IterativeImputer(random_state=42, max_iter=10)
        elif self.strategy == 'adaptive':
            # Choose strategy based on missing percentage
            missing_pct = df[column].isnull().sum() / len(df)
            
            if missing_pct < 0.05:
                # Low missing rate: use mean/median
                if abs(df[column].skew()) > 1:
                    imputer = SimpleImputer(strategy='median')
                else:
                    imputer = SimpleImputer(strategy='mean')
            elif missing_pct < 0.2:
                # Medium missing rate: use KNN
                imputer = KNNImputer(n_neighbors=5)
            else:
                # High missing rate: use iterative imputation
                imputer = IterativeImputer(random_state=42, max_iter=10)
        
        return imputer
    
    def impute_categorical_advanced(self, df, column):
        """Advanced imputation for categorical variables"""
        # Use mode for low missing rates, predict for high missing rates
        missing_pct = df[column].isnull().sum() / len(df)
        
        if missing_pct < 0.1:
            return SimpleImputer(strategy='most_frequent')
        else:
            # For high missing rates, we could use a classifier
            # For simplicity, using mode here
            return SimpleImputer(strategy='most_frequent')
    
    def fit_transform(self, df):
        """Fit and transform the dataset with appropriate imputation strategies"""
        df_imputed = df.copy()
        
        # Analyze missing mechanisms
        missing_patterns = self.analyze_missing_mechanism(df)
        
        # Handle numerical columns
        numerical_cols = df.select_dtypes(include=[np.number]).columns
        for col in numerical_cols:
            if df[col].isnull().any():
                imputer = self.impute_numerical_advanced(df, col)
                
                # Fit and transform
                col_reshaped = df[col].values.reshape(-1, 1)
                df_imputed[col] = imputer.fit_transform(col_reshaped).flatten()
                
                self.imputers[col] = imputer
        
        # Handle categorical columns
        categorical_cols = df.select_dtypes(include=['object']).columns
        for col in categorical_cols:
            if df[col].isnull().any():
                imputer = self.impute_categorical_advanced(df, col)
                
                col_reshaped = df[col].values.reshape(-1, 1)
                df_imputed[col] = imputer.fit_transform(col_reshaped).flatten()
                
                self.imputers[col] = imputer
        
        return df_imputed

class OutlierHandler:
    """
    Comprehensive outlier detection and treatment using multiple methods
    """
    
    def __init__(self):
        self.outlier_detectors = {
            'isolation_forest': IsolationForest(contamination=0.1, random_state=42),
            'elliptic_envelope': EllipticEnvelope(contamination=0.1, random_state=42),
            'one_class_svm': OneClassSVM(nu=0.1)
        }
        self.treatment_methods = {}
    
    def detect_outliers_ensemble(self, df, numerical_cols=None):
        """
        Detect outliers using ensemble of methods
        """
        if numerical_cols is None:
            numerical_cols = df.select_dtypes(include=[np.number]).columns
        
        df_numerical = df[numerical_cols].dropna()
        
        if len(df_numerical) == 0:
            return np.array([])
        
        outlier_scores = {}
        
        # Apply each detection method
        for name, detector in self.outlier_detectors.items():
            try:
                detector.fit(df_numerical)
                predictions = detector.predict(df_numerical)
                # Convert to binary (1 for normal, -1 for outlier)
                outlier_scores[name] = predictions
            except Exception as e:
                print(f"Warning: {name} failed with error: {e}")
                outlier_scores[name] = np.ones(len(df_numerical))
        
        # Ensemble voting (majority vote)
        outlier_matrix = np.column_stack(list(outlier_scores.values()))
        ensemble_predictions = np.apply_along_axis(
            lambda x: 1 if np.sum(x == 1) > np.sum(x == -1) else -1, 
            axis=1, 
            arr=outlier_matrix
        )
        
        # Map back to original dataframe indices
        outlier_mask = np.zeros(len(df), dtype=bool)
        outlier_mask[df_numerical.index] = ensemble_predictions == -1
        
        return outlier_mask
    
    def treat_outliers(self, df, method='cap', outlier_mask=None):
        """
        Treat outliers using specified method
        """
        if outlier_mask is None:
            outlier_mask = self.detect_outliers_ensemble(df)
        
        df_treated = df.copy()
        numerical_cols = df.select_dtypes(include=[np.number]).columns
        
        if method == 'cap':
            # Cap outliers at 95th and 5th percentiles
            for col in numerical_cols:
                Q05 = df[col].quantile(0.05)
                Q95 = df[col].quantile(0.95)
                df_treated[col] = np.clip(df_treated[col], Q05, Q95)
        
        elif method == 'transform':
            # Apply log transformation to reduce outlier impact
            for col in numerical_cols:
                if (df[col] > 0).all():  # Only if all values are positive
                    df_treated[col] = np.log1p(df[col])
        
        elif method == 'remove':
            # Remove outlier rows
            df_treated = df_treated[~outlier_mask]
        
        elif method == 'impute':
            # Replace outliers with median values
            for col in numerical_cols:
                median_val = df[col].median()
                df_treated.loc[outlier_mask, col] = median_val
        
        return df_treated
    
    def analyze_outlier_impact(self, df_original, df_treated):
        """
        Analyze the impact of outlier treatment on data distribution
        """
        numerical_cols = df_original.select_dtypes(include=[np.number]).columns
        
        impact_analysis = {}
        
        for col in numerical_cols:
            original_stats = {
                'mean': df_original[col].mean(),
                'std': df_original[col].std(),
                'skewness': stats.skew(df_original[col].dropna()),
                'kurtosis': stats.kurtosis(df_original[col].dropna())
            }
            
            treated_stats = {
                'mean': df_treated[col].mean(),
                'std': df_treated[col].std(),
                'skewness': stats.skew(df_treated[col].dropna()),
                'kurtosis': stats.kurtosis(df_treated[col].dropna())
            }
            
            impact_analysis[col] = {
                'original': original_stats,
                'treated': treated_stats,
                'mean_change': treated_stats['mean'] - original_stats['mean'],
                'std_change': treated_stats['std'] - original_stats['std']
            }
        
        return impact_analysis

class AdvancedScaler:
    """
    Advanced scaling strategies with automatic selection based on data characteristics
    """
    
    def __init__(self):
        self.scalers = {
            'standard': StandardScaler(),
            'minmax': MinMaxScaler(),
            'robust': RobustScaler(),
            'quantile_uniform': None,  # Will be initialized dynamically
            'quantile_normal': None    # Will be initialized dynamically
        }
        self.selected_scalers = {}
        self.scaling_recommendations = {}
    
    def analyze_scaling_needs(self, df):
        """
        Analyze data characteristics to recommend appropriate scaling method
        """
        numerical_cols = df.select_dtypes(include=[np.number]).columns
        recommendations = {}
        
        for col in numerical_cols:
            data = df[col].dropna()
            
            if len(data) == 0:
                recommendations[col] = 'standard'
                continue
            
            # Calculate distribution characteristics
            skewness = abs(stats.skew(data))
            kurtosis = abs(stats.kurtosis(data))
            outlier_ratio = self._calculate_outlier_ratio(data)
            range_ratio = (data.max() - data.min()) / data.std() if data.std() > 0 else 0
            
            # Decision logic for scaling method
            if outlier_ratio > 0.1:  # High outlier presence
                recommendations[col] = 'robust'
            elif skewness > 2:  # Highly skewed data
                recommendations[col] = 'quantile_uniform'
            elif range_ratio > 100:  # Very large range
                recommendations[col] = 'minmax'
            else:  # Normal-ish distribution
                recommendations[col] = 'standard'
            
            # Store analysis results
            self.scaling_recommendations[col] = {
                'method': recommendations[col],
                'skewness': skewness,
                'kurtosis': kurtosis,
                'outlier_ratio': outlier_ratio,
                'range_ratio': range_ratio
            }
        
        return recommendations
    
    def _calculate_outlier_ratio(self, data):
        """Calculate the ratio of outliers using IQR method"""
        Q1 = data.quantile(0.25)
        Q3 = data.quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        outliers = data[(data < lower_bound) | (data > upper_bound)]
        return len(outliers) / len(data)
    
    def fit_transform_adaptive(self, df):
        """
        Apply adaptive scaling based on data characteristics
        """
        from sklearn.preprocessing import QuantileTransformer
        
        df_scaled = df.copy()
        numerical_cols = df.select_dtypes(include=[np.number]).columns
        
        # Get scaling recommendations
        recommendations = self.analyze_scaling_needs(df)
        
        for col in numerical_cols:
            method = recommendations[col]
            data = df[col].values.reshape(-1, 1)
            
            if method == 'quantile_uniform':
                scaler = QuantileTransformer(output_distribution='uniform', random_state=42)
            elif method == 'quantile_normal':
                scaler = QuantileTransformer(output_distribution='normal', random_state=42)
            else:
                scaler = self.scalers[method]
            
            # Fit and transform
            df_scaled[col] = scaler.fit_transform(data).flatten()
            self.selected_scalers[col] = scaler
        
        return df_scaled
    
    def compare_scaling_methods(self, df, column):
        """
        Compare different scaling methods for a specific column
        """
        from sklearn.preprocessing import QuantileTransformer
        
        data = df[column].dropna().values.reshape(-1, 1)
        
        scalers_to_test = {
            'Original': None,
            'Standard': StandardScaler(),
            'MinMax': MinMaxScaler(),
            'Robust': RobustScaler(),
            'Quantile Uniform': QuantileTransformer(output_distribution='uniform'),
            'Quantile Normal': QuantileTransformer(output_distribution='normal')
        }
        
        results = {}
        
        for name, scaler in scalers_to_test.items():
            if scaler is None:
                scaled_data = data.flatten()
            else:
                scaled_data = scaler.fit_transform(data).flatten()
            
            results[name] = {
                'mean': np.mean(scaled_data),
                'std': np.std(scaled_data),
                'min': np.min(scaled_data),
                'max': np.max(scaled_data),
                'skewness': stats.skew(scaled_data),
                'kurtosis': stats.kurtosis(scaled_data)
            }
        
        return results

class ComprehensiveDataPreprocessor:
    """
    Complete data preprocessing pipeline combining all cleaning steps
    """
    
    def __init__(self, missing_strategy='adaptive', outlier_method='cap', scaling_method='adaptive'):
        self.missing_strategy = missing_strategy
        self.outlier_method = outlier_method
        self.scaling_method = scaling_method
        
        self.imputer = AdvancedImputer(strategy=missing_strategy)
        self.outlier_handler = OutlierHandler()
        self.scaler = AdvancedScaler()
        
        self.preprocessing_report = {}
    
    def fit_transform(self, df):
        """
        Complete preprocessing pipeline
        """
        print("Starting comprehensive data preprocessing...")
        
        # Initial data quality analysis
        analyzer = DataQualityAnalyzer(df)
        initial_quality = analyzer.generate_quality_report()
        
        # Step 1: Handle missing values
        print("\n1. Handling missing values...")
        df_imputed = self.imputer.fit_transform(df)
        missing_reduction = df.isnull().sum().sum() - df_imputed.isnull().sum().sum()
        print(f"   Imputed {missing_reduction} missing values")
        
        # Step 2: Handle outliers
        print("\n2. Handling outliers...")
        outlier_mask = self.outlier_handler.detect_outliers_ensemble(df_imputed)
        df_outliers_treated = self.outlier_handler.treat_outliers(
            df_imputed, method=self.outlier_method, outlier_mask=outlier_mask
        )
        print(f"   Treated {outlier_mask.sum()} outliers using {self.outlier_method} method")
        
        # Step 3: Feature scaling
        print("\n3. Applying feature scaling...")
        if self.scaling_method == 'adaptive':
            df_final = self.scaler.fit_transform_adaptive(df_outliers_treated)
        else:
            # Apply uniform scaling method
            numerical_cols = df_outliers_treated.select_dtypes(include=[np.number]).columns
            df_final = df_outliers_treated.copy()
            
            if self.scaling_method == 'standard':
                scaler = StandardScaler()
            elif self.scaling_method == 'minmax':
                scaler = MinMaxScaler()
            elif self.scaling_method == 'robust':
                scaler = RobustScaler()
            
            for col in numerical_cols:
                df_final[col] = scaler.fit_transform(df_final[col].values.reshape(-1, 1)).flatten()
        
        print("   Applied adaptive scaling based on data characteristics")
        
        # Generate final quality report
        final_analyzer = DataQualityAnalyzer(df_final)
        final_quality = final_analyzer.generate_quality_report()
        
        # Store preprocessing summary
        self.preprocessing_report = {
            'initial_shape': df.shape,
            'final_shape': df_final.shape,
            'missing_values_imputed': missing_reduction,
            'outliers_treated': outlier_mask.sum(),
            'scaling_methods': getattr(self.scaler, 'scaling_recommendations', {}),
            'data_quality_improvement': self._calculate_quality_improvement(df, df_final)
        }
        
        return df_final
    
    def _calculate_quality_improvement(self, df_original, df_processed):
        """Calculate overall data quality improvement metrics"""
        numerical_cols = df_original.select_dtypes(include=[np.number]).columns
        
        improvements = {}
        
        for col in numerical_cols:
            if col in df_processed.columns:
                original_data = df_original[col].dropna()
                processed_data = df_processed[col].dropna()
                
                if len(original_data) > 0 and len(processed_data) > 0:
                    # Calculate coefficient of variation improvement
                    original_cv = original_data.std() / abs(original_data.mean()) if original_data.mean() != 0 else np.inf
                    processed_cv = processed_data.std() / abs(processed_data.mean()) if processed_data.mean() != 0 else np.inf
                    
                    improvements[col] = {
                        'cv_reduction': original_cv - processed_cv,
                        'skewness_reduction': abs(stats.skew(original_data)) - abs(stats.skew(processed_data)),
                        'completeness_improvement': len(processed_data) - len(original_data)
                    }
        
        return improvements
    
    def get_preprocessing_summary(self):
        """Get comprehensive preprocessing summary"""
        return self.preprocessing_report
