## Key Terms and Concepts

- **Time Series**: Sequential data points measured at successive time intervals.
- **Stationarity**: A property where statistical properties (mean, variance) remain constant over time.
- **Seasonality**: Regular patterns that repeat at predictable intervals.
- **Trend**: Long-term movement in the data, showing consistent direction.
- **Autocorrelation**: Correlation between a time series and a lagged version of itself.
- **ARIMA**: Autoregressive Integrated Moving Average, a classical time series forecasting model.
- **Exogenous Variables**: External factors that influence the time series.
- **Feature Engineering**: Creating new variables from existing data to improve model performance.
- **Time Series Cross-Validation**: Validation techniques that respect the temporal ordering of data.
- **Prophet**: Facebook's time series forecasting tool that handles seasonality and holidays.
- **Rolling Window**: A sliding window of fixed size used for feature creation or model training.
- **Forecasting Horizon**: The future time period for which predictions are made.

```python
def create_sample_time_series(start_date='2018-01-01', periods=365*3, freq='D', seasonal_periods=[7, 365], trend=0.05):
    """
    Create a synthetic time series with trend and multiple seasonal components.
    
    Parameters:
    -----------
    start_date : str
        Start date of the time series
    periods : int
        Number of periods to generate
    freq : str
        Frequency of the time series (D=daily, H=hourly, etc.)
    seasonal_periods : list
        List of seasonal periods to include (e.g., 7 for weekly, 365 for yearly)
    trend : float
        Strength of the upward/downward trend
    
    Returns:
    --------
    pd.DataFrame
        DataFrame containing the synthetic time series
    """
    # Create date range
    date_rng = pd.date_range(start=start_date, periods=periods, freq=freq)
    
    # Create base trend
    trend_component = np.arange(periods) * trend
    
    # Create seasonal components
    seasonal_components = np.zeros(periods)
    for period in seasonal_periods:
        # Different amplitude for different seasonal patterns
        amplitude = 10 if period == 7 else 20
        seasonal_components += amplitude * np.sin(2 * np.pi * np.arange(periods) / period)
    
    # Add random noise
    noise = np.random.normal(0, 5, periods)
    
    # Combine components
    values = trend_component + seasonal_components + noise
    
    # Create DataFrame
    df = pd.DataFrame({'date': date_rng, 'value': values})
    df.set_index('date', inplace=True)
    
    return df

def check_stationarity(time_series):
    """
    Perform Augmented Dickey-Fuller test to check for stationarity.
    
    Parameters:
    -----------
    time_series : pd.Series
        Time series to check
    
    Returns:
    --------
    dict
        Dictionary with test results
    """
    result = adfuller(time_series.dropna())
    
    print(f'ADF Statistic: {result[0]:.4f}')
    print(f'p-value: {result[1]:.4f}')
    print('Critical Values:')
    for key, value in result[4].items():
        print(f'\t{key}: {value:.4f}')
    
    # Interpret the result
    if result[1] <= 0.05:
        print("Stationary: The time series is stationary (reject null hypothesis)")
    else:
        print("Non-Stationary: The time series is not stationary (fail to reject null hypothesis)")
    
    return {
        'adf_statistic': result[0],
        'p_value': result[1],
        'critical_values': result[4],
        'is_stationary': result[1] <= 0.05
    }
```

```python
def fit_lstm_model(time_series, n_steps=10, n_features=1, n_epochs=50, batch_size=32):
    """
    Fit an LSTM neural network model to the time series.
    
    Parameters:
    -----------
    time_series : pd.Series
        Time series to model
    n_steps : int
        Number of time steps to look back
    n_features : int
        Number of features (1 for univariate)
    n_epochs : int
        Number of training epochs
    batch_size : int
        Batch size for training
    
    Returns:
    --------
    tuple
        (LSTM model, training history)
    """
    # Scale the data
    scaler = StandardScaler()
    scaled_data = scaler.fit_transform(time_series.values.reshape(-1, 1))
    
    # Create sequences for LSTM
    def create_sequences(data, n_steps):
        X, y = [], []
        for i in range(len(data) - n_steps):
            X.append(data[i:i+n_steps, 0])
            y.append(data[i+n_steps, 0])
        return np.array(X), np.array(y)
    
    X, y = create_sequences(scaled_data, n_steps)
    
    # Reshape X to be [samples, time steps, features]
    X = X.reshape((X.shape[0], X.shape[1], n_features))
    
    # Split data into train and test sets
    train_size = int(len(X) * 0.8)
    X_train, X_test = X[:train_size], X[train_size:]
    y_train, y_test = y[:train_size], y[train_size:]
    
    # Define LSTM model
    model = Sequential()
    model.add(LSTM(50, activation='relu', return_sequences=True, input_shape=(n_steps, n_features)))
    model.add(LSTM(50, activation='relu'))
    model.add(Dropout(0.2))
    model.add(Dense(1))
    model.compile(optimizer='adam', loss='mse')
    
    # Train model
    history = model.fit(
        X_train, y_train,
        epochs=n_epochs,
        batch_size=batch_size,
        validation_data=(X_test, y_test),
        verbose=0
    )
    
    print("\nLSTM Model Summary:")
    model.summary()
    print(f"Final training loss: {history.history['loss'][-1]:.4f}")
    print(f"Final validation loss: {history.history['val_loss'][-1]:.4f}")
    
    return model, history, scaler
```

```python
def create_time_features(df, date_col):
    """
    Create time-based features from datetime index.
    
    Parameters:
    -----------
    df : pd.DataFrame
        DataFrame containing the time series
    date_col : str
        Name of the datetime column
    
    Returns:
    --------
    pd.DataFrame
        DataFrame with added time features
    """
    df_features = df.copy()
    
    # Extract datetime features
    df_features['hour'] = df_features[date_col].dt.hour
    df_features['dayofweek'] = df_features[date_col].dt.dayofweek
    df_features['quarter'] = df_features[date_col].dt.quarter
    df_features['month'] = df_features[date_col].dt.month
    df_features['year'] = df_features[date_col].dt.year
    df_features['dayofyear'] = df_features[date_col].dt.dayofyear
    df_features['dayofmonth'] = df_features[date_col].dt.day
    df_features['weekofyear'] = df_features[date_col].dt.isocalendar().week
    
    # Add cyclical features for periodic time elements
    # This helps with circular nature of time (e.g., Dec 31 is close to Jan 1)
    for col in ['hour', 'dayofweek', 'month', 'dayofyear', 'dayofmonth']:
        max_val = df_features[col].max()
        df_features[f'{col}_sin'] = np.sin(2 * np.pi * df_features[col] / max_val)
        df_features[f'{col}_cos'] = np.cos(2 * np.pi * df_features[col] / max_val)
    
    return df_features

def create_lag_features(df, value_col, lag_periods=[1, 7, 14, 28]):
    """
    Create lag features from time series.
    
    Parameters:
    -----------
    df : pd.DataFrame
        DataFrame containing the time series
    value_col : str
        Name of the value column
    lag_periods : list
        List of lag periods to create
    
    Returns:
    --------
    pd.DataFrame
        DataFrame with added lag features
    """
    df_features = df.copy()
    
    # Create lag features
    for lag in lag_periods:
        df_features[f'lag_{lag}'] = df_features[value_col].shift(lag)
    
    return df_features

def create_rolling_features(df, value_col, windows=[7, 14, 30], functions=['mean', 'std', 'min', 'max']):
    """
    Create rolling window features from time series.
    
    Parameters:
    -----------
    df : pd.DataFrame
        DataFrame containing the time series
    value_col : str
        Name of the value column
    windows : list
        List of window sizes
    functions : list
        List of aggregation functions to apply
    
    Returns:
    --------
    pd.DataFrame
        DataFrame with added rolling features
    """
    df_features = df.copy()
    
    # Create rolling window features
    for window in windows:
        for function in functions:
            if function == 'mean':
                df_features[f'rolling_{window}_{function}'] = df_features[value_col].rolling(window=window).mean()
            elif function == 'std':
                df_features[f'rolling_{window}_{function}'] = df_features[value_col].rolling(window=window).std()
            elif function == 'min':
                df_features[f'rolling_{window}_{function}'] = df_features[value_col].rolling(window=window).min()
            elif function == 'max':
                df_features[f'rolling_{window}_{function}'] = df_features[value_col].rolling(window=window).max()
    
    return df_features

def create_expanding_features(df, value_col, functions=['mean', 'std']):
    """
    Create expanding window features from time series.
    
    Parameters:
    -----------
    df : pd.DataFrame
        DataFrame containing the time series
    value_col : str
        Name of the value column
    functions : list
        List of aggregation functions to apply
    
    Returns:
    --------
    pd.DataFrame
        DataFrame with added expanding features
    """
    df_features = df.copy()
    
    # Create expanding window features
    for function in functions:
        if function == 'mean':
            df_features[f'expanding_{function}'] = df_features[value_col].expanding().mean()
        elif function == 'std':
            df_features[f'expanding_{function}'] = df_features[value_col].expanding().std()
    
    return df_features

def create_fourier_features(df, date_col, value_col, periods=[7, 30, 365]):
    """
    Create Fourier features for capturing seasonality.
    
    Parameters:
    -----------
    df : pd.DataFrame
        DataFrame containing the time series
    date_col : str
        Name of the datetime column
    value_col : str
        Name of the value column
    periods : list
        List of seasonal periods to model
    
    Returns:
    --------
    pd.DataFrame
        DataFrame with added Fourier features
    """
    df_features = df.copy()
    
    # Create a continuous time variable (days since start)
    start_date = df_features[date_col].min()
    df_features['days_since_start'] = (df_features[date_col] - start_date).dt.days
    
    # Create Fourier features for each period
    for period in periods:
        for k in range(1, 3):  # We'll use 2 pairs of sin/cos for each period
            df_features[f'fourier_sin_{period}_{k}'] = np.sin(2 * np.pi * k * df_features['days_since_start'] / period)
            df_features[f'fourier_cos_{period}_{k}'] = np.cos(2 * np.pi * k * df_features['days_since_start'] / period)
    
    return df_features
```

## Cross-Validation in Time Series

Standard cross-validation techniques don't work well with time series due to temporal dependencies:

```python
def time_series_cross_validation(X, y, model, n_splits=5, gap=0):
    """
    Perform time series cross-validation.
    
    Parameters:
    -----------
    X : pd.DataFrame
        Feature DataFrame
    y : pd.Series
        Target variable
    model : object
        Model with fit and predict methods
    n_splits : int
        Number of splits for cross-validation
    gap : int
        Gap between train and test sets
    
    Returns:
    --------
    list
        List of evaluation metrics for each fold
    """
    # Initialize TimeSeriesSplit
    tscv = TimeSeriesSplit(n_splits=n_splits, gap=gap)
    
    # Store results
    rmse_scores = []
    mae_scores = []
    r2_scores = []
    
    # Perform cross-validation
    for i, (train_idx, test_idx) in enumerate(tscv.split(X)):
        # Split data
        X_train, X_test = X.iloc[train_idx], X.iloc[test_idx]
        y_train, y_test = y.iloc[train_idx], y.iloc[test_idx]
        
        # Fit model
        model.fit(X_train, y_train)
        
        # Predict
        y_pred = model.predict(X_test)
        
        # Calculate metrics
        rmse = np.sqrt(mean_squared_error(y_test, y_pred))
        mae = mean_absolute_error(y_test, y_pred)
        r2 = r2_score(y_test, y_pred)
        
        # Store metrics
        rmse_scores.append(rmse)
        mae_scores.append(mae)
        r2_scores.append(r2)
        
        print(f"Fold {i+1}: RMSE = {rmse:.4f}, MAE = {mae:.4f}, R² = {r2:.4f}")
    
    # Calculate average metrics
    avg_rmse = np.mean(rmse_scores)
    avg_mae = np.mean(mae_scores)
    avg_r2 = np.mean(r2_scores)
    
    print(f"\nAverage: RMSE = {avg_rmse:.4f}, MAE = {avg_mae:.4f}, R² = {avg_r2:.4f}")
    
    return {
        'rmse': rmse_scores,
        'mae': mae_scores,
        'r2': r2_scores,
        'avg_rmse': avg_rmse,
        'avg_mae': avg_mae,
        'avg_r2': avg_r2
    }

def expanding_window_cross_validation(X, y, model, initial_window=0.5, step=0.1):

    list
```

```python
        # Store results
        model_results[name] = {

    if hasattr(best_model, 'feature_importances_'):

    print("\nStep 6: Future Predictions")
```

- Account for uncertainty in your forecasts, especially as the prediction horizon extends further into the future
