    def _calculate_skill_match(self, user_skills: Dict[str, int], required_skills: List[str]) -> float:
        """Calculate percentage match for skills"""
        if not required_skills:
            return 0.0
            
        total_skills = len(required_skills)
        matched_skills = 0
        skill_quality = 0
        
        for skill in required_skills:
            # Flexible matching (case insensitive, partial matches)
            for user_skill, level in user_skills.items():
                if skill.lower() in user_skill.lower() or user_skill.lower() in skill.lower():
                    matched_skills += 1
                    skill_quality += level / 10  # Convert to 0-1 scale
                    break
        
        if matched_skills == 0:
            return 0.0
            
        match_percentage = matched_skills / total_skills
        quality_factor = skill_quality / matched_skills if matched_skills > 0 else 0
        
        return (match_percentage * quality_factor) * 100

class JobMarketAnalyzer:
    """Analyze current job market trends in ML/AI"""
    
    def __init__(self):
        self.job_data = []
        self.skill_trends = {}
        self.salary_data = {}
        
    def analyze_job_postings(self, job_descriptions: List[str]) -> Dict[str, Any]:
        """Analyze job postings to extract trends"""
        
        # Extract skills using TF-IDF
        vectorizer = TfidfVectorizer(
            max_features=100,
            stop_words='english',
            ngram_range=(1, 2)
        )
        
        # Preprocess job descriptions
        processed_descriptions = [desc.lower() for desc in job_descriptions]
        tfidf_matrix = vectorizer.fit_transform(processed_descriptions)
        feature_names = vectorizer.get_feature_names_out()
        
        # Calculate skill importance
        skill_scores = tfidf_matrix.mean(axis=0).A1
        skill_importance = dict(zip(feature_names, skill_scores))
        
        # Sort by importance
        top_skills = sorted(skill_importance.items(), key=lambda x: x[1], reverse=True)[:20]
        
        # Cluster job postings
        kmeans = KMeans(n_clusters=5, random_state=42)
        job_clusters = kmeans.fit_predict(tfidf_matrix)
        
        # Analyze clusters
        cluster_analysis = self._analyze_job_clusters(job_clusters, processed_descriptions, feature_names, tfidf_matrix)
        
        return {
            "top_skills": top_skills,
            "job_clusters": cluster_analysis,
            "total_jobs_analyzed": len(job_descriptions),
            "skill_diversity": len(feature_names),
            "market_insights": self._generate_market_insights(top_skills, cluster_analysis)
        }
    
    def _analyze_job_clusters(self, clusters: np.ndarray, descriptions: List[str], 
                             features: np.ndarray, tfidf_matrix: np.ndarray) -> Dict[str, Any]:
        """Analyze job clusters to identify role types"""
        
        cluster_info = {}
        
        for cluster_id in range(max(clusters) + 1):
            cluster_mask = clusters == cluster_id
            cluster_descriptions = [desc for i, desc in enumerate(descriptions) if cluster_mask[i]]
            
            # Get top features for this cluster
            cluster_tfidf = tfidf_matrix[cluster_mask].mean(axis=0).A1
            top_features_idx = cluster_tfidf.argsort()[-10:][::-1]
            top_features = [features[i] for i in top_features_idx]
            
            # Identify cluster theme
            cluster_theme = self._identify_cluster_theme(top_features)
            
            cluster_info[f"Cluster_{cluster_id}"] = {
                "theme": cluster_theme,
                "job_count": sum(cluster_mask),
                "top_skills": top_features[:5],
                "sample_descriptions": cluster_descriptions[:2]
            }
        
        return cluster_info
    
    def _identify_cluster_theme(self, top_features: List[str]) -> str:
        """Identify the theme of a job cluster based on top features"""
        
        themes = {
            "Data Science": ["data", "science", "analytics", "statistical", "research"],
            "Machine Learning Engineering": ["ml", "model", "deployment", "production", "engineering"],
            "Data Engineering": ["pipeline", "etl", "infrastructure", "big data", "spark"],
            "AI Research": ["research", "algorithm", "paper", "phd", "novel"],
            "Business Intelligence": ["business", "dashboard", "reporting", "kpi", "stakeholder"]
        }
        
        feature_text = " ".join(top_features).lower()
        
        theme_scores = {}
        for theme, keywords in themes.items():
            score = sum(1 for keyword in keywords if keyword in feature_text)
            theme_scores[theme] = score
        
        if theme_scores:
            return max(theme_scores, key=theme_scores.get)
        else:
            return "General ML/Data Role"
    
    def _generate_market_insights(self, top_skills: List[Tuple[str, float]], 
                                 clusters: Dict[str, Any]) -> List[str]:
        """Generate actionable market insights"""
        
        insights = []
        
        # Skill demand insights
        if top_skills:
            top_skill = top_skills[0][0]
            insights.append(f"'{top_skill}' is the most in-demand skill across job postings")
        
        # Cluster insights
        largest_cluster = max(clusters.items(), key=lambda x: x[1]["job_count"])
        insights.append(f"'{largest_cluster[1]['theme']}' roles dominate the market with {largest_cluster[1]['job_count']} positions")
        
        # Technology trends
        tech_skills = [skill for skill, _ in top_skills if any(tech in skill.lower() 
                      for tech in ['python', 'sql', 'aws', 'tensorflow', 'pytorch'])]
        if tech_skills:
            insights.append(f"Key technologies in demand: {', '.join(tech_skills[:3])}")
        
        return insights
