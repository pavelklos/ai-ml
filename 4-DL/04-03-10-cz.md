## 2. Vizualizace a atribuce příznaků

Techniky vizualizace příznaků pomáhají pochopit, jaké vzory se neuronové sítě učí, a to syntézou vstupů, které maximalizují aktivace specifických neuronů, kanálů nebo vrstev.

```python
# Vizualizace příznaků pomocí TensorFlow
import tensorflow as tf
from tensorflow.keras.applications import VGG16
import numpy as np

# Načtení předtrénovaného modelu
model = VGG16(weights='imagenet', include_top=False)

# Výběr konkrétní vrstvy k vizualizaci
layer_name = 'block3_conv1'
layer = model.get_layer(name=layer_name)

# Funkce pro vizualizaci filtrů
def visualize_filter(model, layer_name, filter_index, iterations=30, size=150):
    # Vytvoření spojení mezi vstupem a cílovou vrstvou
    layer_output = model.get_layer(layer_name).output
    # Získání výstupu konkrétního filtru
    loss = tf.reduce_mean(layer_output[:, :, :, filter_index])
    
    # Výpočet gradientu vstupního obrázku vzhledem k této ztrátě
    grads = tf.gradients(loss, model.input)[0]
    
    # Normalizační trik
    grads = grads / (tf.sqrt(tf.reduce_mean(tf.square(grads))) + 1e-5)
    
    # Definice funkce pro výpočet ztráty a gradientů
    iterate = tf.function([model.input], [loss, grads])
    
    # Náhodný výchozí bod
    input_img_data = np.random.random((1, size, size, 3)) * 20 + 128.
    
    # Gradientní vzestup
    for i in range(iterations):
        loss_value, grads_value = iterate([input_img_data])
        input_img_data += grads_value * 1.
        
    # Normalizace vygenerovaného obrázku
    img = input_img_data[0]
    img = np.clip(img, 0, 255).astype(np.uint8)
    return img

# Vizualizace prvních několika filtrů z cílové vrstvy
fig, axes = plt.subplots(2, 4, figsize=(12, 6))
axes = axes.flatten()

for i, ax in enumerate(axes):
    if i < 8:  # Vizualizace 8 filtrů
        img = visualize_filter(model, layer_name, i)
        ax.imshow(img)
        ax.set_title(f'Filtr {i}')

plt.tight_layout()
plt.show()
```

Pro atribuci příznaků můžeme použít techniky jako Integrated Gradients nebo GradientShap k určení, které vstupní příznaky nejvíce přispívají k predikci modelu.

```python
# Implementace Integrated Gradients
def integrated_gradients(model, images, target_class_idx, m_steps=50, baseline=None):
    # Pokud není poskytnuta výchozí hodnota, použije se černý obrázek
    if baseline is None:
        baseline = np.zeros_like(images)
        
    # Škálování vstupů a výpočet gradientů
    scaled_inputs = [baseline + (float(i) / m_steps) * (images - baseline) for i in range(0, m_steps + 1)]
    scaled_inputs = np.array(scaled_inputs)
    
    # Získání gradientů
    with tf.GradientTape() as tape:
        scaled_inputs_tf = tf.convert_to_tensor(scaled_inputs)
        tape.watch(scaled_inputs_tf)
        predictions = model(scaled_inputs_tf)
        target_predictions = predictions[:, target_class_idx]
    
    gradients = tape.gradient(target_predictions, scaled_inputs_tf)
    
    # Průměrování gradientů aproximující Riemannův součet
    avg_gradients = np.mean(gradients, axis=0)
    integrated_gradients = (images - baseline) * avg_gradients
    
    return integrated_gradients
```

LIME je technika nezávislá na modelu, která vysvětluje jednotlivé predikce aproximací složitého modelu lokálně jednodušším, interpretovatelným modelem.

## 5. Grad-CAM a vizualizace pozornosti

Grad-CAM používá gradienty tekoucí do poslední konvoluční vrstvy k vytvoření tepelné mapy zvýrazňující důležité oblasti v obrázku.

```python
# Implementace Grad-CAM
def grad_cam(model, img_array, layer_name, class_idx):
    # Vytvoření modelu, který mapuje vstup na aktivace určené vrstvy
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [model.get_layer(layer_name).output, model.output]
    )
    
    # Výpočet gradientů vzhledem k výstupu třídy
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(np.array([img_array]))
        class_output = predictions[:, class_idx]
    
    # Extrakce gradientů
    grads = tape.gradient(class_output, conv_outputs)
    
    # Poolování gradientů
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Vážení kanálů aktivační mapy hodnotami gradientu
    conv_outputs = conv_outputs[0]
    for i in range(pooled_grads.shape[-1]):
        conv_outputs[:, :, i] *= pooled_grads[i]
    
    # Průměrování přes všechny kanály
    heatmap = tf.reduce_mean(conv_outputs, axis=-1)
    
    # Normalizace tepelné mapy
    heatmap = np.maximum(heatmap, 0) / np.max(heatmap)
    
    return heatmap

# Vizualizace Grad-CAM pro vzorový obrázek
def visualize_grad_cam(model, img_path, layer_name='block5_conv3'):
    # Předzpracování obrázku
```

## 6. Destilace modelů pro interpretovatelnost

Destilace znalostí zahrnuje trénování jednodušího, interpretovatelného modelu (žáka), aby napodoboval složitý model hlubokého učení (učitele).

```python
# Implementace destilace modelu
def distill_model(teacher_model, x_train, y_train, input_shape, num_classes,
                  temperature=5.0, alpha=0.1):
    """
    Vytvoření a trénink žákovského modelu destilací znalostí z učitelského modelu
    
    Args:
        teacher_model: Komplexní model, ze kterého se destilují znalosti
        x_train: Trénovací data
        y_train: Trénovací štítky
        input_shape: Tvar vstupních dat
        num_classes: Počet tříd
        temperature: Teplota pro změkčení pravděpodobností
        alpha: Váha pro destilační ztrátu vs. cross-entropy ztrátu
    
    Returns:
        Trénovaný žákovský model
    """
    # Vytvoření jednoduššího žákovského modelu
    student_model = keras.Sequential([
        layers.Dense(32, activation='relu', input_shape=input_shape),
        layers.Dense(num_classes)
    ])
    
    # Získání měkkých cílů z učitelského modelu
    teacher_preds = teacher_model.predict(x_train)
    
    # Vlastní ztrátová funkce kombinující destilaci a cross-entropy
    def distillation_loss(y_true, y_pred):
        # Konverze indexů tříd na one-hot vektory
        y_true_one_hot = tf.one_hot(tf.cast(y_true, tf.int32), depth=num_classes)
        
        # Standardní cross-entropy ztráta
        ce_loss = tf.keras.losses.categorical_crossentropy(
            y_true_one_hot, 
            y_pred, 
            from_logits=True
        )
        
        # Aplikace teplotního škálování
        soft_targets = tf.nn.softmax(teacher_preds / temperature)
        soft_prob = tf.nn.softmax(y_pred / temperature)
        
        # Destilační ztráta
        kl_loss = tf.keras.losses.kullback_leibler_divergence(soft_targets, soft_prob)
        kl_loss = kl_loss * (temperature ** 2)
        
        # Kombinace ztrát
        return alpha * ce_loss + (1 - alpha) * kl_loss
    
    # Kompilace žákovského modelu
    student_model.compile(
        optimizer=keras.optimizers.Adam(),
        loss=distillation_loss,
        metrics=['accuracy']
    )
    
    # Trénink žákovského modelu
    student_model.fit(
        x_train, y_train,
        epochs=10,
        batch_size=64,
        verbose=1
    )
    
    return student_model
```

## 7. Vysvětlení založená na konceptech

Metody založené na konceptech vysvětlují rozhodnutí modelu pomocí vysokoúrovňových konceptů, které jsou pro člověka srozumitelné.

```python
# Implementace testování s vektory aktivace konceptů (TCAV)
def compute_directional_derivatives(model, layer_name, concept_examples, class_id, random_examples):
    """
    Zjednodušená implementace směrových derivací pro konceptové vektory
    
    Args:
        model: Trénovaný model
        layer_name: Název vrstvy k analýze
        concept_examples: Příklady reprezentující koncept
        class_id: Cílová třída k analýze
        random_examples: Náhodné příklady pro srovnání
        
    Returns:
        Skóre citlivosti třídy na koncept
    """
    # Vytvoření modelu, který vrací aktivace cílové vrstvy
    activation_model = keras.Model(
        inputs=model.input,
        outputs=model.get_layer(layer_name).output
    )
    
    # Získání aktivací pro příklady konceptu
    concept_activations = activation_model.predict(concept_examples)
    
    # Získání aktivací pro náhodné příklady
    random_activations = activation_model.predict(random_examples)
    
    # Výpočet vektoru aktivace konceptu (CAV)
    from sklearn.linear_model import SGDClassifier
    
    # Příprava dat pro klasifikátor
    X = np.vstack([concept_activations, random_activations])
    y = np.array([1] * len(concept_activations) + [0] * len(random_activations))
    
    # Trénink lineárního klasifikátoru
    clf = SGDClassifier(alpha=0.01, max_iter=1000, tol=1e-3)
    clf.fit(X.reshape(X.shape[0], -1), y)
    
    # Získání CAV (normála k rozhodovací hranici)
    cav = clf.coef_[0]
    
    # Definice funkce pro výpočet gradientů
    @tf.function
    def compute_gradients(inputs):
        with tf.GradientTape() as tape:
            # Dopředný průchod k cílové vrstvě
            layer_output = activation_model(inputs)
            
            # Přetvarování aktivací
            flat_activations = tf.reshape(layer_output, (tf.shape(layer_output)[0], -1))
            
            # Projekce gradientů na směr CAV
            projection = tf.reduce_sum(flat_activations * tf.convert_to_tensor(cav, dtype=tf.float32), axis=1)
            
            # Získání gradientů projekce vzhledem k predikci třídy
            class_output = model(inputs)[:, class_id]
        
        return tape.gradient(class_output, projection)
    
    # Výpočet citlivosti pomocí několika testovacích příkladů
    test_examples = concept_examples[:5]  # Použití několika příkladů konceptů jako test
    sensitivity = compute_gradients(test_examples).numpy()
    
    # Vrácení průměrné citlivosti
    return np.mean(sensitivity)
```

## 8. Kontrafaktuální vysvětlení

Kontrafaktuální vysvětlení ukazují, jak je třeba změnit vstupy, aby se dosáhlo jiného výsledku, což pomáhá porozumět rozhodovacím hranicím modelu.

```python
# Generování kontrafaktuálního vysvětlení
def generate_counterfactual(model, image, target_class, iterations=1000, lr=0.1):
    """
    Generování kontrafaktuálního vysvětlení pro obrázek
    
    Args:
        model: Trénovaný model
        image: Původní obrázek k modifikaci
        target_class: Cílová třída pro kontrafaktuál
        iterations: Počet optimalizačních iterací
        lr: Rychlost učení
        
    Returns:
        Kontrafaktuální obrázek
    """
    # Vytvoření proměnné k optimalizaci
    counterfactual = tf.Variable(image, dtype=tf.float32)
    
    # Definice ztrátové funkce
    def loss_fn():
        # Predikční ztráta (pro získání cílové třídy)
        pred = model(tf.expand_dims(counterfactual, 0))
        pred_loss = -tf.keras.losses.sparse_categorical_crossentropy(
            tf.constant([target_class]), pred, from_logits=True
        )
        
        # Ztráta vzdálenosti (aby kontrafaktuál zůstal blízko původnímu)
        dist_loss = tf.reduce_sum(tf.square(counterfactual - image))
        
        # Celková ztráta (vyvážení predikce a vzdálenosti)
        return pred_loss + 0.01 * dist_loss
    
    # Optimalizátor
    optimizer = tf.keras.optimizers.Adam(learning_rate=lr)
    
    # Optimalizační smyčka
    for i in range(iterations):
        optimizer.minimize(loss_fn, var_list=[counterfactual])
        
        # Oříznutí hodnot na platný rozsah
        counterfactual.assign(tf.clip_by_value(counterfactual, 0, 255))
        
        # Výpis postupu
        if i % 100 == 0:
            current_pred = model(tf.expand_dims(counterfactual, 0))
            current_class = tf.argmax(current_pred[0]).numpy()
            print(f"Iterace {i}, Aktuální predikce: {current_class}")
            
            if current_class == target_class:
                print("Cílová třída dosažena!")
                break
    
    return counterfactual.numpy()
```

## 9. Hodnocení kvality vysvětlení

Hodnocení kvality vysvětlení je zásadní, ale náročné. Zde jsou metody pro hodnocení kvality vysvětlení:

```python
# Hodnocení věrnosti pro metody atribuce příznaků
def evaluate_faithfulness(model, image, explanation, n_perturb=100):
    """
    Vyhodnocení, jak věrně vysvětlení odpovídá chování modelu
    měřením korelace mezi důležitostí příznaků a dopadem na predikci.
    
    Args:
        model: Trénovaný model
        image: Obrázek k vysvětlení
        explanation: Mapa atribuce příznaků (stejné velikosti jako obrázek)
        n_perturb: Počet perturbačních vzorků
        
    Returns:
        Skóre věrnosti (vyšší je lepší)
    """
    # Původní predikce
    orig_pred = model.predict(np.expand_dims(image, 0))[0]
    orig_class = np.argmax(orig_pred)
    orig_prob = orig_pred[orig_class]
    
    # Zploštění vysvětlení a obrázku
    flat_exp = explanation.flatten()
    flat_img = image.flatten()
    
    # Seřazení pixelů podle důležitosti
    order = np.argsort(np.abs(flat_exp))[::-1]  # Sestupně podle důležitosti
    
    # Perturbace obrázku nahrazením důležitých příznaků náhodnými hodnotami
    perturbation_impact = []
    
    # Vytvoření perturbovaných verzí postupným odstraňováním důležitých pixelů
    step_size = len(flat_img) // n_perturb
    for i in range(0, len(flat_img), step_size):
        # Zkopírování obrázku
        perturbed_img = flat_img.copy()
        
        # Nahrazení top i důležitých pixelů náhodnými hodnotami
        perturbed_indices = order[:i]
        perturbed_img[perturbed_indices] = np.random.random(len(perturbed_indices)) * 255
        
        # Přetvarování a získání predikce
        reshaped_img = perturbed_img.reshape(image.shape)
        perturbed_pred = model.predict(np.expand_dims(reshaped_img, 0))[0]
        perturbed_prob = perturbed_pred[orig_class]
        
        # Zaznamenání změny pravděpodobnosti
        perturbation_impact.append(orig_prob - perturbed_prob)
    
    # Výpočet korelace mezi počtem perturbovaných pixelů a dopadem
    perturb_sizes = np.arange(0, len(flat_img), step_size)[:len(perturbation_impact)]
    correlation = np.corrcoef(perturb_sizes, perturbation_impact)[0, 1]
    
    return correlation
```

Konzistence je další důležitou metrikou:

```python
# Hodnocení konzistence napříč podobnými modely
def evaluate_consistency(models, image, generate_explanation_fn):
    """
    Vyhodnocení konzistence vysvětlení napříč více modely
    
    Args:
        models: Seznam podobných trénovaných modelů
        image: Obrázek k vysvětlení
        generate_explanation_fn: Funkce, která generuje vysvětlení
        
    Returns:
        Průměrná podobnost mezi vysvětleními
    """
    explanations = []
    
    # Generování vysvětlení z každého modelu
    for model in models:
        exp = generate_explanation_fn(model, image)
        explanations.append(exp.flatten())
    
    # Výpočet párových podobností
    similarities = []
    for i in range(len(explanations)):
        for j in range(i+1, len(explanations)):
            # Použití kosinové podobnosti
            similarity = np.dot(explanations[i], explanations[j])
            similarity /= (np.linalg.norm(explanations[i]) * np.linalg.norm(explanations[j]))
            similarities.append(similarity)
    
    # Vrácení průměrné podobnosti
    return np.mean(similarities)
```

```python
def explainability_pipeline(model, data, labels, explanation_methods=None):
    """
    Komplexní XAI pipeline, který aplikuje více technik vysvětlitelnosti
    
    Args:
        model: Trénovaný model k vysvětlení
        data: Vstupní data pro generování vysvětlení
        labels: Pravdivé štítky pro data
        explanation_methods: Seznam metod vysvětlení k použití
        
    Returns:
        Slovník vysvětlení podle metody
    """
    if explanation_methods is None:
        explanation_methods = ['gradcam', 'lime', 'shap']
    
    explanations = {}
    
    # Aplikace vybraných metod vysvětlitelnosti
    for method in explanation_methods:
        if method.lower() == 'gradcam' and isinstance(data, np.ndarray) and len(data.shape) == 4:
            # Pro obrázky použijte GradCAM
            layer_name = None
            # Nalezení poslední konvoluční vrstvy
            for layer in reversed(model.layers):
                if 'conv' in layer.name:
                    layer_name = layer.name
                    break
                    
            if layer_name:
                explanations['gradcam'] = []
                for i, img in enumerate(data[:10]):  # Omezení na prvních 10 obrázků pro efektivitu
                    pred_class = np.argmax(model.predict(np.expand_dims(img, 0)))
                    heatmap = grad_cam(model, img, layer_name, pred_class)
                    explanations['gradcam'].append(heatmap)
        
        elif method.lower() == 'lime':
            # Inicializace vysvětlovače LIME podle typu dat
            if len(data.shape) == 4:  # Obrazová data
                try:
                    from lime import lime_image
                    explainer = lime_image.LimeImageExplainer()
                    explanations['lime'] = []
                    for i, img in enumerate(data[:5]):  # Omezení pro efektivitu
                        exp = explainer.explain_instance(
                            img/255.0,  # Normalizace
                            lambda x: model.predict(x),
                            top_labels=3
                        )
                        explanations['lime'].append(exp)
                except ImportError:
                    print("LIME není nainstalován nebo není kompatibilní s těmito daty")
            elif len(data.shape) == 2:  # Tabulární data
                try:
                    from lime import lime_tabular
                    explainer = lime_tabular.LimeTabularExplainer(
                        data,
                        mode="classification"
                    )
                    explanations['lime'] = []
                    for i, instance in enumerate(data[:20]):
                        exp = explainer.explain_instance(
                            instance, 
                            model.predict
                        )
                        explanations['lime'].append(exp)
                except ImportError:
                    print("LIME není nainstalován nebo není kompatibilní s těmito daty")
        
        elif method.lower() == 'shap':
            try:
                import shap
                # Výběr vysvětlovače podle typu modelu
                if isinstance(model, tf.keras.Model):
                    background = data[:100]  # Použití podmnožiny pro pozadí
                    explainer = shap.DeepExplainer(model, background)
                    shap_values = explainer.shap_values(data[:20])  # Omezení pro efektivitu
                    explanations['shap'] = shap_values
            except ImportError:
                print("SHAP není nainstalován nebo není kompatibilní s tímto modelem")
    
    return explanations
```
