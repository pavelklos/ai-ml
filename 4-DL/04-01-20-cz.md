# Etika a omezení hlubokého učení

## 1. Úvod

Hluboké učení způsobilo revoluci v mnoha oblastech díky působivým schopnostem v rozpoznávání obrazu, zpracování přirozeného jazyka a rozhodování. Tyto výkonné technologie však přinášejí důležité etické otázky a technická omezení, které je třeba řešit.

Tato příručka zkoumá klíčové etické výzvy a omezení systémů hlubokého učení spolu s praktickými přístupy k zmírnění těchto problémů.

## 3. Obavy o soukromí a bezpečnost

### 3.1 Útoky inverzí modelu a odvozování členství

Modely hlubokého učení mohou nechtěně zapamatovat trénovací data, což může potenciálně odhalit citlivé informace:

```python
# Příklad útoku odvozování členství
import tensorflow as tf
import numpy as np

from sklearn.metrics import accuracy_score

def create_attack_model():
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(64, activation='relu', input_shape=(num_classes,)),
        tf.keras.layers.Dense(32, activation='relu'),
        tf.keras.layers.Dense(2, activation='softmax')  # Binární: byl v trénovacích datech nebo nebyl
    ])
    model.compile(optimizer='adam', loss='categorical_crossentropy', metrics=['accuracy'])
    return model

# Trénink útočného modelu na výstupy stínových modelů
def train_attack_model(shadow_train_outputs, shadow_test_outputs):
    # Výstupy stínového tréninku jsou z dat použitých k trénování stínových modelů
    shadow_train_labels = np.ones(len(shadow_train_outputs))  # Značka 1 = byl v trénovacích datech
    
    # Výstupy stínového testu jsou z dat nepoužitých ve stínových modelech
    shadow_test_labels = np.zeros(len(shadow_test_outputs))  # Značka 0 = nebyl v trénovacích datech
    
    # Kombinace dat
    attack_input = np.vstack((shadow_train_outputs, shadow_test_outputs))
    attack_labels = np.hstack((shadow_train_labels, shadow_test_labels))
    
    # One-hot kódování značek
    attack_labels = tf.keras.utils.to_categorical(attack_labels, 2)
    
    # Trénink útočného modelu
    attack_model = create_attack_model()
    attack_model.fit(attack_input, attack_labels, epochs=10, batch_size=32, verbose=0)
    
    return attack_model
```

```python
# Vytvoření modelu s diferenciálním soukromím
def create_private_model(l2_norm_clip=1.0, noise_multiplier=0.5):
    # Použití DP-SGD optimalizátoru
    optimizer = tfp.DPKerasSGDOptimizer(

        noise_multiplier=noise_multiplier,
        num_microbatches=1,
        learning_rate=0.01
    )
    
    # Vytvoření modelu
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(64, activation='relu', input_shape=(input_dim,)),
        tf.keras.layers.Dense(32, activation='relu'),
        tf.keras.layers.Dense(num_classes, activation='softmax')
    ])
    
    # Kompilace modelu
    model.compile(
        optimizer=optimizer,
        loss=tf.keras.losses.CategoricalCrossentropy(
            from_logits=True, reduction=tf.losses.Reduction.NONE
        ),
        metrics=['accuracy']
    )
    
    return model

# Výpočet rozpočtu soukromí
def compute_epsilon(steps, noise_multiplier, batch_size, num_examples):
    from tensorflow_privacy.privacy.analysis.rdp_accountant import compute_rdp
    from tensorflow_privacy.privacy.analysis.rdp_accountant import get_privacy_spent
    
    # Výpočet míry vzorkování
    sampling_probability = batch_size / num_examples
    
    # Výpočet rozpočtu soukromí
    orders = [1 + x / 10.0 for x in range(1, 100)] + list(range(12, 64))
    rdp = compute_rdp(
        q=sampling_probability,
        noise_multiplier=noise_multiplier,
        steps=steps,
        orders=orders
    )
    
    # Převod na epsilon
    delta = 1e-5  # Typická hodnota pro velké datasety
    epsilon = get_privacy_spent(orders, rdp, target_delta=delta)[0]
    
    return epsilon

# Trénink privátního modelu
private_model = create_private_model(l2_norm_clip=1.0, noise_multiplier=1.0)
steps_per_epoch = len(X_train) // batch_size
private_model.fit(X_train, y_train, epochs=10, batch_size=batch_size)

# Výpočet spotřebovaného rozpočtu soukromí
epsilon = compute_epsilon(
    steps=steps_per_epoch * 10,  # 10 epoch
    noise_multiplier=1.0,
    batch_size=batch_size,
    num_examples=len(X_train)
)

print(f"Spotřebovaný rozpočet soukromí: epsilon = {epsilon}")
```

## 4. Environmentální dopady

Modely hlubokého učení, zejména ty velké, mohou mít významnou ekologickou stopu kvůli jejich výpočetním nárokům:

```python
import time
import psutil
import numpy as np
import tensorflow as tf
from tensorflow.keras.applications import ResNet50, MobileNetV2

def measure_power_consumption(model, input_data, iterations=100):
    # Sledování metrik spotřeby energie
    start_time = time.time()
    start_cpu_percent = psutil.cpu_percent()
    
    # Spuštění predikcí vícekrát
    for _ in range(iterations):
        _ = model.predict(input_data)
    
    # Výpočet metrik
    elapsed_time = time.time() - start_time
    average_cpu_percent = psutil.cpu_percent()
    
    # Odhad spotřeby energie (toto je zjednodušená aproximace)
    # Přesnější měření by vyžadovalo specializovaný hardware
    system_power_draw = 0.1  # Přibližný odběr energie v kW (upravte podle vašeho systému)
    energy_consumed = system_power_draw * elapsed_time / 3600  # kWh
    
    return {
        "inference_time": elapsed_time / iterations,
        "cpu_utilization": average_cpu_percent,
        "total_energy": energy_consumed,
        "carbon_estimate": energy_consumed * 0.5  # Hrubý odhad CO2 v kg (závisí na vaší elektrické síti)
    }

# Porovnání environmentálního dopadu různých architektur modelů
input_data = np.random.random((100, 224, 224, 3))

# Velký model
large_model = ResNet50(weights=None)
large_metrics = measure_power_consumption(large_model, input_data)

# Menší, efektivnější model
efficient_model = MobileNetV2(weights=None)
efficient_metrics = measure_power_consumption(efficient_model, input_data)

print("Metriky velkého modelu:")
print(large_metrics)
print("\nMetriky efektivního modelu:")
print(efficient_metrics)

print(f"\nÚspora energie: {(1 - efficient_metrics['total_energy'] / large_metrics['total_energy']) * 100:.2f}%")
print(f"Snížení emisí uhlíku: {(large_metrics['carbon_estimate'] - efficient_metrics['carbon_estimate']) * 1000:.2f} g CO2")
```

```python
# Implementace Grad-CAM pro vizualizaci
def make_gradcam_heatmap(img_array, model, last_conv_layer_name, pred_index=None):
    # Vytvoření modelu, který mapuje vstupní obrázek na:
    # 1) Aktivace poslední konvoluční vrstvy
    # 2) Konečné predikce tříd
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [model.get_layer(last_conv_layer_name).output, model.output]
    )

    # Výpočet gradientu nejlépe predikované třídy pro vstupní obrázek
    with tf.GradientTape() as tape:
        last_conv_layer_output, preds = grad_model(img_array)
        if pred_index is None:
            pred_index = tf.argmax(preds[0])
        class_channel = preds[:, pred_index]

    # Gradient výstupu vzhledem k aktivacím poslední konvoluční vrstvy
    grads = tape.gradient(class_channel, last_conv_layer_output)

    # Vektor průměrné intenzity gradientu přes kanál
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))

    # Váhování kanálů podle významnosti gradientu
    last_conv_layer_output = last_conv_layer_output[0]
    heatmap = tf.reduce_sum(tf.multiply(pooled_grads, last_conv_layer_output), axis=-1)
    heatmap = tf.squeeze(heatmap)

    # Normalizace teplotní mapy
    heatmap = tf.maximum(heatmap, 0) / tf.reduce_max(heatmap)
    
    return heatmap.numpy()

# Generování aktivační mapy třídy
last_conv_layer = 'conv5_block3_out'  # Poslední konvoluční vrstva v ResNet50
heatmap = make_gradcam_heatmap(x, model, last_conv_layer)

# Zobrazení teplotní mapy nad obrázkem
def save_and_display_gradcam(img_path, heatmap, alpha=0.4):
    # Načtení původního obrázku
    img = image.load_img(img_path)
    img = image.img_to_array(img)

    # Změna velikosti teplotní mapy na velikost původního obrázku
    import cv2
    heatmap = cv2.resize(heatmap, (img.shape[1], img.shape[0]))
    
    # Převod teplotní mapy na RGB
    heatmap = np.uint8(255 * heatmap)
    heatmap = cm.jet(heatmap)[..., :3] * 255
    
    # Překrytí teplotní mapy na obrázek
    superimposed_img = heatmap * alpha + img
    superimposed_img = np.clip(superimposed_img, 0, 255).astype("uint8")
    
    # Zobrazení obrázků
    fig, (ax1, ax2, ax3) = plt.subplots(1, 3, figsize=(15, 5))
    ax1.imshow(img.astype('uint8'))
    ax1.set_title('Původní obrázek')
    ax1.axis('off')
    
    ax2.imshow(cv2.cvtColor(heatmap.astype('uint8'), cv2.COLOR_BGR2RGB))
    ax2.set_title('Teplotní mapa Grad-CAM')
    ax2.axis('off')
    
    ax3.imshow(cv2.cvtColor(superimposed_img, cv2.COLOR_BGR2RGB))
    ax3.set_title('Překrytí')
    ax3.axis('off')
    
    plt.tight_layout()
    plt.show()
    
    return heatmap, superimposed_img

save_and_display_gradcam(img_path, heatmap)
```

### 6.1 Adversariální příklady

Modely hlubokého učení mohou být oklamány malými, nezjistitelnými perturbacemi vstupů:

```python

# Vytvoření adversariálního příkladu pomocí metody Fast Gradient Sign Method (FGSM)
def create_adversarial_example(model, image, label, epsilon=0.01):
    image_tensor = tf.convert_to_tensor(image, dtype=tf.float32)
    
    with tf.GradientTape() as tape:
        tape.watch(image_tensor)
        prediction = model(image_tensor)
        loss = tf.keras.losses.sparse_categorical_crossentropy(label, prediction)
    
    # Získání gradientu ztráty vzhledem ke vstupnímu obrázku
    gradient = tape.gradient(loss, image_tensor)
    
    # Vytvoření perturbace: epsilon * sign(gradient)
    perturbation = epsilon * tf.sign(gradient)
    
    # Vytvoření adversariálního příkladu
    adversarial_example = image_tensor + perturbation
    
    # Ujistěte se, že zůstáváme v platném rozsahu obrázku [0, 1]
    adversarial_example = tf.clip_by_value(adversarial_example, -1, 1)
    
    return adversarial_example.numpy()

# Cílení na jinou třídu (např. třída 910: 'psací stroj')
target_class = 910

# Vytvoření adversariálního příkladu
adv_x = create_adversarial_example(model, x, target_class, epsilon=0.1)

# Získání predikce na adversariálním příkladu
```

### 6.2 Detekce vzorků mimo distribuci

Modely hlubokého učení mohou dělat sebevědomé predikce na zcela irelevantní vstupy:

```python
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.preprocessing import image
from tensorflow.keras.applications.resnet50 import preprocess_input, decode_predictions
from sklearn.metrics import roc_curve, auc

# Načtení předtrénovaného modelu
model = ResNet50(weights='imagenet')

# Funkce pro získání predikce modelu a jistoty
def predict_image(model, img_path):
    img = image.load_img(img_path, target_size=(224, 224))
    x = image.img_to_array(img)
    x = np.expand_dims(x, axis=0)
    x = preprocess_input(x)
    
    preds = model.predict(x)
    confidence = np.max(preds[0])
    label = decode_predictions(preds, top=1)[0][0][1]
    
    return {
        'label': label,
        'confidence': confidence,
        'softmax': preds[0]
    }

# Funkce pro detekci vzorků mimo distribuci pomocí softmax pravděpodobností
def detect_ood(in_distribution_softmax, test_softmax, method='entropy'):
    if method == 'max_prob':
        # Použití záporné maximální pravděpodobnosti jako skóre anomálií
        in_dist_scores = -np.max(in_distribution_softmax, axis=1)
        test_scores = -np.max(test_softmax, axis=1)
    elif method == 'entropy':
        # Použití entropie jako skóre anomálií
        in_dist_scores = -np.sum(in_distribution_softmax * np.log(in_distribution_softmax + 1e-10), axis=1)
        test_scores = -np.sum(test_softmax * np.log(test_softmax + 1e-10), axis=1)
    
    # Výpočet ROC křivky a AUC
    # Kombinace skóre a vytvoření štítků (0 pro in-distribution, 1 pro OOD)
    all_scores = np.concatenate([in_dist_scores, test_scores])
    all_labels = np.concatenate([np.zeros(len(in_dist_scores)), np.ones(len(test_scores))])
    
    fpr, tpr, thresholds = roc_curve(all_labels, all_scores)
    roc_auc = auc(fpr, tpr)
    
    return {
        'in_dist_scores': in_dist_scores,
        'test_scores': test_scores,
        'fpr': fpr,
        'tpr': tpr,
        'auc': roc_auc,
        'thresholds': thresholds
    }

# Příklad použití:
# 1. Sbírejte predikce na in-distribution datech (ImageNet validace)
# 2. Sbírejte predikce na out-of-distribution datech (např. SVHN, náhodný šum)
# 3. Porovnejte metody detekce

# Pro demonstraci budeme simulovat s náhodnými daty:
# In-distribution softmax hodnoty (typicky sebevědomější)
in_dist_softmax = np.random.dirichlet(np.ones(1000) * 0.1, size=100)

# Out-of-distribution softmax hodnoty (typicky rovnoměrnější)
ood_softmax = np.random.dirichlet(np.ones(1000) * 1.0, size=100)

# Detekce OOD vzorků pomocí metody entropie
results = detect_ood(in_dist_softmax, ood_softmax, method='entropy')

# Vykreslení výsledků
plt.figure(figsize=(10, 8))
```

### 7.2 Metriky spravedlivosti a monitorování

Průběžné sledování spravedlivosti modelu v produkci:

```python
import pandas as pd
import numpy as np
from sklearn.metrics import confusion_matrix, classification_report
import matplotlib.pyplot as plt
import seaborn as sns

class FairnessMonitor:
    """Nástroj pro monitorování metrik spravedlivosti v klasifikačních modelech."""
    
    def __init__(self):
        self.metrics_history = {}
        
    def add_evaluation_run(self, run_id, y_true, y_pred, protected_attributes, threshold=0.5):
        """
        Přidání nového evaluačního běhu a výpočet metrik spravedlivosti.
        
        Args:
            run_id (str): Identifikátor pro tento evaluační běh
            y_true (array): Skutečné značky
            y_pred (array): Predikované pravděpodobnosti nebo značky
            protected_attributes (dict): Slovník mapující názvy atributů na hodnoty pro každý vzorek
            threshold (float): Klasifikační práh pro predikované pravděpodobnosti
        """
        if isinstance(y_pred[0], float):
            # Převod pravděpodobností na binární predikce pomocí prahu
            y_pred_binary = (np.array(y_pred) >= threshold).astype(int)
        else:
            y_pred_binary = y_pred
            
        metrics = {}
        
        # Výpočet celkových metrik
        tn, fp, fn, tp = confusion_matrix(y_true, y_pred_binary).ravel()
        overall_precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        overall_recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        overall_fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
        
        metrics['overall'] = {
            'precision': overall_precision,
            'recall': overall_recall,
            'false_positive_rate': overall_fpr,
            'samples': len(y_true)
        }
        
        # Výpočet metrik pro každou skupinu chráněného atributu
        for attr_name, attr_values in protected_attributes.items():
            metrics[attr_name] = {}
            unique_values = np.unique(attr_values)
            
            for value in unique_values:
                # Výběr vzorků pro tuto skupinu
                mask = np.array(attr_values) == value
                if sum(mask) == 0:
                    continue
                    
                group_true = np.array(y_true)[mask]
                group_pred = np.array(y_pred_binary)[mask]
                
                # Výpočet metrik pro tuto skupinu
                tn, fp, fn, tp = confusion_matrix(group_true, group_pred).ravel()
                precision = tp / (tp + fp) if (tp + fp) > 0 else 0
                recall = tp / (tp + fn) if (tp + fn) > 0 else 0
                fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
                
                metrics[attr_name][value] = {
                    'precision': precision,
                    'recall': recall,
                    'false_positive_rate': fpr,
                    'samples': sum(mask)
                }
        
        # Uložení výsledků
        self.metrics_history[run_id] = metrics
        
        return metrics
    
    def get_disparity_metrics(self, run_id, attr_name):
        """
        Výpočet metrik disparity pro chráněný atribut.
        
        Args:
            run_id (str): Identifikátor pro evaluační běh
            attr_name (str): Název chráněného atributu
        
        Returns:
            dict: Metriky disparity pro atribut
        """
        if run_id not in self.metrics_history:
            raise ValueError(f"Run ID {run_id} nebyl nalezen")
            
        if attr_name not in self.metrics_history[run_id]:
            raise ValueError(f"Atribut {attr_name} nebyl nalezen v běhu {run_id}")
        
        metrics = self.metrics_history[run_id][attr_name]
        overall = self.metrics_history[run_id]['overall']
        
        # Výpočet disparit mezi skupinami
        disparities = {}
        for metric_name in ['precision', 'recall', 'false_positive_rate']:
            values = [group_metrics[metric_name] for group_metrics in metrics.values() if group_metrics['samples'] > 0]
            
            if values:
                min_val = min(values)
                max_val = max(values)
                
                disparities[f'{metric_name}_disparity_ratio'] = min_val / max_val if max_val > 0 else 1
                disparities[f'{metric_name}_disparity_difference'] = max_val - min_val
        
        return disparities
    
    def plot_metric_comparison(self, run_id, attr_name, metric_name='precision'):
        """
        Vykreslení porovnání metriky napříč různými skupinami.
        
        Args:
            run_id (str): Identifikátor pro evaluační běh
            attr_name (str): Název chráněného atributu
            metric_name (str): Název metriky pro vykreslení
        """
        if run_id not in self.metrics_history:
            raise ValueError(f"Run ID {run_id} nebyl nalezen")
            
        if attr_name not in self.metrics_history[run_id]:
            raise ValueError(f"Atribut {attr_name} nebyl nalezen v běhu {run_id}")
        
        metrics = self.metrics_history[run_id][attr_name]
        overall = self.metrics_history[run_id]['overall'][metric_name]
        
        # Extrakce hodnot pro vykreslení
        groups = list(metrics.keys())
        values = [metrics[group][metric_name] for group in groups]
        samples = [metrics[group]['samples'] for group in groups]
        
        # Vytvoření sloupcového grafu
        plt.figure(figsize=(10, 6))
        bars = plt.bar(groups, values, alpha=0.7)
        
        # Přidání celkové linky
        plt.axhline(y=overall, color='r', linestyle='--', label=f'Celkový {metric_name}: {overall:.3f}')
        
        # Přidání popisků a titulku
        plt.xlabel('Skupina')
        plt.ylabel(metric_name.capitalize())
        plt.title(f'{metric_name.capitalize()} podle {attr_name}')
        
        # Přidání velikostí vzorků nad sloupce
        for i, bar in enumerate(bars):
            plt.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.01, 
                    f'n={samples[i]}', ha='center')
        
        plt.legend()
        plt.tight_layout()
        plt.show()
        
        # Výpis metrik disparity
        disparities = self.get_disparity_metrics(run_id, attr_name)
        ratio = disparities.get(f'{metric_name}_disparity_ratio', 'N/A')
        difference = disparities.get(f'{metric_name}_disparity_difference', 'N/A')
        
        print(f"Poměr disparity {metric_name}: {ratio}")
        print(f"Rozdíl disparity {metric_name}: {difference}")

# Příklad použití
monitor = FairnessMonitor()

# Simulace evaluačních dat
np.random.seed(42)
y_true = np.random.binomial(1, 0.3, 1000)

# Vytvoření simulovaných predikcí se zkreslením
gender = np.random.choice(['muž', 'žena'], 1000)
age_group = np.random.choice(['18-25', '26-40', '41-65', '65+'], 1000)

# Simulace zkreslených predikcí (nižší úplnost pro ženy a starší lidi)
base_prob = 0.7
gender_bias = {'muž': 0.1, 'žena': -0.1}
age_bias = {'18-25': 0.1, '26-40': 0.05, '41-65': -0.05, '65+': -0.2}

y_pred_prob = np.zeros(1000)
for i in range(1000):
    if y_true[i] == 1:
        # Pravděpodobnost true positive se liší podle skupiny
        prob = base_prob + gender_bias[gender[i]] + age_bias[age_group[i]]
        y_pred_prob[i] = np.random.binomial(1, prob)
    else:
        # Pravděpodobnost false positive
        prob = 0.1  # Základní míra false positive
        y_pred_prob[i] = np.random.binomial(1, prob)

# Přidání evaluačního běhu
metrics = monitor.add_evaluation_run(
    run_id='model_v1', 
    y_true=y_true, 
    y_pred=y_pred_prob, 
    protected_attributes={'gender': gender, 'age_group': age_group}
)

# Vykreslení porovnání
monitor.plot_metric_comparison('model_v1', 'gender', 'recall')
monitor.plot_metric_comparison('model_v1', 'age_group', 'recall')
```
