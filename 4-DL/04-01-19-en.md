# Visualizing Neural Networks

## 1. Introduction to Neural Network Visualization

Neural network visualization is essential for understanding, debugging, and improving deep learning models. It transforms abstract mathematical constructs into intuitive visual representations that help researchers and practitioners gain insights into how models learn and make decisions.

Key benefits of neural network visualization:
- **Interpretability**: Reveals what features the network learns
- **Debugging**: Helps identify issues in model architecture or training
- **Communication**: Makes complex models accessible to non-experts
- **Trust**: Builds confidence in model decisions through transparency

```python
        super(SimpleCNN, self).__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3)
        self.pool = nn.MaxPool2d(kernel_size=2)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3)

        x = self.softmax(self.fc2(x))
        return x
```

```python
import matplotlib.pyplot as plt
import numpy as np

def plot_conv_weights(model, layer_index, max_filters=64):
    """Plot convolutional filters from the specified layer"""
    # Get layer weights
    filters = model.layers[layer_index].get_weights()[0]
    
    # Normalize filter values to 0-1
    f_min, f_max = filters.min(), filters.max()
    filters = (filters - f_min) / (f_max - f_min)
    
    # Get the number of filters
    n_filters = min(max_filters, filters.shape[3])
    
    # Plot filters
    fig = plt.figure(figsize=(12, 8))
    for i in range(n_filters):
        ax = fig.add_subplot(n_filters // 8 + 1, 8, i + 1)
        
        # For RGB filters
        if filters.shape[2] == 3:
            # Convert the 3-channel filter to RGB
            f_img = filters[:, :, :, i]
            ax.imshow(f_img)
        else:
            # For single channel filters
            f_img = filters[:, :, 0, i]
            ax.imshow(f_img, cmap='gray')
            
        ax.set_xticks([])
        ax.set_yticks([])
    
    plt.tight_layout()
    plt.show()
```

```python
def visualize_activations(model, img, layer_names=None):
    """Visualize activations of intermediate layers for a single image"""
    from tensorflow.keras.models import Model
    
    # If no layer names provided, get all layers with output shape
    if layer_names is None:
        layer_names = [layer.name for layer in model.layers if len(layer.output_shape) > 1]
    
    # Create models to extract intermediate layer activations
    activation_models = [Model(inputs=model.input, outputs=model.get_layer(name).output)
                        for name in layer_names]
    
    # Get activations
    activations = [activation_model.predict(np.expand_dims(img, axis=0))[0]
                  for activation_model in activation_models]
    
    # Plot activations
    for layer_name, activation in zip(layer_names, activations):
        # For convolutional layers
        if len(activation.shape) == 3:
            n_features = min(64, activation.shape[2])  # Number of features to plot
```

### 4.2 Maximizing Layer Activations

```python
def generate_pattern(model, layer_name, filter_index, iterations=30, step=1.0):
    """Generate a pattern that maximizes the activation of a specific filter in a layer"""
    # Build a loss function that maximizes the activation of a filter
    layer_output = model.get_layer(layer_name).output
    loss = tf.reduce_mean(layer_output[:, :, :, filter_index])
    
    # Compute gradients with respect to the input
    grads = tf.gradients(loss, model.input)[0]
    # Normalize the gradients
    grads = grads / (tf.sqrt(tf.reduce_mean(tf.square(grads))) + 1e-5)
    
    # Create a function to compute loss and gradients
    iterate = tf.keras.backend.function([model.input], [loss, grads])
    
    # Start from a gray image with some noise
    input_img_data = np.random.random((1, 28, 28, 1)) * 0.2 + 0.5
    
    # Run gradient ascent
    for i in range(iterations):
        loss_value, grads_value = iterate([input_img_data])
        input_img_data += grads_value * step
        
        # Clip to valid image values
        input_img_data = np.clip(input_img_data, 0, 1)
    
    return input_img_data[0]
```

## 5. Gradient and Attention Visualization

### 5.1 Grad-CAM Visualization

Grad-CAM (Gradient-weighted Class Activation Mapping) helps visualize what regions in an image are important for classification.

```python
def grad_cam(model, img, class_idx, layer_name=None):
    """
    Generate Grad-CAM visualization for a specific class prediction
    
    Args:
        model: Trained keras model
        img: Input image (preprocessed for the model)
        class_idx: Index of the class to generate Grad-CAM for
        layer_name: Name of the last convolutional layer (if None, will try to find it)
    """
    # If no layer specified, try to find the last conv layer
    if layer_name is None:
        for layer in reversed(model.layers):
            if isinstance(layer, tf.keras.layers.Conv2D):
                layer_name = layer.name
                print(f"Using layer: {layer_name}")
                break
    
    # Ensure the image has batch dimension
    if len(img.shape) == 3:
        img = np.expand_dims(img, axis=0)
    
    # Create a model that maps the input image to:
    # 1) The activations of the last conv layer
    # 2) The final class predictions
    grad_model = tf.keras.models.Model(
        inputs=[model.inputs],
        outputs=[model.get_layer(layer_name).output, model.output]
    )
    
    # Compute gradients
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img)
        class_output = predictions[:, class_idx]
    
    # Extract gradients for the class with respect to the feature map
    grads = tape.gradient(class_output, conv_outputs)
    
    # Global average pooling of gradients
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Weight feature maps by importance
    conv_outputs = conv_outputs[0]
    heatmap = tf.reduce_sum(tf.multiply(pooled_grads, conv_outputs), axis=-1)
    
    # Normalize heatmap
    heatmap = tf.maximum(heatmap, 0) / tf.reduce_max(heatmap)
    heatmap = heatmap.numpy()
    
    # Resize heatmap to original image size
    img_shape = img[0].shape[:2]  # Get original image dimensions
    import cv2
    heatmap = cv2.resize(heatmap, (img_shape[1], img_shape[0]))
    
    # Convert heatmap to RGB and apply colormap
    heatmap = np.uint8(255 * heatmap)
    heatmap = cv2.applyColorMap(heatmap, cv2.COLORMAP_JET)
    
    # Combine original image and heatmap
    original_img = img[0] * 255  # Assuming values are 0-1
    original_img = original_img.astype(np.uint8)
    
    # If grayscale, convert to RGB
    if len(original_img.shape) == 2 or original_img.shape[2] == 1:
        original_img = cv2.cvtColor(original_img, cv2.COLOR_GRAY2BGR)
    
    # Overlay heatmap on original image
    superimposed_img = heatmap * 0.4 + original_img
    superimposed_img = np.clip(superimposed_img, 0, 255).astype(np.uint8)
    
    # Display images
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))
    
    axes[0].imshow(original_img)
    axes[0].set_title("Original Image")
    axes[0].axis('off')
    
    axes[1].imshow(cv2.cvtColor(heatmap, cv2.COLOR_BGR2RGB))
    axes[1].set_title("Grad-CAM Heatmap")
    axes[1].axis('off')
    
    axes[2].imshow(cv2.cvtColor(superimposed_img, cv2.COLOR_BGR2RGB))
    axes[2].set_title("Grad-CAM Overlay")
    axes[2].axis('off')
    
    plt.tight_layout()
    plt.show()
    
    return heatmap, superimposed_img

# Load a test image and apply Grad-CAM
(_, _), (x_test, y_test) = mnist.load_data()
sample_img = x_test[0]
sample_img = sample_img.reshape(28, 28, 1) / 255.0

# Predict class
predictions = model.predict(np.expand_dims(sample_img, axis=0))
predicted_class = np.argmax(predictions[0])

print(f"Predicted class: {predicted_class}")

```

### 6.1 Deep Dream Algorithm

```python
def deep_dream(model, img, layer_name, iterations=20, step=0.01, octave_scale=1.4, num_octaves=3):
    """Implement a simplified version of DeepDream algorithm"""
    # Get the symbolic outputs of layer outputs
    layer_dict = dict([(layer.name, layer) for layer in model.layers])
    loss = tf.reduce_mean(layer_dict[layer_name].output)
    
    # Compute the gradients of the layer with respect to the input image
    grads = tf.gradients(loss, model.input)[0]
    
    # Normalize gradients
    grads = grads / (tf.sqrt(tf.reduce_mean(tf.square(grads))) + 1e-5)
    
    # Set up function to compute loss and gradients
    iterate = tf.keras.backend.function([model.input], [loss, grads])
    
    # Generate multi-scale versions of the image
    octave_shapes = []
    original_shape = img.shape[1:3]
    
    for i in range(num_octaves):
        octave_shape = (original_shape[0] // (octave_scale ** i),
                         original_shape[1] // (octave_scale ** i))
        octave_shapes.append(octave_shape)
    
    octaves = []
    for i in range(num_octaves):
        octave_shape = octave_shapes[i]
        octave = tf.image.resize(img, [int(octave_shape[0]), int(octave_shape[1])])
        octaves.append(octave.numpy())
    
    detail = np.zeros_like(octaves[-1])
    
    # Scale input image to the desired shape
    for octave_idx in range(num_octaves):
        octave = octaves[num_octaves - octave_idx - 1]
        
        if octave_idx > 0:
            # Upscale detail to match current octave shape
            detail = tf.image.resize(detail, [octave.shape[1], octave.shape[2]]).numpy()
            octave = octave + detail
        
        # Process this octave
        for i in range(iterations):
            loss_value, grads_value = iterate([octave])
            octave += grads_value * step
        
        # Extract detail from this octave
        detail = octave - tf.image.resize(
            octaves[num_octaves - octave_idx - 2] if octave_idx < num_octaves - 1 else img,
            [octave.shape[1], octave.shape[2]]).numpy()
    
    # Clip to valid image values
    result = np.clip(octave, 0, 1)
    return result
```

### 7.1 t-SNE for Feature Visualization

```python
def visualize_features_tsne(model, data, labels, layer_name=None, perplexity=30):
    """
    Visualize high-dimensional features from a specific layer using t-SNE
    
    Args:
        model: Trained model
        data: Input data
        labels: Labels for coloring points
        layer_name: Name of the layer to extract features from
        perplexity: t-SNE perplexity parameter
    """
    from sklearn.manifold import TSNE
    import tensorflow as tf
    
    # If no layer name specified, use the layer before the final layer
    if layer_name is None:
        layer_name = model.layers[-2].name
    
    # Create a model that outputs the specified layer
    feature_model = tf.keras.models.Model(
        inputs=model.input,
        outputs=model.get_layer(layer_name).output
    )
    
    # Extract features
    features = feature_model.predict(data)
    
    # Flatten features if necessary
    if len(features.shape) > 2:
        features = features.reshape(features.shape[0], -1)
    
    # Apply t-SNE for dimensionality reduction
    print(f"Applying t-SNE on {features.shape[1]}-dimensional data...")
    tsne = TSNE(n_components=2, perplexity=perplexity, random_state=42)
    tsne_result = tsne.fit_transform(features)
    
    # Plot results
    plt.figure(figsize=(12, 10))
    scatter = plt.scatter(tsne_result[:, 0], tsne_result[:, 1], c=labels, cmap='tab10', alpha=0.7)
    plt.colorbar(scatter, label="Class")
    plt.title(f"t-SNE visualization of features from layer '{layer_name}'")
    plt.xlabel("t-SNE feature 1")
    plt.ylabel("t-SNE feature 2")
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

# Apply t-SNE on MNIST dataset
```

```python
def setup_tensorboard(model, log_dir='logs'):
    """Set up TensorBoard callback and visualizations"""
    import os
    import datetime
    
    # Create a log directory with timestamp
    current_time = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
    log_dir = os.path.join(log_dir, current_time)
    os.makedirs(log_dir, exist_ok=True)
    
    # Create TensorBoard callback
    tensorboard_callback = tf.keras.callbacks.TensorBoard(
        log_dir=log_dir,
        histogram_freq=1,  # Save histograms of weights
        write_graph=True,  # Visualize the model graph
        write_images=True,  # Visualize any image summary
        update_freq='epoch',  # Update every epoch
        profile_batch=0  # No profiling to keep things simple
    )
    
    # Add embedding visualizations for the first layer's weights
    weights = model.layers[0].get_weights()[0]
    
    if len(weights.shape) == 4:  # For convolutional layers
        # Reshape 4D filters to 2D (n_filters, n_channels*filter_height*filter_width)
        n_filters = weights.shape[3]
        flat_weights = weights.reshape(weights.shape[0] * weights.shape[1] * weights.shape[2], n_filters).T
    else:  # For dense layers
        flat_weights = weights
    
    # Create a summary writer
    writer = tf.summary.create_file_writer(os.path.join(log_dir, 'embeddings'))
    
    # Save the embedding
    with writer.as_default():
        tf.summary.scalar("loss", 0.0, step=0)  # Dummy value
        writer.flush()
    
    # Print instructions
    print(f"TensorBoard logs will be saved to {log_dir}")
    print("Start TensorBoard with: tensorboard --logdir=logs")
    
    return tensorboard_callback
```

```python
def visualize_lime_explanation(model, image, true_label=None, class_names=None):
    """
    Visualize LIME explanation for an image classification
    
    Args:
        model: Trained model
        image: Input image
        true_label: Actual class label (optional)
        class_names: List of class names (optional)
    """
    import lime
    from lime import lime_image
    
    # Define prediction function
    def predict_fn(images):
        # Preprocess if needed
        processed = images.copy()
        if processed.max() > 1.0:  # Normalize if not already
            processed = processed / 255.0
            
        # Make prediction
        preds = model.predict(processed)
        return preds
    
    # Create explainer
    explainer = lime_image.LimeImageExplainer()
    
    # Create explanation
    explanation = explainer.explain_instance(
        image.astype('double'), 
        predict_fn,
        top_labels=5, 
        hide_color=0, 
        num_samples=1000
    )
    
    # Get prediction and explanation
    pred = model.predict(np.expand_dims(image, axis=0))[0]

        class_names = [str(i) for i in range(len(pred))]
    
    # Get title text
    title_text = f"LIME Explanation (Predicted: {class_names[predicted_class]}"

        title_text += f", Actual: {class_names[true_label]})"
    else:
        title_text += ")"
    
    # Plot the explanation
    plt.figure(figsize=(12, 6))
```

```python
def create_interactive_activation_viewer(model, sample_images, layer_names=None):
    """
    Create an interactive activation viewer using ipywidgets
    (Run this in a Jupyter notebook)
    
    Args:
        model: Trained model
        sample_images: List of sample images
        layer_names: Names of layers to visualize (if None, use all conv/dense layers)
    """
    import ipywidgets as widgets
    from IPython.display import display
    
    # Get default layer names if not provided
    if layer_names is None:
        layer_names = []
        for layer in model.layers:
            if isinstance(layer, (tf.keras.layers.Conv2D, tf.keras.layers.Dense)):
                layer_names.append(layer.name)
    
    # Create activation models for each layer
    activation_models = {}
    for layer_name in layer_names:
        activation_models[layer_name] = tf.keras.models.Model(
            inputs=model.input,
            outputs=model.get_layer(layer_name).output
        )
    
    # Create widgets
    layer_dropdown = widgets.Dropdown(
        options=layer_names,
        description='Layer:',
        value=layer_names[0]
    )
    
    image_slider = widgets.IntSlider(
        value=0,
        min=0,
        max=len(sample_images) - 1,
        description='Image:',
        continuous_update=False
    )
    
    filter_slider = widgets.IntSlider(
        value=0,
        min=0,
        max=31,  # Default max, will be updated
        description='Filter:',
        continuous_update=False
    )
    
    # Create output widget to display visualizations
    output = widgets.Output()
    
    # Define update function
    def update_visualization(_=None):
        layer_name = layer_dropdown.value
        img_idx = image_slider.value
        filter_idx = filter_slider.value
        
        # Get activations
        img = sample_images[img_idx]
        activations = activation_models[layer_name].predict(np.expand_dims(img, axis=0))[0]
        
        # Update filter slider max based on number of filters
        if len(activations.shape) == 3:  # Conv layer
            filter_slider.max = activations.shape[2] - 1
            filter_idx = min(filter_idx, activations.shape[2] - 1)
            
            with output:
                output.clear_output(wait=True)
                plt.figure(figsize=(12, 5))
                
                # Display original image
                plt.subplot(1, 2, 1)
                plt.imshow(img.squeeze(), cmap='gray')
                plt.title("Original Image")
                plt.axis('off')
                
                # Display activation
                plt.subplot(1, 2, 2)
                plt.imshow(activations[:, :, filter_idx], cmap='viridis')
                plt.title(f"Activation for {layer_name}, Filter {filter_idx}")
                plt.axis('off')
                
                plt.tight_layout()
                plt.show()
        else:
            # For dense layers
            with output:
                output.clear_output(wait=True)
                plt.figure(figsize=(10, 6))
                plt.bar(range(activations.shape[0]), activations)
                plt.title(f"Activations for {layer_name}")
                plt.xlabel("Neuron Index")
                plt.ylabel("Activation")
                plt.tight_layout()
                plt.show()
    
    # Set up widget event handlers
    layer_dropdown.observe(update_visualization, names='value')
    image_slider.observe(update_visualization, names='value')
    filter_slider.observe(update_visualization, names='value')
    
    # Create layout
    controls = widgets.HBox([layer_dropdown, image_slider, filter_slider])
    app = widgets.VBox([controls, output])
    
    # Initial update
    update_visualization()
    
    return app
```
