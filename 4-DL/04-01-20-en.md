# Ethics and Limitations of Deep Learning

## 1. Introduction

Deep learning has revolutionized many fields with impressive capabilities in image recognition, natural language processing, and decision-making. However, these powerful technologies come with important ethical considerations and technical limitations that must be addressed.

This guide explores the key ethical challenges and limitations of deep learning systems, along with practical approaches to mitigating these issues.

```python
def compute_epsilon(steps, noise_multiplier, batch_size, num_examples):
    from tensorflow_privacy.privacy.analysis.rdp_accountant import compute_rdp
    from tensorflow_privacy.privacy.analysis.rdp_accountant import get_privacy_spent
    
    # Calculate the sampling rate
    sampling_probability = batch_size / num_examples
    
    # Calculate the privacy budget
    orders = [1 + x / 10.0 for x in range(1, 100)] + list(range(12, 64))
    rdp = compute_rdp(
        q=sampling_probability,
        noise_multiplier=noise_multiplier,
        steps=steps,
        orders=orders
    )
    
    # Convert to epsilon
    delta = 1e-5  # A typical value for large datasets
    epsilon = get_privacy_spent(orders, rdp, target_delta=delta)[0]
    
    return epsilon
```

```python
# Implementing Grad-CAM for visualization
def make_gradcam_heatmap(img_array, model, last_conv_layer_name, pred_index=None):
    # Create a model that maps the input image to the activations of the last conv layer
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [model.get_layer(last_conv_layer_name).output, model.output]
    )

    # Compute the gradient of the top predicted class for the input image
    with tf.GradientTape() as tape:
        last_conv_layer_output, preds = grad_model(img_array)
        if pred_index is None:
            pred_index = tf.argmax(preds[0])
        class_channel = preds[:, pred_index]

    # Gradient of the output with respect to the last conv layer output
    grads = tape.gradient(class_channel, last_conv_layer_output)

    # Vector of mean intensity of gradient over a channel
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))

    # Weight the channels by the gradient importance
    last_conv_layer_output = last_conv_layer_output[0]
    heatmap = last_conv_layer_output @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)

    # Normalize the heatmap
    heatmap = tf.maximum(heatmap, 0) / tf.reduce_max(heatmap)
    
    return heatmap.numpy()

# Generate class activation heatmap
last_conv_layer = 'conv5_block3_out'  # Last convolutional layer in ResNet50
heatmap = make_gradcam_heatmap(x, model, last_conv_layer)

# Display heatmap over image
def save_and_display_gradcam(img_path, heatmap, alpha=0.4):
    # Load the original image
    img = image.load_img(img_path)
    img = image.img_to_array(img)

    # Resize the heatmap to match the original image
    import cv2
    heatmap = cv2.resize(heatmap, (img.shape[1], img.shape[0]))
    
    # Convert heatmap to RGB
    heatmap = np.uint8(255 * heatmap)
    heatmap = cm.jet(heatmap)[..., :3] * 255
    
    # Superimpose the heatmap on the image
    superimposed_img = heatmap * alpha + img
    superimposed_img = np.clip(superimposed_img, 0, 255).astype("uint8")
    
    # Display the images
    fig, (ax1, ax2, ax3) = plt.subplots(1, 3, figsize=(15, 5))
    ax1.imshow(img.astype('uint8'))
    ax1.set_title('Original Image')
    ax1.axis('off')
    
    ax2.imshow(heatmap.astype('uint8'))
    ax2.set_title('Grad-CAM Heatmap')
    ax2.axis('off')
    
    ax3.imshow(superimposed_img)
    ax3.set_title('Superimposed')
    ax3.axis('off')
```

## 6. Technical Limitations

### 6.1 Adversarial Examples

Deep learning models can be fooled by small, imperceptible perturbations to inputs:

```python
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.applications import MobileNetV2
from tensorflow.keras.applications.mobilenet_v2 import preprocess_input, decode_predictions
from tensorflow.keras.preprocessing import image

# Load pre-trained model

# Load an image
img_path = 'panda.jpg'

x = image.img_to_array(img)
x = np.expand_dims(x, axis=0)
x = preprocess_input(x)

# Get original prediction
preds = model.predict(x)
original_class = np.argmax(preds[0])
print("Original prediction:", decode_predictions(preds, top=3)[0])

# Create adversarial example using Fast Gradient Sign Method (FGSM)
def create_adversarial_example(model, image, label, epsilon=0.01):
    image_tensor = tf.convert_to_tensor(image, dtype=tf.float32)
    
    with tf.GradientTape() as tape:
        tape.watch(image_tensor)
        prediction = model(image_tensor)
        loss = tf.keras.losses.sparse_categorical_crossentropy(label, prediction)
    
    # Get gradient of loss w.r.t the input image
    gradient = tape.gradient(loss, image_tensor)
    
    # Create perturbation: epsilon * sign(gradient)
    perturbation = epsilon * tf.sign(gradient)
    
    # Create adversarial example
    adversarial_example = image_tensor + perturbation
    
    # Make sure we stay in valid image range [0, 1]
    adversarial_example = tf.clip_by_value(adversarial_example, -1, 1)
    
    return adversarial_example.numpy()

# Target a different class (e.g., class 910: 'typewriter')
target_class = 910

# Create adversarial example
adv_x = create_adversarial_example(model, x, target_class, epsilon=0.1)

# Get prediction on adversarial example
adv_preds = model.predict(adv_x)
print("Adversarial prediction:", decode_predictions(adv_preds, top=3)[0])

# Visualize
def display_images(original, adversarial, original_pred, adversarial_pred):
```

### 6.2 Out-of-Distribution Detection

Deep learning models can make confident predictions on completely irrelevant inputs:

```python
import tensorflow as tf
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.preprocessing import image
from tensorflow.keras.applications.resnet50 import preprocess_input, decode_predictions
from sklearn.metrics import roc_curve, auc

# Load pre-trained model
model = ResNet50(weights='imagenet')

# Function to get model prediction and confidence
def predict_image(model, img_path):
    img = image.load_img(img_path, target_size=(224, 224))
    x = image.img_to_array(img)
    x = np.expand_dims(x, axis=0)
    x = preprocess_input(x)
    
    preds = model.predict(x)
    confidence = np.max(preds[0])
    label = decode_predictions(preds, top=1)[0][0][1]
    
    return {
        'label': label,
        'confidence': confidence,
        'softmax': preds[0]
    }

# Function to detect out-of-distribution samples using softmax probabilities
def detect_ood(in_distribution_softmax, test_softmax, method='entropy'):
    if method == 'max_prob':
        # Use negative max probability as anomaly score
        in_dist_scores = -np.max(in_distribution_softmax, axis=1)
        test_scores = -np.max(test_softmax, axis=1)
    elif method == 'entropy':
        # Use entropy as anomaly score
        in_dist_scores = -np.sum(in_distribution_softmax * np.log(in_distribution_softmax + 1e-10), axis=1)
        test_scores = -np.sum(test_softmax * np.log(test_softmax + 1e-10), axis=1)
    
    # Calculate ROC curve and AUC
    # Combine scores and create labels (0 for in-distribution, 1 for OOD)
    all_scores = np.concatenate([in_dist_scores, test_scores])
    all_labels = np.concatenate([np.zeros(len(in_dist_scores)), np.ones(len(test_scores))])
    
    fpr, tpr, thresholds = roc_curve(all_labels, all_scores)
    roc_auc = auc(fpr, tpr)
    
    return {
        'in_dist_scores': in_dist_scores,
        'test_scores': test_scores,
        'fpr': fpr,
        'tpr': tpr,
        'auc': roc_auc,
        'thresholds': thresholds
    }

# Example usage:

# 2. Collect predictions on out-of-distribution data (e.g., SVHN, random noise)
```

```python
        Args:
            run_id (str): Identifier for this evaluation run
            y_true (array): True labels
            y_pred (array): Predicted probabilities or labels
            protected_attributes (dict): Dict mapping attribute names to values for each sample
            threshold (float): Classification threshold for predicted probabilities
        """
        if isinstance(y_pred[0], float):
            # Convert probabilities to binary predictions using threshold
            y_pred_binary = (np.array(y_pred) >= threshold).astype(int)
        else:
            y_pred_binary = y_pred
            
        metrics = {}
        
        # Calculate overall metrics
        tn, fp, fn, tp = confusion_matrix(y_true, y_pred_binary).ravel()
        overall_precision = tp / (tp + fp) if (tp + fp) > 0 else 0
        overall_recall = tp / (tp + fn) if (tp + fn) > 0 else 0
        overall_fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
        
        metrics['overall'] = {
            'precision': overall_precision,
            'recall': overall_recall,
            'false_positive_rate': overall_fpr,
            'samples': len(y_true)
        }
        
        # Calculate metrics for each protected attribute group
        for attr_name, attr_values in protected_attributes.items():
            metrics[attr_name] = {}
            unique_values = np.unique(attr_values)
            
            for value in unique_values:
                # Select samples for this group
                mask = np.array(attr_values) == value
                if sum(mask) == 0:
                    continue
                    
                group_true = np.array(y_true)[mask]
                group_pred = np.array(y_pred_binary)[mask]
                
                # Calculate metrics for this group
                tn, fp, fn, tp = confusion_matrix(group_true, group_pred).ravel()
                precision = tp / (tp + fp) if (tp + fp) > 0 else 0
                recall = tp / (tp + fn) if (tp + fn) > 0 else 0
                fpr = fp / (fp + tn) if (fp + tn) > 0 else 0
                
                metrics[attr_name][value] = {
                    'precision': precision,
                    'recall': recall,
                    'false_positive_rate': fpr,
                    'samples': sum(mask)
                }
        
        # Store results
        self.metrics_history[run_id] = metrics
        
        return metrics
    
    def get_disparity_metrics(self, run_id, attr_name):
        """
        Calculate disparity metrics for a protected attribute.
        
        Args:
            run_id (str): Identifier for the evaluation run
            attr_name (str): Name of the protected attribute
        
        Returns:
            dict: Disparity metrics for the attribute
        """
        if run_id not in self.metrics_history:
            raise ValueError(f"Run ID {run_id} not found")
            
        if attr_name not in self.metrics_history[run_id]:
            raise ValueError(f"Attribute {attr_name} not found in run {run_id}")
        
        metrics = self.metrics_history[run_id][attr_name]
        overall = self.metrics_history[run_id]['overall']
        
        # Calculate disparities between groups
        disparities = {}
        for metric_name in ['precision', 'recall', 'false_positive_rate']:
            values = [group_metrics[metric_name] for group_metrics in metrics.values() if group_metrics['samples'] > 0]
            
            if values:
                min_val = min(values)
                max_val = max(values)
                
                disparities[f'{metric_name}_disparity_ratio'] = min_val / max_val if max_val > 0 else 1
                disparities[f'{metric_name}_disparity_difference'] = max_val - min_val
        
        return disparities
    
    def plot_metric_comparison(self, run_id, attr_name, metric_name='precision'):
        """
        Plot comparison of a metric across different groups.
        
        Args:
            run_id (str): Identifier for the evaluation run
            attr_name (str): Name of the protected attribute
            metric_name (str): Name of the metric to plot
        """
        if run_id not in self.metrics_history:
            raise ValueError(f"Run ID {run_id} not found")
            
        if attr_name not in self.metrics_history[run_id]:
            raise ValueError(f"Attribute {attr_name} not found in run {run_id}")
        
        metrics = self.metrics_history[run_id][attr_name]
        overall = self.metrics_history[run_id]['overall'][metric_name]
        
        # Extract values for plotting
        groups = list(metrics.keys())
        values = [metrics[group][metric_name] for group in groups]
        samples = [metrics[group]['samples'] for group in groups]
        
        # Create bar plot
        plt.figure(figsize=(10, 6))
        bars = plt.bar(groups, values, alpha=0.7)
        
        # Add overall line
        plt.axhline(y=overall, color='r', linestyle='--', label=f'Overall {metric_name}: {overall:.3f}')
        
        # Add labels and title
        plt.xlabel('Group')
        plt.ylabel(metric_name.capitalize())
        plt.title(f'{metric_name.capitalize()} by {attr_name}')
        
        # Add sample sizes above bars
        for i, bar in enumerate(bars):
            plt.text(bar.get_x() + bar.get_width()/2, bar.get_height() + 0.01, 
                    f'n={samples[i]}', ha='center')
        
        plt.legend()
        plt.tight_layout()
        plt.show()
        
        # Print disparity metrics
        disparities = self.get_disparity_metrics(run_id, attr_name)
        ratio = disparities.get(f'{metric_name}_disparity_ratio', 'N/A')
        difference = disparities.get(f'{metric_name}_disparity_difference', 'N/A')
        
        print(f"{metric_name.capitalize()} disparity ratio: {ratio}")
        print(f"{metric_name.capitalize()} disparity difference: {difference}")
```

## 8. Future Directions and Limitations

Deep learning continues to evolve rapidly, but several fundamental limitations remain:

1. **Data dependence**: Deep learning models require large amounts of high-quality data and struggle with limited data scenarios.

2. **Computational requirements**: State-of-the-art models demand significant computational resources, making them inaccessible to many.

3. **Causal reasoning**: Current models excel at pattern recognition but struggle with causal reasoning and common-sense understanding.

4. **Adaptability**: Models trained for specific tasks typically don't generalize well to new domains without retraining.

5. **Verification and validation**: There's a lack of robust methods for formal verification of deep learning systems.
