# Hodnocení modelů pro hluboké učení

## 1. Úvod do hodnocení modelů

Hodnocení modelů je klíčovým krokem v pracovním postupu hlubokého učení, který nám pomáhá pochopit, jak dobře naše modely fungují. Na rozdíl od tradičního strojového učení mají modely hlubokého učení jedinečné výzvy při hodnocení kvůli:

- Velkému počtu parametrů
- Tendenci k přetrénování
- Nárokům na výpočetní zdroje
- Specifickým potřebám hodnocení pro různé domény

Tento průvodce prozkoumává komplexní techniky pro hodnocení modelů hlubokého učení s praktickými implementacemi v Pythonu.

## 3. Metriky klasifikace

### 3.1 Základní metriky (přesnost, preciznost, úplnost, F1)

```python
def evaluate_classification_model(model, X_test, y_test, threshold=0.5):
    """Vyhodnotí klasifikační model pomocí více metrik"""
    # Získání predikcí
    if isinstance(model, tf.keras.Model):  # TensorFlow model
        y_pred_prob = model.predict(X_test)
        y_pred = (y_pred_prob > threshold).astype(int)
    else:  # PyTorch model
        model.eval()
        with torch.no_grad():
            y_pred_prob = model(torch.FloatTensor(X_test)).numpy()
            y_pred = (y_pred_prob > threshold).astype(int)
    
    # Výpočet metrik
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, average='weighted')
    recall = recall_score(y_test, y_pred, average='weighted')
    f1 = f1_score(y_test, y_pred, average='weighted')
    
    # Výpis výsledků
    print(f"Přesnost (accuracy): {accuracy:.4f}")
    print(f"Preciznost (precision): {precision:.4f}")
    print(f"Úplnost (recall): {recall:.4f}")
    print(f"F1 skóre: {f1:.4f}")
    
    # Zobrazení klasifikační zprávy
    print("\nKlasifikační zpráva:")
    print(classification_report(y_test, y_pred))
    
    return {
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall,
        'f1': f1,
        'y_pred': y_pred,
        'y_pred_prob': y_pred_prob
    }
```

```python
def plot_confusion_matrix(y_true, y_pred, class_names=None):
    """Vykreslí matici záměn s možností přizpůsobení"""
    cm = confusion_matrix(y_true, y_pred)
    
    plt.figure(figsize=(10, 8))

    plt.title('Matice záměn')
    plt.show()
    
    # Normalizovaná matice záměn
    cm_norm = cm.astype('float') / cm.sum(axis=1)[:, np.newaxis]
    plt.figure(figsize=(10, 8))
    sns.heatmap(cm_norm, annot=True, fmt='.2f', cmap='Blues', 
                xticklabels=class_names, yticklabels=class_names)
```

### 3.3 ROC křivka a AUC

ROC (Receiver Operating Characteristic) křivky vizualizují kompromis mezi mírou pravdivě pozitivních výsledků a mírou falešně pozitivních výsledků:

```python
def plot_roc_curve(y_true, y_scores, multi_class='ovr'):
    """Vykreslí ROC křivku pro binární nebo multiclass problémy"""
    
    plt.figure(figsize=(10, 8))
    
    # Binární klasifikace
    if len(np.unique(y_true)) == 2:
        fpr, tpr, _ = roc_curve(y_true, y_scores)
        roc_auc = auc(fpr, tpr)
        
        plt.plot(fpr, tpr, color='darkorange', lw=2, 
                 label=f'ROC křivka (plocha = {roc_auc:.2f})')
        plt.plot([0, 1], [0, 1], color='navy', lw=2, linestyle='--')
        plt.xlabel('Míra falešně pozitivních')
        plt.ylabel('Míra pravdivě pozitivních')
        plt.title('Křivka ROC (Receiver Operating Characteristic)')
        plt.legend(loc='lower right')
        
    # Multiclass klasifikace
    else:
        # Přístup one-vs-rest (jedna křivka na třídu)
        n_classes = len(np.unique(y_true))
        y_true_bin = tf.keras.utils.to_categorical(y_true, n_classes)
        
        for i in range(n_classes):
            fpr, tpr, _ = roc_curve(y_true_bin[:, i], y_scores[:, i])
            roc_auc = auc(fpr, tpr)
            plt.plot(fpr, tpr, lw=2, 
                     label=f'Třída {i} (plocha = {roc_auc:.2f})')
        
        plt.plot([0, 1], [0, 1], 'k--', lw=2)
        plt.xlabel('Míra falešně pozitivních')
        plt.ylabel('Míra pravdivě pozitivních')
        plt.title('Multi-class ROC křivka (One-vs-Rest)')
        plt.legend(loc='lower right')
    
    plt.show()
```

```python
def evaluate_regression_model(model, X_test, y_test):
    """Vyhodnotí regresní model pomocí více metrik"""
    # Získání predikcí
    if isinstance(model, tf.keras.Model):  # TensorFlow model
        y_pred = model.predict(X_test).flatten()
    else:  # PyTorch model
        model.eval()
        with torch.no_grad():
            y_pred = model(torch.FloatTensor(X_test)).numpy().flatten()
    
    # Výpočet metrik
    mse = mean_squared_error(y_test, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_test, y_pred)
    r2 = r2_score(y_test, y_pred)
    
    # Výpis výsledků
    print(f"Střední kvadratická chyba (MSE): {mse:.4f}")
    print(f"Odmocnina střední kvadratické chyby (RMSE): {rmse:.4f}")
    print(f"Střední absolutní chyba (MAE): {mae:.4f}")
    print(f"Koeficient determinace (R²): {r2:.4f}")
    
    # Graf skutečných vs. predikovaných hodnot
    plt.figure(figsize=(10, 6))
    plt.scatter(y_test, y_pred, alpha=0.5)
    plt.plot([min(y_test), max(y_test)], [min(y_test), max(y_test)], 'r--')
    plt.xlabel('Skutečnost')
    plt.ylabel('Predikce')
    plt.title('Skutečné vs. predikované hodnoty')
    plt.tight_layout()
    plt.show()
    
    # Graf reziduí
    residuals = y_test - y_pred
    plt.figure(figsize=(10, 6))
    plt.scatter(y_pred, residuals, alpha=0.5)
    plt.axhline(y=0, color='r', linestyle='--')
    plt.xlabel('Predikce')
    plt.ylabel('Rezidua')
    plt.title('Graf reziduí')
    plt.tight_layout()
    plt.show()
    
    return {
        'mse': mse,
        'rmse': rmse,
        'mae': mae,
        'r2': r2,
        'y_pred': y_pred
    }
```

## 5. Křížová validace pro hluboké učení

Křížová validace pomáhá robustněji hodnotit výkon modelu, ale vyžaduje speciální zacházení pro hluboké učení:

```python
def deep_learning_kfold_cv(model_fn, X, y, n_splits=5, epochs=10, batch_size=32):
    """Provede k-fold křížovou validaci s modely hlubokého učení
    
    Args:
        model_fn: Funkce, která vrací zkompilovaný model
        X: Vstupní vlastnosti
        y: Cílové hodnoty
        n_splits: Počet rozdělení
        epochs: Počet epoch pro každé rozdělení
        batch_size: Velikost dávky pro trénování
        
    Returns:
        Seznam metrik pro každé rozdělení
    """
    from sklearn.model_selection import KFold
    
    kfold = KFold(n_splits=n_splits, shuffle=True, random_state=42)
    metrics = []
    
    for fold, (train_idx, val_idx) in enumerate(kfold.split(X, y)):
        print(f"\nRozdělení {fold+1}/{n_splits}")
        
        # Rozdělení dat
        X_train, X_val = X[train_idx], X[val_idx]
        y_train, y_val = y[train_idx], y[val_idx]
        
        # Vytvoření a trénování nového modelu pro každé rozdělení
        model = model_fn()
        
        # Trénování modelu
        history = model.fit(
            X_train, y_train,
            validation_data=(X_val, y_val),
            epochs=epochs,
            batch_size=batch_size,
            verbose=1
        )
        
        # Vyhodnocení modelu
        val_loss, val_metric = model.evaluate(X_val, y_val, verbose=0)
        
        # Uložení metrik
        metrics.append({
            'val_loss': val_loss,
            'val_metric': val_metric,
            'history': history.history
        })
        
    # Výpis průměrného výkonu
    avg_val_metric = np.mean([m['val_metric'] for m in metrics])
    print(f"\nPrůměrná validační metrika napříč všemi rozděleními: {avg_val_metric:.4f}")
    
    return metrics
```

## 6. Křivky učení a detekce přetrénování

Křivky učení nám pomáhají vizualizovat průběh trénování modelu a detekovat přetrénování:

```python
def plot_learning_curves(history, metrics=['loss'], figsize=(12, 6)):
    """Vykreslí křivky učení z historie tréninku
    
    Args:
        history: Objekt Historie z model.fit() nebo slovník s metrikami
        metrics: Seznam metrik k vykreslení
        figsize: Velikost obrázku
    """
    plt.figure(figsize=figsize)
    
    for i, metric in enumerate(metrics):
        plt.subplot(1, len(metrics), i+1)
        
        # Vykreslení trénovacích metrik
        plt.plot(history[metric], label=f'Trénovací {metric}')
        
        # Vykreslení validačních metrik, pokud jsou dostupné
        val_metric = f'val_{metric}'
        if val_metric in history:
            plt.plot(history[val_metric], label=f'Validační {metric}')
        
        plt.title(f'{metric.capitalize()} během epoch')
        plt.xlabel('Epochy')
        plt.ylabel(metric.capitalize())
        plt.legend()
        plt.grid(True)
    
    plt.tight_layout()
    plt.show()
    
    # Kontrola příznaků přetrénování
    for metric in metrics:
        val_metric = f'val_{metric}'
        if val_metric in history:
            train_final = history[metric][-1]
            val_final = history[val_metric][-1]
            
            # Jednoduchá heuristika: pokud je validační výkon výrazně horší
            if metric == 'loss' and val_final > 1.1 * train_final:
                print(f"Varování: Možné přetrénování detekováno. Konečný {val_metric}={val_final:.4f} je {val_final/train_final:.2f}x vyšší než {metric}={train_final:.4f}")
            elif metric != 'loss' and val_final < 0.9 * train_final:
                print(f"Varování: Možné přetrénování detekováno. Konečný {val_metric}={val_final:.4f} je {val_final/train_final:.2f}x nižší než {metric}={train_final:.4f}")
```

## 7. Vizualizace výkonu modelu

### 7.1 Mapy aktivace tříd (CAM)

Vizualizace toho, na co se model zaměřuje při klasifikaci obrázků:

```python
def visualize_cam(model, img, layer_name=None, class_idx=None, figsize=(15, 5)):
    """Vizualizuje mapy aktivace tříd pro CNN modely
    
    Args:
        model: Trénovaný model Keras
        img: Vstupní obrázek (předzpracovaný pro model)
        layer_name: Název poslední konvoluční vrstvy
        class_idx: Index třídy, pro kterou generovat CAM
                  (výchozí je nejvyšší predikce)
    """
    # Ujistíme se, že obrázek má dimenze dávky
    if len(img.shape) == 3:
        img = np.expand_dims(img, axis=0)
    
    # Pokud není specifikována vrstva, pokusíme se najít poslední konvoluční vrstvu
    if layer_name is None:
        for layer in reversed(model.layers):
            if 'conv' in layer.name:
                layer_name = layer.name
                print(f"Použití vrstvy: {layer_name}")
                break
    
    # Získání predikcí
    preds = model.predict(img)
    
    # Pokud není specifikována třída, použijeme predikovanou třídu
    if class_idx is None:
        class_idx = np.argmax(preds[0])
    
    # Získání názvu třídy, pokud je k dispozici
    class_output = f"Třída {class_idx}"
    if hasattr(model, 'classes'):
        class_output = f"Třída {model.classes[class_idx]} ({class_idx})"
    
    # Získání poslední konvoluční vrstvy
    last_conv_layer = model.get_layer(layer_name)
    
    # Vytvoření modelu, který mapuje vstupní obrázek na aktivace a predikce
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [last_conv_layer.output, model.output]
    )
    
    # Výpočet gradientu nejvyšší predikované třídy vzhledem k aktivacím
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img)
        class_channel = predictions[:, class_idx]
    
    # Gradient výstupu třídy vzhledem k aktivacím mapy příznaků
    grads = tape.gradient(class_channel, conv_outputs)
    
    # Vektor střední intenzity gradientu přes mapu příznaků
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Vážení map příznaků podle důležitosti pro třídu
    conv_outputs = conv_outputs[0]
    heatmap = conv_outputs @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)
    
    # Normalizace heatmapy
    heatmap = tf.maximum(heatmap, 0) / tf.math.reduce_max(heatmap)
    heatmap = heatmap.numpy()
    
    # Zobrazení původního obrázku
    plt.figure(figsize=figsize)
    
    # Původní obrázek
    plt.subplot(1, 3, 1)
    # V případě potřeby denormalizace obrázku
    img_to_show = img[0].copy()
    if img_to_show.max() <= 1.0:
        img_to_show = img_to_show * 255
    plt.imshow(img_to_show.astype('uint8'))
    plt.title(f"Původní - Predikce: {class_output}")
    
    # Heatmapa
    plt.subplot(1, 3, 2)
    plt.imshow(heatmap, cmap='jet')
    plt.title("Mapa aktivace tříd")
    plt.colorbar()
    
    # Překrývající se
    plt.subplot(1, 3, 3)
    
    # Změna velikosti heatmapy tak, aby odpovídala rozměrům původního obrázku
    import cv2
    heatmap_resized = cv2.resize(heatmap, (img.shape[2], img.shape[1]))
    heatmap_colored = np.uint8(255 * heatmap_resized)
    heatmap_colored = cv2.applyColorMap(heatmap_colored, cv2.COLORMAP_JET)
    
    # Převod na RGB, pokud je to potřeba
    if img_to_show.shape[-1] == 1:  # Stupně šedi
        img_to_show = cv2.cvtColor(img_to_show.astype('uint8'), cv2.COLOR_GRAY2RGB)
    
    # Překrytí heatmapy na původní obrázek
    superimposed = heatmap_colored * 0.4 + img_to_show
    superimposed = np.clip(superimposed, 0, 255).astype('uint8')
    
    plt.imshow(superimposed)
    plt.title("Překrytí heatmapy")
    plt.tight_layout()
    plt.show()
```

### 7.2 t-SNE pro vizualizaci příznaků

Vizualizace, jak váš model odděluje třídy ve vysokorozměrném prostoru:

```python
def visualize_embeddings_tsne(model, X, y, layer_name=None, perplexity=30, figsize=(12, 10)):
    """Vizualizace vnoření modelu pomocí t-SNE
    
    Args:
        model: Trénovaný model
        X: Vstupní data
        y: Štítky
        layer_name: Název vrstvy pro extrakci vnoření
                   (výchozí je vrstva před poslední vrstvou)
        perplexity: Parametr perplexity t-SNE
    """
    from sklearn.manifold import TSNE
    
    # Pokud není specifikován název vrstvy, pokusíme se získat předposlední vrstvu
    if layer_name is None:
        if isinstance(model, tf.keras.Model):
            layer_name = model.layers[-2].name
            print(f"Použití vrstvy: {layer_name}")
    
    # Získání modelu pro extrakci příznaků
    if isinstance(model, tf.keras.Model):
        feature_model = tf.keras.models.Model(
            inputs=model.input,
            outputs=model.get_layer(layer_name).output
        )
        # Extrakce příznaků
        features = feature_model.predict(X)
    else:  # PyTorch model
        # Pro PyTorch by to vyžadovalo vlastní implementaci v závislosti na architektuře modelu
        pass
    
    # Zploštění příznaků, pokud je to nutné
    if len(features.shape) > 2:
        features = features.reshape(features.shape[0], -1)
    
    # Aplikace t-SNE
    print("Aplikace redukce dimenzionality t-SNE...")
    tsne = TSNE(n_components=2, perplexity=perplexity, random_state=42)
    tsne_results = tsne.fit_transform(features)
    
    # Vykreslení výsledků
    plt.figure(figsize=figsize)
    
    # Určení, zda máme číselné nebo kategorické štítky
    unique_labels = np.unique(y)
    is_categorical = len(unique_labels) <= 20  # Libovolný práh
    
    if is_categorical:
        # Použití diskrétní barevné mapy pro kategorická data
        scatter = plt.scatter(tsne_results[:, 0], tsne_results[:, 1], c=y, cmap='tab10', alpha=0.7)
        plt.colorbar(label="Třída")
        plt.legend(*scatter.legend_elements(), title="Třídy")
    else:
        # Použití spojité barevné mapy pro číselná data
        scatter = plt.scatter(tsne_results[:, 0], tsne_results[:, 1], c=y, cmap='viridis', alpha=0.7)
        plt.colorbar(label="Hodnota")
    
    plt.title(f't-SNE vizualizace příznaků z {layer_name}')
    plt.xlabel('t-SNE dimenze 1')
    plt.ylabel('t-SNE dimenze 2')
    plt.tight_layout()
    plt.show()
```

```python
def evaluate_model_calibration(y_true, y_prob, n_bins=10):

        calibration_error = np.mean((prob_true - prob_pred) ** 2)

        y_true_bin = tf.keras.utils.to_categorical(y_true, n_classes)

            prob_true, prob_pred = calibration_curve(y_true_bin[:, i], 
                                                   y_prob[:, i], 
                                                   n_bins=n_bins)
            plt.plot(prob_pred, prob_true, marker='o', linewidth=2, 
                     label=f'Třída {i}')
        
        plt.plot([0, 1], [0, 1], 'k--', label='Perfektně kalibrovaný')
        plt.xlabel('Střední předpovězená pravděpodobnost')
        plt.ylabel('Podíl pozitivních')
        plt.title('Kalibrační křivky pro multi-třídní')
        plt.legend()
        plt.grid(True)
    
    plt.show()

        # Vytvoření košů predikci
        bin_edges = np.linspace(0, 1, n_bins + 1)
        bin_indices = np.digitize(y_prob, bin_edges) - 1

        # Výpočet přesnosti v každém koši
        bin_accs = np.array([np.mean(y_true[bin_indices == i])

        # Výpočet očekávané kalibrační chyby (ECE)

        plt.text(0.05, 0.9, f'Očekávaná kalibrační chyba: {ece:.4f}', 
                 transform=plt.gca().transAxes, fontsize=12)
        
        plt.xlabel('Koš')
        plt.ylabel('Hodnota')
        plt.title('Diagram spolehlivosti')
        plt.legend()
        plt.grid(True)
        
    plt.tight_layout()
    plt.show()
```

```python
def compare_models_statistically(models, model_names, X_test, y_test, n_bootstrap=1000, alpha=0.05):
    """Porovná více modelů pomocí bootstrap samplingu
    
    Args:
        models: Seznam trénovaných modelů
        model_names: Seznam názvů modelů
        X_test: Testovací příznaky
        y_test: Testovací štítky
        n_bootstrap: Počet bootstrap vzorků
        alpha: Úroveň významnosti
    """
    import scipy.stats as stats
    
    n_models = len(models)
    n_samples = len(y_test)
    
    # Získání predikcí z každého modelu
    all_predictions = []
    for model in models:
        if isinstance(model, tf.keras.Model):
            y_pred = model.predict(X_test)
            if len(y_pred.shape) > 1 and y_pred.shape[1] > 1:
                y_pred = np.argmax(y_pred, axis=1)
            all_predictions.append(y_pred)
        else:
            # Pro PyTorch modely
            model.eval()
            with torch.no_grad():
                y_pred = model(torch.FloatTensor(X_test)).numpy()
                if len(y_pred.shape) > 1 and y_pred.shape[1] > 1:
                    y_pred = np.argmax(y_pred, axis=1)
                all_predictions.append(y_pred)
    
    # Výpočet metrik výkonu
    base_metrics = []
    for i in range(n_models):
        accuracy = accuracy_score(y_test, all_predictions[i])
        base_metrics.append(accuracy)
        print(f"{model_names[i]} - Přesnost: {accuracy:.4f}")
    
    # Provedení bootstrap samplingu
    print("\nProvádění bootstrap samplingu...")
    bootstrap_metrics = np.zeros((n_bootstrap, n_models))
    
    for b in range(n_bootstrap):
        # Vzorkování s nahrazením
        indices = np.random.choice(n_samples, n_samples, replace=True)
        y_bootstrap = y_test[indices]
        
        for i in range(n_models):
            y_pred_bootstrap = all_predictions[i][indices]
            bootstrap_metrics[b, i] = accuracy_score(y_bootstrap, y_pred_bootstrap)
    
    # Výpočet intervalů spolehlivosti
    ci_lower = np.percentile(bootstrap_metrics, alpha/2*100, axis=0)
    ci_upper = np.percentile(bootstrap_metrics, (1-alpha/2)*100, axis=0)
    
    print(f"\nBootstrap 95% intervaly spolehlivosti:")
    for i in range(n_models):
        print(f"{model_names[i]}: [{ci_lower[i]:.4f}, {ci_upper[i]:.4f}]")
    
    # Provedení párového porovnání
    print("\nPárová porovnání modelů:")
    print("-"*60)
    print(f"{'Model 1':<15} {'Model 2':<15} {'Rozdíl':<10} {'p-hodnota':<10} {'Významný':<10}")
    print("-"*60)
    
    for i in range(n_models):
        for j in range(i+1, n_models):
            # Výpočet rozdílů pro každý bootstrap vzorek
            diffs = bootstrap_metrics[:, i] - bootstrap_metrics[:, j]
            
            # Výpočet p-hodnoty pro dvoustranný test
            if np.mean(diffs) >= 0:
                p_value = 2 * np.mean(diffs <= 0)
            else:
                p_value = 2 * np.mean(diffs >= 0)
            
            # Určení významnosti
            is_significant = p_value < alpha
            significance = "Ano" if is_significant else "Ne"
            
            print(f"{model_names[i]:<15} {model_names[j]:<15} "
                  f"{base_metrics[i]-base_metrics[j]:.4f}   {p_value:.4f}    {significance:<10}")
    
    # Vizualizace výkonu modelu s intervaly spolehlivosti
    plt.figure(figsize=(12, 6))
    
    x = np.arange(n_models)
    width = 0.6
    
    plt.bar(x, base_metrics, width, label='Přesnost', color='skyblue')
    plt.errorbar(x, base_metrics, yerr=[base_metrics-ci_lower, ci_upper-base_metrics],
                 fmt='o', color='blue', capsize=10)
    
    plt.xlabel('Modely')
    plt.ylabel('Přesnost')
    plt.title('Porovnání výkonu modelů s intervaly spolehlivosti')
    plt.xticks(x, model_names, rotation=45, ha='right')
    plt.ylim(min(ci_lower)*0.95, max(ci_upper)*1.05)
    plt.grid(True, axis='y', linestyle='--', alpha=0.7)
    plt.tight_layout()
    plt.show()
```

```python
        generated_images = generated_images / 255.0
    
    n_samples = min(len(real_images), len(generated_images))
    
    # Výpočet SSIM
    ssim_values = []
    for i in range(n_samples):
        if real_images[i].shape[-1] == 1:
            # Černobílé obrázky
            s = ssim(real_images[i].squeeze(), generated_images[i].squeeze(),
                    data_range=1.0)
        else:
            # RGB obrázky - výpočet SSIM pro každý kanál a průměrování
            s = np.mean([ssim(real_images[i][:,:,j], generated_images[i][:,:,j],
                             data_range=1.0) for j in range(real_images[i].shape[-1])])
        ssim_values.append(s)
    
    # Průměrný SSIM
    avg_ssim = np.mean(ssim_values)
    
    # Metriky založené na příznacích (s použitím předtrénovaného modelu)
    try:
        # Načtení InceptionV3 pro extrakci příznaků
        from tensorflow.keras.applications.inception_v3 import InceptionV3, preprocess_input
        
        # Načtení modelu
        model = InceptionV3(include_top=False, pooling='avg')
        
        # Extrakce příznaků
        def extract_features(images):
            # Změna velikosti na vstupní velikost InceptionV3
            resized = []
            for img in images:
                if img.shape[-1] == 1:  # Z černobílé na RGB
                    img = np.repeat(img, 3, axis=-1)
                resized.append(tf.image.resize(img, (299, 299)).numpy())
            
            preprocessed = preprocess_input(np.array(resized) * 255.0)  # Škálování zpět na [0, 255]
            features = model.predict(preprocessed, batch_size=32)
            return features
        
        real_features = extract_features(real_images[:n_samples])
        gen_features = extract_features(generated_images[:n_samples])
        
        # Výpočet FID (Fréchet Inception Distance)
        # Výpočet průměru a kovariance
        mu_real = np.mean(real_features, axis=0)
        sigma_real = np.cov(real_features, rowvar=False)
        
        mu_gen = np.mean(gen_features, axis=0)
        sigma_gen = np.cov(gen_features, rowvar=False)
        
        # Výpočet FID
        diff = mu_real - mu_gen
        covmean, _ = linalg.sqrtm(sigma_real.dot(sigma_gen), disp=False)
        if np.iscomplexobj(covmean):
            covmean = covmean.real
        
        fid = diff.dot(diff) + np.trace(sigma_real + sigma_gen - 2*covmean)
        
        print(f"Fréchet Inception Distance (FID): {fid:.4f}")
        print(f"Průměrný SSIM: {avg_ssim:.4f}")
        
        # Zobrazení vzorových obrázků
        n_display = min(5, n_samples)
        plt.figure(figsize=(15, 6))
        
        for i in range(n_display):
            # Skutečný obrázek
            plt.subplot(2, n_display, i+1)
            plt.imshow(real_images[i])
            plt.title(f"Skutečný {i+1}")
            plt.axis('off')
            
            # Generovaný obrázek
            plt.subplot(2, n_display, i+n_display+1)
            plt.imshow(generated_images[i])
            plt.title(f"Generovaný {i+1}")
            plt.axis('off')
        
        plt.tight_layout()
        plt.show()
        
        return {
            'fid': fid,
            'ssim': avg_ssim,
            'ssim_individual': ssim_values
        }
    
    except Exception as e:
        print(f"Chyba při výpočtu FID: {e}")
        print(f"Průměrný SSIM: {avg_ssim:.4f}")
        
        return {
            'ssim': avg_ssim,
            'ssim_individual': ssim_values
        }
```

```python
def evaluate_nlp_model(y_true, y_pred, task='text_classification'):
    """Vyhodnocení výkonu NLP modelu
    
    Args:
        y_true: Pravdivé štítky nebo text
        y_pred: Predikované štítky nebo text
        task: Typ NLP úlohy ('text_classification', 'text_generation', 
              'machine_translation', 'summarization')
    """
    if task == 'text_classification':
        # Použití standardních klasifikačních metrik

        # Pokus o použití metrik BLEU, ROUGE a dalších metrik pro generování textu
        try:
            from nltk.translate.bleu_score import corpus_bleu, sentence_bleu
            from nltk.translate.bleu_score import SmoothingFunction
            from rouge import Rouge

            # Předpokládáme, že y_true a y_pred jsou seznamy řetězců

            hypotheses = [text.split() for text in y_pred]

            smoothie = SmoothingFunction().method1
            bleu_scores = [sentence_bleu([ref[0]], hyp, smoothing_function=smoothie) 
                          for ref, hyp in zip(references, hypotheses)]
            
            # Korpusové BLEU
            corpus_bleu_score = corpus_bleu(references, hypotheses,

            # Výpočet ROUGE skóre
            rouge = Rouge()
            rouge_scores = rouge.get_scores([' '.join(hyp) for hyp in hypotheses],
                                           [' '.join(ref[0]) for ref in references],
                                           avg=True)
            
            print("\nROUGE skóre:")

            print("\nPříklady:")

                print(f"\nPříklad {i+1}:")
                print(f"Reference: {y_true[i]}")
                print(f"Hypotéza: {y_pred[i]}")
                print(f"BLEU: {bleu_scores[i]:.4f}")
                
            return {
                'corpus_bleu': corpus_bleu_score,
                'bleu_scores': bleu_scores,
                'rouge_scores': rouge_scores
            }
```

## 15. Časté chyby při hodnocení modelů hlubokého učení

1. **Únik dat**: Kdy informace z testovacího souboru ovlivňují trénink, což způsobuje uměle vysoký výkon.

2. **Přeučení na validační sadu**: Když ladění hyperparametrů způsobí, že model se příliš dobře přizpůsobí validační sadě.

3. **Ignorování statistické významnosti**: Malé rozdíly ve výkonu nemusí být významné.

4. **Nesprávné zacházení s nevyváženými daty**: Používání pouze přesnosti pro nevyvážené datasety.

5. **Nedostatek reprodukovatelnosti**: Nenastavení náhodných seedů nebo nedostatečná dokumentace postupů hodnocení.

6. **Nevhodné metriky**: Používání metrik, které nejsou v souladu s obchodním nebo výzkumným cílem.

7. **Selektivní výběr výsledků**: Uvádění pouze metrik, které dávají model v dobrém světle.
