## 2. Předzpracování dat časových řad

Správné předzpracování je klíčové pro efektivní predikci časových řad pomocí hlubokého učení. Mezi běžné techniky patří:

1. **Převzorkování**: Změna frekvence časové řady
2. **Škálování**: Normalizace nebo standardizace hodnot
3. **Vytváření oken**: Vytváření vstupně-výstupních sekvencí
4. **Inženýrství příznaků**: Extrakce relevantních příznaků jako proměnné zpoždění nebo příznaky založené na datu

```python
# Předzpracování dat pro predikci časových řad pomocí hlubokého učení

# Škálování dat
scaler = MinMaxScaler(feature_range=(0, 1))
scaled_data = scaler.fit_transform(df['Passengers'].values.reshape(-1, 1))

# Vytvoření sekvencí (oken) pro učení s učitelem
def create_sequences(data, seq_length):
    """Vytvoření sekvencí pro učení s učitelem z dat časových řad."""
    xs, ys = [], []
    for i in range(len(data) - seq_length):
        x = data[i:i+seq_length]
        y = data[i+seq_length]
        xs.append(x)
        ys.append(y)
    return np.array(xs), np.array(ys)

# Definice délky sekvence (období zpětného pohledu)
seq_length = 12  # Použití 12 měsíců k predikci následujícího měsíce

# Změna tvaru vstupních dat na [vzorky, časové kroky, příznaky]

# Rozdělení dat na trénovací a testovací sady
train_size = int(len(X) * 0.8)
X_train, X_test = X[:train_size], X[train_size:]

print(f"Tvar trénovacích dat: {X_train.shape}")
print(f"Tvar testovacích dat: {X_test.shape}")
```

## 3. Rekurentní neuronové sítě (RNN) pro časové řady

Rekurentní neuronové sítě (RNN) jsou navrženy pro práci se sekvenčními daty udržováním paměti předchozích vstupů. Jednoduché RNN dokáží zachytit krátkodobé závislosti, ale mají potíže s delšími sekvencemi kvůli problému mizejícího gradientu.

```python
# Vytvoření jednoduchého RNN modelu pro predikci časových řad
def build_simple_rnn_model(input_shape):
    model = keras.Sequential([
        keras.layers.SimpleRNN(64, input_shape=input_shape, return_sequences=True),
        keras.layers.SimpleRNN(32),
        keras.layers.Dense(1)
    ])
    model.compile(optimizer='adam', loss='mse')
    return model

# Vytvoření a trénování RNN modelu
```

## 5. Temporální konvoluční sítě (TCN)

Temporální konvoluční sítě (TCN) používají dilatované kauzální konvoluce ke zpracování sekvenčních dat. Nabízejí výhody oproti RNN včetně paralelizovatelného tréninku, flexibilní velikosti receptivního pole a stabilních gradientů.

```python
# Implementace TCN pro predikci časových řad
from tensorflow.keras.layers import Conv1D, BatchNormalization, Activation, Add, Input, Dense
from tensorflow.keras.models import Model

def residual_block(x, filters, kernel_size, dilation_rate):
    """TCN reziduální blok s dilatovanou kauzální konvolucí."""
    # Přeskočené spojení
    residual = x
    
    # První dilatovaná kauzální konvoluce
    conv1 = Conv1D(filters, kernel_size, padding='causal', dilation_rate=dilation_rate)(x)
    norm1 = BatchNormalization()(conv1)
    act1 = Activation('relu')(norm1)
    
    # Druhá dilatovaná kauzální konvoluce
    conv2 = Conv1D(filters, kernel_size, padding='causal', dilation_rate=dilation_rate)(act1)
    norm2 = BatchNormalization()(conv2)
    act2 = Activation('relu')(norm2)
    
    # Zpracování přeskočeného spojení (sjednocení dimenzí)
    if x.shape[-1] != filters:
        residual = Conv1D(filters, 1, padding='same')(x)
    
    # Přidání přeskočeného spojení
    output = Add()([residual, act2])
    return Activation('relu')(output)

def build_tcn_model(input_shape, filters=64, kernel_size=3, num_layers=3):
    """Vytvoření TCN modelu s více reziduálními bloky."""
    inputs = Input(shape=input_shape)
    x = inputs
    
    # Vytvoření TCN bloků se zvyšujícími se rychlostmi dilatace
    for i in range(num_layers):
        dilation_rate = 2 ** i
        x = residual_block(x, filters, kernel_size, dilation_rate)
    
    # Globální průměrné poolování a výstupní vrstva
    x = keras.layers.GlobalAveragePooling1D(data_format="channels_first")(x)
    outputs = Dense(1)(x)
    
    model = Model(inputs, outputs)
    model.compile(optimizer='adam', loss='mse')
    return model

# Vytvoření a trénování TCN modelu

print(tcn_model.summary())

# Trénování s předčasným zastavením

    X_train, y_train,
    epochs=100,
    batch_size=16,
```

## 6. Mechanismy pozornosti pro časové řady

Mechanismy pozornosti umožňují modelům zaměřit se na relevantní části vstupní sekvence při vytváření predikcí. To je zvláště užitečné pro časové řady s důležitými vzory v různých časových bodech.

```python
# Implementace mechanismu pozornosti pro časové řady
from tensorflow.keras.layers import Attention, Dense, LSTM, Input, Concatenate, Multiply, Permute, RepeatVector

def build_attention_lstm_model(input_shape):
    """Vytvoření LSTM modelu s mechanismem pozornosti."""
    # Vstupní vrstva
    inputs = Input(shape=input_shape)
    
    # LSTM vrstva
    lstm_out = LSTM(64, return_sequences=True)(inputs)
    
    # Mechanismus pozornosti
    attention = Dense(1, activation='tanh')(lstm_out)
    attention = Permute((2, 1))(attention)
    attention = Activation('softmax')(attention)
    
    # Aplikace pozornosti na výstup LSTM
    context = Multiply()([lstm_out, Permute((2, 1))(attention)])
    context = Permute((2, 1))(context)
    
    # Součet přes časovou dimenzi
    context = Lambda(lambda x: K.sum(x, axis=1))(context)
    
    # Výstupní vrstva
    output = Dense(1)(context)
    
    # Vytvoření a kompilace modelu
    model = Model(inputs=inputs, outputs=output)
    model.compile(optimizer='adam', loss='mse')
    return model

# Alternativní implementace pomocí Keras Attention vrstvy
def build_simple_attention_model(input_shape):
    inputs = Input(shape=input_shape)
    lstm_out = LSTM(64, return_sequences=True)(inputs)
    attention_layer = Attention()([lstm_out, lstm_out])
    flattened = keras.layers.Flatten()(attention_layer)
    output = Dense(1)(flattened)
    
    model = Model(inputs=inputs, outputs=output)
    model.compile(optimizer='adam', loss='mse')
    return model
```

## 7. Transformer modely pro data časových řad

Transformer modely, původně navržené pro zpracování přirozeného jazyka, se staly stále populárnějšími pro predikci časových řad díky své schopnosti zachytit globální závislosti a efektivně zpracovávat sekvence paralelně.

```python
# Implementace Transformer modelu pro predikci časových řad
import tensorflow as tf
from tensorflow.keras import layers

def transformer_encoder(inputs, head_size, num_heads, ff_dim, dropout=0):
    """Transformer enkodér blok s multi-head self-attention."""
    # Normalizace a Multi-Head Attention
    x = layers.LayerNormalization(epsilon=1e-6)(inputs)
    x = layers.MultiHeadAttention(
        key_dim=head_size, num_heads=num_heads, dropout=dropout
    )(x, x)
    res = x + inputs
    
    # Feed Forward síť
    x = layers.LayerNormalization(epsilon=1e-6)(res)
    x = layers.Conv1D(filters=ff_dim, kernel_size=1, activation="relu")(x)
    x = layers.Dropout(dropout)(x)
    x = layers.Conv1D(filters=inputs.shape[-1], kernel_size=1)(x)
    return x + res

def build_transformer_model(input_shape, head_size=256, num_heads=4, ff_dim=4, num_transformer_blocks=4, mlp_units=[128], dropout=0.1):
    """Vytvoření Transformer modelu pro predikci časových řad."""
    inputs = layers.Input(shape=input_shape)
    x = inputs
    
    # Přidání pozičního kódování
    positional_encoding = PositionalEncoding(input_shape[0], input_shape[1])(x)
    x = layers.Add()([x, positional_encoding])
    
    # Transformer bloky
    for _ in range(num_transformer_blocks):
        x = transformer_encoder(x, head_size, num_heads, ff_dim, dropout)
    
    # Závěrečné zpracování
    x = layers.GlobalAveragePooling1D(data_format="channels_first")(x)
    for dim in mlp_units:
        x = layers.Dense(dim, activation="relu")(x)
        x = layers.Dropout(dropout)(x)
    
    outputs = layers.Dense(1)(x)
    
    model = keras.Model(inputs, outputs)
    model.compile(optimizer='adam', loss='mse')
    return model

# Poziční kódování pro Transformer
class PositionalEncoding(layers.Layer):
    def __init__(self, max_len, d_model):
        super(PositionalEncoding, self).__init__()
        self.max_len = max_len
        self.d_model = d_model
    
    def build(self, input_shape):
        pe = np.zeros((self.max_len, self.d_model))
        position = np.arange(0, self.max_len)[:, np.newaxis]
        div_term = np.exp(np.arange(0, self.d_model, 2) * -(np.log(10000.0) / self.d_model))
        
        pe[:, 0::2] = np.sin(position * div_term)
        pe[:, 1::2] = np.cos(position * div_term)
        
        self.pe = tf.cast(pe[np.newaxis, ...], dtype=tf.float32)
        
    def call(self, inputs):
        return self.pe[:, :inputs.shape[1], :]

# Vytvoření a trénování Transformer modelu
transformer_model = build_transformer_model((X_train.shape[1], X_train.shape[2]))
print(transformer_model.summary())

# Trénování s předčasným zastavením

    X_train, y_train,
    epochs=100,
    batch_size=16,
    validation_split=0.2,

    verbose=1
)

# Vyhodnocení Transformer modelu

# Vykreslení predikcí Transformer modelu
```

## 8. Vícerozměrná a vícekroková predikce

## 9. Ladění hyperparametrů a hodnocení modelů

Efektivní ladění hyperparametrů a správné hodnocení modelů jsou nezbytné pro vývoj přesných modelů predikce časových řad. Techniky jako křížová validace musí být pro časové řady upraveny, aby se zabránilo úniku dat.

```python

import keras_tuner as kt
from sklearn.model_selection import TimeSeriesSplit

# Definice funkce pro vytvoření modelu s laditelnými hyperparametry
def build_tunable_lstm(hp):
    model = keras.Sequential()
    
    # Ladění počtu jednotek v LSTM vrstvě
    lstm_units = hp.Int('lstm_units', min_value=32, max_value=128, step=32)
    
    # Ladění míry dropout
    dropout_rate = hp.Float('dropout_rate', min_value=0.0, max_value=0.5, step=0.1)
    
    # Ladění počtu vrstev
    num_layers = hp.Int('num_layers', 1, 3)
    
    # První LSTM vrstva
    model.add(keras.layers.LSTM(
        units=lstm_units,
        return_sequences=num_layers > 1,
        input_shape=(X_train.shape[1], X_train.shape[2])
    ))
    model.add(keras.layers.Dropout(dropout_rate))
    
    # Další LSTM vrstvy
    for i in range(num_layers - 1):
        return_sequences = i < num_layers - 2
        model.add(keras.layers.LSTM(units=lstm_units, return_sequences=return_sequences))
        model.add(keras.layers.Dropout(dropout_rate))
    
    # Výstupní vrstva
    model.add(keras.layers.Dense(1))
    
    # Ladění rychlosti učení
    learning_rate = hp.Float('learning_rate', min_value=1e-4, max_value=1e-2, sampling='log')
    
    # Kompilace modelu
    model.compile(
        optimizer=keras.optimizers.Adam(learning_rate=learning_rate),
        loss='mse'
    )
    
    return model

# Pokročilé metriky hodnocení pro časové řady
def evaluate_forecast(y_true, y_pred):
    """Výpočet různých metrik hodnocení pro predikci časových řad."""
    # Root Mean Squared Error
    rmse = np.sqrt(mean_squared_error(y_true, y_pred))
    
    # Mean Absolute Error
    mae = mean_absolute_error(y_true, y_pred)
    
    # Mean Absolute Percentage Error
    mape = np.mean(np.abs((y_true - y_pred) / y_true)) * 100
    
    # Symmetric Mean Absolute Percentage Error
    smape = np.mean(200 * np.abs(y_pred - y_true) / (np.abs(y_pred) + np.abs(y_true)))
    
    # Koeficient determinace (R²)
    r2 = 1 - (np.sum((y_true - y_pred) ** 2) / np.sum((y_true - np.mean(y_true)) ** 2))
    
    return {
        'RMSE': rmse,
        'MAE': mae,
        'MAPE': mape,
        'SMAPE': smape,
        'R²': r2
    }
```

## 10. Shrnutí a osvědčené postupy

Hluboké učení významně pokročilo v možnostech predikce časových řad, nabízející výkonné nástroje pro zachycení komplexních časových vzorů. Zde je shrnutí klíčových bodů a osvědčených postupů:

1. **Předzpracování dat** je pro predikci časových řad zásadní:
   - Zajistěte stacionaritu nebo vhodně zpracujte nestacionaritu
   - Škálujte data pro zlepšení tréninku modelu
   - Vytvářejte vhodná sekvenční okna pro učení s učitelem
   - Navrhujte užitečné příznaky jako proměnné zpoždění a příznaky založené na datu

2. **Výběr modelu** závisí na specifických charakteristikách vaší časové řady:
   - **Jednoduché RNN**: Pro velmi krátké sekvence s omezenými časovými závislostmi
   - **LSTM/GRU**: Pro zachycení dlouhodobějších závislostí
   - **TCN**: Pro efektivní paralelní zpracování a přesné řízení velikosti receptivního pole
   - **Pozornost/Transformers**: Pro velmi dlouhé sekvence a zachycení komplexních vztahů

3. **Strategie tréninku**:
   - Používejte správnou křížovou validaci pro časové řady, abyste zabránili úniku dat
   - Aplikujte předčasné zastavení, abyste zabránili přeučení
   - Experimentujte s plánováním rychlosti učení pro lepší konvergenci
   - Upravujte velikost dávky podle velikosti vašeho datasetu

4. **Zpracování komplexních úloh predikce**:
   - Pro vícerozměrná data zajistěte správné škálování příznaků
   - Pro vícekrokovou predikci vybírejte mezi přímými a rekurzivními strategiemi
   - Zvažte metody souborů (ensemble methods) pro kombinování predikcí z více modelů

5. **Hlediska hodnocení**:
   - Používejte více metrik (RMSE, MAE, MAPE) pro komplexní hodnocení
   - Zvažte kritéria hodnocení specifická pro danou doménu
   - Vizualizujte predikce spolu se skutečnými hodnotami pro intuitivní posouzení

```python
# Příklad: Vytvoření praktické pipeline pro predikci časových řad
def time_series_forecasting_pipeline(data, target_column=None, sequence_length=10, 
                                    forecast_horizon=1, test_size=0.2, model_type='lstm'):
    """
    Kompletní pipeline pro predikci časových řad pomocí hlubokého učení.
    
    Args:
        data: DataFrame obsahující data časové řady
        target_column: Sloupec k predikci (None pro jednorozměrnou řadu)
        sequence_length: Počet časových kroků použitých pro predikci
        forecast_horizon: Počet kroků k predikci
        test_size: Podíl dat k použití pro testování
        model_type: Typ modelu k použití ('lstm', 'gru', 'tcn', 'transformer')
        
    Returns:
        Slovník obsahující model, historii, predikce a metriky hodnocení
    """
    # Příprava dat
    if target_column is not None:
        # Vícerozměrný případ
        values = data.values
    else:
        # Jednorozměrný případ
        values = data.values.reshape(-1, 1)
    
    # Škálování dat
    scaler = MinMaxScaler(feature_range=(0, 1))
    scaled_data = scaler.fit_transform(values)
    
    # Vytvoření sekvencí
    X, y = create_sequences(scaled_data, sequence_length, forecast_horizon)
    
    # Rozdělení dat
    split_idx = int(len(X) * (1 - test_size))
    X_train, X_test = X[:split_idx], X[split_idx:]
    y_train, y_test = y[:split_idx], y[split_idx:]
    
    # Přetvarování pro RNN, pokud je potřeba
    if len(X_train.shape) < 3:
        X_train = X_train.reshape(X_train.shape[0], X_train.shape[1], 1)
        X_test = X_test.reshape(X_test.shape[0], X_test.shape[1], 1)
    
    # Výběr typu modelu
    if model_type == 'lstm':
        model = build_lstm_model((X_train.shape[1], X_train.shape[2]))
    elif model_type == 'gru':
        model = build_gru_model((X_train.shape[1], X_train.shape[2]))
    elif model_type == 'tcn':
        model = build_tcn_model((X_train.shape[1], X_train.shape[2]))
    elif model_type == 'transformer':
        model = build_transformer_model((X_train.shape[1], X_train.shape[2]))
    else:
        raise ValueError(f"Neznámý typ modelu: {model_type}")
    
    # Předčasné zastavení
    early_stopping = keras.callbacks.EarlyStopping(
        monitor='val_loss', patience=10, restore_best_weights=True
    )
    
    # Trénování modelu
    history = model.fit(
        X_train, y_train,
        epochs=100,
        batch_size=16,
        validation_split=0.2,
        callbacks=[early_stopping],
        verbose=1
    )
    
    # Vytvoření predikcí
    y_pred = model.predict(X_test)
    
    # Příprava pro inverzní škálování
    if forecast_horizon == 1:
        # Přetvarování pro jednokrokové předpovědi
        if target_column is not None:
            # Pro vícerozměrné, musíme vytvořit celý řádek s nulami kromě cíle
            y_test_temp = np.zeros((y_test.shape[0], values.shape[1]))
            y_pred_temp = np.zeros((y_pred.shape[0], values.shape[1]))
            
            target_idx = data.columns.get_loc(target_column)
            y_test_temp[:, target_idx] = y_test.ravel()
            y_pred_temp[:, target_idx] = y_pred.ravel()
            
            y_test_inv = scaler.inverse_transform(y_test_temp)[:, target_idx]
            y_pred_inv = scaler.inverse_transform(y_pred_temp)[:, target_idx]
        else:
            # Pro jednorozměrné
            y_test_inv = scaler.inverse_transform(y_test.reshape(-1, 1)).ravel()
            y_pred_inv = scaler.inverse_transform(y_pred.reshape(-1, 1)).ravel()
    else:
        # Zpracování vícekrokových předpovědí obdobně (s více přetvářením)
        # Tato implementace by závisela na tom, jak je strukturována proměnná y
        pass
    
    # Vyhodnocení
    metrics = evaluate_forecast(y_test_inv, y_pred_inv)
    
    # Vrácení výsledků
    return {
        'model': model,
        'history': history,
        'predictions': y_pred_inv,
        'actual': y_test_inv,
        'metrics': metrics
    }
```
