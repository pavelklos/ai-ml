## Introduction

TensorFlow is an open-source machine learning framework developed by Google that provides a comprehensive ecosystem of tools for building and deploying deep learning models. Keras is a high-level neural networks API that runs on top of TensorFlow, designed to enable fast experimentation with deep neural networks.

TensorFlow 2.x integrates Keras as its official high-level API, providing an easy-to-use interface while maintaining the flexibility and power of TensorFlow. This combination offers:

- Simple, user-friendly APIs for model building
- Seamless GPU/TPU acceleration
- Efficient data pipelines
- Production-ready deployment options
- Rich ecosystem of tools and extensions

## TensorFlow Basics

TensorFlow's core concept is the tensor - a multi-dimensional array. Let's explore some basic tensor operations:

```python
# Creating tensors
scalar = tf.constant(5)
vector = tf.constant([1, 2, 3])
matrix = tf.constant([[1, 2], [3, 4]])
tensor_3d = tf.constant([[[1, 2], [3, 4]], [[5, 6], [7, 8]]])

print(f"Scalar shape: {scalar.shape}")
print(f"Vector shape: {vector.shape}")
print(f"Matrix shape: {matrix.shape}")
print(f"3D Tensor shape: {tensor_3d.shape}")

# Basic operations
a = tf.constant([[1, 2], [3, 4]])
b = tf.constant([[5, 6], [7, 8]])

# Element-wise addition
print(f"a + b = \n{a + b}")

# Matrix multiplication
print(f"a @ b = \n{a @ b}")  # or tf.matmul(a, b)

# Automatic differentiation with GradientTape
x = tf.Variable(3.0)
with tf.GradientTape() as tape:
    y = x**2  # y = x²
  
dy_dx = tape.gradient(y, x)
print(f"dy/dx at x=3: {dy_dx.numpy()}")  # Should be 2x = 6
```

## Building Models with Keras

Keras offers multiple ways to build models. The two main approaches are the Sequential API and the Functional API.

### Sequential API

The Sequential API is the simplest way to build a model in Keras:

```python
# Sequential model for MNIST classification
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),  # Convert 2D images to 1D vectors
    keras.layers.Dense(128, activation='relu'),   # Hidden layer with 128 neurons
    keras.layers.Dense(10)                        # Output layer with 10 neurons (one per digit)
])

# View model summary
model.summary()
```

### Functional API

The Functional API allows for more complex model architectures:

```python
# Functional API for a similar model
inputs = keras.Input(shape=(28, 28))
x = keras.layers.Flatten()(inputs)
x = keras.layers.Dense(128, activation='relu')(x)
outputs = keras.layers.Dense(10)(x)

# Create model by specifying inputs and outputs
model_functional = keras.Model(inputs=inputs, outputs=outputs)

# View model summary
model_functional.summary()
```

### Subclassing

For maximum flexibility, you can subclass the `keras.Model` class:

```python
class MyModel(keras.Model):
    def __init__(self):
        super(MyModel, self).__init__()
        self.flatten = keras.layers.Flatten()
        self.dense1 = keras.layers.Dense(128, activation='relu')
        self.dense2 = keras.layers.Dense(10)
        
    def call(self, inputs):
        x = self.flatten(inputs)
        x = self.dense1(x)
        return self.dense2(x)

# Create an instance of the model
model_subclassed = MyModel()

# Compile the model
model_subclassed.compile(
    optimizer='adam',
    loss=tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)
```

### Custom Layers

You can create custom layers by subclassing the `keras.layers.Layer` class:

```python
class MyCustomLayer(keras.layers.Layer):
    def __init__(self, units=32):
        super(MyCustomLayer, self).__init__()
        self.units = units
        
    def build(self, input_shape):
        # Create trainable weights
        self.w = self.add_weight(
            shape=(input_shape[-1], self.units),
            initializer='random_normal',
            trainable=True
        )
        self.b = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            trainable=True
        )
        
    def call(self, inputs):
        # Forward pass logic
        return tf.matmul(inputs, self.w) + self.b
    
    def get_config(self):
        # For serialization
        config = super(MyCustomLayer, self).get_config()
        config.update({'units': self.units})
        return config

# Use custom layer in a model
custom_model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    MyCustomLayer(128),
    keras.layers.Activation('relu'),
    keras.layers.Dense(10)
])

custom_model.summary()
```

## Training and Evaluation

Compiling and training Keras models is straightforward:

```python
# Prepare MNIST data for training
(x_train, y_train), (x_test, y_test) = mnist.load_data()
x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

# Reshape for CNN (add channel dimension)
x_train = x_train.reshape(-1, 28, 28, 1)
x_test = x_test.reshape(-1, 28, 28, 1)

# Compile the CNN model
cnn_model.compile(
    optimizer=keras.optimizers.Adam(learning_rate=0.001),
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Train the model with validation data
history = cnn_model.fit(
    x_train, y_train,
    batch_size=128,
    epochs=5,
    validation_split=0.1,
    verbose=1
)

# Evaluate on test data
test_loss, test_acc = cnn_model.evaluate(x_test, y_test, verbose=2)
print(f"\nTest accuracy: {test_acc:.4f}")

# Visualize training history
plt.figure(figsize=(12, 4))
```

### Using Callbacks

Keras callbacks provide a way to customize the training process:

```python
# Common callbacks
from tensorflow.keras.callbacks import (
    EarlyStopping, ModelCheckpoint, ReduceLROnPlateau, 
    TensorBoard, CSVLogger
)

# Configure callbacks
callbacks = [
    # Stop training when validation loss stops improving
    EarlyStopping(
        monitor='val_loss',
        patience=3,
        restore_best_weights=True
    ),
    
    # Save model checkpoints
    ModelCheckpoint(
        filepath='best_model.h5',
        monitor='val_accuracy',
        save_best_only=True
    ),
    
    # Reduce learning rate when plateauing
    ReduceLROnPlateau(
        monitor='val_loss',
        factor=0.1,
        patience=2
    ),
    
    # Log training metrics to CSV
    CSVLogger('training_log.csv')
]

# Train with callbacks
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

history = model.fit(
    x_train.reshape(-1, 28, 28), y_train,
    epochs=10,
    validation_split=0.1,
    callbacks=callbacks,
    verbose=1
)
```

## Saving and Loading Models

TensorFlow offers several options for saving and loading models:

```python
# 1. Save entire model (architecture + weights + optimizer state)
model.save('complete_model.h5')  # HDF5 format
model.save('complete_model')     # SavedModel format (directory)

# 2. Save only weights
model.save_weights('model_weights.h5')

# 3. Save only architecture (as JSON)
model_json = model.to_json()
with open('model_architecture.json', 'w') as f:
    f.write(model_json)

# Loading models
# 1. Load complete model
loaded_model = keras.models.load_model('complete_model.h5')

# 2. Load weights into existing model
model_new = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])
model_new.load_weights('model_weights.h5')

# 3. Load architecture from JSON and then weights
from tensorflow.keras.models import model_from_json

with open('model_architecture.json', 'r') as f:
    loaded_model_json = f.read()
    
reconstructed_model = model_from_json(loaded_model_json)
reconstructed_model.load_weights('model_weights.h5')
```

### TensorFlow SavedModel Format

The SavedModel format is the recommended way to save models for production:

```python
# Save in SavedModel format
model.save('saved_model_dir')

# Load from SavedModel format
loaded_model = tf.keras.models.load_model('saved_model_dir')

# Convert to TensorFlow Lite format
converter = tf.lite.TFLiteConverter.from_saved_model('saved_model_dir')
tflite_model = converter.convert()

# Save TFLite model to file
with open('model.tflite', 'wb') as f:
    f.write(tflite_model)
```

## TensorFlow Datasets

The `tf.data` API provides efficient data loading and preprocessing pipelines:

```python
# Create a dataset from tensors
dataset = tf.data.Dataset.from_tensor_slices((x_train, y_train))

# Basic dataset operations
dataset = dataset.shuffle(buffer_size=10000)  # Shuffle data
dataset = dataset.batch(32)                   # Create batches of 32
dataset = dataset.prefetch(tf.data.AUTOTUNE)  # Prefetch for performance

# Data transformations with map
def preprocess(image, label):
    image = tf.cast(image, tf.float32) / 255.0  # Normalize
    # Add data augmentation
    image = tf.image.random_flip_left_right(image)
    return image, label

dataset = dataset.map(preprocess, num_parallel_calls=tf.data.AUTOTUNE)

# Training with tf.data
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28, 1)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

model.fit(
    dataset,
    epochs=5,
    verbose=1
)
```

## Practical Example: Building a Classifier

Let's build a complete classifier for the Fashion MNIST dataset:

```python
# Load Fashion MNIST dataset
from tensorflow.keras.datasets import fashion_mnist

# Class names for reference
class_names = ['T-shirt/top', 'Trouser', 'Pullover', 'Dress', 'Coat',
               'Sandal', 'Shirt', 'Sneaker', 'Bag', 'Ankle boot']

# Load and preprocess data
(x_train, y_train), (x_test, y_test) = fashion_mnist.load_data()
x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

# Visualize some examples
plt.figure(figsize=(10, 10))
for i in range(25):
    plt.subplot(5, 5, i+1)
    plt.imshow(x_train[i], cmap='gray')
    plt.title(class_names[y_train[i]])
    plt.axis('off')
plt.tight_layout()
plt.show()

# Create data pipeline with tf.data
train_dataset = tf.data.Dataset.from_tensor_slices((x_train, y_train))
train_dataset = train_dataset.shuffle(buffer_size=60000)
train_dataset = train_dataset.batch(64).prefetch(tf.data.AUTOTUNE)

test_dataset = tf.data.Dataset.from_tensor_slices((x_test, y_test))
test_dataset = test_dataset.batch(64).prefetch(tf.data.AUTOTUNE)

# Build a CNN model
model = keras.Sequential([
    # Reshape from (28, 28) to (28, 28, 1)
    keras.layers.Reshape((28, 28, 1), input_shape=(28, 28)),
    
    # First convolutional block
    keras.layers.Conv2D(32, (3, 3), activation='relu', padding='same'),
    keras.layers.BatchNormalization(),
    keras.layers.Conv2D(32, (3, 3), activation='relu', padding='same'),
    keras.layers.MaxPooling2D((2, 2)),
    keras.layers.Dropout(0.25),
    
    # Second convolutional block
    keras.layers.Conv2D(64, (3, 3), activation='relu', padding='same'),
    keras.layers.BatchNormalization(),
    keras.layers.Conv2D(64, (3, 3), activation='relu', padding='same'),
    keras.layers.MaxPooling2D((2, 2)),
    keras.layers.Dropout(0.25),
    
    # Flatten and dense layers
    keras.layers.Flatten(),
    keras.layers.Dense(512, activation='relu'),
    keras.layers.BatchNormalization(),
    keras.layers.Dropout(0.5),
    keras.layers.Dense(10)  # 10 classes
])

# Compile model with learning rate scheduling
lr_schedule = keras.optimizers.schedules.ExponentialDecay(
    initial_learning_rate=1e-3,
    decay_steps=10000,
    decay_rate=0.9
)
optimizer = keras.optimizers.Adam(learning_rate=lr_schedule)

model.compile(
    optimizer=optimizer,
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Train with callbacks
callbacks = [
    keras.callbacks.EarlyStopping(patience=5, restore_best_weights=True),
    keras.callbacks.ModelCheckpoint(
        'fashion_mnist_best.h5',
        save_best_only=True
    ),
    keras.callbacks.TensorBoard(log_dir='./logs')
]

history = model.fit(
    train_dataset,
    epochs=25,
    validation_data=test_dataset,
    callbacks=callbacks
)

# Evaluate final model
test_loss, test_acc = model.evaluate(test_dataset)
print(f"Test accuracy: {test_acc:.4f}")

# Make predictions and visualize results
predictions = model.predict(test_dataset)
predicted_labels = np.argmax(predictions, axis=1)

# Plot some predictions
plt.figure(figsize=(12, 12))
for i in range(25):
    plt.subplot(5, 5, i+1)
    plt.imshow(x_test[i], cmap='gray')
    predicted_class = class_names[predicted_labels[i]]
    actual_class = class_names[y_test[i]]
    color = 'green' if predicted_labels[i] == y_test[i] else 'red'
    plt.title(f"Pred: {predicted_class}\nActual: {actual_class}", 
              color=color, fontsize=9)
    plt.axis('off')
plt.tight_layout()
plt.show()

# Plot confusion matrix
from sklearn.metrics import confusion_matrix
import seaborn as sns

cm = confusion_matrix(y_test, predicted_labels)
plt.figure(figsize=(10, 8))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=class_names,
            yticklabels=class_names)
plt.xlabel('Predicted')
plt.ylabel('True')
plt.title('Confusion Matrix')
plt.show()
```

## Custom Training Loops

For advanced use cases, you can create custom training loops with the TensorFlow GradientTape API:

```python
# Define model
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

# Loss function and optimizer
loss_fn = keras.losses.SparseCategoricalCrossentropy(from_logits=True)
optimizer = keras.optimizers.Adam(learning_rate=0.001)

# Metrics
train_acc_metric = keras.metrics.SparseCategoricalAccuracy()
val_acc_metric = keras.metrics.SparseCategoricalAccuracy()

# Custom training loop
@tf.function  # Compile for performance
def train_step(x, y):
    with tf.GradientTape() as tape:
        # Forward pass
        logits = model(x, training=True)
        # Calculate loss
        loss_value = loss_fn(y, logits)
        
    # Calculate gradients
    grads = tape.gradient(loss_value, model.trainable_weights)
    # Apply gradients
    optimizer.apply_gradients(zip(grads, model.trainable_weights))
    
    # Update metrics
    train_acc_metric.update_state(y, logits)
    
    return loss_value

@tf.function
def val_step(x, y):
    # Forward pass (no training)
    val_logits = model(x, training=False)
    # Update metrics
    val_acc_metric.update_state(y, val_logits)

# Training loop
epochs = 5
batch_size = 64

# Create datasets
train_dataset = tf.data.Dataset.from_tensor_slices((x_train.reshape(-1, 28, 28), y_train))
train_dataset = train_dataset.shuffle(buffer_size=60000).batch(batch_size)

val_dataset = tf.data.Dataset.from_tensor_slices((x_test.reshape(-1, 28, 28), y_test))
val_dataset = val_dataset.batch(batch_size)

# Training loop
for epoch in range(epochs):
    print(f"\nStart of epoch {epoch + 1}")
    
    # Training
    train_acc_metric.reset_states()
    total_loss = 0
    num_batches = 0
    
    for step, (x_batch, y_batch) in enumerate(train_dataset):
        loss_value = train_step(x_batch, y_batch)
        total_loss += loss_value
        num_batches += 1
        
        if step % 200 == 0:
            print(f"  Training - Step {step}: Loss = {loss_value:.4f}")
    
    # Calculate average training loss
    avg_loss = total_loss / num_batches
    
    # Validation
    val_acc_metric.reset_states()
    for x_batch, y_batch in val_dataset:
        val_step(x_batch, y_batch)
    
    # Print metrics
    train_acc = train_acc_metric.result()
    val_acc = val_acc_metric.result()
    
    print(f"Epoch {epoch + 1} - "
          f"Loss: {avg_loss:.4f}, "
          f"Training Accuracy: {train_acc:.4f}, "
          f"Validation Accuracy: {val_acc:.4f}")
```

## TensorBoard for Visualization

TensorBoard is a visualization tool for TensorFlow that helps monitor training:

```python
# TensorBoard setup
import datetime

# Create log directory with timestamp
log_dir = "logs/fit/" + datetime.datetime.now().strftime("%Y%m%d-%H%M%S")

# Create TensorBoard callback
tensorboard_callback = tf.keras.callbacks.TensorBoard(
    log_dir=log_dir,
    histogram_freq=1,  # Log histogram of weights
    update_freq='epoch'  # Update every epoch
)

# Define a model
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu', name='dense_1'),
    keras.layers.Dropout(0.2, name='dropout'),
    keras.layers.Dense(10, name='dense_2')
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Train with TensorBoard callback
model.fit(
    x_train.reshape(-1, 28, 28), y_train,
    epochs=5,
    validation_split=0.1,
    callbacks=[tensorboard_callback]
)

# How to launch TensorBoard
# In terminal/command prompt:
# tensorboard --logdir logs/fit
```
