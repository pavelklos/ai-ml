## Introduction to PyTorch

PyTorch is an open-source deep learning framework developed by Facebook's AI Research lab (FAIR). It provides a flexible and intuitive interface for building and training neural networks, with strong GPU acceleration support. PyTorch is known for its:

- **Dynamic computation graph** - Unlike static graph frameworks, PyTorch builds the computational graph on-the-fly during execution, making it more intuitive for debugging and prototyping
- **Python integration** - PyTorch feels like a natural extension of Python
- **Strong research community** - Widely adopted in academic and research settings
- **Production readiness** - With TorchScript and integrations for deployment

## PyTorch Tensors

Tensors are the fundamental data structure in PyTorch, similar to NumPy arrays but with additional capabilities like GPU support and automatic differentiation.

### Creating Tensors

```python
import torch
import numpy as np

# Create a tensor directly from data
x = torch.tensor([1, 2, 3, 4])
print(f"Tensor from data: {x}")

# Create tensor from numpy array
np_array = np.array([1, 2, 3, 4])
x_np = torch.from_numpy(np_array)
print(f"Tensor from NumPy: {x_np}")

# Create tensors with specific properties
zeros = torch.zeros(3, 4)  # 3x4 tensor of zeros
ones = torch.ones(2, 3, 4)  # 2x3x4 tensor of ones
rand = torch.rand(2, 3)  # 2x3 tensor with random values from uniform distribution [0,1)
randn = torch.randn(2, 3)  # 2x3 tensor with random values from standard normal distribution

print(f"Zeros tensor shape: {zeros.shape}")
print(f"Ones tensor shape: {ones.shape}")
print(f"Random tensor: \n{rand}")
```

### Tensor Operations

```python
# Basic operations
a = torch.tensor([1, 2, 3])
b = torch.tensor([4, 5, 6])

print(f"a + b: {a + b}")  # Element-wise addition
print(f"a * b: {a * b}")  # Element-wise multiplication
print(f"a @ b: {a @ b}")  # Dot product (here, it's the same as torch.dot(a, b))

# Matrix operations
m1 = torch.tensor([[1, 2], [3, 4]])
m2 = torch.tensor([[5, 6], [7, 8]])
print(f"Matrix multiplication: \n{torch.matmul(m1, m2)}")

# Reshaping tensors
x = torch.tensor([[1, 2], [3, 4], [5, 6]])
print(f"Original shape: {x.shape}")

x_reshaped = x.reshape(2, 3)
print(f"Reshaped: \n{x_reshaped}")
print(f"New shape: {x_reshaped.shape}")

x_transposed = x.t()  # Transpose
print(f"Transposed: \n{x_transposed}")
print(f"Transposed shape: {x_transposed.shape}")

# Indexing and slicing (similar to NumPy)
print(f"First row: {x[0]}")
print(f"First column: {x[:, 0]}")
print(f"Sub-matrix: \n{x[0:2, :]}")
```

## Autograd - Automatic Differentiation

One of PyTorch's key features is automatic differentiation through its `autograd` package. This allows automatic computation of gradients for all operations on tensors.

```python
# Create tensors with gradient tracking
x = torch.tensor([2.0], requires_grad=True)
y = torch.tensor([3.0], requires_grad=True)

# Perform operations
z = x**2 + y**3

# Compute gradients
z.backward()

# Display gradients (dz/dx and dz/dy)
print(f"dz/dx: {x.grad}")  # Should be 2*x = 4
print(f"dz/dy: {y.grad}")  # Should be 3*y^2 = 27

# Example with vector output (sum is used to get scalar output)
x = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)
y = x * 2
z = y.sum()
z.backward()
print(f"x: {x}")
print(f"x.grad: {x.grad}")  # Should be [2.0, 2.0, 2.0]
```

### Stopping Gradient Tracking

```python
# Temporarily stop gradient tracking
x = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)

with torch.no_grad():
    y = x * 2
    
print(f"y.requires_grad: {y.requires_grad}")  # Should be False

# Detach a tensor from its computation history
y = x * 2
z = y.detach()
print(f"z.requires_grad: {z.requires_grad}")  # Should be False
```

## Building Neural Networks

PyTorch provides the `nn` module for building neural networks, which contains common layers, activation functions, and other components.

### Using nn.Module

```python
import torch.nn as nn
import torch.nn.functional as F

# Define a simple neural network
class SimpleNN(nn.Module):
    def __init__(self):
        super(SimpleNN, self).__init__()
        # Define layers
        self.fc1 = nn.Linear(10, 50)  # Input: 10, Output: 50
        self.fc2 = nn.Linear(50, 20)  # Input: 50, Output: 20
        self.fc3 = nn.Linear(20, 2)   # Input: 20, Output: 2
        
    def forward(self, x):
        # Define forward pass
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Create an instance of the model
model = SimpleNN()
print(model)

# Generate random input and perform a forward pass
input_data = torch.randn(5, 10)  # 5 samples, 10 features each
output = model(input_data)
print(f"Input shape: {input_data.shape}")
print(f"Output shape: {output.shape}")
```

### Layers and Activation Functions

```python
# Common layers in PyTorch
linear = nn.Linear(10, 5)  # Fully connected layer
conv = nn.Conv2d(3, 16, kernel_size=3, padding=1)  # 2D Convolutional layer
pool = nn.MaxPool2d(2)  # Max pooling
dropout = nn.Dropout(0.5)  # Dropout for regularization
batch_norm = nn.BatchNorm1d(10)  # Batch normalization

# Common activation functions
x = torch.randn(10)
relu_out = F.relu(x)
sigmoid_out = torch.sigmoid(x)
tanh_out = torch.tanh(x)
softmax_out = F.softmax(x, dim=0)

print(f"ReLU output: {relu_out}")
print(f"Sigmoid output: {sigmoid_out}")
print(f"Softmax output: {softmax_out}")
```

## Loss Functions and Optimizers

PyTorch provides various loss functions and optimization algorithms for training neural networks.

### Common Loss Functions

```python
import torch.nn as nn

# Binary Cross Entropy Loss
bce_loss = nn.BCELoss()
inputs = torch.randn(3, requires_grad=True)
targets = torch.empty(3).random_(2)  # Binary targets (0 or 1)
loss = bce_loss(torch.sigmoid(inputs), targets)
print(f"BCE Loss: {loss.item()}")

# Cross Entropy Loss (for classification)
ce_loss = nn.CrossEntropyLoss()
inputs = torch.randn(3, 5, requires_grad=True)  # 3 samples, 5 classes
targets = torch.empty(3, dtype=torch.long).random_(5)  # Class indices
loss = ce_loss(inputs, targets)
print(f"CrossEntropy Loss: {loss.item()}")

# Mean Squared Error (for regression)
mse_loss = nn.MSELoss()
inputs = torch.randn(3, requires_grad=True)
targets = torch.randn(3)
loss = mse_loss(inputs, targets)
print(f"MSE Loss: {loss.item()}")
```

### Optimizers

```python
import torch.optim as optim

# Define a simple model
model = SimpleNN()

# Stochastic Gradient Descent (SGD)
optimizer_sgd = optim.SGD(model.parameters(), lr=0.01, momentum=0.9)

# Adam optimizer
optimizer_adam = optim.Adam(model.parameters(), lr=0.001, betas=(0.9, 0.999))

# RMSprop
optimizer_rmsprop = optim.RMSprop(model.parameters(), lr=0.01)

# Using an optimizer
optimizer = optim.SGD(model.parameters(), lr=0.01)
# Zero the parameter gradients
optimizer.zero_grad()
# Forward pass with random data
outputs = model(torch.randn(5, 10))
# Compute loss
loss = torch.sum(outputs)
# Backward pass
loss.backward()
# Update weights
optimizer.step()
```

## Basic Model Training

Let's put everything together to train a simple model on a synthetic dataset.

```python
import torch
import torch.nn as nn
import torch.optim as optim
import matplotlib.pyplot as plt

# Generate synthetic data
X = torch.linspace(-1, 1, 100).reshape(-1, 1)
y = 3 * X.pow(2) + 2 + 0.2 * torch.randn(X.size())

# Visualize data
plt.scatter(X.numpy(), y.numpy())
plt.title("Synthetic Regression Dataset")
plt.xlabel("X")
plt.ylabel("y")
plt.show()

# Define model
class RegressionModel(nn.Module):
    def __init__(self):
        super(RegressionModel, self).__init__()
        self.fc1 = nn.Linear(1, 10)
        self.fc2 = nn.Linear(10, 10)
        self.fc3 = nn.Linear(10, 1)
        
    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Create model, loss function, and optimizer
model = RegressionModel()
criterion = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)

# Training loop
epochs = 200
losses = []

for epoch in range(epochs):
    # Forward pass
    y_pred = model(X)
    loss = criterion(y_pred, y)
    losses.append(loss.item())
    
    # Backward pass and optimization
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    
    # Print progress
    if (epoch + 1) % 20 == 0:
        print(f'Epoch {epoch+1}/{epochs}, Loss: {loss.item():.4f}')

# Plot loss curve
plt.plot(losses)
plt.title("Training Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.show()

# Visualize predictions
with torch.no_grad():
    y_pred = model(X)
    
plt.scatter(X.numpy(), y.numpy(), label='Data')
plt.plot(X.numpy(), y_pred.numpy(), 'r-', label='Predictions')
plt.title("Predictions vs Actual")
plt.xlabel("X")
plt.ylabel("y")
plt.legend()
plt.show()
```

## Saving and Loading Models

PyTorch allows you to save and load trained models, which is essential for deployment or resuming training.

```python
# Saving a model
model_path = "model.pth"

# Method 1: Save the entire model
torch.save(model, model_path)
# To load the entire model
loaded_model = torch.load(model_path)

# Method 2: Save only the model parameters (recommended)
torch.save(model.state_dict(), "model_state_dict.pth")
# To load the model parameters
new_model = RegressionModel()
new_model.load_state_dict(torch.load("model_state_dict.pth"))
new_model.eval()  # Set to evaluation mode

# Saving and loading checkpoints (for resuming training)
checkpoint = {
    'epoch': epoch,
    'model_state_dict': model.state_dict(),
    'optimizer_state_dict': optimizer.state_dict(),
    'loss': loss
}
torch.save(checkpoint, "checkpoint.pth")

# To load checkpoint
checkpoint = torch.load("checkpoint.pth")
model.load_state_dict(checkpoint['model_state_dict'])
optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
epoch = checkpoint['epoch']
loss = checkpoint['loss']
```

## Using GPU Acceleration

One of the main advantages of PyTorch is the ability to use GPU acceleration for faster training.

```python
# Check if GPU is available
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Using device: {device}")

# Create a model and move it to GPU
model = RegressionModel().to(device)

# Move data to GPU
X_gpu = X.to(device)
y_gpu = y.to(device)

# Training loop with GPU
for epoch in range(10):
    # Forward pass
    y_pred = model(X_gpu)
    loss = criterion(y_pred, y_gpu)
    
    # Backward pass and optimization
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    
    print(f'Epoch {epoch+1}/10, Loss: {loss.item():.4f}')

# Important: When working with tensors from GPU, remember to move them back to CPU for numpy operations
with torch.no_grad():
    y_pred_gpu = model(X_gpu)
    y_pred_cpu = y_pred_gpu.cpu().numpy()
```

## Practical Example

Let's create a more realistic example by training a classifier on the MNIST dataset.

```python
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt

# Device configuration
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# Hyperparameters
batch_size = 100
learning_rate = 0.001
num_epochs = 5

# MNIST dataset
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])

train_dataset = torchvision.datasets.MNIST(root='./data', 
                                          train=True, 
                                          transform=transform,
                                          download=True)

test_dataset = torchvision.datasets.MNIST(root='./data', 
                                         train=False, 
                                         transform=transform)

train_loader = DataLoader(dataset=train_dataset, 
                         batch_size=batch_size, 
                         shuffle=True)

test_loader = DataLoader(dataset=test_dataset, 
                        batch_size=batch_size, 
                        shuffle=False)

# Visualize some examples
examples = iter(train_loader)
samples, labels = next(examples)
print(f"Sample shape: {samples.shape}")

plt.figure(figsize=(10, 5))
for i in range(10):
    plt.subplot(2, 5, i + 1)
    plt.imshow(samples[i][0], cmap='gray')
    plt.title(f'Label: {labels[i].item()}')
    plt.axis('off')
plt.tight_layout()
plt.show()

# Convolutional Neural Network model
class ConvNet(nn.Module):
    def __init__(self):
        super(ConvNet, self).__init__()
        self.layer1 = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=5, padding=2),
            nn.BatchNorm2d(16),
            nn.ReLU(),
            nn.MaxPool2d(2)
        )
        self.layer2 = nn.Sequential(
            nn.Conv2d(16, 32, kernel_size=5, padding=2),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(2)
        )
        self.fc = nn.Linear(7*7*32, 10)
        
    def forward(self, x):
        out = self.layer1(x)
        out = self.layer2(out)
        out = out.view(out.size(0), -1)  # Flatten
        out = self.fc(out)
        return out

# Initialize model
model = ConvNet().to(device)

# Loss and optimizer
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Training function
def train(model, train_loader, criterion, optimizer, num_epochs):
    total_step = len(train_loader)
    
    for epoch in range(num_epochs):
        for i, (images, labels) in enumerate(train_loader):
            # Move tensors to device
            images = images.to(device)
            labels = labels.to(device)
            
            # Forward pass
            outputs = model(images)
            loss = criterion(outputs, labels)
            
            # Backward and optimize
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            if (i+1) % 100 == 0:
                print(f'Epoch [{epoch+1}/{num_epochs}], Step [{i+1}/{total_step}], Loss: {loss.item():.4f}')

# Evaluation function
def evaluate(model, test_loader):
    model.eval()
    with torch.no_grad():
        correct = 0
        total = 0
        for images, labels in test_loader:
            images = images.to(device)
            labels = labels.to(device)
            outputs = model(images)
            _, predicted = torch.max(outputs.data, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
        
        print(f'Accuracy on test images: {100 * correct / total:.2f}%')

# Train the model
train(model, train_loader, criterion, optimizer, num_epochs)

# Test the model
evaluate(model, test_loader)

# Save the model
torch.save(model.state_dict(), 'mnist_model.pth')
```
