## 1. Introduction to Reinforcement Learning

Reinforcement Learning (RL) is a machine learning paradigm focused on decision making and optimal behavior in interactive environments. Unlike supervised learning, an agent doesn't have access to correct answers but learns from feedback through environment interaction.

```python
# Basic concepts of reinforcement learning
import numpy as np
import gym

# Create a simple environment
env = gym.make('CartPole-v1')

# Core components of reinforcement learning
observation_space = env.observation_space  # State space
action_space = env.action_space            # Action space
rewards = []                               # Rewards

# Simple environment walkthrough with random actions
observation = env.reset()
done = False
total_reward = 0

while not done:
    action = env.action_space.sample()  # Random action
    observation, reward, done, info = env.step(action)
    total_reward += reward

print(f"Total reward with random behavior: {total_reward}")
```

### Key Components of Reinforcement Learning:

1. **Agent**: The entity that learns and makes decisions
2. **Environment**: The world with which the agent interacts
3. **State**: Information about the current situation
4. **Action**: Possible decisions by the agent
5. **Reward**: Feedback from the environment
6. **Policy**: Strategy by which the agent selects actions
7. **Value Function**: Estimate of future rewards

## 2. Fundamentals of Deep Neural Networks for RL

Deep neural networks serve as function approximators in reinforcement learning to estimate value functions or policies, especially when dealing with high-dimensional state spaces like images.

```python
import torch
import torch.nn as nn
import torch.nn.functional as F

class SimpleDQN(nn.Module):
    """A simple deep Q-network"""
    def __init__(self, input_size, output_size):
        super(SimpleDQN, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, 128)
        self.fc3 = nn.Linear(128, output_size)
        
    def forward(self, x):
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.fc3(x)  # Q-values for each action

# Example usage
state_size = 4    # For CartPole environment
action_size = 2   # For CartPole environment
model = SimpleDQN(state_size, action_size)

# Process a sample state
sample_state = torch.FloatTensor(np.random.rand(1, state_size))
q_values = model(sample_state)
print(f"Q-values for sample state: {q_values}")
```

Key considerations when using neural networks for RL:

- **Non-stationarity**: Training data distribution changes as the agent learns
- **Sample efficiency**: Maximizing learning from limited environment interactions
- **Credit assignment**: Determining which actions led to delayed rewards
- **Exploration-exploitation tradeoff**: Balancing discovery with exploitation of known rewards

## 3. Q-Learning and Deep Q-Networks (DQN)

Q-Learning is a value-based RL algorithm that learns the value of actions in different states. Deep Q-Networks extend this by using neural networks to approximate the Q-function.

```python
import torch.optim as optim
from collections import deque
import random

class DQNAgent:
    def __init__(self, state_size, action_size):
        self.state_size = state_size
        self.action_size = action_size
        self.memory = deque(maxlen=10000)  # Experience replay buffer
        
        self.gamma = 0.99    # Discount factor
        self.epsilon = 1.0   # Exploration rate
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
        
        # Q-Networks
        self.model = SimpleDQN(state_size, action_size)
        self.target_model = SimpleDQN(state_size, action_size)
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)
        
        # Update target network
        self.update_target_model()
        
    def update_target_model(self):
        """Copy weights from model to target_model"""
        self.target_model.load_state_dict(self.model.state_dict())
        
    def remember(self, state, action, reward, next_state, done):
        """Store experience in replay memory"""
        self.memory.append((state, action, reward, next_state, done))
        
    def act(self, state):
        """Select action using epsilon-greedy policy"""
        if np.random.rand() <= self.epsilon:
            return random.randrange(self.action_size)  # Exploration
            
        # Exploitation: choose best action based on Q-values
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        with torch.no_grad():
            q_values = self.model(state_tensor)
        return q_values.argmax().item()
    
    def replay(self, batch_size):
        """Train on random minibatch from replay memory"""
        if len(self.memory) < batch_size:
            return
            
        # Sample random minibatch
        minibatch = random.sample(self.memory, batch_size)
        
        for state, action, reward, next_state, done in minibatch:
            target = reward
            state_tensor = torch.FloatTensor(state).unsqueeze(0)
            next_state_tensor = torch.FloatTensor(next_state).unsqueeze(0)
            
            if not done:
                # Compute target Q-value using target network (Double DQN)
                with torch.no_grad():
                    next_q_values = self.target_model(next_state_tensor)
                target = reward + self.gamma * next_q_values.max().item()
                
            # Update Q-value for the action taken
            current_q = self.model(state_tensor)
            target_q = current_q.clone().detach()
            target_q[0][action] = target
            
            # Train the model
            self.optimizer.zero_grad()
            loss = F.mse_loss(current_q, target_q)
            loss.backward()
            self.optimizer.step()
            
        # Decay exploration rate
        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay
```

### Key DQN Innovations:

1. **Experience Replay**: Storing and randomly sampling past experiences to break correlations in sequential data
2. **Target Network**: Using a separate network for target value computation to improve training stability
3. **Double DQN**: Decoupling action selection from evaluation to reduce overestimation bias
4. **Dueling DQN**: Separately estimating state value and action advantages

## 4. Policy Gradient Methods

Policy gradient methods directly optimize the policy without learning a value function. They work by updating policy parameters in the direction of greater expected reward.

```python
class PolicyNetwork(nn.Module):
    """Simple policy network for policy gradient methods"""
    def __init__(self, input_size, output_size):
        super(PolicyNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 128)
        self.fc2 = nn.Linear(128, output_size)
        
    def forward(self, x):
        x = F.relu(self.fc1(x))
        # Output probability distribution over actions
        return F.softmax(self.fc2(x), dim=1)

class REINFORCE:
    """REINFORCE (Monte Carlo Policy Gradient) algorithm"""
    def __init__(self, state_size, action_size):
        self.policy = PolicyNetwork(state_size, action_size)
        self.optimizer = optim.Adam(self.policy.parameters(), lr=0.01)
        self.gamma = 0.99
        
    def select_action(self, state):
        """Sample action from policy"""
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        action_probs = self.policy(state_tensor)
        action_distribution = torch.distributions.Categorical(action_probs)
        action = action_distribution.sample()
        log_prob = action_distribution.log_prob(action)
        return action.item(), log_prob
    
    def update_policy(self, rewards, log_probs):
        """Update policy parameters using policy gradient"""
        # Calculate discounted returns
        returns = []
        discounted_return = 0
        
        # Calculate returns in reverse order
        for reward in reversed(rewards):
            discounted_return = reward + self.gamma * discounted_return
            returns.insert(0, discounted_return)
            
        returns = torch.FloatTensor(returns)
        
        # Normalize returns for training stability
        returns = (returns - returns.mean()) / (returns.std() + 1e-9)
        
        # Calculate policy loss
        policy_loss = []
        for log_prob, R in zip(log_probs, returns):
            # Negative because we're performing gradient ascent
            policy_loss.append(-log_prob * R)
        
        # Backpropagation
        self.optimizer.zero_grad()
        policy_loss = torch.cat(policy_loss).sum()
        policy_loss.backward()
        self.optimizer.step()
```

### Advantages of Policy Gradient Methods:

1. **Direct optimization**: Directly optimizes the policy for the objective
2. **Stochastic policies**: Naturally handles stochastic policies, important for exploration
3. **Continuous actions**: Works well with continuous action spaces
4. **Simple update rule**: Conceptually straightforward gradient updates

## 5. Actor-Critic Architectures

Actor-Critic methods combine value-based and policy-based approaches. The actor (policy network) decides which actions to take, while the critic (value network) evaluates those actions.

```python
class ActorCritic(nn.Module):
    """Combined actor-critic network"""
    def __init__(self, state_size, action_size):
        super(ActorCritic, self).__init__()
        
        # Common feature extraction layers
        self.fc1 = nn.Linear(state_size, 128)
        
        # Actor head (policy)
        self.actor = nn.Linear(128, action_size)
        
        # Critic head (value function)
        self.critic = nn.Linear(128, 1)
    
    def forward(self, x):
        """Forward pass through the network"""
        features = F.relu(self.fc1(x))
        
        # Actor: Action probability distribution
        action_probs = F.softmax(self.actor(features), dim=-1)
        
        # Critic: State value estimate
        state_value = self.critic(features)
        
        return action_probs, state_value
    
    def act(self, state):
        """Sample action from current policy"""
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        action_probs, _ = self(state_tensor)
        
        # Create a distribution and sample
        dist = torch.distributions.Categorical(action_probs)
        action = dist.sample()
        
        return action.item(), dist.log_prob(action), action_probs

class A2CAgent:
    """Advantage Actor-Critic (A2C) agent"""
    def __init__(self, state_size, action_size):
        self.model = ActorCritic(state_size, action_size)
        self.optimizer = optim.Adam(self.model.parameters(), lr=0.001)
        self.gamma = 0.99
        
    def update(self, states, actions, rewards, next_states, dones):
        """Update actor and critic networks"""
        # Convert lists to tensors
        states = torch.FloatTensor(states)
        actions = torch.LongTensor(actions)
        rewards = torch.FloatTensor(rewards)
        next_states = torch.FloatTensor(next_states)
        dones = torch.FloatTensor(dones)
        
        # Get action probabilities and state values
        _, next_state_values = self.model(next_states)
        action_probs, state_values = self.model(states)
        
        # Index selected actions
        dist = torch.distributions.Categorical(action_probs)
        log_probs = dist.log_prob(actions)
        entropy = dist.entropy()  # For exploration bonus
        
        # Calculate advantages
        next_state_values[dones] = 0  # Terminal states have zero future reward
        advantages = rewards + self.gamma * next_state_values.detach() - state_values
        
        # Calculate losses
        critic_loss = advantages.pow(2).mean()  # MSE
        actor_loss = -(log_probs * advantages.detach()).mean()  # Policy gradient
        entropy_loss = -0.01 * entropy.mean()  # Exploration bonus
        
        # Combined loss
        loss = actor_loss + critic_loss + entropy_loss
        
        # Update networks
        self.optimizer.zero_grad()
        loss.backward()
        self.optimizer.step()
        
        return loss.item()
```

### Actor-Critic Benefits:

1. **Reduced variance**: Using a critic reduces the variance of policy gradient updates
2. **Sample efficiency**: More efficient use of experience data
3. **Online learning**: Can learn from individual transitions without full episodes
4. **Combined strengths**: Leverages benefits of both value-based and policy-based methods

## 6. DQN Implementation in Python

Here's a complete implementation of a Deep Q-Network agent training on the CartPole environment:

```python
import gym
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
from collections import deque
import random
import matplotlib.pyplot as plt

# Deep Q-Network model
class DQN(nn.Module):
    def __init__(self, state_size, action_size):
        super(DQN, self).__init__()
        self.fc1 = nn.Linear(state_size, 64)
        self.fc2 = nn.Linear(64, 64)
        self.fc3 = nn.Linear(64, action_size)
        
    def forward(self, x):
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        return self.fc3(x)

# DQN Agent with experience replay and target network
class DQNAgent:
    def __init__(self, state_size, action_size):
        self.state_size = state_size
        self.action_size = action_size
        self.memory = deque(maxlen=2000)
        self.batch_size = 64
        
        # Hyperparameters
        self.gamma = 0.99        # Discount factor
        self.epsilon = 1.0       # Exploration rate
        self.epsilon_min = 0.01
        self.epsilon_decay = 0.995
        self.tau = 0.001         # Soft update parameter
        
        # Networks
        self.policy_net = DQN(state_size, action_size)
        self.target_net = DQN(state_size, action_size)
        self.target_net.load_state_dict(self.policy_net.state_dict())
        self.target_net.eval()  # Set target network to evaluation mode
        
        self.optimizer = optim.Adam(self.policy_net.parameters(), lr=0.001)
        
    def remember(self, state, action, reward, next_state, done):
        """Store transition in replay memory"""
        self.memory.append((state, action, reward, next_state, done))
        
    def act(self, state):
        """Select action using epsilon-greedy policy"""
        if random.random() <= self.epsilon:
            return random.randrange(self.action_size)
        else:
            state_tensor = torch.FloatTensor(state).unsqueeze(0)
            with torch.no_grad():
                q_values = self.policy_net(state_tensor)
            return q_values.argmax().item()
        
    def replay(self):
        """Train on batch from replay memory"""
        if len(self.memory) < self.batch_size:
            return
            
        # Sample random batch
        minibatch = random.sample(self.memory, self.batch_size)
        
        # Prepare batch data
        states, actions, rewards, next_states, dones = zip(*minibatch)
        
        states = torch.FloatTensor(np.array(states))
        actions = torch.LongTensor(actions).unsqueeze(1)
        rewards = torch.FloatTensor(rewards).unsqueeze(1)
        next_states = torch.FloatTensor(np.array(next_states))
        dones = torch.FloatTensor(dones).unsqueeze(1)
        
        # Compute current Q values
        current_q = self.policy_net(states).gather(1, actions)
        
        # Compute target Q values (Double DQN)
        with torch.no_grad():
            # Select actions using policy network
            policy_argmax = self.policy_net(next_states).argmax(dim=1, keepdim=True)
            # Evaluate using target network
            next_q = self.target_net(next_states).gather(1, policy_argmax)
            target_q = rewards + (1 - dones) * self.gamma * next_q
        
        # Compute loss and update
        loss = F.smooth_l1_loss(current_q, target_q)
        self.optimizer.zero_grad()
        loss.backward()
        # Gradient clipping to prevent exploding gradients
        for param in self.policy_net.parameters():
            param.grad.data.clamp_(-1, 1)
        self.optimizer.step()
        
        # Soft update target network
        for target_param, policy_param in zip(self.target_net.parameters(), 
                                              self.policy_net.parameters()):
            target_param.data.copy_(
                self.tau * policy_param.data + (1 - self.tau) * target_param.data
            )
        
        # Decay epsilon
        if self.epsilon > self.epsilon_min:
            self.epsilon *= self.epsilon_decay
            
# Training function
def train_dqn(env_name='CartPole-v1', episodes=500):
    env = gym.make(env_name)
    state_size = env.observation_space.shape[0]
    action_size = env.action_space.n
    
    agent = DQNAgent(state_size, action_size)
    scores = []
    
    for episode in range(episodes):
        state = env.reset()
        score = 0
        done = False
        
        while not done:
            # Choose and perform action
            action = agent.act(state)
            next_state, reward, done, _ = env.step(action)
            
            # Store transition and learn
            agent.remember(state, action, reward, next_state, done)
            agent.replay()
            
            state = next_state
            score += reward
            
        scores.append(score)
        
        print(f"Episode: {episode+1}, Score: {score}, Epsilon: {agent.epsilon:.2f}")
        
        # If CartPole is solved (average score >= 195 over 100 episodes)
        if len(scores) > 100 and np.mean(scores[-100:]) >= 195:
            print(f"CartPole solved in {episode+1} episodes!")
            break
            
    return scores

# Run training
if __name__ == "__main__":
    scores = train_dqn()
    
    # Plot results
    plt.figure(figsize=(10, 6))
    plt.plot(scores)
    plt.title('DQN Training on CartPole')
    plt.xlabel('Episode')
    plt.ylabel('Score')
    plt.show()
```

This implementation includes key DQN features:
- Experience replay buffer
- Target network with soft updates
- Double DQN algorithm
- Epsilon-greedy exploration

## 7. Policy Gradient Implementation in Python

Now let's implement a REINFORCE (Monte Carlo Policy Gradient) algorithm:

```python
import gym
import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F
import matplotlib.pyplot as plt
from torch.distributions import Categorical

# Policy network
class PolicyNetwork(nn.Module):
    def __init__(self, state_size, action_size):
        super(PolicyNetwork, self).__init__()
        self.fc1 = nn.Linear(state_size, 128)
        self.fc2 = nn.Linear(128, action_size)
        
    def forward(self, x):
        x = F.relu(self.fc1(x))
        return F.softmax(self.fc2(x), dim=1)
    
    def act(self, state):
        """Sample action from policy"""
        state_tensor = torch.FloatTensor(state).unsqueeze(0)
        probs = self.forward(state_tensor)
        m = Categorical(probs)
        action = m.sample()
        return action.item(), m.log_prob(action)

# REINFORCE Agent
class REINFORCEAgent:
    def __init__(self, state_size, action_size):
        self.policy = PolicyNetwork(state_size, action_size)
        self.optimizer = optim.Adam(self.policy.parameters(), lr=0.001)
        self.gamma = 0.99
        
    def update_policy(self, rewards, log_probs):
        """Update policy using policy gradient"""
        # Calculate discounted rewards
        discounted_rewards = []
        R = 0
        for r in reversed(rewards):
            R = r + self.gamma * R
            discounted_rewards.insert(0, R)
            
        # Convert to tensor and normalize
        discounted_rewards = torch.FloatTensor(discounted_rewards)
        discounted_rewards = (discounted_rewards - discounted_rewards.mean()) / \
                             (discounted_rewards.std() + 1e-9)
        
        # Calculate policy loss
        policy_loss = []
        for log_prob, R in zip(log_probs, discounted_rewards):
            policy_loss.append(-log_prob * R)
            
        # Backpropagation
        self.optimizer.zero_grad()
        policy_loss = torch.cat(policy_loss).sum()
        policy_loss.backward()
        self.optimizer.step()
        
        return policy_loss.item()

# Training function
def train_reinforce(env_name='CartPole-v1', episodes=1000):
    env = gym.make(env_name)
    state_size = env.observation_space.shape[0]
    action_size = env.action_space.n
    
    agent = REINFORCEAgent(state_size, action_size)
    scores = []
    
    for episode in range(episodes):
        state = env.reset()
        log_probs = []
        rewards = []
        score = 0
        done = False
        
        # Collect trajectory
        while not done:
            # Select action
            action, log_prob = agent.policy.act(state)
            
            # Take action
            next_state, reward, done, _ = env.step(action)
            
            # Store experience
            log_probs.append(log_prob)
            rewards.append(reward)
            
            state = next_state
            score += reward
        
        # Update policy after episode completes
        loss = agent.update_policy(rewards, log_probs)
        scores.append(score)
        
        print(f"Episode: {episode+1}, Score: {score}, Loss: {loss:.4f}")
        
        # If CartPole is solved
        if len(scores) > 100 and np.mean(scores[-100:]) >= 195:
            print(f"CartPole solved in {episode+1} episodes!")
            break
            
    return scores

# Run training
if __name__ == "__main__":
    scores = train_reinforce()
    
    # Plot results
    plt.figure(figsize=(10, 6))
    plt.plot(scores)
    plt.title('REINFORCE Training on CartPole')
    plt.xlabel('Episode')
    plt.ylabel('Score')
    plt.show()
```

This implementation demonstrates:
- Policy network that outputs action probabilities
- Monte Carlo return calculation
- Policy gradient update
- Reward normalization for training stability

```python
# Example: Trading Strategy with RL
import pandas as pd
import numpy as np
import yfinance as yf
import matplotlib.pyplot as plt

class TradingEnvironment:
    """Simple trading environment for RL"""
    def __init__(self, data, initial_balance=10000):
        self.data = data
        self.initial_balance = initial_balance
        self.reset()
        
    def reset(self):
        self.balance = self.initial_balance
        self.position = 0  # Number of shares
        self.current_step = 0
        self.done = False
        state = self._get_state()
        return state
        
    def _get_state(self):
        """Create state representation for the agent"""
        # Simple state: price features, current balance, current position
        if self.current_step >= len(self.data) - 1:
            return None
            
        # Use price, 5-day return, 10-day return as features
        price = self.data.iloc[self.current_step]['Close']
        if self.current_step >= 5:
            return_5d = price / self.data.iloc[self.current_step-5]['Close'] - 1
        else:
            return_5d = 0
            
        if self.current_step >= 10:
            return_10d = price / self.data.iloc[self.current_step-10]['Close'] - 1
        else:
            return_10d = 0
        
        state = np.array([
            return_5d, return_10d,
            self.balance / self.initial_balance,
            self.position * price / self.initial_balance  # Portfolio value
        ])
        
        return state
        
    def step(self, action):
        """Take action and return new state, reward, done"""
        # Action: 0=sell, 1=hold, 2=buy
        if self.done:
            return self._get_state(), 0, self.done, {}
            
        # Get current price
        current_price = self.data.iloc[self.current_step]['Close']
        
        # Execute action
        if action == 0:  # Sell
            if self.position > 0:
                self.balance += self.position * current_price
                self.position = 0
        elif action == 2:  # Buy
            if self.balance > 0:
                shares_to_buy = self.balance / current_price
                self.position += shares_to_buy
                self.balance = 0
                
        # Move to next step
        self.current_step += 1
        
        # Calculate reward (change in portfolio value)
        old_portfolio_value = self.balance + self.position * current_price
        
        if self.current_step >= len(self.data) - 1:
            self.done = True
        
        if not self.done:
            new_price = self.data.iloc[self.current_step]['Close']
            new_portfolio_value = self.balance + self.position * new_price
            reward = (new_portfolio_value / old_portfolio_value) - 1
        else:
            reward = 0
            
        return self._get_state(), reward, self.done, {}
```
