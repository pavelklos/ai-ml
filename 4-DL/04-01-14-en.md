# Understanding Hyperparameters in Deep Learning

## 1. Introduction to Hyperparameters

Hyperparameters are configuration variables that govern the training process of machine learning models. Unlike model parameters (weights and biases) which are learned during training, hyperparameters must be set before training begins. They control aspects like model complexity, learning speed, and regularization strength.

### Key Concepts:

- **Parameters vs Hyperparameters**: Parameters are learned from data, hyperparameters are set by the practitioner
- **Impact**: Hyperparameter selection dramatically affects model performance, training speed, and convergence
- **No Universal Values**: Optimal hyperparameters vary across datasets, architectures, and tasks

## 2. Common Hyperparameters

### 2.1 Learning Rate

The learning rate controls how much the model parameters are updated during training. It's arguably the most important hyperparameter.

```python
import tensorflow as tf
import matplotlib.pyplot as plt
import numpy as np

# Example: Effect of learning rate on training
learning_rates = [0.1, 0.01, 0.001, 0.0001]
histories = []

for lr in learning_rates:
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(64, activation='relu', input_shape=(784,)),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.SGD(learning_rate=lr),
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    # Load data (MNIST)
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
    x_train = x_train.reshape(-1, 784).astype('float32') / 255.0
    x_test = x_test.reshape(-1, 784).astype('float32') / 255.0
    
    # Train for just a few epochs to demonstrate the effect
    history = model.fit(
        x_train, y_train, epochs=5, batch_size=128, 
        validation_split=0.2, verbose=0
    )
    histories.append(history.history)

# Plot learning curves
plt.figure(figsize=(12, 5))
for i, lr in enumerate(learning_rates):
    plt.subplot(1, 2, 1)
    plt.plot(histories[i]['loss'], label=f'LR: {lr}')
    plt.title('Training Loss')
    plt.xlabel('Epoch')
    plt.ylabel('Loss')
    plt.legend()
    
    plt.subplot(1, 2, 2)
    plt.plot(histories[i]['accuracy'], label=f'LR: {lr}')
    plt.title('Training Accuracy')
    plt.xlabel('Epoch')
    plt.ylabel('Accuracy')
    plt.legend()

plt.tight_layout()
plt.show()
```

### 2.3 Number of Epochs

An epoch is a complete pass through the entire training dataset. The number of epochs controls how many times the learning algorithm will work through the entire dataset.


## 3. Hyperparameter Tuning Methods

### 3.1 Grid Search

Exhaustively tries all combinations of hyperparameters from predefined sets.

```python
from sklearn.model_selection import GridSearchCV
from tensorflow.keras.wrappers.scikit_learn import KerasClassifier

# Define the model-building function
def build_model(learning_rate=0.001, dropout_rate=0.2, units=64):
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(units, activation='relu', input_shape=(784,)),
        tf.keras.layers.Dropout(dropout_rate),
        tf.keras.layers.Dense(32, activation='relu'),
        tf.keras.layers.Dropout(dropout_rate),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=learning_rate),
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model

# Create a scikit-learn compatible model
model = KerasClassifier(build_fn=build_model, verbose=0)

# Define the grid search parameters
param_grid = {
    'learning_rate': [0.001, 0.01],
    'dropout_rate': [0.2, 0.4],
    'units': [32, 64],
    'batch_size': [32, 64],
    'epochs': [10, 15]
}

# Set up and run grid search
grid = GridSearchCV(estimator=model, param_grid=param_grid, cv=3)
grid_result = grid.fit(x_train[:1000], y_train[:1000])  # Using subset for demonstration

# Display results
print(f"Best: {grid_result.best_score_:.4f} using {grid_result.best_params_}")
```

```python
# Learning rate scheduler
def lr_scheduler(epoch, lr):
    decay_rate = 0.1
    decay_step = 10
    if epoch % decay_step == 0 and epoch:
        return lr * decay_rate
    return lr

lr_callback = tf.keras.callbacks.LearningRateScheduler(lr_scheduler)

model = tf.keras.Sequential([
    tf.keras.layers.Dense(64, activation='relu', input_shape=(784,)),
    tf.keras.layers.Dense(32, activation='relu'),
```

### 4.3 PyTorch Learning Rate Finder

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
import matplotlib.pyplot as plt
import numpy as np

# Convert data to PyTorch tensors
x_train_tensor = torch.FloatTensor(x_train)
y_train_tensor = torch.LongTensor(y_train)

# Create dataset and dataloader
train_dataset = TensorDataset(x_train_tensor, y_train_tensor)
train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)

# Define a simple network
class SimpleNN(nn.Module):
    def __init__(self):
        super(SimpleNN, self).__init__()
        self.fc1 = nn.Linear(784, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 10)
        self.relu = nn.ReLU()
        
    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Learning rate finder implementation
def find_learning_rate(model, train_loader, optimizer, criterion, start_lr=1e-7, end_lr=10, num_iterations=100):
    model.train()
    lrs = np.exp(np.linspace(np.log(start_lr), np.log(end_lr), num_iterations))
    losses = []
    
    # Set initial learning rate
    for param_group in optimizer.param_groups:
        param_group['lr'] = start_lr
    
    for i, (inputs, targets) in enumerate(train_loader):
        if i >= num_iterations:
            break
            
        # Update learning rate
        for param_group in optimizer.param_groups:
            param_group['lr'] = lrs[i]
            
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, targets)
        loss.backward()
        optimizer.step()
        
        losses.append(loss.item())
        
    return lrs, losses

# Create model, optimizer and loss function
model = SimpleNN()
optimizer = optim.SGD(model.parameters(), lr=0.01)
criterion = nn.CrossEntropyLoss()

# Find optimal learning rate
lrs, losses = find_learning_rate(model, train_loader, optimizer, criterion)

# Plot results
plt.figure(figsize=(10, 6))
plt.plot(lrs, losses)
plt.xscale('log')
plt.xlabel('Learning Rate')
plt.ylabel('Loss')
plt.title('Learning Rate vs Loss')
plt.grid(True)
plt.show()

# The optimal learning rate is typically where the loss decreases the most steeply
```

### 5.1 TensorFlow / Keras Implementation

```python
# Creating a model with hyperparameters
def create_model(hyperparams):
    model = tf.keras.Sequential()
    
    # Input layer
    model.add(tf.keras.layers.Dense(
        hyperparams['units_layer1'],
        activation=hyperparams['activation'],
        kernel_regularizer=tf.keras.regularizers.l2(hyperparams['l2_reg']),
        input_shape=(784,)
    ))
    
    model.add(tf.keras.layers.Dropout(hyperparams['dropout_rate']))
    
    # Hidden layer
    model.add(tf.keras.layers.Dense(
        hyperparams['units_layer2'],
        activation=hyperparams['activation'],
        kernel_regularizer=tf.keras.regularizers.l2(hyperparams['l2_reg'])
    ))
    
    model.add(tf.keras.layers.Dropout(hyperparams['dropout_rate']))
    
    # Output layer
    model.add(tf.keras.layers.Dense(10, activation='softmax'))
    
    # Compile the model
    optimizer_name = hyperparams['optimizer'].lower()
    learning_rate = hyperparams['learning_rate']
    
    if optimizer_name == 'adam':
        optimizer = tf.keras.optimizers.Adam(learning_rate=learning_rate)
    elif optimizer_name == 'sgd':
        optimizer = tf.keras.optimizers.SGD(learning_rate=learning_rate, momentum=0.9)
    elif optimizer_name == 'rmsprop':
        optimizer = tf.keras.optimizers.RMSprop(learning_rate=learning_rate)
    else:
        raise ValueError(f"Unsupported optimizer: {optimizer_name}")
    
    model.compile(
        optimizer=optimizer,
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model

# Define hyperparameters
hyperparams = {
```

### 5.2 PyTorch Implementation

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

class CustomNN(nn.Module):
    def __init__(self, hyperparams):
        super(CustomNN, self).__init__()
        self.fc1 = nn.Linear(784, hyperparams['hidden_size1'])
        self.fc2 = nn.Linear(hyperparams['hidden_size1'], hyperparams['hidden_size2'])
        self.fc3 = nn.Linear(hyperparams['hidden_size2'], 10)
        
        if hyperparams['activation'] == 'relu':
            self.activation = nn.ReLU()
        elif hyperparams['activation'] == 'tanh':
            self.activation = nn.Tanh()
        elif hyperparams['activation'] == 'leaky_relu':
            self.activation = nn.LeakyReLU(0.1)
        else:
            raise ValueError(f"Unsupported activation: {hyperparams['activation']}")
            
        self.dropout = nn.Dropout(hyperparams['dropout_rate'])
        
    def forward(self, x):
        x = self.activation(self.fc1(x))
        x = self.dropout(x)
        x = self.activation(self.fc2(x))
        x = self.dropout(x)
        x = self.fc3(x)
        return x

# Define hyperparameters
hyperparams = {
    'hidden_size1': 128,
    'hidden_size2': 64,
    'activation': 'relu',
    'dropout_rate': 0.3,
    'learning_rate': 0.001,
    'weight_decay': 1e-4,  # L2 regularization
    'batch_size': 64,
    'epochs': 10
}

# Prepare data for PyTorch
x_train_tensor = torch.FloatTensor(x_train)
y_train_tensor = torch.LongTensor(y_train)

# Create dataset and dataloader
train_dataset = TensorDataset(x_train_tensor, y_train_tensor)
train_loader = DataLoader(
    train_dataset, 
    batch_size=hyperparams['batch_size'], 
    shuffle=True
)

# Create model
model = CustomNN(hyperparams)

# Define loss function and optimizer
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(
    model.parameters(), 
    lr=hyperparams['learning_rate'],
    weight_decay=hyperparams['weight_decay']
)

# Training loop
losses = []
for epoch in range(hyperparams['epochs']):
    running_loss = 0.0
    for i, (inputs, labels) in enumerate(train_loader):
        # Zero the parameter gradients
        optimizer.zero_grad()
        
        # Forward + backward + optimize
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        # Print statistics
        running_loss += loss.item()
        
    epoch_loss = running_loss / len(train_loader)
    losses.append(epoch_loss)
    print(f'Epoch {epoch+1}/{hyperparams["epochs"]}, Loss: {epoch_loss:.4f}')
    
# Plot training loss
plt.figure(figsize=(10, 5))
plt.plot(range(1, hyperparams['epochs']+1), losses)
plt.title('Training Loss')
plt.xlabel('Epoch')
plt.ylabel('Loss')
plt.grid(True)
plt.show()
```

## 6. Best Practices for Hyperparameter Tuning

1. **Start with defaults**: Begin with established baseline values for your problem type
2. **Focus on important hyperparameters first**: Learning rate, batch size, model architecture
3. **Use logarithmic scales**: For learning rate and regularization parameters
4. **Monitor validation metrics**: Use early stopping to prevent overfitting
5. **Understand interactions**: Some hyperparameters interact (e.g., learning rate and batch size)
6. **Document your experiments**: Keep track of what you've tried and the results
7. **Automate when possible**: Use libraries like Optuna, Hyperopt, or Ray Tune
