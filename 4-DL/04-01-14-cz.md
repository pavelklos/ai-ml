# Porozumění hyperparametrům v hlubokém učení

## 1. Úvod do hyperparametrů

Hyperparametry jsou konfigurační proměnné, které řídí proces trénování modelů strojového učení. Na rozdíl od parametrů modelu (váhy a prahy), které se učí během tréninku, musí být hyperparametry nastaveny před začátkem tréninku. Řídí aspekty jako je složitost modelu, rychlost učení a síla regularizace.

### Klíčové koncepty:

- **Parametry vs Hyperparametry**: Parametry se učí z dat, hyperparametry nastavuje vývojář
- **Dopad**: Výběr hyperparametrů dramaticky ovlivňuje výkon modelu, rychlost trénování a konvergenci
- **Neexistují univerzální hodnoty**: Optimální hyperparametry se liší napříč datovými sadami, architekturami a úlohami

## 2. Běžné hyperparametry

### 2.1 Rychlost učení (Learning Rate)

Rychlost učení určuje, jak moc jsou parametry modelu aktualizovány během tréninku. Je pravděpodobně nejdůležitějším hyperparametrem.

```python
import tensorflow as tf
import matplotlib.pyplot as plt
import numpy as np

# Příklad: Vliv rychlosti učení na trénink
learning_rates = [0.1, 0.01, 0.001, 0.0001]
histories = []

for lr in learning_rates:
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(64, activation='relu', input_shape=(784,)),
        tf.keras.layers.Dense(64, activation='relu'),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.SGD(learning_rate=lr),
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    # Načtení dat (MNIST)
    (x_train, y_train), (x_test, y_test) = tf.keras.datasets.mnist.load_data()
    x_train = x_train.reshape(-1, 784).astype('float32') / 255.0
    x_test = x_test.reshape(-1, 784).astype('float32') / 255.0
    
    # Trénování pouze několik epoch pro demonstraci efektu
    history = model.fit(
        x_train, y_train, epochs=5, batch_size=128, 
        validation_split=0.2, verbose=0
    )
    histories.append(history.history)

# Vykreslení křivek učení
plt.figure(figsize=(12, 5))
for i, lr in enumerate(learning_rates):
    plt.subplot(1, 2, 1)
    plt.plot(histories[i]['loss'], label=f'LR: {lr}')
    plt.title('Trénovací ztráta')
    plt.xlabel('Epocha')
    plt.ylabel('Ztráta')
    plt.legend()
    
    plt.subplot(1, 2, 2)
    plt.plot(histories[i]['accuracy'], label=f'LR: {lr}')
    plt.title('Trénovací přesnost')
    plt.xlabel('Epocha')
    plt.ylabel('Přesnost')
    plt.legend()

plt.tight_layout()
plt.show()
```

## 3. Metody ladění hyperparametrů

### 3.1 Grid Search (Mřížkové prohledávání)

Vyčerpávající zkoušení všech kombinací hyperparametrů z předem definovaných sad.

```python
from sklearn.model_selection import GridSearchCV
from tensorflow.keras.wrappers.scikit_learn import KerasClassifier

# Definice funkce pro vytvoření modelu
def build_model(learning_rate=0.001, dropout_rate=0.2, units=64):
    model = tf.keras.Sequential([
        tf.keras.layers.Dense(units, activation='relu', input_shape=(784,)),
        tf.keras.layers.Dropout(dropout_rate),
        tf.keras.layers.Dense(32, activation='relu'),
        tf.keras.layers.Dropout(dropout_rate),
        tf.keras.layers.Dense(10, activation='softmax')
    ])
    
    model.compile(
        optimizer=tf.keras.optimizers.Adam(learning_rate=learning_rate),
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model

# Vytvoření modelu kompatibilního se scikit-learn
model = KerasClassifier(build_fn=build_model, verbose=0)

# Definice parametrů pro grid search
param_grid = {
    'learning_rate': [0.001, 0.01],
    'dropout_rate': [0.2, 0.4],
    'units': [32, 64],
    'batch_size': [32, 64],
    'epochs': [10, 15]
}

# Nastavení a spuštění grid search
```

```python
def lr_scheduler(epoch, lr):
    decay_rate = 0.1
    decay_step = 10
    if epoch % decay_step == 0 and epoch:
        return lr * decay_rate
    return lr

lr_callback = tf.keras.callbacks.LearningRateScheduler(lr_scheduler)
```

### 4.3 PyTorch hledač rychlosti učení

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset
import matplotlib.pyplot as plt
import numpy as np

# Převod dat na PyTorch tenzory
x_train_tensor = torch.FloatTensor(x_train)
y_train_tensor = torch.LongTensor(y_train)

# Vytvoření datasetu a dataloaderu
train_dataset = TensorDataset(x_train_tensor, y_train_tensor)
train_loader = DataLoader(train_dataset, batch_size=64, shuffle=True)

# Definice jednoduché sítě
class SimpleNN(nn.Module):
    def __init__(self):
        super(SimpleNN, self).__init__()
        self.fc1 = nn.Linear(784, 64)
        self.fc2 = nn.Linear(64, 32)
        self.fc3 = nn.Linear(32, 10)
        self.relu = nn.ReLU()
        
    def forward(self, x):
        x = self.relu(self.fc1(x))
        x = self.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Implementace hledače rychlosti učení
def find_learning_rate(model, train_loader, optimizer, criterion, start_lr=1e-7, end_lr=10, num_iterations=100):
    model.train()
    lrs = np.exp(np.linspace(np.log(start_lr), np.log(end_lr), num_iterations))
    losses = []
    
    # Nastavení počáteční rychlosti učení
    for param_group in optimizer.param_groups:
        param_group['lr'] = start_lr
    
    for i, (inputs, targets) in enumerate(train_loader):
        if i >= num_iterations:
            break
            
        # Aktualizace rychlosti učení
        for param_group in optimizer.param_groups:
            param_group['lr'] = lrs[i]
            
        optimizer.zero_grad()
        outputs = model(inputs)
        loss = criterion(outputs, targets)
        loss.backward()
        optimizer.step()
        
        losses.append(loss.item())
        
    return lrs, losses

# Vytvoření modelu, optimalizátoru a ztrátové funkce
model = SimpleNN()
optimizer = optim.SGD(model.parameters(), lr=0.01)
criterion = nn.CrossEntropyLoss()

# Hledání optimální rychlosti učení
lrs, losses = find_learning_rate(model, train_loader, optimizer, criterion)

# Vykreslení výsledků
plt.figure(figsize=(10, 6))
plt.plot(lrs, losses)
plt.xscale('log')
plt.xlabel('Rychlost učení')
plt.ylabel('Ztráta')
plt.title('Rychlost učení vs. Ztráta')
plt.grid(True)
plt.show()

# Optimální rychlost učení je typicky tam, kde ztráta klesá nejstrmějším způsobem
```

```python
def create_model(hyperparams):
    model = tf.keras.Sequential()
    
    # Vstupní vrstva
    model.add(tf.keras.layers.Dense(
        hyperparams['units_layer1'],
        activation=hyperparams['activation'],
        kernel_regularizer=tf.keras.regularizers.l2(hyperparams['l2_reg']),
        input_shape=(784,)
    ))

    ))
    
    model.add(tf.keras.layers.Dropout(hyperparams['dropout_rate']))
    
    # Výstupní vrstva
    model.add(tf.keras.layers.Dense(10, activation='softmax'))
    
    # Kompilace modelu
    optimizer_name = hyperparams['optimizer'].lower()
    learning_rate = hyperparams['learning_rate']
    
    if optimizer_name == 'adam':
        optimizer = tf.keras.optimizers.Adam(learning_rate=learning_rate)
    elif optimizer_name == 'sgd':
        optimizer = tf.keras.optimizers.SGD(learning_rate=learning_rate, momentum=0.9)

        optimizer = tf.keras.optimizers.RMSprop(learning_rate=learning_rate)
    else:
        raise ValueError(f"Nepodporovaný optimalizátor: {optimizer_name}")
    
    model.compile(
        optimizer=optimizer,
        loss='sparse_categorical_crossentropy',
        metrics=['accuracy']
    )
    
    return model
```

### 5.2 Implementace v PyTorch

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, TensorDataset

class CustomNN(nn.Module):
    def __init__(self, hyperparams):
        super(CustomNN, self).__init__()
        self.fc1 = nn.Linear(784, hyperparams['hidden_size1'])
        self.fc2 = nn.Linear(hyperparams['hidden_size1'], hyperparams['hidden_size2'])
        self.fc3 = nn.Linear(hyperparams['hidden_size2'], 10)
        
        if hyperparams['activation'] == 'relu':
            self.activation = nn.ReLU()
        elif hyperparams['activation'] == 'tanh':
            self.activation = nn.Tanh()
        elif hyperparams['activation'] == 'leaky_relu':
            self.activation = nn.LeakyReLU(0.1)
        else:
            raise ValueError(f"Nepodporovaná aktivace: {hyperparams['activation']}")
            
        self.dropout = nn.Dropout(hyperparams['dropout_rate'])
        
    def forward(self, x):
        x = self.activation(self.fc1(x))
        x = self.dropout(x)
        x = self.activation(self.fc2(x))
        x = self.dropout(x)
        x = self.fc3(x)
        return x

# Definice hyperparametrů
hyperparams = {
    'hidden_size1': 128,
    'hidden_size2': 64,
    'activation': 'relu',
    'dropout_rate': 0.3,
    'learning_rate': 0.001,
    'weight_decay': 1e-4,  # L2 regularizace
    'batch_size': 64,
    'epochs': 10
}

# Příprava dat pro PyTorch
x_train_tensor = torch.FloatTensor(x_train)
y_train_tensor = torch.LongTensor(y_train)

# Vytvoření datasetu a dataloaderu
train_dataset = TensorDataset(x_train_tensor, y_train_tensor)
train_loader = DataLoader(
    train_dataset, 
    batch_size=hyperparams['batch_size'], 
    shuffle=True
)

# Vytvoření modelu
model = CustomNN(hyperparams)

# Definice ztrátové funkce a optimalizátoru
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(
    model.parameters(), 
    lr=hyperparams['learning_rate'],
    weight_decay=hyperparams['weight_decay']
)

# Trénovací smyčka
losses = []
for epoch in range(hyperparams['epochs']):
    running_loss = 0.0
    for i, (inputs, labels) in enumerate(train_loader):
        # Vynulování gradientů parametrů
        optimizer.zero_grad()
        
        # Dopředný průchod + zpětný průchod + optimalizace
        outputs = model(inputs)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        # Statistiky
        running_loss += loss.item()
        
    epoch_loss = running_loss / len(train_loader)
    losses.append(epoch_loss)
    print(f'Epocha {epoch+1}/{hyperparams["epochs"]}, Ztráta: {epoch_loss:.4f}')
    
# Vykreslení trénovací ztráty
plt.figure(figsize=(10, 5))
plt.plot(range(1, hyperparams['epochs']+1), losses)
plt.title('Trénovací ztráta')
plt.xlabel('Epocha')
plt.ylabel('Ztráta')
plt.grid(True)
plt.show()
```
