## Úvod do PyTorch

PyTorch je open-source framework pro hluboké učení vyvinutý výzkumnou laboratoří Facebook AI Research (FAIR). Poskytuje flexibilní a intuitivní rozhraní pro vytváření a trénování neuronových sítí s výraznou podporou akcelerace GPU. PyTorch je známý pro:

- **Dynamický výpočetní graf** - Na rozdíl od frameworků se statickým grafem, PyTorch buduje výpočetní graf za běhu, což usnadňuje ladění a prototypování
- **Integrace s Pythonem** - PyTorch se chová jako přirozené rozšíření Pythonu
- **Silná výzkumná komunita** - Široce používán v akademickém a výzkumném prostředí
- **Připravenost pro produkci** - Díky TorchScript a integracím pro nasazení

## PyTorch tenzory

Tenzory jsou základní datovou strukturou v PyTorch, podobně jako NumPy pole, ale s dalšími funkcemi jako je podpora GPU a automatická diferenciace.

### Vytváření tenzorů

```python
import torch
import numpy as np

# Vytvoření tenzoru přímo z dat
x = torch.tensor([1, 2, 3, 4])
print(f"Tenzor z dat: {x}")

# Vytvoření tenzoru z numpy pole
np_array = np.array([1, 2, 3, 4])
x_np = torch.from_numpy(np_array)
print(f"Tenzor z NumPy: {x_np}")

# Vytvoření tenzorů se specifickými vlastnostmi
zeros = torch.zeros(3, 4)  # 3x4 tenzor nul
ones = torch.ones(2, 3, 4)  # 2x3x4 tenzor jedniček
rand = torch.rand(2, 3)  # 2x3 tenzor s náhodnými hodnotami z rovnoměrného rozdělení [0,1)
randn = torch.randn(2, 3)  # 2x3 tenzor s náhodnými hodnotami ze standardního normálního rozdělení

print(f"Tvar tenzoru nul: {zeros.shape}")
print(f"Tvar tenzoru jedniček: {ones.shape}")
print(f"Náhodný tenzor: \n{rand}")
```

### Operace s tenzory

```python
# Základní operace
a = torch.tensor([1, 2, 3])
b = torch.tensor([4, 5, 6])

print(f"a + b: {a + b}")  # Sčítání prvek po prvku
print(f"a * b: {a * b}")  # Násobení prvek po prvku
print(f"a @ b: {a @ b}")  # Skalární součin (zde stejný jako torch.dot(a, b))

# Maticové operace
m1 = torch.tensor([[1, 2], [3, 4]])
m2 = torch.tensor([[5, 6], [7, 8]])
print(f"Maticové násobení: \n{torch.matmul(m1, m2)}")

# Změna tvaru tenzorů
x = torch.tensor([[1, 2], [3, 4], [5, 6]])
print(f"Původní tvar: {x.shape}")

x_reshaped = x.reshape(2, 3)
print(f"Po změně tvaru: \n{x_reshaped}")
print(f"Nový tvar: {x_reshaped.shape}")

x_transposed = x.t()  # Transpozice
print(f"Po transpozici: \n{x_transposed}")
print(f"Tvar po transpozici: {x_transposed.shape}")

# Indexování a výběr částí (podobně jako v NumPy)
print(f"První řádek: {x[0]}")
print(f"První sloupec: {x[:, 0]}")
print(f"Podmatice: \n{x[0:2, :]}")
```

## Autograd - automatická diferenciace

Jednou z klíčových funkcí PyTorch je automatická diferenciace pomocí balíčku `autograd`. To umožňuje automatický výpočet gradientů pro všechny operace s tenzory.

```python
# Vytvoření tenzorů se sledováním gradientu
x = torch.tensor([2.0], requires_grad=True)
y = torch.tensor([3.0], requires_grad=True)

# Provedení operací
z = x**2 + y**3

# Výpočet gradientů
z.backward()

# Zobrazení gradientů (dz/dx a dz/dy)
print(f"dz/dx: {x.grad}")  # Mělo by být 2*x = 4
print(f"dz/dy: {y.grad}")  # Mělo by být 3*y^2 = 27

# Příklad s vektorovým výstupem (sum se používá k získání skalárního výstupu)
x = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)
y = x * 2
z = y.sum()
z.backward()
print(f"x: {x}")
print(f"x.grad: {x.grad}")  # Mělo by být [2.0, 2.0, 2.0]
```

### Zastavení sledování gradientu

```python
# Dočasné zastavení sledování gradientu
x = torch.tensor([1.0, 2.0, 3.0], requires_grad=True)

with torch.no_grad():
    y = x * 2
    
print(f"y.requires_grad: {y.requires_grad}")  # Mělo by být False

# Odpojení tenzoru od jeho výpočetní historie
y = x * 2
z = y.detach()
print(f"z.requires_grad: {z.requires_grad}")  # Mělo by být False
```

## Vytváření neuronových sítí

PyTorch poskytuje modul `nn` pro vytváření neuronových sítí, který obsahuje běžné vrstvy, aktivační funkce a další komponenty.

### Použití nn.Module

```python
import torch.nn as nn
import torch.nn.functional as F

# Definice jednoduché neuronové sítě
class SimpleNN(nn.Module):
    def __init__(self):
        super(SimpleNN, self).__init__()
        # Definice vrstev
        self.fc1 = nn.Linear(10, 50)  # Vstup: 10, Výstup: 50
        self.fc2 = nn.Linear(50, 20)  # Vstup: 50, Výstup: 20
        self.fc3 = nn.Linear(20, 2)   # Vstup: 20, Výstup: 2
        
    def forward(self, x):
        # Definice dopředného průchodu
        x = F.relu(self.fc1(x))
        x = F.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Vytvoření instance modelu
model = SimpleNN()
print(model)

# Generování náhodného vstupu a provedení dopředného průchodu
input_data = torch.randn(5, 10)  # 5 vzorků, 10 příznaků každý
output = model(input_data)
print(f"Tvar vstupu: {input_data.shape}")
print(f"Tvar výstupu: {output.shape}")
```

### Vrstvy a aktivační funkce

```python
# Běžné vrstvy v PyTorch
linear = nn.Linear(10, 5)  # Plně propojená vrstva

pool = nn.MaxPool2d(2)  # Max pooling

```

## Ztrátové funkce a optimalizátory

PyTorch poskytuje různé ztrátové funkce a optimalizační algoritmy pro trénování neuronových sítí.

### Běžné ztrátové funkce

```python
import torch.nn as nn

# Binární křížová entropie
bce_loss = nn.BCELoss()
inputs = torch.randn(3, requires_grad=True)
targets = torch.empty(3).random_(2)  # Binární cíle (0 nebo 1)
loss = bce_loss(torch.sigmoid(inputs), targets)
print(f"BCE ztráta: {loss.item()}")

# Křížová entropie (pro klasifikaci)
ce_loss = nn.CrossEntropyLoss()
inputs = torch.randn(3, 5, requires_grad=True)  # 3 vzorky, 5 tříd
targets = torch.empty(3, dtype=torch.long).random_(5)  # Indexy tříd
loss = ce_loss(inputs, targets)
print(f"CrossEntropy ztráta: {loss.item()}")

# Střední kvadratická chyba (pro regresi)
mse_loss = nn.MSELoss()
inputs = torch.randn(3, requires_grad=True)
targets = torch.randn(3)
loss = mse_loss(inputs, targets)
print(f"MSE ztráta: {loss.item()}")
```

### Optimalizátory

```python
import torch.optim as optim

# Definice jednoduchého modelu
model = SimpleNN()

# Stochastický gradientní sestup (SGD)
optimizer_sgd = optim.SGD(model.parameters(), lr=0.01, momentum=0.9)

# Adam optimalizátor
optimizer_adam = optim.Adam(model.parameters(), lr=0.001, betas=(0.9, 0.999))

# RMSprop
optimizer_rmsprop = optim.RMSprop(model.parameters(), lr=0.01)

# Použití optimalizátoru
optimizer = optim.SGD(model.parameters(), lr=0.01)
# Vynulování gradientů parametrů
optimizer.zero_grad()
# Dopředný průchod s náhodnými daty
outputs = model(torch.randn(5, 10))
# Výpočet ztráty
loss = torch.sum(outputs)
# Zpětný průchod
loss.backward()
# Aktualizace vah
optimizer.step()
```

## Základní trénování modelu

Pojďme všechno spojit dohromady a natrénovat jednoduchý model na syntetických datech.

```python
import torch
import torch.nn as nn
import torch.optim as optim
import matplotlib.pyplot as plt

# Generování syntetických dat
X = torch.linspace(-1, 1, 100).reshape(-1, 1)
y = 3 * X.pow(2) + 2 + 0.2 * torch.randn(X.size())

# Vizualizace dat
plt.scatter(X.numpy(), y.numpy())
plt.title("Syntetický regresní dataset")
plt.xlabel("X")
plt.ylabel("y")
plt.show()

# Definice modelu
class RegressionModel(nn.Module):
    def __init__(self):
        super(RegressionModel, self).__init__()
        self.fc1 = nn.Linear(1, 10)
        self.fc2 = nn.Linear(10, 10)
        self.fc3 = nn.Linear(10, 1)
        
    def forward(self, x):
        x = torch.relu(self.fc1(x))
        x = torch.relu(self.fc2(x))
        x = self.fc3(x)
        return x

# Vytvoření modelu, ztrátové funkce a optimalizátoru
model = RegressionModel()
criterion = nn.MSELoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)

# Trénovací smyčka
epochs = 200
losses = []

for epoch in range(epochs):
    # Dopředný průchod
    y_pred = model(X)
    loss = criterion(y_pred, y)
    losses.append(loss.item())
    
    # Zpětný průchod a optimalizace
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    
    # Výpis postupu
    if (epoch + 1) % 20 == 0:
        print(f'Epocha {epoch+1}/{epochs}, Ztráta: {loss.item():.4f}')

# Vykreslení křivky ztráty
plt.plot(losses)
plt.title("Trénovací ztráta")
plt.xlabel("Epocha")
plt.ylabel("Ztráta")
plt.show()

# Vizualizace predikcí
with torch.no_grad():
    y_pred = model(X)
    
plt.scatter(X.numpy(), y.numpy(), label='Data')
plt.plot(X.numpy(), y_pred.numpy(), 'r-', label='Predikce')
plt.title("Predikce vs Skutečnost")
plt.xlabel("X")
plt.ylabel("y")
plt.legend()
plt.show()
```

## Ukládání a načítání modelů

PyTorch umožňuje ukládat a načítat natrénované modely, což je nezbytné pro nasazení nebo pokračování v tréninku.

```python
# Ukládání modelu
model_path = "model.pth"

# Metoda 1: Uložení celého modelu
torch.save(model, model_path)
# Pro načtení celého modelu
loaded_model = torch.load(model_path)

# Metoda 2: Uložení pouze parametrů modelu (doporučeno)
torch.save(model.state_dict(), "model_state_dict.pth")
# Pro načtení parametrů modelu
new_model = RegressionModel()
new_model.load_state_dict(torch.load("model_state_dict.pth"))
new_model.eval()  # Nastavení do režimu vyhodnocování

# Ukládání a načítání checkpointů (pro pokračování v tréninku)
checkpoint = {
    'epoch': epoch,
    'model_state_dict': model.state_dict(),
    'optimizer_state_dict': optimizer.state_dict(),
    'loss': loss
}
torch.save(checkpoint, "checkpoint.pth")

# Pro načtení checkpointu
checkpoint = torch.load("checkpoint.pth")
model.load_state_dict(checkpoint['model_state_dict'])
optimizer.load_state_dict(checkpoint['optimizer_state_dict'])
epoch = checkpoint['epoch']
loss = checkpoint['loss']
```

## Využití GPU akcelerace

Jednou z hlavních výhod PyTorch je možnost využití GPU akcelerace pro rychlejší trénování.

```python
# Kontrola dostupnosti GPU
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print(f"Používané zařízení: {device}")

# Vytvoření modelu a přesunutí na GPU
model = RegressionModel().to(device)

# Přesunutí dat na GPU
X_gpu = X.to(device)
y_gpu = y.to(device)

# Trénovací smyčka s GPU
for epoch in range(10):
    # Dopředný průchod
    y_pred = model(X_gpu)
    loss = criterion(y_pred, y_gpu)
    
    # Zpětný průchod a optimalizace
    optimizer.zero_grad()
    loss.backward()
    optimizer.step()
    
    print(f'Epocha {epoch+1}/10, Ztráta: {loss.item():.4f}')

# Důležité: Při práci s tenzory z GPU nezapomeňte je přesunout zpět na CPU pro numpy operace
with torch.no_grad():
    y_pred_gpu = model(X_gpu)
    y_pred_cpu = y_pred_gpu.cpu().numpy()
```

## Praktický příklad

Vytvořme realističtější příklad tréninkem klasifikátoru na datasetu MNIST.

```python
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision
import torchvision.transforms as transforms
from torch.utils.data import DataLoader
import matplotlib.pyplot as plt

# Konfigurace zařízení
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# Hyperparametry
batch_size = 100
learning_rate = 0.001
num_epochs = 5

# MNIST dataset
transform = transforms.Compose([
    transforms.ToTensor(),
    transforms.Normalize((0.1307,), (0.3081,))
])

train_dataset = torchvision.datasets.MNIST(root='./data', 
                                          train=True, 
                                          transform=transform,
                                          download=True)

test_dataset = torchvision.datasets.MNIST(root='./data', 
                                         train=False, 
                                         transform=transform)

train_loader = DataLoader(dataset=train_dataset, 
                         batch_size=batch_size, 
                         shuffle=True)

test_loader = DataLoader(dataset=test_dataset, 
                        batch_size=batch_size, 
                        shuffle=False)

# Vizualizace některých příkladů
examples = iter(train_loader)
samples, labels = next(examples)
print(f"Tvar vzorku: {samples.shape}")

plt.figure(figsize=(10, 5))
for i in range(10):
    plt.subplot(2, 5, i + 1)
    plt.imshow(samples[i][0], cmap='gray')
    plt.title(f'Štítek: {labels[i].item()}')
    plt.axis('off')
plt.tight_layout()
plt.show()

# Model konvoluční neuronové sítě
class ConvNet(nn.Module):
    def __init__(self):
        super(ConvNet, self).__init__()
        self.layer1 = nn.Sequential(
            nn.Conv2d(1, 16, kernel_size=5, padding=2),
            nn.BatchNorm2d(16),
            nn.ReLU(),
            nn.MaxPool2d(2)
        )
        self.layer2 = nn.Sequential(
            nn.Conv2d(16, 32, kernel_size=5, padding=2),
            nn.BatchNorm2d(32),
            nn.ReLU(),
            nn.MaxPool2d(2)
        )
        self.fc = nn.Linear(7*7*32, 10)
        
    def forward(self, x):
        out = self.layer1(x)
        out = self.layer2(out)
        out = out.view(out.size(0), -1)  # Zploštění
        out = self.fc(out)
        return out

# Inicializace modelu
model = ConvNet().to(device)

# Ztrátová funkce a optimalizátor
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=learning_rate)

# Trénovací funkce
def train(model, train_loader, criterion, optimizer, num_epochs):
    total_step = len(train_loader)
    
    for epoch in range(num_epochs):
        for i, (images, labels) in enumerate(train_loader):
            # Přesunutí tenzorů na zařízení
            images = images.to(device)
            labels = labels.to(device)
            
            # Dopředný průchod
            outputs = model(images)
            loss = criterion(outputs, labels)
            
            # Zpětný průchod a optimalizace
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()
            
            if (i+1) % 100 == 0:
                print(f'Epocha [{epoch+1}/{num_epochs}], Krok [{i+1}/{total_step}], Ztráta: {loss.item():.4f}')

# Hodnotící funkce
def evaluate(model, test_loader):
    model.eval()
    with torch.no_grad():
        correct = 0
        total = 0
        for images, labels in test_loader:
            images = images.to(device)
            labels = labels.to(device)
            outputs = model(images)
            _, predicted = torch.max(outputs.data, 1)
            total += labels.size(0)
            correct += (predicted == labels).sum().item()
        
        print(f'Přesnost na testovacích obrázcích: {100 * correct / total:.2f}%')

# Trénování modelu
train(model, train_loader, criterion, optimizer, num_epochs)

# Testování modelu
evaluate(model, test_loader)

# Uložení modelu
torch.save(model.state_dict(), 'mnist_model.pth')
```
