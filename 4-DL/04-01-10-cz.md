### Vlastní dataset v PyTorch

```python
import torch
from torch.utils.data import Dataset
from PIL import Image
import os
import pandas as pd

class CustomImageDataset(Dataset):
    def __init__(self, annotations_file, img_dir, transform=None):
        self.img_labels = pd.read_csv(annotations_file)
        self.img_dir = img_dir
        self.transform = transform

    def __len__(self):
        return len(self.img_labels)

    def __getitem__(self, idx):
        img_path = os.path.join(self.img_dir, self.img_labels.iloc[idx, 0])
        image = Image.open(img_path).convert('RGB')
        label = self.img_labels.iloc[idx, 1]
        
        if self.transform:
            image = self.transform(image)
            
        return image, label

# Použití
transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])
])

custom_dataset = CustomImageDataset(
    annotations_file='labels.csv',
    img_dir='images/',
    transform=transform
)
```

### Vlastní dataset v TensorFlow

```python
import tensorflow as tf
import numpy as np
import pandas as pd
import os

def load_image(file_path, label):
    # Načtení obrázku
    img = tf.io.read_file(file_path)
    img = tf.image.decode_jpeg(img, channels=3)
    
    # Předzpracování
    img = tf.image.resize(img, [224, 224])
    img = img / 255.0  # Normalizace na [0,1]
    
    return img, label

def create_dataset(img_dir, annotations_file, batch_size=32, shuffle=True):
    df = pd.read_csv(annotations_file)
    file_paths = [os.path.join(img_dir, fname) for fname in df.iloc[:, 0]]
    labels = df.iloc[:, 1].values
    
    # Vytvoření TensorFlow datasetu
    dataset = tf.data.Dataset.from_tensor_slices((file_paths, labels))
    
    # Aplikace transformací
    dataset = dataset.map(load_image, num_parallel_calls=tf.data.AUTOTUNE)
    
    if shuffle:
        dataset = dataset.shuffle(buffer_size=1000)
        
    dataset = dataset.batch(batch_size)
    dataset = dataset.prefetch(tf.data.AUTOTUNE)
    
    return dataset

# Použití
train_dataset = create_dataset('images/', 'train_labels.csv', batch_size=32)
val_dataset = create_dataset('images/', 'val_labels.csv', batch_size=32, shuffle=False)
```

```python
import torch
import torchvision.transforms as transforms
from PIL import Image
import matplotlib.pyplot as plt

def load_and_preprocess_image(image_path):
    # Definice transformací
    transform = transforms.Compose([
        transforms.Resize(256),
        transforms.CenterCrop(224),
        transforms.ToTensor(),
        transforms.Normalize(mean=[0.485, 0.456, 0.406], std=[0.229, 0.224, 0.225])

    # Přidání rozměru dávky

    return image_tensor
```

### Obrazová datová pipeline s TensorFlow

```python
import tensorflow as tf
import matplotlib.pyplot as plt

def create_image_dataset(image_dir, image_size=(224, 224), batch_size=32):
    dataset = tf.keras.utils.image_dataset_from_directory(
        image_dir,
        image_size=image_size,
        batch_size=batch_size,
        label_mode='categorical'
    )
    
    # Definice předzpracování
    normalization_layer = tf.keras.layers.Rescaling(1./255)
    
    # Aplikace předzpracování
    dataset = dataset.map(lambda x, y: (normalization_layer(x), y))
    
    # Optimalizace výkonu
    dataset = dataset.cache().prefetch(buffer_size=tf.data.AUTOTUNE)
    
    return dataset

# Vytvoření datasetů
train_dataset = create_image_dataset('data/train')
```

## Práce s textovými daty

Textová data vyžadují speciální předzpracování pro modely hlubokého učení.

### Zpracování textu s PyTorch

```python
import torch
from torchtext.data.utils import get_tokenizer
from torchtext.vocab import build_vocab_from_iterator
import io

# Vzorová textová data
text_data = [
    "Deep learning is a subset of machine learning",
    "Neural networks have revolutionized AI",
    "PyTorch and TensorFlow are popular deep learning frameworks"
]

# Vytvoření tokenizátoru
tokenizer = get_tokenizer('basic_english')

# Tokenizace textu
def yield_tokens(data_iter):
    for text in data_iter:
        yield tokenizer(text)

# Vytvoření slovníku
vocab = build_vocab_from_iterator(yield_tokens(text_data), specials=['<unk>'])
vocab.set_default_index(vocab['<unk>'])

# Převod textu na indexy
def text_pipeline(text):
    return [vocab[token] for token in tokenizer(text)]

# Vytvoření numerických reprezentací
processed_data = [torch.tensor(text_pipeline(text)) for text in text_data]

# Dávkování s pomocí paddingu
from torch.nn.utils.rnn import pad_sequence

# Padding sekvencí na stejnou délku
padded_sequences = pad_sequence(processed_data, batch_first=True)

print("Velikost slovníku:", len(vocab))
print("Text na indexy:", text_pipeline("Deep learning frameworks are powerful"))
print("Tvar zarovnaných sekvencí:", padded_sequences.shape)
```

Velké datasety vyžadují speciální zpracování, aby se předešlo problémům s pamětí a optimalizovala se rychlost tréninku.

### Zpracování velkých datasetů v PyTorch

```python
import os
import torch
from torch.utils.data import Dataset, DataLoader, IterableDataset

# IterableDataset pro streamování velkých dat
class StreamingDataset(IterableDataset):
    def __init__(self, data_path, chunk_size=1000):
        self.data_path = data_path
        self.chunk_size = chunk_size
        
    def line_mapper(self, line):
        # Převod řádku textu na příznaky a štítek
        # Toto je jen příklad, přizpůsobte vašemu formátu dat
        values = line.strip().split(',')
        features = torch.tensor([float(x) for x in values[:-1]])
        label = torch.tensor(int(values[-1]))
        return features, label
        
    def __iter__(self):
        file_iter = open(self.data_path, 'r')
        mapped_iter = map(self.line_mapper, file_iter)
        return mapped_iter

# Vytvoření streamovacího datasetu
stream_dataset = StreamingDataset('large_dataset.csv')
stream_loader = DataLoader(stream_dataset, batch_size=32, num_workers=4)

# Zpracování dat v dávkách, aby nedošlo k problémům s pamětí
for batch_idx, (data, target) in enumerate(stream_loader):
    # Zpracování jedné dávky najednou
    pass
```

### Zpracování velkých datasetů v TensorFlow

```python
import tensorflow as tf

def parse_csv_line(line):
    # Parsování řádku z CSV souboru
    # Upravte defaults a record_defaults podle vašich dat
    fields = tf.io.decode_csv(line, record_defaults=[[0.0]] * 10 + [[0]])
    features = fields[:-1]
    label = fields[-1]
    return features, label

def create_large_dataset(file_path, batch_size=32):
    # Vytvoření datasetu z velkého souboru
    dataset = tf.data.TextLineDataset(file_path)
    
    # Přeskočení hlavičkového řádku, pokud je potřeba
    dataset = dataset.skip(1)
    
    # Parsování každého řádku
    dataset = dataset.map(parse_csv_line, num_parallel_calls=tf.data.AUTOTUNE)
    
    # Dávkování a prefetch
    dataset = dataset.batch(batch_size)
    dataset = dataset.prefetch(tf.data.AUTOTUNE)
    
    return dataset

# Použití velkého datasetu
large_dataset = create_large_dataset('large_dataset.csv')

# Trénování modelu s datasetem
model = tf.keras.Sequential([
    tf.keras.layers.Dense(64, activation='relu'),
    tf.keras.layers.Dense(1)
])

model.compile(optimizer='adam', loss='mse')
model.fit(large_dataset, epochs=5)
```

```python
val_loader = DataLoader(cifar_val, batch_size=128, shuffle=False, num_workers=2)
test_loader = DataLoader(cifar_test, batch_size=128, shuffle=False, num_workers=2)

# 4. Definice jednoduché CNN sítě
class SimpleCNN(nn.Module):
    def __init__(self):
        super(SimpleCNN, self).__init__()
        self.conv1 = nn.Conv2d(3, 32, 3, padding=1)
        self.conv2 = nn.Conv2d(32, 64, 3, padding=1)
        self.pool = nn.MaxPool2d(2, 2)
        self.fc1 = nn.Linear(64 * 8 * 8, 512)
        self.fc2 = nn.Linear(512, 10)
        self.relu = nn.ReLU()
        
    def forward(self, x):
        x = self.pool(self.relu(self.conv1(x)))
        x = self.pool(self.relu(self.conv2(x)))
        x = x.view(-1, 64 * 8 * 8)
        x = self.relu(self.fc1(x))
        x = self.fc2(x)
        return x

# 5. Inicializace modelu, ztrátové funkce a optimalizátoru
model = SimpleCNN()
criterion = nn.CrossEntropyLoss()
optimizer = optim.Adam(model.parameters(), lr=0.001)

# 6. Trénovací funkce
def train_epoch(model, loader, criterion, optimizer, device='cpu'):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    
    for data, targets in loader:
        data, targets = data.to(device), targets.to(device)
        
        # Vynulování gradientů parametrů
        optimizer.zero_grad()
        
        # Dopředný průchod + zpětný průchod + optimalizace
        outputs = model(data)
        loss = criterion(outputs, targets)
        loss.backward()
        optimizer.step()
        
        # Statistiky
        running_loss += loss.item()
        _, predicted = outputs.max(1)
        total += targets.size(0)
        correct += predicted.eq(targets).sum().item()
    
    return running_loss / len(loader), 100. * correct / total

# 7. Validační funkce
def validate(model, loader, criterion, device='cpu'):

    running_loss = 0.0

            _, predicted = outputs.max(1)
            total += targets.size(0)
            correct += predicted.eq(targets).sum().item()
    
    return running_loss / len(loader), 100. * correct / total

# 8. Trénovací smyčka
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

epochs = 5

    train_accs.append(train_acc)
    val_losses.append(val_loss)
    val_accs.append(val_acc)
    
    print(f"Epocha {epoch+1}/{epochs}")
    print(f"Trénovací ztráta: {train_loss:.4f}, Trénovací přesnost: {train_acc:.2f}%")
```

### Pipeline pro zpracování přirozeného jazyka

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, Dataset, random_split
from torchtext.vocab import GloVe
import numpy as np

# 1. Definice vzorových textových dat
texts = [
    "machine learning is fascinating",
    "deep neural networks are powerful",
    "data preprocessing is important",
    "python is great for deep learning",
    "transformers have revolutionized nlp",
    "convolutional networks work well for images",
    "recurrent networks process sequences",
    "attention mechanisms improve performance",
    "transfer learning saves training time",
    "hyperparameter tuning is crucial"
]

# Štítky (binární pro tento příklad)
labels = [1, 1, 0, 1, 1, 0, 0, 1, 0, 0]

# 2. Vytvoření jednoduchého tokenizátoru
def tokenize(text):
    return text.lower().split()

# 3. Vytvoření slovníku
vocab = {}
for text in texts:
    for word in tokenize(text):
        if word not in vocab:
            vocab[word] = len(vocab)

# 4. Vytvoření embeddings (normálně bychom použili předtrénované jako GloVe)
embedding_dim = 50
word_embeddings = np.random.rand(len(vocab), embedding_dim)

# 5. Definice textového datasetu
class TextDataset(Dataset):
    def __init__(self, texts, labels, vocab, max_length=10):
        self.texts = texts
        self.labels = labels
        self.vocab = vocab
        self.max_length = max_length
        
    def __len__(self):
        return len(self.texts)
    
    def __getitem__(self, idx):
        text = self.texts[idx]
        label = self.labels[idx]
        
        # Převod textu na indexy
        tokens = tokenize(text)
        indices = [self.vocab.get(word, len(self.vocab)-1) for word in tokens]
        
        # Padding nebo oříznutí na max_length
        if len(indices) < self.max_length:
            indices = indices + [0] * (self.max_length - len(indices))
        else:
            indices = indices[:self.max_length]
            
        return torch.tensor(indices), torch.tensor(label, dtype=torch.float32)

# 6. Vytvoření datasetu a rozdělení
dataset = TextDataset(texts, labels, vocab)
train_size = int(0.8 * len(dataset))
val_size = len(dataset) - train_size
train_dataset, val_dataset = random_split(dataset, [train_size, val_size])

# 7. Vytvoření data loaderů
train_loader = DataLoader(train_dataset, batch_size=2, shuffle=True)
val_loader = DataLoader(val_dataset, batch_size=2)

# 8. Definice jednoduché RNN sítě
class SimpleRNN(nn.Module):
    def __init__(self, vocab_size, embedding_dim, hidden_dim):
        super(SimpleRNN, self).__init__()
        self.embedding = nn.Embedding(vocab_size, embedding_dim)
        self.rnn = nn.GRU(embedding_dim, hidden_dim, batch_first=True)
        self.fc = nn.Linear(hidden_dim, 1)
        
    def forward(self, x):
        embedded = self.embedding(x)
        output, hidden = self.rnn(embedded)
        # Vezmi poslední skrytý stav
        return torch.sigmoid(self.fc(hidden[-1]))

# 9. Inicializace modelu, ztrátové funkce a optimalizátoru
model = SimpleRNN(len(vocab), embedding_dim, 64)
criterion = nn.BCELoss()
optimizer = optim.Adam(model.parameters(), lr=0.01)

# 10. Trénovací funkce
def train_epoch(model, loader, criterion, optimizer):
    model.train()
    running_loss = 0.0
    correct = 0
    total = 0
    
    for data, targets in loader:
        # Dopředný průchod
        outputs = model(data).squeeze()
        loss = criterion(outputs, targets)
        
        # Zpětný průchod a optimalizace
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        # Statistiky
        running_loss += loss.item()
        predicted = (outputs > 0.5).float()
        total += targets.size(0)
        correct += (predicted == targets).sum().item()
    
    return running_loss / len(loader), 100. * correct / total

# 11. Trénovací smyčka
epochs = 20

for epoch in range(epochs):
    train_loss, train_acc = train_epoch(model, train_loader, criterion, optimizer)
    print(f"Epocha {epoch+1}/{epochs}")
    print(f"Trénovací ztráta: {train_loss:.4f}, Trénovací přesnost: {train_acc:.2f}%")

# 12. Inference příklad
def predict_sentiment(text, model, vocab):
    model.eval()
    tokens = tokenize(text)
    indices = [vocab.get(word, len(vocab)-1) for word in tokens]
    
    # Padding nebo oříznutí
    max_length = 10
    if len(indices) < max_length:
        indices = indices + [0] * (max_length - len(indices))
    else:
        indices = indices[:max_length]
    
    # Převod na tenzor a predikce
    with torch.no_grad():
        input_tensor = torch.tensor([indices])
        output = model(input_tensor).item()
    
    return output, "Pozitivní" if output > 0.5 else "Negativní"

# Vyzkoušejte predikci
test_text = "neural networks are amazing"
probability, sentiment = predict_sentiment(test_text, model, vocab)
print(f"Text: '{test_text}'")
print(f"Sentiment: {sentiment} (Pravděpodobnost: {probability:.2f})")
```
