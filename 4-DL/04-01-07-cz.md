## Implementace neuronové sítě od základů

Implementujme neuronovou síť pouze pomocí NumPy, abychom pochopili základy:

```python
class SimpleNeuralNetwork:
    def __init__(self, input_size, hidden_size, output_size):
        # Inicializace vah malými náhodnými hodnotami
        self.W1 = np.random.randn(input_size, hidden_size) * 0.01
        self.b1 = np.zeros((1, hidden_size))
        self.W2 = np.random.randn(hidden_size, output_size) * 0.01
        self.b2 = np.zeros((1, output_size))
        
    def relu(self, Z):
        return np.maximum(0, Z)
    
    def relu_derivative(self, Z):
        return Z > 0
    
    def sigmoid(self, Z):
        return 1 / (1 + np.exp(-Z))
    
    def sigmoid_derivative(self, Z):
        return self.sigmoid(Z) * (1 - self.sigmoid(Z))
        
    def forward(self, X):
        # První vrstva
        self.Z1 = np.dot(X, self.W1) + self.b1
        self.A1 = self.relu(self.Z1)
        
        # Druhá vrstva
        self.Z2 = np.dot(self.A1, self.W2) + self.b2
        self.A2 = self.sigmoid(self.Z2)
        
        return self.A2
    
    def compute_cost(self, A2, Y):
        m = Y.shape[0]
        # Binární cross entropy ztráta
        cost = -np.sum(Y * np.log(A2) + (1 - Y) * np.log(1 - A2)) / m
        return cost
    
    def backward(self, X, Y):
        m = X.shape[0]
        
        # Výpočet gradientů
        dZ2 = self.A2 - Y
        dW2 = np.dot(self.A1.T, dZ2) / m
        db2 = np.sum(dZ2, axis=0, keepdims=True) / m
        
        dZ1 = np.dot(dZ2, self.W2.T) * self.relu_derivative(self.Z1)
        dW1 = np.dot(X.T, dZ1) / m
        db1 = np.sum(dZ1, axis=0, keepdims=True) / m
        
        return dW1, db1, dW2, db2
    
    def update_parameters(self, dW1, db1, dW2, db2, learning_rate):
        # Aktualizace vah a biasů
        self.W1 -= learning_rate * dW1
        self.b1 -= learning_rate * db1
        self.W2 -= learning_rate * dW2
        self.b2 -= learning_rate * db2
    
    def train(self, X, Y, learning_rate=0.1, epochs=1000):
        costs = []
        
        for epoch in range(epochs):
            # Dopředný průchod
            A2 = self.forward(X)
            
            # Výpočet ztráty
            cost = self.compute_cost(A2, Y)
            
            # Zpětný průchod
            dW1, db1, dW2, db2 = self.backward(X, Y)
            
            # Aktualizace parametrů
            self.update_parameters(dW1, db1, dW2, db2, learning_rate)
            
            # Záznam ztráty
            if epoch % 100 == 0:
                costs.append(cost)
                print(f"Epocha {epoch}, Ztráta: {cost}")
        
        return costs
    
    def predict(self, X):
        return (self.forward(X) > 0.5).astype(int)
    
    def accuracy(self, X, Y):
        predictions = self.predict(X)
        return np.mean(predictions == Y)

# Úprava tvaru Y pro neuronovou síť
```

## Vytvoření neuronové sítě pomocí PyTorch

Implementujme stejnou síť pomocí PyTorch:

```python
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import TensorDataset, DataLoader

# Převod dat na PyTorch tenzory
X_train_tensor = torch.FloatTensor(X_train)
y_train_tensor = torch.FloatTensor(y_train.reshape(-1, 1))
X_test_tensor = torch.FloatTensor(X_test)
y_test_tensor = torch.FloatTensor(y_test.reshape(-1, 1))

# Vytvoření datasetu a dataloaderu
train_dataset = TensorDataset(X_train_tensor, y_train_tensor)
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)

# Definice neuronové sítě
class SimpleNN(nn.Module):
    def __init__(self):
        super(SimpleNN, self).__init__()
        self.layer1 = nn.Linear(2, 10)
        self.layer2 = nn.Linear(10, 1)
        self.relu = nn.ReLU()
        self.sigmoid = nn.Sigmoid()
        
    def forward(self, x):
        x = self.relu(self.layer1(x))
        x = self.sigmoid(self.layer2(x))
        return x

# Inicializace modelu, ztrátové funkce a optimalizátoru
model_pytorch = SimpleNN()
criterion = nn.BCELoss()  # Binární Cross Entropy Ztráta
optimizer = optim.Adam(model_pytorch.parameters(), lr=0.01)

# Trénovací smyčka
epochs = 100
losses = []

for epoch in range(epochs):
    epoch_loss = 0
    for inputs, targets in train_loader:
        # Dopředný průchod
        outputs = model_pytorch(inputs)
        loss = criterion(outputs, targets)
        
        # Zpětný průchod a optimalizace
        optimizer.zero_grad()
        loss.backward()
        optimizer.step()
        
        epoch_loss += loss.item()
    
    avg_loss = epoch_loss / len(train_loader)
    losses.append(avg_loss)
    
    if epoch % 10 == 0:
        print(f'Epocha {epoch}, Ztráta: {avg_loss:.4f}')

# Vykreslení trénovací ztráty
plt.figure(figsize=(10, 6))
plt.plot(losses)
plt.title('Trénovací ztráta')
plt.xlabel('Epocha')
plt.ylabel('Ztráta')
plt.show()

# Vyhodnocení modelu
model_pytorch.eval()
with torch.no_grad():
    # Trénovací přesnost
    train_outputs = model_pytorch(X_train_tensor)
    train_preds = (train_outputs > 0.5).float()
    train_acc = (train_preds == y_train_tensor).float().mean()
    
    # Testovací přesnost
    test_outputs = model_pytorch(X_test_tensor)
    test_preds = (test_outputs > 0.5).float()
    test_acc = (test_preds == y_test_tensor).float().mean()
    
print(f"Trénovací přesnost: {train_acc.item() * 100:.2f}%")
print(f"Testovací přesnost: {test_acc.item() * 100:.2f}%")
```

```python
def plot_decision_boundary(X, y, model, title, model_type='keras'):
    h = 0.02  # Velikost kroku v mřížce
    x_min, x_max = X[:, 0].min() - 1, X[:, 0].max() + 1
    y_min, y_max = X[:, 1].min() - 1, X[:, 1].max() + 1
    xx, yy = np.meshgrid(np.arange(x_min, x_max, h), np.arange(y_min, y_max, h))
    
    # Predikce na mřížce
    if model_type == 'keras':
        Z = model.predict(np.c_[xx.ravel(), yy.ravel()]).reshape(xx.shape)

        with torch.no_grad():
```

```python
# Kontrola explodujících gradientů v vlastní neuronové síti
def check_gradients(model, X, Y):
    # Dopředný průchod
    model.forward(X)
    
    # Zpětný průchod
    dW1, db1, dW2, db2 = model.backward(X, Y)
    
    # Kontrola velikosti gradientů
    grad_magnitudes = {
        'dW1': np.mean(np.abs(dW1)),
        'db1': np.mean(np.abs(db1)),
        'dW2': np.mean(np.abs(dW2)),
        'db2': np.mean(np.abs(db2))
    }
    
    print("Velikosti gradientů:")
    for name, magnitude in grad_magnitudes.items():
        status = "OK"
        if magnitude > 10.0:
            status = "VAROVÁNÍ: Možný explodující gradient"
        elif magnitude < 1e-10:
            status = "VAROVÁNÍ: Možný mizející gradient"
        print(f"{name}: {magnitude:.6f} - {status}")
    
    return grad_magnitudes
```

```python
def initialize_parameters_zeros(layer_dims):
    """
    Inicializace všech vah na nuly
    """
    parameters = {}
    L = len(layer_dims)
    
    for l in range(1, L):
        parameters[f'W{l}'] = np.zeros((layer_dims[l-1], layer_dims[l]))
        parameters[f'b{l}'] = np.zeros((1, layer_dims[l]))

def initialize_parameters_random(layer_dims):
    """
    Náhodná inicializace vah (může vést k explodujícím/mizejícím gradientům)
    """
    parameters = {}
    L = len(layer_dims)
    
    for l in range(1, L):
        parameters[f'W{l}'] = np.random.randn(layer_dims[l-1], layer_dims[l]) * 10
        parameters[f'b{l}'] = np.zeros((1, layer_dims[l]))
        
    return parameters

def initialize_parameters_he(layer_dims):
    """
    He inicializace (vhodná pro ReLU)
    """
    parameters = {}
    L = len(layer_dims)
    
    for l in range(1, L):
        parameters[f'W{l}'] = np.random.randn(layer_dims[l-1], layer_dims[l]) * np.sqrt(2. / layer_dims[l-1])
        parameters[f'b{l}'] = np.zeros((1, layer_dims[l]))
        
    return parameters
```

```python
# Funkce pro hledání rychlosti učení pro TensorFlow/Keras
def lr_finder(X_train, y_train, min_lr=1e-5, max_lr=1, steps=100, epochs=5):
    from tensorflow.keras.callbacks import LambdaCallback
    from tensorflow.keras.models import Sequential
    from tensorflow.keras.layers import Dense
    import numpy as np
    
    # Vytvoření modelu
    model = Sequential([
        Dense(10, activation='relu', input_shape=(X_train.shape[1],)),
        Dense(1, activation='sigmoid')
    ])
    
    # Kompilace modelu s počáteční nízkou rychlostí učení
    model.compile(
        optimizer='adam',
        loss='binary_crossentropy',
        metrics=['accuracy']
    )
    
    # Rychlosti učení k vyzkoušení
    lr_schedule = np.geomspace(min_lr, max_lr, steps)
    learning_rates = []
    losses = []
    
    # Callback pro zaznamenání rychlosti učení a ztráty po každé dávce
    lr_callback = LambdaCallback(
        on_batch_end=lambda batch, logs: 
            learning_rates.append(float(tf.keras.backend.get_value(
                model.optimizer.lr))) if batch == 0 else None
    )
    
    # Trénování modelu s exponenciálně rostoucí rychlostí učení
    for i, lr in enumerate(lr_schedule):
        tf.keras.backend.set_value(model.optimizer.lr, lr)
        print(f"\nRychlost učení: {lr:.7f}")
        
        history = model.fit(
            X_train, y_train,
            epochs=1,
            batch_size=32,
            verbose=0,
            callbacks=[lr_callback]
        )
        
        losses.append(history.history['loss'][0])
    
    # Vykreslení ztráty vs. rychlost učení
    plt.figure(figsize=(10, 6))
    plt.plot(learning_rates, losses)
    plt.xscale('log')
    plt.xlabel('Rychlost učení (logaritmické měřítko)')
    plt.ylabel('Ztráta')
    plt.title('Rychlost učení vs. Ztráta')
```

```python
# Vlastní ztrátová funkce v TensorFlow
def focal_loss(gamma=2., alpha=0.25):
    def loss_fn(y_true, y_pred):
        pt = tf.where(tf.equal(y_true, 1), y_pred, 1 - y_pred)
        return -alpha * tf.pow(1. - pt, gamma) * tf.math.log(pt + 1e-10)
    return loss_fn

# Vlastní ztrátová funkce v PyTorch
class FocalLoss(nn.Module):
    def __init__(self, gamma=2.0, alpha=0.25):
        super(FocalLoss, self).__init__()
        self.gamma = gamma
        self.alpha = alpha
        
    def forward(self, inputs, targets):
        BCE_loss = nn.BCELoss(reduction='none')(inputs, targets)
        pt = torch.exp(-BCE_loss)
        F_loss = self.alpha * (1-pt)**self.gamma * BCE_loss
        return torch.mean(F_loss)
```

```python
# Implementace SGD s momentem od základů
def sgd_momentum(parameters, grads, velocity, learning_rate=0.01, momentum=0.9):
    """
    Stochastický gradientní sestup s aktualizací momentu
    """
    for key in parameters:
        # Aktualizace rychlosti
        velocity[key] = momentum * velocity[key] - learning_rate * grads[key]
        # Aktualizace parametrů
        parameters[key] += velocity[key]
    
    return parameters, velocity
```
