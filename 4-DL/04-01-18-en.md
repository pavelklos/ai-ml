# Model Evaluation for Deep Learning

## 1. Introduction to Model Evaluation

Model evaluation is a crucial step in the deep learning workflow that helps us understand how well our models perform. Unlike traditional machine learning, deep learning models often have unique evaluation challenges due to:

- Large number of parameters
- Tendency to overfit
- Computational resource requirements
- Domain-specific evaluation needs

This guide explores comprehensive techniques for evaluating deep learning models with practical Python implementations.

## 3. Classification Metrics

### 3.1 Basic Metrics (Accuracy, Precision, Recall, F1)

```python
def evaluate_classification_model(model, X_test, y_test, threshold=0.5):
    """Evaluate classification model with multiple metrics"""
    # Get predictions
    if isinstance(model, tf.keras.Model):  # TensorFlow model
        y_pred_prob = model.predict(X_test)
        y_pred = (y_pred_prob > threshold).astype(int)
    else:  # PyTorch model
        model.eval()
        with torch.no_grad():
            y_pred_prob = model(torch.FloatTensor(X_test)).numpy()
            y_pred = (y_pred_prob > threshold).astype(int)
    
    # Calculate metrics
    accuracy = accuracy_score(y_test, y_pred)
    precision = precision_score(y_test, y_pred, average='weighted')
    recall = recall_score(y_test, y_pred, average='weighted')
    f1 = f1_score(y_test, y_pred, average='weighted')
    
    # Print results
    print(f"Accuracy: {accuracy:.4f}")
    print(f"Precision: {precision:.4f}")
    print(f"Recall: {recall:.4f}")
    print(f"F1 Score: {f1:.4f}")
    
    # Display classification report
    print("\nClassification Report:")
    print(classification_report(y_test, y_pred))
    
    return {
        'accuracy': accuracy,
        'precision': precision,
        'recall': recall,
        'f1': f1,
        'y_pred': y_pred,
        'y_pred_prob': y_pred_prob
    }
```

### 3.3 ROC Curve and AUC

ROC (Receiver Operating Characteristic) curves visualize the trade-off between true positive rate and false positive rate:

```python
def plot_roc_curve(y_true, y_scores, multi_class='ovr'):
    """Plot ROC curve for binary or multiclass problems"""
    
    plt.figure(figsize=(10, 8))
    
    # Binary classification
    if len(np.unique(y_true)) == 2:
        fpr, tpr, _ = roc_curve(y_true, y_scores)
        roc_auc = auc(fpr, tpr)
        
        plt.plot(fpr, tpr, color='darkorange', lw=2, 
                 label=f'ROC curve (area = {roc_auc:.2f})')
        plt.plot([0, 1], [0, 1], color='navy', lw=2, linestyle='--')
        plt.xlabel('False Positive Rate')
        plt.ylabel('True Positive Rate')
        plt.title('Receiver Operating Characteristic (ROC) Curve')
        plt.legend(loc='lower right')
        
    # Multiclass classification
    else:
        # One-vs-Rest approach (one curve per class)
        n_classes = len(np.unique(y_true))
        y_true_bin = tf.keras.utils.to_categorical(y_true, n_classes)
        
        for i in range(n_classes):
            fpr, tpr, _ = roc_curve(y_true_bin[:, i], y_scores[:, i])
            roc_auc = auc(fpr, tpr)
            plt.plot(fpr, tpr, lw=2, 
                     label=f'Class {i} (area = {roc_auc:.2f})')
        
        plt.plot([0, 1], [0, 1], 'k--', lw=2)
        plt.xlabel('False Positive Rate')
        plt.ylabel('True Positive Rate')
        plt.title('Multi-class ROC Curve (One-vs-Rest)')
        plt.legend(loc='lower right')
    
    plt.show()
```

## 4. Regression Metrics

For regression tasks in deep learning (like image generation quality, time series prediction), we use different metrics:

```python
def evaluate_regression_model(model, X_test, y_test):
    """Evaluate regression model with multiple metrics"""
    # Get predictions
    if isinstance(model, tf.keras.Model):  # TensorFlow model
        y_pred = model.predict(X_test).flatten()
    else:  # PyTorch model
        model.eval()
        with torch.no_grad():
            y_pred = model(torch.FloatTensor(X_test)).numpy().flatten()
    
    # Calculate metrics
    mse = mean_squared_error(y_test, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_test, y_pred)
    r2 = r2_score(y_test, y_pred)
    
    # Print results
    print(f"Mean Squared Error: {mse:.4f}")
    print(f"Root Mean Squared Error: {rmse:.4f}")
    print(f"Mean Absolute Error: {mae:.4f}")
    print(f"R² Score: {r2:.4f}")
    
    # Plot actual vs predicted
    plt.figure(figsize=(10, 6))
    plt.scatter(y_test, y_pred, alpha=0.5)
    plt.plot([min(y_test), max(y_test)], [min(y_test), max(y_test)], 'r--')
    plt.xlabel('Actual')
    plt.ylabel('Predicted')
    plt.title('Actual vs Predicted Values')
    plt.tight_layout()
    plt.show()
    
    # Plot residuals
    residuals = y_test - y_pred
    plt.figure(figsize=(10, 6))
    plt.scatter(y_pred, residuals, alpha=0.5)
    plt.axhline(y=0, color='r', linestyle='--')
    plt.xlabel('Predicted')
    plt.ylabel('Residuals')
    plt.title('Residual Plot')
    plt.tight_layout()
    plt.show()
    
    return {
        'mse': mse,
        'rmse': rmse,
        'mae': mae,
        'r2': r2,
        'y_pred': y_pred
    }
```

## 5. Cross-Validation for Deep Learning

Cross-validation helps assess model performance more robustly, but requires special handling for deep learning:

```python
def deep_learning_kfold_cv(model_fn, X, y, n_splits=5, epochs=10, batch_size=32):
    """Perform k-fold cross validation with deep learning models
    
    Args:
        model_fn: A function that returns a compiled model
        X: Input features
        y: Target values
        n_splits: Number of folds
        epochs: Number of epochs for each fold
        batch_size: Batch size for training
        
    Returns:
        List of metrics for each fold
    """
    from sklearn.model_selection import KFold
    
    kfold = KFold(n_splits=n_splits, shuffle=True, random_state=42)
    metrics = []
    
    for fold, (train_idx, val_idx) in enumerate(kfold.split(X, y)):
        print(f"\nFold {fold+1}/{n_splits}")
        
        # Split data
        X_train, X_val = X[train_idx], X[val_idx]
        y_train, y_val = y[train_idx], y[val_idx]
        
        # Create and train a new model for each fold
        model = model_fn()
        
        # Train model
        history = model.fit(
            X_train, y_train,
            validation_data=(X_val, y_val),
            epochs=epochs,
            batch_size=batch_size,
            verbose=1
        )
        
        # Evaluate model
        val_loss, val_metric = model.evaluate(X_val, y_val, verbose=0)
        
        # Store metrics
        metrics.append({
            'val_loss': val_loss,
            'val_metric': val_metric,
            'history': history.history
        })
        
    # Print average performance
    avg_val_metric = np.mean([m['val_metric'] for m in metrics])
    print(f"\nAverage validation metric across all folds: {avg_val_metric:.4f}")
    
    return metrics
```

## 6. Learning Curves and Overfitting Detection

Learning curves help us visualize model training progress and detect overfitting:

```python
def plot_learning_curves(history, metrics=['loss'], figsize=(12, 6)):
    """Plot learning curves from training history
    
    Args:
        history: History object from model.fit() or dict with metrics
        metrics: List of metrics to plot
        figsize: Figure size
    """
    plt.figure(figsize=figsize)
    
    for i, metric in enumerate(metrics):
        plt.subplot(1, len(metrics), i+1)
        
        # Plot training metrics
        plt.plot(history[metric], label=f'Training {metric}')
        
        # Plot validation metrics if available
        val_metric = f'val_{metric}'
        if val_metric in history:
            plt.plot(history[val_metric], label=f'Validation {metric}')
        
        plt.title(f'{metric.capitalize()} Over Epochs')
        plt.xlabel('Epochs')
        plt.ylabel(metric.capitalize())
        plt.legend()
        plt.grid(True)
    
    plt.tight_layout()
    plt.show()
    
    # Check for signs of overfitting
    for metric in metrics:
        val_metric = f'val_{metric}'
        if val_metric in history:
            train_final = history[metric][-1]
            val_final = history[val_metric][-1]
            
            # Simple heuristic: if validation performance is significantly worse
            if metric == 'loss' and val_final > 1.1 * train_final:
                print(f"Warning: Possible overfitting detected. Final {val_metric}={val_final:.4f} is {val_final/train_final:.2f}x higher than {metric}={train_final:.4f}")
            elif metric != 'loss' and val_final < 0.9 * train_final:
                print(f"Warning: Possible overfitting detected. Final {val_metric}={val_final:.4f} is {val_final/train_final:.2f}x lower than {metric}={train_final:.4f}")
```

## 7. Model Performance Visualization

### 7.1 Class Activation Maps (CAM)

Visualizing where your model focuses attention in image classification:

```python
def visualize_cam(model, img, layer_name=None, class_idx=None, figsize=(15, 5)):
    """Visualize class activation maps for CNN models
    
    Args:
        model: Trained Keras model
        img: Input image (preprocessed for model)
        layer_name: Name of the last convolutional layer
        class_idx: Index of the class to generate CAM for
                  (defaults to highest prediction)
    """
    # Ensure the image has batch dimension
    if len(img.shape) == 3:
        img = np.expand_dims(img, axis=0)
    
    # If no layer specified, try to find the last conv layer
    if layer_name is None:
        for layer in reversed(model.layers):
            if 'conv' in layer.name:
                layer_name = layer.name
                print(f"Using layer: {layer_name}")
                break
    
    # Get predictions
    preds = model.predict(img)
    
    # If no class specified, use the predicted class
    if class_idx is None:
        class_idx = np.argmax(preds[0])
    
    # Get the class name if available
    class_output = f"Class {class_idx}"
    if hasattr(model, 'classes'):
        class_output = f"Class {model.classes[class_idx]} ({class_idx})"
    
    # Get the last convolutional layer
    last_conv_layer = model.get_layer(layer_name)
    
    # Create a model that maps the input image to activations and predictions
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [last_conv_layer.output, model.output]
    )
    
    # Compute gradient of top predicted class with respect to activations
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img)
        class_channel = predictions[:, class_idx]
    
    # Gradient of class output with respect to feature map activations
    grads = tape.gradient(class_channel, conv_outputs)
    
    # Vector of mean intensity of gradient over feature map
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Weight feature maps by importance to class
    conv_outputs = conv_outputs[0]
    heatmap = conv_outputs @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)
    
    # Normalize heatmap
    heatmap = tf.maximum(heatmap, 0) / tf.math.reduce_max(heatmap)
    heatmap = heatmap.numpy()
    
    # Display original image
    plt.figure(figsize=figsize)
    
    # Original image
    plt.subplot(1, 3, 1)
    # Denormalize the image if necessary
    img_to_show = img[0].copy()
    if img_to_show.max() <= 1.0:
        img_to_show = img_to_show * 255
    plt.imshow(img_to_show.astype('uint8'))
    plt.title(f"Original - Prediction: {class_output}")
    
    # Heatmap
    plt.subplot(1, 3, 2)
    plt.imshow(heatmap, cmap='jet')
    plt.title("Class Activation Map")
    plt.colorbar()
    
    # Superimposed
    plt.subplot(1, 3, 3)
    
    # Resize heatmap to match original image dimensions
    import cv2
    heatmap_resized = cv2.resize(heatmap, (img.shape[2], img.shape[1]))
    heatmap_colored = np.uint8(255 * heatmap_resized)
    heatmap_colored = cv2.applyColorMap(heatmap_colored, cv2.COLORMAP_JET)
    
    # Convert to RGB if needed
    if img_to_show.shape[-1] == 1:  # Grayscale
        img_to_show = cv2.cvtColor(img_to_show.astype('uint8'), cv2.COLOR_GRAY2RGB)
    
    # Superimpose heatmap on original image
    superimposed = heatmap_colored * 0.4 + img_to_show
    superimposed = np.clip(superimposed, 0, 255).astype('uint8')
    
    plt.imshow(superimposed)
    plt.title("Heatmap Overlay")
    plt.tight_layout()
    plt.show()
```

```python
def evaluate_imbalanced_classification(y_true, y_pred, y_score=None):
    """Comprehensive evaluation for imbalanced classification tasks"""
    from sklearn.metrics import balanced_accuracy_score, matthews_corrcoef
    from sklearn.metrics import precision_recall_fscore_support, roc_auc_score
    
    # Basic metrics
    accuracy = accuracy_score(y_true, y_pred)
    balanced_acc = balanced_accuracy_score(y_true, y_pred)
    
    # Class-wise metrics
    precision, recall, f1, support = precision_recall_fscore_support(
        y_true, y_pred, average=None
    )
    
    # Micro and macro averages
    precision_macro = precision_score(y_true, y_pred, average='macro')
    recall_macro = recall_score(y_true, y_pred, average='macro')
    f1_macro = f1_score(y_true, y_pred, average='macro')
    
    # Matthews correlation coefficient (good for imbalanced datasets)
    mcc = matthews_corrcoef(y_true, y_pred)
    
    # AUC if probability scores are provided
    auc_score = None
    if y_score is not None:
        if len(np.unique(y_true)) == 2:  # Binary classification
            auc_score = roc_auc_score(y_true, y_score)
        else:  # Multi-class
            try:
                # One-hot encode for multi-class ROC AUC
                y_true_bin = tf.keras.utils.to_categorical(y_true)
                auc_score = roc_auc_score(y_true_bin, y_score, multi_class='ovr')
            except:
                pass
    
    # Print results
    print(f"Standard Accuracy: {accuracy:.4f}")
    print(f"Balanced Accuracy: {balanced_acc:.4f}")
    print(f"Matthews Correlation Coefficient: {mcc:.4f}")
    
    if auc_score is not None:
        print(f"ROC AUC Score: {auc_score:.4f}")
    
    # Print per-class results
    unique_classes = np.unique(y_true)
    print("\nPer-class metrics:")
    print(f"{'Class':<10} {'Precision':<10} {'Recall':<10} {'F1':<10} {'Support':<10}")

    for i, cls in enumerate(unique_classes):

    print("\nAverage metrics:")
    print(f"Macro Precision: {precision_macro:.4f}")
    print(f"Macro Recall: {recall_macro:.4f}")
    print(f"Macro F1: {f1_macro:.4f}")
    
    # Plot confusion matrix
    conf_mat = confusion_matrix(y_true, y_pred)
    plt.figure(figsize=(10, 8))
    
    # Normalize by row (true labels) for better visualization of imbalanced data
```

## 10. Model Calibration

Calibration measures how well a model's predicted probabilities match observed frequencies:

```python
def evaluate_model_calibration(y_true, y_prob, n_bins=10):
    """Evaluate and visualize model calibration
    
    Args:
        y_true: True binary labels
        y_prob: Predicted probabilities
        n_bins: Number of bins for calibration curve
    """
    from sklearn.calibration import calibration_curve
    
    plt.figure(figsize=(10, 8))
    
    # For binary classification
    if len(y_prob.shape) == 1 or y_prob.shape[1] == 1:
        # Compute calibration curve
        prob_true, prob_pred = calibration_curve(y_true, y_prob, n_bins=n_bins)
        
        # Plot calibration curve
        plt.plot(prob_pred, prob_true, marker='o', linewidth=2, label='Model')
        plt.plot([0, 1], [0, 1], 'k--', label='Perfectly Calibrated')
        
        # Calculate and plot calibration error
        calibration_error = np.mean((prob_true - prob_pred) ** 2)
        plt.text(0.05, 0.9, f'Calibration Error: {calibration_error:.4f}', 
                 transform=plt.gca().transAxes, fontsize=12)
        
        plt.xlabel('Mean Predicted Probability')
        plt.ylabel('Fraction of Positives')
        plt.title('Calibration Curve')
        plt.legend()
        plt.grid(True)
    
    # For multi-class, we calculate calibration for each class
    else:
        n_classes = y_prob.shape[1]
        y_true_bin = tf.keras.utils.to_categorical(y_true, n_classes)
        
        for i in range(n_classes):
            prob_true, prob_pred = calibration_curve(y_true_bin[:, i], 
                                                   y_prob[:, i], 
                                                   n_bins=n_bins)
            plt.plot(prob_pred, prob_true, marker='o', linewidth=2, 
                     label=f'Class {i}')
        
        plt.plot([0, 1], [0, 1], 'k--', label='Perfectly Calibrated')
        plt.xlabel('Mean Predicted Probability')
        plt.ylabel('Fraction of Positives')
        plt.title('Calibration Curves for Multi-class')
        plt.legend()
        plt.grid(True)
    
    plt.show()
    
    # Reliability diagram (histogram of predictions and their accuracy)
    plt.figure(figsize=(12, 6))
    
    if len(y_prob.shape) == 1 or y_prob.shape[1] == 1:
        # Create bins of predictions
        bin_edges = np.linspace(0, 1, n_bins + 1)
        bin_indices = np.digitize(y_prob, bin_edges) - 1
        bin_indices = np.minimum(bin_indices, n_bins - 1)  # Cap at max bin
        
        # Calculate accuracy in each bin
        bin_accs = np.array([np.mean(y_true[bin_indices == i]) 
                           for i in range(n_bins) 
                           if np.sum(bin_indices == i) > 0])
        
        # Calculate mean prediction in each bin
        bin_preds = np.array([np.mean(y_prob[bin_indices == i]) 
                            for i in range(n_bins) 
                            if np.sum(bin_indices == i) > 0])
        
        # Count samples in each bin
        bin_counts = np.array([np.sum(bin_indices == i) 
                             for i in range(n_bins) 
                             if np.sum(bin_indices == i) > 0])
        
        # Plot reliability diagram
        plt.bar(range(len(bin_counts)), bin_counts / np.sum(bin_counts), 
                alpha=0.3, label='Fraction of Samples')
        
        plt.plot(range(len(bin_accs)), bin_accs, 'ro-', label='Accuracy')
        plt.plot(range(len(bin_preds)), bin_preds, 'bs-', label='Mean Prediction')
        
        # Calculate Expected Calibration Error (ECE)
        ece = np.sum(bin_counts / np.sum(bin_counts) * np.abs(bin_accs - bin_preds))
        plt.text(0.05, 0.9, f'Expected Calibration Error: {ece:.4f}', 
                 transform=plt.gca().transAxes, fontsize=12)
        
        plt.xlabel('Bin')
        plt.ylabel('Value')
        plt.title('Reliability Diagram')
        plt.legend()
        plt.grid(True)
        
    plt.tight_layout()
    plt.show()
```

Statistical comparison of model performance:

```python
def compare_models_statistically(models, model_names, X_test, y_test, n_bootstrap=1000, alpha=0.05):
    """Compare multiple models using bootstrap sampling
    
    Args:
        models: List of trained models
        model_names: List of model names
        X_test: Test features
        y_test: Test labels
        n_bootstrap: Number of bootstrap samples
        alpha: Significance level
    """
    import scipy.stats as stats
    
    n_models = len(models)
    n_samples = len(y_test)
    
    # Get predictions from each model
    all_predictions = []
    for model in models:
        if isinstance(model, tf.keras.Model):
            y_pred = model.predict(X_test)
            if len(y_pred.shape) > 1 and y_pred.shape[1] > 1:
                y_pred = np.argmax(y_pred, axis=1)
            all_predictions.append(y_pred)
        else:
            # For PyTorch models
            model.eval()
            with torch.no_grad():
                y_pred = model(torch.FloatTensor(X_test)).numpy()
                if len(y_pred.shape) > 1 and y_pred.shape[1] > 1:
                    y_pred = np.argmax(y_pred, axis=1)
                all_predictions.append(y_pred)
    
    # Calculate performance metrics
    base_metrics = []
    for i in range(n_models):
        accuracy = accuracy_score(y_test, all_predictions[i])
        base_metrics.append(accuracy)
        print(f"{model_names[i]} - Accuracy: {accuracy:.4f}")
    
    # Perform bootstrap sampling
    print("\nPerforming bootstrap sampling...")
    bootstrap_metrics = np.zeros((n_bootstrap, n_models))
    
    for b in range(n_bootstrap):
        # Sample with replacement
        indices = np.random.choice(n_samples, n_samples, replace=True)
        y_bootstrap = y_test[indices]
        
        for i in range(n_models):
            y_pred_bootstrap = all_predictions[i][indices]
            bootstrap_metrics[b, i] = accuracy_score(y_bootstrap, y_pred_bootstrap)
    
    # Compute confidence intervals
    ci_lower = np.percentile(bootstrap_metrics, alpha/2*100, axis=0)
    ci_upper = np.percentile(bootstrap_metrics, (1-alpha/2)*100, axis=0)
    
    print(f"\nBootstrap 95% Confidence Intervals:")
    for i in range(n_models):
        print(f"{model_names[i]}: [{ci_lower[i]:.4f}, {ci_upper[i]:.4f}]")
    
    # Perform pairwise comparisons
    print("\nPairwise Model Comparisons:")
    print("-"*60)
    print(f"{'Model 1':<15} {'Model 2':<15} {'Diff':<10} {'p-value':<10} {'Significant':<10}")
    print("-"*60)
    
    for i in range(n_models):
        for j in range(i+1, n_models):
            # Compute differences for each bootstrap sample
            diffs = bootstrap_metrics[:, i] - bootstrap_metrics[:, j]
            
            # Compute p-value for two-sided test
            if np.mean(diffs) >= 0:
                p_value = 2 * np.mean(diffs <= 0)
            else:
                p_value = 2 * np.mean(diffs >= 0)
            
            # Determine significance
            is_significant = p_value < alpha
            significance = "Yes" if is_significant else "No"
            
            print(f"{model_names[i]:<15} {model_names[j]:<15} "
                  f"{base_metrics[i]-base_metrics[j]:.4f}   {p_value:.4f}    {significance:<10}")
    
    # Visualize model performance with confidence intervals
    plt.figure(figsize=(12, 6))
    
    x = np.arange(n_models)
    width = 0.6
    
    plt.bar(x, base_metrics, width, label='Accuracy', color='skyblue')
    plt.errorbar(x, base_metrics, yerr=[base_metrics-ci_lower, ci_upper-base_metrics],
                 fmt='o', color='blue', capsize=10)
    
    plt.xlabel('Models')
    plt.ylabel('Accuracy')
    plt.title('Model Performance Comparison with Confidence Intervals')
    plt.xticks(x, model_names, rotation=45, ha='right')
    plt.ylim(min(ci_lower)*0.95, max(ci_upper)*1.05)
    plt.grid(True, axis='y', linestyle='--', alpha=0.7)
    plt.tight_layout()
    plt.show()
```

```python
def evaluate_image_generation(real_images, generated_images, resize_to=None):
    """Evaluate image generation quality
    
    Args:
        real_images: Array of real images (n_samples, height, width, channels)
        generated_images: Array of generated images
        resize_to: Optional tuple to resize images before evaluation
    """
    from scipy import linalg
    from skimage.metrics import structural_similarity as ssim
    import tensorflow as tf
    
    # Resize if needed
    if resize_to is not None:
        # Resize real images
        real_resized = []
        for img in real_images:
            real_resized.append(tf.image.resize(img, resize_to).numpy())
        real_images = np.array(real_resized)
        
        # Resize generated images
        gen_resized = []
        for img in generated_images:
            gen_resized.append(tf.image.resize(img, resize_to).numpy())
        generated_images = np.array(gen_resized)
    
    # Ensure images are in correct range [0, 1]
    if real_images.max() > 1.0:
        real_images = real_images / 255.0
    if generated_images.max() > 1.0:
        generated_images = generated_images / 255.0
    
    n_samples = min(len(real_images), len(generated_images))
    
    # Calculate SSIM
    ssim_values = []
    for i in range(n_samples):
        if real_images[i].shape[-1] == 1:
            # Grayscale images
            s = ssim(real_images[i].squeeze(), generated_images[i].squeeze(),
                    data_range=1.0)
        else:
            # RGB images - calculate SSIM for each channel and average
            s = np.mean([ssim(real_images[i][:,:,j], generated_images[i][:,:,j],
                             data_range=1.0) for j in range(real_images[i].shape[-1])])
        ssim_values.append(s)
    
    # Average SSIM
    avg_ssim = np.mean(ssim_values)
    
    # Feature-based metrics (using a pre-trained model)
    try:
        # Load InceptionV3 for feature extraction
        from tensorflow.keras.applications.inception_v3 import InceptionV3, preprocess_input
        
        # Load model
        model = InceptionV3(include_top=False, pooling='avg')
        
        # Extract features
        def extract_features(images):
            # Resize to InceptionV3 input size
            resized = []
            for img in images:
                if img.shape[-1] == 1:  # Grayscale to RGB
                    img = np.repeat(img, 3, axis=-1)
                resized.append(tf.image.resize(img, (299, 299)).numpy())
            
            preprocessed = preprocess_input(np.array(resized) * 255.0)  # Scale back to [0, 255]
            features = model.predict(preprocessed, batch_size=32)
            return features
        
        real_features = extract_features(real_images[:n_samples])
        gen_features = extract_features(generated_images[:n_samples])
        
        # Calculate FID (Fréchet Inception Distance)
        # Calculate mean and covariance
        mu_real = np.mean(real_features, axis=0)
        sigma_real = np.cov(real_features, rowvar=False)
        
        mu_gen = np.mean(gen_features, axis=0)
        sigma_gen = np.cov(gen_features, rowvar=False)
        
        # Calculate FID
        diff = mu_real - mu_gen
        covmean, _ = linalg.sqrtm(sigma_real.dot(sigma_gen), disp=False)
        if np.iscomplexobj(covmean):
            covmean = covmean.real
        
        fid = diff.dot(diff) + np.trace(sigma_real + sigma_gen - 2*covmean)
        
        print(f"Fréchet Inception Distance (FID): {fid:.4f}")
        print(f"Average SSIM: {avg_ssim:.4f}")
        
        # Display sample images
        n_display = min(5, n_samples)
        plt.figure(figsize=(15, 6))
        
        for i in range(n_display):
            # Real image
            plt.subplot(2, n_display, i+1)
            plt.imshow(real_images[i])
            plt.title(f"Real {i+1}")
            plt.axis('off')
            
            # Generated image
            plt.subplot(2, n_display, i+n_display+1)
            plt.imshow(generated_images[i])
            plt.title(f"Generated {i+1}")
            plt.axis('off')
        
        plt.tight_layout()
        plt.show()
        
        return {
            'fid': fid,
            'ssim': avg_ssim,
            'ssim_individual': ssim_values
        }
    
    except Exception as e:
        print(f"Error calculating FID: {e}")
        print(f"Average SSIM: {avg_ssim:.4f}")
        
        return {
            'ssim': avg_ssim,
            'ssim_individual': ssim_values
        }
```

## 15. Common Pitfalls in Deep Learning Evaluation

1. **Data Leakage**: When information from the test set influences training, causing artificially high performance.

2. **Overfitting to the Validation Set**: When hyperparameter tuning causes the model to fit the validation set too well.

3. **Ignoring Statistical Significance**: Small performance differences might not be significant.

4. **Improper Handling of Imbalanced Data**: Using accuracy alone for imbalanced datasets.

5. **Lack of Reproducibility**: Not setting random seeds or not documenting evaluation procedures.

6. **Inappropriate Metrics**: Using metrics that don't align with the business or research goal.

7. **Cherry-Picking Results**: Only reporting metrics that make the model look good.
