# Vizualizace neuronových sítí

## 1. Úvod do vizualizace neuronových sítí

Vizualizace neuronových sítí je nezbytná pro pochopení, ladění a vylepšování modelů hlubokého učení. Transformuje abstraktní matematické konstrukce do intuitivních vizuálních reprezentací, které pomáhají výzkumníkům a praktikům získat vhled do toho, jak se modely učí a jak rozhodují.

Klíčové výhody vizualizace neuronových sítí:
- **Interpretovatelnost**: Odhaluje, jaké příznaky se síť naučila
- **Ladění**: Pomáhá identifikovat problémy v architektuře modelu nebo tréninku
- **Komunikace**: Činí složité modely přístupné i pro neodborníky
- **Důvěra**: Buduje důvěru v rozhodnutí modelu díky transparentnosti

class SimpleCNN(nn.Module):
    def __init__(self):
        super(SimpleCNN, self).__init__()
        self.conv1 = nn.Conv2d(1, 32, kernel_size=3)
        self.pool = nn.MaxPool2d(kernel_size=2)
        self.conv2 = nn.Conv2d(32, 64, kernel_size=3)
        self.fc1 = nn.Linear(64 * 5 * 5, 128)
        self.fc2 = nn.Linear(128, 10)
        self.relu = nn.ReLU()
        self.softmax = nn.Softmax(dim=1)
        
    def forward(self, x):
        x = self.pool(self.relu(self.conv1(x)))
        x = self.pool(self.relu(self.conv2(x)))
        x = x.view(-1, 64 * 5 * 5)
        x = self.relu(self.fc1(x))
        x = self.softmax(self.fc2(x))
        return x

# Vytvoření modelu a zkušebního vstupu

def plot_conv_weights(model, layer_index, max_filters=64):
    """Vykreslení konvolučních filtrů ze zadané vrstvy"""
    # Získání vah vrstvy
    filters = model.layers[layer_index].get_weights()[0]
    
    # Normalizace hodnot filtrů na 0-1
    f_min, f_max = filters.min(), filters.max()
    filters = (filters - f_min) / (f_max - f_min)

    n_filters = min(max_filters, filters.shape[3])

    fig = plt.figure(figsize=(12, 8))

        ax = fig.add_subplot(n_filters // 8 + 1, 8, i + 1)

        if filters.shape[2] == 3:
            # Převod 3kanálového filtru na RGB
            f_img = filters[:, :, :, i]
            ax.imshow(f_img)
        else:
            # Pro jednokanálové filtry
            f_img = filters[:, :, 0, i]
            ax.imshow(f_img, cmap='gray')

        ax.set_yticks([])

def visualize_activations(model, img, layer_names=None):
    """Vizualizace aktivací mezilehlých vrstev pro jeden obrázek"""
    from tensorflow.keras.models import Model
    
    # Pokud není zadáno žádné jméno vrstvy, získáme všechny vrstvy s výstupním tvarem
    if layer_names is None:
        layer_names = [layer.name for layer in model.layers if len(layer.output_shape) > 1]
    
    # Vytvoření modelů pro extrakci aktivací mezilehlých vrstev
    activation_models = [Model(inputs=model.input, outputs=model.get_layer(name).output)
                        for name in layer_names]
    
    # Získání aktivací
    activations = [activation_model.predict(np.expand_dims(img, axis=0))[0]
                  for activation_model in activation_models]
    
    # Vykreslení aktivací
    for layer_name, activation in zip(layer_names, activations):
        # Pro konvoluční vrstvy
        if len(activation.shape) == 3:
            n_features = min(64, activation.shape[2])  # Počet příznaků k vykreslení
            size = activation.shape[0]  # Velikost mapy příznaků
            
            # Vytvoření mřížky aktivačních map
            fig, axes = plt.subplots(n_features // 8 + 1, 8, figsize=(15, n_features // 8 * 2))
            fig.suptitle(f"Aktivační mapy pro vrstvu: {layer_name}")
            
            for i in range(n_features):
                row, col = i // 8, i % 8
                if n_features // 8 + 1 > 1:
                    ax = axes[row, col]
                else:
                    ax = axes[col]
                ax.imshow(activation[:, :, i], cmap='viridis')
                ax.set_xticks([])
                ax.set_yticks([])

            plt.subplots_adjust(top=0.9)

def generate_pattern(model, layer_name, filter_index, iterations=30, step=1.0):
    """Generování vzoru, který maximalizuje aktivaci specifického filtru ve vrstvě"""
    # Vytvoření ztrátové funkce, která maximalizuje aktivaci filtru
    layer_output = model.get_layer(layer_name).output
    loss = tf.reduce_mean(layer_output[:, :, :, filter_index])
    
    # Výpočet gradientů vzhledem ke vstupu
    grads = tf.gradients(loss, model.input)[0]
    # Normalizace gradientů
    grads = grads / (tf.sqrt(tf.reduce_mean(tf.square(grads))) + 1e-5)
    
    # Vytvoření funkce pro výpočet ztráty a gradientů
    iterate = tf.keras.backend.function([model.input], [loss, grads])
    
    # Začátek z šedého obrázku s trochou šumu
    input_img_data = np.random.random((1, 28, 28, 1)) * 0.2 + 0.5
    
    # Spuštění gradientního vzestupu
    for i in range(iterations):
        loss_value, grads_value = iterate([input_img_data])
        input_img_data += grads_value * step
        
        # Oříznutí na platné hodnoty obrázku
        input_img_data = np.clip(input_img_data, 0, 1)
    
    return input_img_data[0]

## 5. Vizualizace gradientů a pozornosti

### 5.1 Grad-CAM vizualizace

Grad-CAM (Gradient-weighted Class Activation Mapping) pomáhá vizualizovat, které oblasti v obrázku jsou důležité pro klasifikaci.

```python
def grad_cam(model, img, class_idx, layer_name=None):
    """
    Generování vizualizace Grad-CAM pro konkrétní predikci třídy
    
    Args:
        model: Natrénovaný model Keras
        img: Vstupní obrázek (předzpracovaný pro model)
        class_idx: Index třídy, pro kterou generovat Grad-CAM
        layer_name: Název poslední konvoluční vrstvy (pokud None, pokusí se ji najít)
    """
    # Pokud není zadána žádná vrstva, pokus se najít poslední konvoluční vrstvu
    if layer_name is None:
        for layer in reversed(model.layers):
            if isinstance(layer, tf.keras.layers.Conv2D):
                layer_name = layer.name
                print(f"Použití vrstvy: {layer_name}")
                break
    
    # Ujistěte se, že obrázek má rozměr dávky
    if len(img.shape) == 3:
        img = np.expand_dims(img, axis=0)
    
    # Vytvoření modelu, který mapuje vstupní obrázek na:
    # 1) Aktivace poslední konvoluční vrstvy
    # 2) Finální predikce tříd
    grad_model = tf.keras.models.Model(
        inputs=[model.inputs],
        outputs=[model.get_layer(layer_name).output, model.output]
    )
    
    # Výpočet gradientů
    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img)
        class_output = predictions[:, class_idx]
    
    # Extrakce gradientů pro třídu vzhledem k mapě příznaků
    grads = tape.gradient(class_output, conv_outputs)
    
    # Globální průměrování gradientů
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Váhování map příznaků podle důležitosti
    conv_outputs = conv_outputs[0]
    heatmap = tf.reduce_sum(tf.multiply(pooled_grads, conv_outputs), axis=-1)
    
    # Normalizace teplotní mapy
    heatmap = tf.maximum(heatmap, 0) / tf.reduce_max(heatmap)
    heatmap = heatmap.numpy()
    
    # Změna velikosti teplotní mapy na původní velikost obrázku
    img_shape = img[0].shape[:2]  # Získání původních rozměrů obrázku
    import cv2
    heatmap = cv2.resize(heatmap, (img_shape[1], img_shape[0]))
    
    # Převod teplotní mapy na RGB a aplikace barevné mapy
    heatmap = np.uint8(255 * heatmap)
    heatmap = cv2.applyColorMap(heatmap, cv2.COLORMAP_JET)
    
    # Kombinace původního obrázku a teplotní mapy
    original_img = img[0] * 255  # Předpokládáme hodnoty 0-1
    original_img = original_img.astype(np.uint8)
    
    # Pokud je obrázek černobílý, převod na RGB
    if len(original_img.shape) == 2 or original_img.shape[2] == 1:
        original_img = cv2.cvtColor(original_img, cv2.COLOR_GRAY2BGR)
    
    # Překrytí teplotní mapy na původní obrázek
    superimposed_img = heatmap * 0.4 + original_img
    superimposed_img = np.clip(superimposed_img, 0, 255).astype(np.uint8)
    
    # Zobrazení obrázků
    fig, axes = plt.subplots(1, 3, figsize=(15, 5))
    
    axes[0].imshow(original_img)
    axes[0].set_title("Původní obrázek")
    axes[0].axis('off')
    
    axes[1].imshow(cv2.cvtColor(heatmap, cv2.COLOR_BGR2RGB))
    axes[1].set_title("Teplotní mapa Grad-CAM")
    axes[1].axis('off')
    
    axes[2].imshow(cv2.cvtColor(superimposed_img, cv2.COLOR_BGR2RGB))
    axes[2].set_title("Překrytí Grad-CAM")
    axes[2].axis('off')
    
    plt.tight_layout()
    plt.show()
    
    return heatmap, superimposed_img

# Načtení testovacího obrázku a aplikace Grad-CAM
(_, _), (x_test, y_test) = mnist.load_data()
sample_img = x_test[0]
sample_img = sample_img.reshape(28, 28, 1) / 255.0

# Predikce třídy

predicted_class = np.argmax(predictions[0])

```

def deep_dream(model, img, layer_name, iterations=20, step=0.01, octave_scale=1.4, num_octaves=3):
    """Implementace zjednodušené verze algoritmu DeepDream"""
    # Získání symbolických výstupů vrstev
    layer_dict = dict([(layer.name, layer) for layer in model.layers])
    loss = tf.reduce_mean(layer_dict[layer_name].output)
    
    # Výpočet gradientů vrstvy vzhledem ke vstupnímu obrázku
    grads = tf.gradients(loss, model.input)[0]
    
    # Normalizace gradientů
    grads = grads / (tf.sqrt(tf.reduce_mean(tf.square(grads))) + 1e-5)
    
    # Nastavení funkce pro výpočet ztráty a gradientů
    iterate = tf.keras.backend.function([model.input], [loss, grads])
    
    # Generování verzí obrázku v různých měřítkách
    octave_shapes = []
    original_shape = img.shape[1:3]
    
    for i in range(num_octaves):
        octave_shape = (original_shape[0] // (octave_scale ** i),
                         original_shape[1] // (octave_scale ** i))
        octave_shapes.append(octave_shape)
    
    octaves = []
    for i in range(num_octaves):
        octave_shape = octave_shapes[i]
        octave = tf.image.resize(img, [int(octave_shape[0]), int(octave_shape[1])])
        octaves.append(octave.numpy())
    
    detail = np.zeros_like(octaves[-1])
    
    # Změna měřítka vstupního obrázku na požadovaný tvar
    for octave_idx in range(num_octaves):
        octave = octaves[num_octaves - octave_idx - 1]
        
        if octave_idx > 0:
            # Zvětšení detailu tak, aby odpovídal aktuálnímu octave
            detail = tf.image.resize(detail, [octave.shape[1], octave.shape[2]]).numpy()
            octave = octave + detail
        
        # Zpracování tohoto octave
        for i in range(iterations):
            loss_value, grads_value = iterate([octave])
            octave += grads_value * step
        
        # Extrakce detailu z tohoto octave
        detail = octave - tf.image.resize(
            octaves[num_octaves - octave_idx - 2] if octave_idx < num_octaves - 1 else img,
            [octave.shape[1], octave.shape[2]]).numpy()
    
    # Oříznutí na platné hodnoty obrázku
    result = np.clip(octave, 0, 1)
    return result

def visualize_features_tsne(model, data, labels, layer_name=None, perplexity=30):
    """
    Vizualizace vysocedimenzionálních příznaků z konkrétní vrstvy pomocí t-SNE
    
    Args:
        model: Natrénovaný model
        data: Vstupní data
        labels: Štítky pro obarvení bodů
        layer_name: Název vrstvy pro extrakci příznaků
        perplexity: Parametr perplexity t-SNE
    """
    from sklearn.manifold import TSNE
    import tensorflow as tf
    
    # Pokud není zadáno žádné jméno vrstvy, použijte vrstvu před poslední vrstvou
    if layer_name is None:
        layer_name = model.layers[-2].name
    
    # Vytvoření modelu, který vypisuje specifikovanou vrstvu
    feature_model = tf.keras.models.Model(
        inputs=model.input,
        outputs=model.get_layer(layer_name).output
    )
    
    # Extrakce příznaků
    features = feature_model.predict(data)
    
    # Zploštění příznaků, pokud je to nutné
    if len(features.shape) > 2:
        features = features.reshape(features.shape[0], -1)
    
    # Aplikace t-SNE pro redukci dimenzí
    print(f"Aplikace t-SNE na {features.shape[1]}-dimenzionální data...")
    tsne = TSNE(n_components=2, perplexity=perplexity, random_state=42)
    tsne_result = tsne.fit_transform(features)
    
    # Vykreslení výsledků
    plt.figure(figsize=(12, 10))
    scatter = plt.scatter(tsne_result[:, 0], tsne_result[:, 1], c=labels, cmap='tab10', alpha=0.7)
    plt.colorbar(scatter, label="Třída")
    plt.title(f"t-SNE vizualizace příznaků z vrstvy '{layer_name}'")
    plt.xlabel("t-SNE příznak 1")
    plt.ylabel("t-SNE příznak 2")
    plt.grid(alpha=0.3)
    plt.tight_layout()
    plt.show()

    # Vytvoření modelu, který vypisuje specifikovanou vrstvu
    feature_model = tf.keras.models.Model(
        inputs=model.input,
        outputs=model.get_layer(layer_name).output
    )
    
    # Extrakce příznaků
    features = feature_model.predict(data)
    
    # Zploštění příznaků, pokud je to nutné
    if len(features.shape) > 2:
        features = features.reshape(features.shape[0], -1)
    
    # Aplikace UMAP pro redukci dimenzí
    print(f"Aplikace UMAP na {features.shape[1]}-dimenzionální data...")
    reducer = umap.UMAP(random_state=42)
    umap_result = reducer.fit_transform(features)
    
    # Vykreslení výsledků
    plt.figure(figsize=(12, 10))
    scatter = plt.scatter(umap_result[:, 0], umap_result[:, 1], c=labels, cmap='tab10', alpha=0.7)
    plt.colorbar(scatter, label="Třída")

def setup_tensorboard(model, log_dir='logs'):
    """Nastavení TensorBoard callbacku a vizualizací"""
    import os
    import datetime
    
    # Vytvoření adresáře pro logy s časovou značkou
    current_time = datetime.datetime.now().strftime("%Y%m%d-%H%M%S")
    log_dir = os.path.join(log_dir, current_time)
    os.makedirs(log_dir, exist_ok=True)
    
    # Vytvoření TensorBoard callbacku
    tensorboard_callback = tf.keras.callbacks.TensorBoard(
        log_dir=log_dir,
        histogram_freq=1,  # Ukládání histogramů vah
        write_graph=True,  # Vizualizace grafu modelu
        write_images=True,  # Vizualizace jakéhokoliv image summary
        update_freq='epoch',  # Aktualizace v každé epoše
        profile_batch=0  # Žádné profilování pro jednoduchost
    )
    
    # Přidání vizualizací embedingů pro váhy první vrstvy
    weights = model.layers[0].get_weights()[0]
    
    if len(weights.shape) == 4:  # Pro konvoluční vrstvy
        # Změna tvaru 4D filtrů na 2D (n_filters, n_channels*filter_height*filter_width)
        n_filters = weights.shape[3]
        flat_weights = weights.reshape(weights.shape[0] * weights.shape[1] * weights.shape[2], n_filters).T
    else:  # Pro plně propojené vrstvy
        flat_weights = weights
    
    # Vytvoření summary writeru
    writer = tf.summary.create_file_writer(os.path.join(log_dir, 'embeddings'))
    
    # Uložení embedingu
    with writer.as_default():
        tf.summary.scalar("loss", 0.0, step=0)  # Dummy hodnota
        writer.flush()
    
    # Výpis instrukcí
    print(f"TensorBoard logy budou uloženy do {log_dir}")
    print("Spusťte TensorBoard pomocí: tensorboard --logdir=logs")
    
    return tensorboard_callback

def visualize_shap_values(model, data, class_names=None, max_display=10):
    """
    Vizualizace SHAP hodnot pro vysvětlení predikcí modelu
    
    Args:
        model: Natrénovaný model
        data: Vstupní data
        class_names: Názvy tříd
        max_display: Maximální počet příznaků k zobrazení
    """
    import shap
    
    # Vytvoření vysvětlovače
    explainer = shap.DeepExplainer(model, data[:100])  # Použití podmnožiny dat jako pozadí
    
    # Výpočet SHAP hodnot pro několik vzorků
    sample_to_explain = data[:10]  # Vysvětlení prvních 10 vzorků
    shap_values = explainer.shap_values(sample_to_explain)
    
    # Vizualizace vysvětlení první predikce
    plt.figure(figsize=(12, 6))
    
    # Pokud jsou data obrázku (podobná MNIST)
    if len(data.shape) == 4 and data.shape[3] in [1, 3]:
        # Pro multi-class modely je shap_values seznam polí pro každou třídu
        if isinstance(shap_values, list):
            # Určení predikované třídy pro první vzorek
            predictions = model.predict(sample_to_explain)
            predicted_class = np.argmax(predictions[0])
            
            # Získání názvu třídy, pokud je dostupný
            class_name = class_names[predicted_class] if class_names else f"Třída {predicted_class}"
            
            # Zobrazení vysvětlení pro predikovanou třídu
            shap.image_plot(shap_values[predicted_class][0:1], sample_to_explain[0:1],
                           show=False)
            plt.title(f"SHAP hodnoty pro {class_name}")
        else:
            # Pro binární klasifikaci
            shap.image_plot(shap_values[0:1], sample_to_explain[0:1], show=False)
    else:
        # Pro tabulární data
        shap.summary_plot(shap_values, sample_to_explain, max_display=max_display, show=False)
    
    plt.tight_layout()
    plt.show()

def create_interactive_activation_viewer(model, sample_images, layer_names=None):
    """
    Vytvoření interaktivního prohlížeče aktivací pomocí ipywidgets
    (Spusťte v Jupyter notebooku)
    
    Args:
        model: Natrénovaný model
        sample_images: Seznam vzorových obrázků
        layer_names: Názvy vrstev k vizualizaci (pokud None, použijí se všechny konv/husté vrstvy)
    """
    import ipywidgets as widgets
    from IPython.display import display
    
    # Získání výchozích názvů vrstev, pokud nejsou zadány
    if layer_names is None:
        layer_names = []
        for layer in model.layers:
            if isinstance(layer, (tf.keras.layers.Conv2D, tf.keras.layers.Dense)):
                layer_names.append(layer.name)
    
    # Vytvoření aktivačních modelů pro každou vrstvu
    activation_models = {}
    for layer_name in layer_names:
        activation_models[layer_name] = tf.keras.models.Model(
            inputs=model.input,
            outputs=model.get_layer(layer_name).output
        )
    
    # Vytvoření widgetů
    layer_dropdown = widgets.Dropdown(
        options=layer_names,
        description='Vrstva:',
        value=layer_names[0]
    )
    
    image_slider = widgets.IntSlider(
        value=0,
        min=0,
        max=len(sample_images) - 1,
        description='Obrázek:',
        continuous_update=False
    )
    
    filter_slider = widgets.IntSlider(
        value=0,
        min=0,
        max=31,  # Výchozí max, bude aktualizováno
        description='Filtr:',
        continuous_update=False
    )
    
    # Vytvoření výstupního widgetu pro zobrazení vizualizací
    output = widgets.Output()
    
    # Definice funkce pro aktualizaci
    def update_visualization(_=None):
        layer_name = layer_dropdown.value
        img_idx = image_slider.value
        filter_idx = filter_slider.value
        
        # Získání aktivací
        img = sample_images[img_idx]
        activations = activation_models[layer_name].predict(np.expand_dims(img, axis=0))[0]
        
        # Aktualizace maximální hodnoty slideru filtrů podle počtu filtrů
        if len(activations.shape) == 3:  # Konvoluční vrstva
            filter_slider.max = activations.shape[2] - 1
            filter_idx = min(filter_idx, activations.shape[2] - 1)
            
            with output:
                output.clear_output(wait=True)
                plt.figure(figsize=(12, 5))
                
                # Zobrazení původního obrázku
                plt.subplot(1, 2, 1)
                plt.imshow(img.squeeze(), cmap='gray')
                plt.title("Původní obrázek")
                plt.axis('off')
                
                # Zobrazení aktivace
                plt.subplot(1, 2, 2)
                plt.imshow(activations[:, :, filter_idx], cmap='viridis')
                plt.title(f"Aktivace pro {layer_name}, Filtr {filter_idx}")
                plt.axis('off')
                
                plt.tight_layout()
                plt.show()
        else:
            # Pro husté vrstvy
            with output:
                output.clear_output(wait=True)
                plt.figure(figsize=(10, 6))
                plt.bar(range(activations.shape[0]), activations)
                plt.title(f"Aktivace pro {layer_name}")
                plt.xlabel("Index neuronu")
                plt.ylabel("Aktivace")
                plt.tight_layout()
                plt.show()
    
    # Nastavení obslužných rutin událostí widgetů
    layer_dropdown.observe(update_visualization, names='value')
    image_slider.observe(update_visualization, names='value')
    filter_slider.observe(update_visualization, names='value')
    
    # Vytvoření rozložení
    controls = widgets.HBox([layer_dropdown, image_slider, filter_slider])
    app = widgets.VBox([controls, output])
    
    # Počáteční aktualizace
    update_visualization()
    
    return app
