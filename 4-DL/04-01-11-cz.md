# Klasifikace obrazu pomocí neuronových sítí

## 1. Úvod do klasifikace obrazu

Klasifikace obrazu je základní úlohou v počítačovém vidění, kde algoritmus identifikuje, do které kategorie (z množiny kategorií) obraz patří. Například klasifikace, zda obrázek obsahuje kočku, psa nebo ptáka.

Neuronové sítě, zejména konvoluční neuronové sítě (CNN), způsobily revoluci v klasifikaci obrazu díky dosažení pozoruhodné přesnosti na komplexních datových sadách.

## 2. Pochopení neuronových sítí pro klasifikaci obrazu

### Co činí CNN efektivními pro zpracování obrazu?

- **Lokální konektivita**: Neurony se propojují pouze s malou oblastí vstupního obrazu
- **Sdílení parametrů**: Stejný filtr je aplikován v celém obrazu
- **Operace sdružování (pooling)**: Redukují prostorové dimenze při zachování důležitých informací
- **Hierarchické učení příznaků**: Od jednoduchých hran ke komplexním tvarům

## 3. Klíčové komponenty CNN

- **Konvoluční vrstvy**: Extrahují příznaky pomocí filtrů
- **Pooling vrstvy**: Redukují dimenzionalitu
- **Aktivační funkce**: Přidávají nelinearitu (ReLU atd.)
- **Plně propojené vrstvy**: Provádějí klasifikaci na základě extrahovaných příznaků
- **Softmax**: Generuje pravděpodobnosti tříd

## 4. Příprava dat

```python
# Import potřebných knihoven
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.datasets import cifar10
from tensorflow.keras.utils import to_categorical
from tensorflow.keras.preprocessing.image import ImageDataGenerator

# Načtení datové sady (CIFAR-10 jako příklad)
(X_train, y_train), (X_test, y_test) = cifar10.load_data()

# Prozkoumání tvaru dat
print(f"Tvar trénovacích obrázků: {X_train.shape}")  # (50000, 32, 32, 3)
print(f"Tvar testovacích obrázků: {X_test.shape}")   # (10000, 32, 32, 3)

# Normalizace hodnot pixelů na rozsah 0 až 1
X_train = X_train.astype('float32') / 255.0
X_test = X_test.astype('float32') / 255.0

# One-hot kódování štítků
num_classes = 10
y_train = to_categorical(y_train, num_classes)
y_test = to_categorical(y_test, num_classes)

# Vizualizace některých obrázků
class_names = ['letadlo', 'automobil', 'pták', 'kočka', 'jelen', 
               'pes', 'žába', 'kůň', 'loď', 'nákladní auto']

plt.figure(figsize=(10, 10))
for i in range(25):
    plt.subplot(5, 5, i + 1)
    plt.imshow(X_train[i])
    plt.title(class_names[np.argmax(y_train[i])])
    plt.axis('off')
plt.tight_layout()
plt.show()

# Augmentace dat pro lepší generalizaci
datagen = ImageDataGenerator(
    rotation_range=15,
    width_shift_range=0.1,
    height_shift_range=0.1,
    horizontal_flip=True,
    zoom_range=0.1
)
datagen.fit(X_train)
```

## 5. Tvorba CNN modelu pro klasifikaci obrazu

```python
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout
from tensorflow.keras.optimizers import Adam

# Definice architektury modelu
model = Sequential([
    # První konvoluční blok
    Conv2D(32, (3, 3), padding='same', activation='relu', input_shape=(32, 32, 3)),
    Conv2D(32, (3, 3), activation='relu'),
    MaxPooling2D(pool_size=(2, 2)),
    Dropout(0.25),
    
    # Druhý konvoluční blok
    Conv2D(64, (3, 3), padding='same', activation='relu'),
    Conv2D(64, (3, 3), activation='relu'),
    MaxPooling2D(pool_size=(2, 2)),
    Dropout(0.25),
    
    # Zplošťovací a plně propojené vrstvy
    Flatten(),
    Dense(512, activation='relu'),
    Dropout(0.5),
    Dense(num_classes, activation='softmax')
])

# Kompilace modelu
model.compile(
    optimizer=Adam(learning_rate=0.001),
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

# Zobrazení souhrnu modelu
model.summary()
```

## 6. Trénování modelu

```python
# Trénování modelu
batch_size = 64
epochs = 30

history = model.fit(
    datagen.flow(X_train, y_train, batch_size=batch_size),
    steps_per_epoch=len(X_train) // batch_size,
    epochs=epochs,
    validation_data=(X_test, y_test),
    verbose=1
)

# Vykreslení přesnosti a ztráty při trénování a validaci
```

## 8. Použití předtrénovaných modelů (Transfer Learning)

```python
from tensorflow.keras.applications import VGG16, ResNet50
from tensorflow.keras.models import Model
from tensorflow.keras.layers import GlobalAveragePooling2D

# Načtení předtrénovaného modelu (bez klasifikační vrstvy nahoře)
base_model = ResNet50(weights='imagenet', include_top=False, input_shape=(224, 224, 3))

# Zmrazení vrstev základního modelu
for layer in base_model.layers:
    layer.trainable = False

# Přidání vlastních vrstev pro naši klasifikační úlohu
x = base_model.output
x = GlobalAveragePooling2D()(x)
x = Dense(256, activation='relu')(x)
x = Dropout(0.5)(x)
predictions = Dense(num_classes, activation='softmax')(x)

# Vytvoření modelu s transfer learningem
transfer_model = Model(inputs=base_model.input, outputs=predictions)

# Kompilace modelu
transfer_model.compile(
    optimizer=Adam(learning_rate=0.0001),
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

# Musíme změnit velikost našich obrázků pro předtrénovaný model
from tensorflow.keras.preprocessing.image import ImageDataGenerator

# Příklad použití generátorů dat s změnou velikosti obrázků pro reálnou datovou sadu
train_datagen = ImageDataGenerator(
    rescale=1./255,
    rotation_range=20,
    width_shift_range=0.2,
    height_shift_range=0.2,
    horizontal_flip=True,
    fill_mode='nearest'
)

test_datagen = ImageDataGenerator(rescale=1./255)

# Příklad cest - nahraďte vašimi skutečnými cestami k datům
train_generator = train_datagen.flow_from_directory(
    'cesta/k/trenovacim_datum',
    target_size=(224, 224),
    batch_size=32,
    class_mode='categorical'
)

validation_generator = test_datagen.flow_from_directory(
    'cesta/k/validacnim_datum',
    target_size=(224, 224),
    batch_size=32,
    class_mode='categorical'
)

# Trénování modelu s transfer learningem
history = transfer_model.fit(
    train_generator,
    steps_per_epoch=train_generator.samples // 32,
    epochs=10,
    validation_data=validation_generator,
    validation_steps=validation_generator.samples // 32
)
```

## 10. Pokročilé techniky

### Grad-CAM vizualizace

```python
import tensorflow as tf
import cv2

def make_gradcam_heatmap(img_array, model, last_conv_layer_name, pred_index=None):
    # Nejprve vytvoříme model, který mapuje vstup na aktivace poslední konvoluční vrstvy
    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [model.get_layer(last_conv_layer_name).output, model.output]
    )
    
    # Poté vypočítáme gradient nejvíce predikované třídy
    with tf.GradientTape() as tape:
        last_conv_layer_output, preds = grad_model(img_array)
        if pred_index is None:
            pred_index = tf.argmax(preds[0])
        class_channel = preds[:, pred_index]
            
    # Gradient výstupu vzhledem k výstupu poslední konvoluční vrstvy
    grads = tape.gradient(class_channel, last_conv_layer_output)
    
    # Vektor střední intenzity gradientu přes kanály feature map
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    
    # Vážení výstupní feature mapy gradienty
    last_conv_layer_output = last_conv_layer_output[0]
    heatmap = last_conv_layer_output @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)
    
    # Normalizace teplotní mapy
    heatmap = tf.maximum(heatmap, 0) / tf.math.reduce_max(heatmap)
    
    return heatmap.numpy()

# Příklad použití
img = X_test[0]  # Příklad obrázku
img_array = np.expand_dims(img, axis=0)

# Vygenerování teplotní mapy (předpokládáme, že 'conv2d_2' je název poslední konvoluční vrstvy)
heatmap = make_gradcam_heatmap(img_array, model, 'conv2d_2')

# Zobrazení původního obrázku a teplotní mapy
plt.figure(figsize=(10, 4))
plt.subplot(1, 2, 1)
plt.imshow(img)
plt.title("Původní obrázek")
plt.axis('off')

plt.subplot(1, 2, 2)
plt.imshow(img)
plt.imshow(heatmap, alpha=0.6, cmap='jet')
plt.title("Grad-CAM teplotní mapa")
plt.axis('off')
plt.tight_layout()
plt.show()
```
