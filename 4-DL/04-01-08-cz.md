## Úvod

TensorFlow je open-source framework pro strojové učení vyvinutý společností Google, který poskytuje komplexní ekosystém nástrojů pro vytváření a nasazování modelů hlubokého učení. Keras je vysoko-úrovňové API pro neuronové sítě, které běží nad TensorFlow a je navrženo pro rychlé experimentování s hlubokými neuronovými sítěmi.

TensorFlow 2.x integruje Keras jako své oficiální vysoko-úrovňové API, poskytující jednoduché rozhraní při zachování flexibility a výkonu TensorFlow. Tato kombinace nabízí:

- Jednoduché, uživatelsky přívětivé API pro vytváření modelů
- Bezproblémové využití GPU/TPU akcelerace
- Efektivní datové pipeline
- Možnosti nasazení do produkce
- Bohatý ekosystém nástrojů a rozšíření

## Základy TensorFlow

Základním konceptem TensorFlow je tenzor - vícerozměrné pole. Prozkoumejme některé základní operace s tenzory:

```python
# Vytváření tenzorů
scalar = tf.constant(5)
vector = tf.constant([1, 2, 3])
matrix = tf.constant([[1, 2], [3, 4]])
tensor_3d = tf.constant([[[1, 2], [3, 4]], [[5, 6], [7, 8]]])

print(f"Tvar skaláru: {scalar.shape}")
print(f"Tvar vektoru: {vector.shape}")
print(f"Tvar matice: {matrix.shape}")
print(f"Tvar 3D tenzoru: {tensor_3d.shape}")

# Základní operace
a = tf.constant([[1, 2], [3, 4]])
b = tf.constant([[5, 6], [7, 8]])

# Operace po prvcích
print(f"a + b = \n{a + b}")

# Maticové násobení
print(f"a @ b = \n{a @ b}")  # nebo tf.matmul(a, b)

# Automatická diferenciace pomocí GradientTape
x = tf.Variable(3.0)
with tf.GradientTape() as tape:
    y = x**2  # y = x²
  
dy_dx = tape.gradient(y, x)
print(f"dy/dx pro x=3: {dy_dx.numpy()}")  # Mělo by být 2x = 6
```

## Vytváření modelů pomocí Keras

Keras nabízí několik způsobů vytváření modelů. Dva hlavní přístupy jsou Sequential API a Functional API.

### Sequential API

Sequential API je nejjednodušší způsob, jak vytvořit model v Keras:

```python
# Sekvenční model pro klasifikaci MNIST
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),  # Převedení 2D obrázků na 1D vektory
    keras.layers.Dense(128, activation='relu'),   # Skrytá vrstva se 128 neurony
    keras.layers.Dense(10)                        # Výstupní vrstva s 10 neurony (jeden pro každou číslici)
])

# Zobrazení souhrnu modelu
model.summary()
```

### Functional API

Functional API umožňuje vytváření složitějších architektur modelů:

```python
# Functional API pro podobný model
inputs = keras.Input(shape=(28, 28))
x = keras.layers.Flatten()(inputs)
x = keras.layers.Dense(128, activation='relu')(x)
outputs = keras.layers.Dense(10)(x)

# Vytvoření modelu specifikováním vstupů a výstupů
model_functional = keras.Model(inputs=inputs, outputs=outputs)

# Zobrazení souhrnu modelu
model_functional.summary()
```

### Dědění třídy Model

Pro maximální flexibilitu můžete dědit z třídy `keras.Model`:

```python
class MyModel(keras.Model):
    def __init__(self):
        super(MyModel, self).__init__()
        self.flatten = keras.layers.Flatten()
        self.dense1 = keras.layers.Dense(128, activation='relu')
        self.dense2 = keras.layers.Dense(10)
        
    def call(self, inputs):
        x = self.flatten(inputs)
        x = self.dense1(x)
        return self.dense2(x)

# Vytvoření instance modelu
model_subclassed = MyModel()

# Kompilace modelu
model_subclassed.compile(
    optimizer='adam',
    loss=tf.keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)
```

### Vlastní vrstvy

Můžete vytvářet vlastní vrstvy děděním z třídy `keras.layers.Layer`:

```python
class MyCustomLayer(keras.layers.Layer):
    def __init__(self, units=32):
        super(MyCustomLayer, self).__init__()
        self.units = units
        
    def build(self, input_shape):
        # Vytvoření trénovatelných vah
        self.w = self.add_weight(
            shape=(input_shape[-1], self.units),
            initializer='random_normal',
            trainable=True
        )
        self.b = self.add_weight(
            shape=(self.units,),
            initializer='zeros',
            trainable=True
        )
        
    def call(self, inputs):
        # Logika dopředného průchodu
        return tf.matmul(inputs, self.w) + self.b
    
    def get_config(self):
        # Pro serializaci
        config = super(MyCustomLayer, self).get_config()
        config.update({'units': self.units})
        return config

# Použití vlastní vrstvy v modelu
custom_model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    MyCustomLayer(128),
    keras.layers.Activation('relu'),
    keras.layers.Dense(10)
])

custom_model.summary()
```

## Trénování a vyhodnocování

Kompilace a trénování modelů v Keras je přímočaré:

```python
# Příprava dat MNIST pro trénování
(x_train, y_train), (x_test, y_test) = mnist.load_data()
x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

# Změna tvaru pro CNN (přidání rozměru kanálu)
x_train = x_train.reshape(-1, 28, 28, 1)
x_test = x_test.reshape(-1, 28, 28, 1)

# Kompilace CNN modelu
cnn_model.compile(
    optimizer=keras.optimizers.Adam(learning_rate=0.001),
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Trénování modelu s validačními daty
history = cnn_model.fit(
    x_train, y_train,
    batch_size=128,
    epochs=5,
    validation_split=0.1,
    verbose=1
)

# Vyhodnocení na testovacích datech
test_loss, test_acc = cnn_model.evaluate(x_test, y_test, verbose=2)
print(f"\nTestovací přesnost: {test_acc:.4f}")

# Vizualizace historie trénování
plt.figure(figsize=(12, 4))

plt.subplot(1, 2, 1)
plt.plot(history.history['accuracy'], label='Trénovací přesnost')
plt.plot(history.history['val_accuracy'], label='Validační přesnost')
plt.xlabel('Epocha')
plt.ylabel('Přesnost')
plt.legend()

plt.subplot(1, 2, 2)
plt.plot(history.history['loss'], label='Trénovací ztráta')
plt.plot(history.history['val_loss'], label='Validační ztráta')
plt.xlabel('Epocha')
plt.ylabel('Ztráta')
plt.legend()

plt.tight_layout()
plt.show()
```

### Použití callbacků

Keras callbacky poskytují způsob, jak přizpůsobit proces trénování:

```python
# Běžné callbacky
from tensorflow.keras.callbacks import (
    EarlyStopping, ModelCheckpoint, ReduceLROnPlateau, 
    TensorBoard, CSVLogger
)

# Konfigurace callbacků
callbacks = [
    # Zastavení trénování, když se validační ztráta přestane zlepšovat
    EarlyStopping(
        monitor='val_loss',
        patience=3,
        restore_best_weights=True
    ),
    
    # Uložení checkpointů modelu
    ModelCheckpoint(
        filepath='best_model.h5',
        monitor='val_accuracy',
        save_best_only=True
    ),
    
    # Snížení learning rate při dosažení plateau
    ReduceLROnPlateau(
        monitor='val_loss',
        factor=0.1,
        patience=2
    ),
    
    # Logování metrik trénování do CSV
    CSVLogger('training_log.csv')
]

# Trénování s callbacky
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

history = model.fit(
    x_train.reshape(-1, 28, 28), y_train,
    epochs=10,
    validation_split=0.1,
    callbacks=callbacks,
    verbose=1
)
```

# 1. Načtení celého modelu
loaded_model = keras.models.load_model('complete_model.h5')

model_new = keras.Sequential([

## TensorFlow datasety

API `tf.data` poskytuje efektivní způsoby načítání a předzpracování dat:

```python
# Vytvoření datasetu z tenzorů
dataset = tf.data.Dataset.from_tensor_slices((x_train, y_train))

# Základní operace s datasetem
dataset = dataset.shuffle(buffer_size=10000)  # Zamíchání dat
dataset = dataset.batch(32)                   # Vytvoření dávek po 32 prvcích
dataset = dataset.prefetch(tf.data.AUTOTUNE)  # Prefetch pro lepší výkon

# Transformace dat pomocí map
def preprocess(image, label):
    image = tf.cast(image, tf.float32) / 255.0  # Normalizace
    # Přidání augmentace dat
    image = tf.image.random_flip_left_right(image)
    return image, label

dataset = dataset.map(preprocess, num_parallel_calls=tf.data.AUTOTUNE)

# Trénování s tf.data
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28, 1)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

model.fit(
    dataset,
    epochs=5,
    verbose=1
)
```

## Praktický příklad: Vytvoření klasifikátoru

Pojďme vytvořit kompletní klasifikátor pro dataset Fashion MNIST:

```python
# Načtení datasetu Fashion MNIST
from tensorflow.keras.datasets import fashion_mnist

# Názvy tříd pro referenci
class_names = ['Tričko/top', 'Kalhoty', 'Svetr', 'Šaty', 'Kabát',
               'Sandály', 'Košile', 'Tenisky', 'Taška', 'Kotníková obuv']

# Načtení a předzpracování dat
(x_train, y_train), (x_test, y_test) = fashion_mnist.load_data()
x_train = x_train.astype('float32') / 255.0
x_test = x_test.astype('float32') / 255.0

# Vizualizace některých příkladů
plt.figure(figsize=(10, 10))
for i in range(25):
    plt.subplot(5, 5, i+1)
    plt.imshow(x_train[i], cmap='gray')
    plt.title(class_names[y_train[i]])
    plt.axis('off')
plt.tight_layout()
plt.show()

# Vytvoření datové pipeline s tf.data
train_dataset = tf.data.Dataset.from_tensor_slices((x_train, y_train))
train_dataset = train_dataset.shuffle(buffer_size=60000)
train_dataset = train_dataset.batch(64).prefetch(tf.data.AUTOTUNE)

test_dataset = tf.data.Dataset.from_tensor_slices((x_test, y_test))
test_dataset = test_dataset.batch(64).prefetch(tf.data.AUTOTUNE)

# Vytvoření CNN modelu
model = keras.Sequential([
    # Změna tvaru z (28, 28) na (28, 28, 1)
    keras.layers.Reshape((28, 28, 1), input_shape=(28, 28)),
    
    # První konvoluční blok
    keras.layers.Conv2D(32, (3, 3), activation='relu', padding='same'),
    keras.layers.BatchNormalization(),
    keras.layers.Conv2D(32, (3, 3), activation='relu', padding='same'),
    keras.layers.MaxPooling2D((2, 2)),
    keras.layers.Dropout(0.25),
    
    # Druhý konvoluční blok
    keras.layers.Conv2D(64, (3, 3), activation='relu', padding='same'),
    keras.layers.BatchNormalization(),
    keras.layers.Conv2D(64, (3, 3), activation='relu', padding='same'),
    keras.layers.MaxPooling2D((2, 2)),
    keras.layers.Dropout(0.25),
    
    # Flatten a dense vrstvy
    keras.layers.Flatten(),
    keras.layers.Dense(512, activation='relu'),
    keras.layers.BatchNormalization(),
    keras.layers.Dropout(0.5),
    keras.layers.Dense(10)  # 10 tříd
])

# Kompilace modelu s plánováním learning rate
lr_schedule = keras.optimizers.schedules.ExponentialDecay(
    initial_learning_rate=1e-3,
    decay_steps=10000,
    decay_rate=0.9
)
optimizer = keras.optimizers.Adam(learning_rate=lr_schedule)

model.compile(
    optimizer=optimizer,
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Trénování s callbacky
callbacks = [
    keras.callbacks.EarlyStopping(patience=5, restore_best_weights=True),
    keras.callbacks.ModelCheckpoint(
        'fashion_mnist_best.h5',
        save_best_only=True
    ),
    keras.callbacks.TensorBoard(log_dir='./logs')
]

history = model.fit(
    train_dataset,
    epochs=25,
    validation_data=test_dataset,
    callbacks=callbacks
)

# Vyhodnocení finálního modelu
test_loss, test_acc = model.evaluate(test_dataset)
print(f"Testovací přesnost: {test_acc:.4f}")

# Predikce a vizualizace výsledků
predictions = model.predict(test_dataset)
predicted_labels = np.argmax(predictions, axis=1)

# Vizualizace některých predikcí
plt.figure(figsize=(12, 12))
for i in range(25):
    plt.subplot(5, 5, i+1)
    plt.imshow(x_test[i], cmap='gray')
    predicted_class = class_names[predicted_labels[i]]
    actual_class = class_names[y_test[i]]
    color = 'green' if predicted_labels[i] == y_test[i] else 'red'
    plt.title(f"Předpověď: {predicted_class}\nSkutečnost: {actual_class}", 
              color=color, fontsize=9)
    plt.axis('off')
plt.tight_layout()
plt.show()

# Vykreslení matice záměn
from sklearn.metrics import confusion_matrix
import seaborn as sns

cm = confusion_matrix(y_test, predicted_labels)
plt.figure(figsize=(10, 8))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues',
            xticklabels=class_names,
            yticklabels=class_names)
plt.xlabel('Předpovězeno')
plt.ylabel('Skutečnost')
plt.title('Matice záměn')
plt.show()
```

## Vlastní trénovací smyčky

Pro pokročilé případy můžete vytvořit vlastní trénovací smyčky pomocí TensorFlow GradientTape API:

```python
# Definice modelu
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dense(10)
])

# Ztrátová funkce a optimalizátor
loss_fn = keras.losses.SparseCategoricalCrossentropy(from_logits=True)
optimizer = keras.optimizers.Adam(learning_rate=0.001)

# Metriky
train_acc_metric = keras.metrics.SparseCategoricalAccuracy()
val_acc_metric = keras.metrics.SparseCategoricalAccuracy()

# Vlastní trénovací smyčka
@tf.function  # Kompilace pro lepší výkon
def train_step(x, y):
    with tf.GradientTape() as tape:
        # Dopředný průchod
        logits = model(x, training=True)
        # Výpočet ztráty
        loss_value = loss_fn(y, logits)
        
    # Výpočet gradientů
    grads = tape.gradient(loss_value, model.trainable_weights)
    # Aplikace gradientů
    optimizer.apply_gradients(zip(grads, model.trainable_weights))
    
    # Aktualizace metrik
    train_acc_metric.update_state(y, logits)
    
    return loss_value

@tf.function
def val_step(x, y):
    # Dopředný průchod (bez trénování)
    val_logits = model(x, training=False)
    # Aktualizace metrik
    val_acc_metric.update_state(y, val_logits)

# Trénovací smyčka
epochs = 5
batch_size = 64

# Vytvoření datasetů
train_dataset = tf.data.Dataset.from_tensor_slices((x_train.reshape(-1, 28, 28), y_train))
train_dataset = train_dataset.shuffle(buffer_size=60000).batch(batch_size)

val_dataset = tf.data.Dataset.from_tensor_slices((x_test.reshape(-1, 28, 28), y_test))
val_dataset = val_dataset.batch(batch_size)

# Trénovací smyčka
for epoch in range(epochs):
    print(f"\nZačátek epochy {epoch + 1}")
    
    # Trénování
    train_acc_metric.reset_states()
    total_loss = 0
    num_batches = 0
    
    for step, (x_batch, y_batch) in enumerate(train_dataset):
        loss_value = train_step(x_batch, y_batch)
        total_loss += loss_value
        num_batches += 1
        
        if step % 200 == 0:
            print(f"  Trénování - Krok {step}: Ztráta = {loss_value:.4f}")
    
    # Výpočet průměrné trénovací ztráty
    avg_loss = total_loss / num_batches
    
    # Validace
    val_acc_metric.reset_states()
    for x_batch, y_batch in val_dataset:
        val_step(x_batch, y_batch)
    
    # Výpis metrik
    train_acc = train_acc_metric.result()
    val_acc = val_acc_metric.result()
    
    print(f"Epocha {epoch + 1} - "
          f"Ztráta: {avg_loss:.4f}, "
          f"Trénovací přesnost: {train_acc:.4f}, "
          f"Validační přesnost: {val_acc:.4f}")
```

## TensorBoard pro vizualizaci

TensorBoard je nástroj pro vizualizaci TensorFlow, který pomáhá sledovat průběh trénování:

```python
# Nastavení TensorBoard
import datetime

# Vytvoření adresáře pro logy s časovým razítkem
log_dir = "logs/fit/" + datetime.datetime.now().strftime("%Y%m%d-%H%M%S")

# Vytvoření TensorBoard callbacku
tensorboard_callback = tf.keras.callbacks.TensorBoard(
    log_dir=log_dir,
    histogram_freq=1,  # Logování histogramu vah
    update_freq='epoch'  # Aktualizace každou epochu
)

# Definice modelu
model = keras.Sequential([
    keras.layers.Flatten(input_shape=(28, 28)),
    keras.layers.Dense(128, activation='relu', name='dense_1'),
    keras.layers.Dropout(0.2, name='dropout'),
    keras.layers.Dense(10, name='dense_2')
])

model.compile(
    optimizer='adam',
    loss=keras.losses.SparseCategoricalCrossentropy(from_logits=True),
    metrics=['accuracy']
)

# Trénování s TensorBoard callbackem
model.fit(
    x_train.reshape(-1, 28, 28), y_train,
    epochs=5,
    validation_split=0.1,
    callbacks=[tensorboard_callback]
)

# Jak spustit TensorBoard
# V terminálu/příkazovém řádku:
# tensorboard --logdir logs/fit
```

### Profilování s TensorBoard

# Vytvoření profilovacího callbacku
profile_batch = '500,520'  # Profilování od dávky 500 do 520
profiler_callback = tf.keras.callbacks.TensorBoard(

# Trénování s povoleným profilerem
model.fit(
    x_train.reshape(-1, 28, 28), y_train,
