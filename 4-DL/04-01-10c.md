## Ukázky kódu

### Vytvoření vlastního datasetu v PyTorch:

```python
class SyntheticDataset(Dataset):
    def __init__(self, num_samples=1000, feature_dim=10, transform=None):
        # Generování náhodných features a labels pro demonstraci
        self.features = torch.randn(num_samples, feature_dim)
        self.labels = torch.randint(0, 2, (num_samples,))
        self.transform = transform
        
    def __len__(self):
        return len(self.features)
    
    def __getitem__(self, idx):
        x = self.features[idx]
        y = self.labels[idx]
        
        if self.transform:
            x = self.transform(x)
            
        return x, y
```

### Vytvoření datasetu v TensorFlow:

```python
def create_tf_synthetic_dataset(num_samples=1000, feature_dim=10, batch_size=32):
    # Generování náhodných features a labels
    features = tf.random.normal((num_samples, feature_dim))
    labels = tf.random.uniform((num_samples,), 0, 2, dtype=tf.int32)
    
    # Vytvoření TensorFlow datasetu
    dataset = tf.data.Dataset.from_tensor_slices((features, labels))
    
    # Shuffle a batching datasetu
    dataset = dataset.shuffle(buffer_size=num_samples)
    dataset = dataset.batch(batch_size)
    
    return dataset
```

### Řešení nevyváženého datasetu pomocí přesamplování:

```python
def random_undersample(X, y, ratio=1.0):
    """Náhodné podsamplovávání majoritní třídy"""
    # Identifikace tříd a jejich indexů
    classes, counts = np.unique(y, return_counts=True)
    minority_class = classes[np.argmin(counts)]
    majority_class = classes[np.argmax(counts)]
    
    # Získání indexů pro každou třídu
    minority_indices = np.where(y == minority_class)[0]
    majority_indices = np.where(y == majority_class)[0]
    
    # Výpočet počtu vzorků k ponechání z majoritní třídy
    n_minority = len(minority_indices)
    n_majority_to_keep = int(n_minority * ratio)
    
    # Náhodný výběr vzorků z majoritní třídy
    majority_indices_to_keep = np.random.choice(majority_indices, size=n_majority_to_keep, replace=False)
    
    # Kombinace minoritních a vybraných majoritních indexů
    indices_to_keep = np.concatenate([minority_indices, majority_indices_to_keep])
    
    # Vrácení vyváženého datasetu
    return X[indices_to_keep], y[indices_to_keep]
```
