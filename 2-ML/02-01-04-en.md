## 1. Introduction to Linear Regression

Linear regression is one of the most fundamental and widely used algorithms in machine learning. It establishes a relationship between a dependent variable (target) and one or more independent variables (features) by fitting a linear equation to the observed data.

```python
# Import essential libraries for linear regression

```

Linear regression aims to find the best-fitting line (or hyperplane in multiple dimensions) that minimizes the difference between observed and predicted values. The algorithm is intuitive, computationally efficient, and serves as a building block for more complex algorithms.

## 2. Simple Linear Regression

Simple linear regression involves predicting a target variable based on a single feature. The model can be represented as:

y = β₀ + β₁x + ε

Where:
- y is the target variable
- x is the feature variable
- β₀ is the y-intercept (bias term)
- β₁ is the slope (coefficient)
- ε is the error term


## 4. Mathematical Foundations

Linear regression is solved by minimizing the sum of squared errors between the observed and predicted values. This is known as the Ordinary Least Squares (OLS) method.

The cost function J(θ) for linear regression is:

J(θ) = (1/2m) * Σ(h_θ(x^(i)) - y^(i))²

Where:
- m is the number of training examples
- h_θ(x) is the hypothesis function (prediction)
- y is the actual value

```python
# Implementing the cost function manually
def cost_function(X, y, theta):
    m = len(y)
    predictions = X.dot(theta)
    cost = (1/(2*m)) * np.sum(np.square(predictions - y))
    return cost

# For simple linear regression example
X_with_bias = np.c_[np.ones((X_train.shape[0], 1)), X_train]  # Add intercept term
theta = np.random.randn(2, 1)  # Random initial parameters

# Calculate cost
initial_cost = cost_function(X_with_bias, y_train, theta)
print(f"Initial Cost with random parameters: {initial_cost:.4f}")

# Gradient descent can be implemented to find the optimal parameters
def gradient_descent(X, y, theta, alpha, iterations):
    m = len(y)
    cost_history = np.zeros(iterations)
    theta_history = np.zeros((iterations, theta.shape[0]))
    
    for i in range(iterations):
        predictions = X.dot(theta)
        errors = predictions - y
        gradient = (1/m) * X.T.dot(errors)
        theta = theta - alpha * gradient
        theta_history[i, :] = theta.T
        cost_history[i] = cost_function(X, y, theta)
    
    return theta, cost_history, theta_history

# Run gradient descent
alpha = 0.01  # Learning rate
iterations = 1000
theta, cost_history, theta_history = gradient_descent(X_with_bias, y_train, theta, alpha, iterations)

print(f"Optimized Parameters:")
print(f"  θ₀ (Intercept): {theta[0][0]:.4f}")
print(f"  θ₁ (Slope): {theta[1][0]:.4f}")
print(f"Final Cost: {cost_history[-1]:.4f}")

# Plot the cost function convergence
plt.figure(figsize=(10, 6))
plt.plot(range(iterations), cost_history)
plt.title('Cost Function Convergence')
plt.xlabel('Iterations')
plt.ylabel('Cost')
plt.show()
```

## 5. Implementing Linear Regression from Scratch

Linear regression can be implemented from scratch using NumPy to better understand its inner workings.

```python
# A complete Linear Regression class implemented from scratch
class LinearRegressionFromScratch:
    def __init__(self, learning_rate=0.01, n_iterations=1000):
        self.learning_rate = learning_rate
        self.n_iterations = n_iterations
        self.weights = None
        self.bias = None
        self.cost_history = []
    
    def fit(self, X, y):
        # Initialize parameters
        n_samples, n_features = X.shape
        self.weights = np.zeros(n_features)
        self.bias = 0
        
        # Gradient descent
        for i in range(self.n_iterations):
            # Linear model prediction
            y_pred = np.dot(X, self.weights) + self.bias
            
            # Compute gradients
            dw = (1/n_samples) * np.dot(X.T, (y_pred - y))
            db = (1/n_samples) * np.sum(y_pred - y)
            
            # Update parameters
            self.weights = self.weights - self.learning_rate * dw
            self.bias = self.bias - self.learning_rate * db
            
            # Calculate cost
            cost = (1/(2*n_samples)) * np.sum(np.square(y_pred - y))
            self.cost_history.append(cost)
    
    def predict(self, X):
        return np.dot(X, self.weights) + self.bias

# Use our implementation on the simple linear regression example
X_train_flat = X_train.reshape(-1)  # Flatten for our implementation

# Print the parameters
```

## 7. Model Evaluation Metrics

To evaluate the performance of a linear regression model, we use various metrics:

```python
# Common regression evaluation metrics
from sklearn.metrics import mean_squared_error, mean_absolute_error, r2_score, explained_variance_score

# Assuming we have actual values y_test and predictions y_pred
def evaluate_regression(y_true, y_pred):
    mse = mean_squared_error(y_true, y_pred)
    rmse = np.sqrt(mse)
    mae = mean_absolute_error(y_true, y_pred)
    r2 = r2_score(y_true, y_pred)
    ev = explained_variance_score(y_true, y_pred)
    
    print(f"Mean Squared Error (MSE): {mse:.4f}")
    print(f"Root Mean Squared Error (RMSE): {rmse:.4f}")
    print(f"Mean Absolute Error (MAE): {mae:.4f}")
    print(f"R² Score: {r2:.4f}")
    print(f"Explained Variance Score: {ev:.4f}")
    
    return {
        'MSE': mse,
        'RMSE': rmse,
        'MAE': mae,
        'R2': r2,
        'EV': ev
    }
```

## 8. Assumptions and Diagnostics

Linear regression relies on several key assumptions. Violations of these assumptions can lead to unreliable results.

```python
# Checking linear regression assumptions
def check_assumptions(X, y, model, y_pred):
    residuals = y - y_pred
    
    fig, axes = plt.subplots(2, 2, figsize=(15, 10))
    
    # 1. Linearity - Residuals should be randomly distributed around zero
    axes[0, 0].scatter(y_pred, residuals, alpha=0.5)
    axes[0, 0].axhline(y=0, color='r', linestyle='--')
    axes[0, 0].set_title('Linearity Check: Residuals vs Fitted Values')
    axes[0, 0].set_xlabel('Predicted Values')
    axes[0, 0].set_ylabel('Residuals')
    
    # 2. Homoscedasticity - Variance of residuals should be constant
    axes[0, 1].scatter(y_pred, np.abs(residuals), alpha=0.5)
    axes[0, 1].set_title('Homoscedasticity Check: |Residuals| vs Fitted Values')
    axes[0, 1].set_xlabel('Predicted Values')
    axes[0, 1].set_ylabel('|Residuals|')
    
    # 3. Normality - Residuals should be normally distributed
    from scipy import stats
    
    # QQ-plot
    stats.probplot(residuals, dist="norm", plot=axes[1, 0])
    axes[1, 0].set_title('Normality Check: QQ-plot')
    
    # Histogram
    axes[1, 1].hist(residuals, bins=30, alpha=0.7, density=True)
    xmin, xmax = axes[1, 1].get_xlim()
    x = np.linspace(xmin, xmax, 100)
    p = stats.norm.pdf(x, np.mean(residuals), np.std(residuals))
    axes[1, 1].plot(x, p, 'k', linewidth=2)
    axes[1, 1].set_title('Normality Check: Residual Distribution')
    axes[1, 1].set_xlabel('Residuals')
    axes[1, 1].set_ylabel('Density')
    
    plt.tight_layout()
    plt.show()
    
    # 4. Multicollinearity (for multiple regression)
    if X.shape[1] > 1:
        # Compute correlation matrix
        corr_matrix = pd.DataFrame(X).corr().abs()
        
        # Plot heatmap
        plt.figure(figsize=(10, 8))
        sns.heatmap(corr_matrix, annot=True, cmap='coolwarm')
        plt.title('Multicollinearity Check: Feature Correlation Matrix')
        plt.show()
        
        # VIF (Variance Inflation Factor)
        from statsmodels.stats.outliers_influence import variance_inflation_factor
        
        # Add a constant to the independent variables
        from statsmodels.tools.tools import add_constant
        X_with_const = add_constant(X)
        
        # Calculate VIF
        vif_data = pd.DataFrame()
        vif_data["Feature"] = X_with_const.columns
        vif_data["VIF"] = [variance_inflation_factor(X_with_const.values, i) 
                           for i in range(X_with_const.shape[1])]
        
        print("Variance Inflation Factors:")
        print(vif_data)

# Apply the diagnostics to the California housing data
# Let's use a subset of features for clearer visualization
X_subset = X_test.iloc[:, :3]  # First 3 features
X_subset_scaled = scaler.transform(X_subset)

# Fit a model on this subset
subset_model = LinearRegression()
subset_model.fit(X_subset_scaled, y_test)
y_subset_pred = subset_model.predict(X_subset_scaled)

# Check assumptions
check_assumptions(X_subset_scaled, y_test, subset_model, y_subset_pred)
```

## 9. Regularization Techniques

Regularization helps prevent overfitting by adding a penalty term to the cost function.

```python
# Implementing Ridge (L2) and Lasso (L1) regularization
from sklearn.linear_model import Ridge, Lasso, ElasticNet
from sklearn.preprocessing import PolynomialFeatures
from sklearn.pipeline import make_pipeline

# Create a more complex dataset to demonstrate overfitting
np.random.seed(0)
X_complex = 6 * np.random.rand(200, 1) - 3
y_complex = 0.5 * X_complex**2 + X_complex + 2 + np.random.randn(200, 1)

# Split the data
X_train_complex, X_test_complex, y_train_complex, y_test_complex = train_test_split(
    X_complex, y_complex, test_size=0.3, random_state=42)

# Create polynomial features to induce overfitting
poly = PolynomialFeatures(degree=10)
X_train_poly = poly.fit_transform(X_train_complex)
X_test_poly = poly.transform(X_test_complex)

# Regular Linear Regression with polynomial features (prone to overfitting)
lr_poly = LinearRegression()
lr_poly.fit(X_train_poly, y_train_complex)
y_train_pred_lr = lr_poly.predict(X_train_poly)
y_test_pred_lr = lr_poly.predict(X_test_poly)

# Ridge Regression (L2 regularization)
ridge = Ridge(alpha=1.0)  # alpha is the regularization strength
ridge.fit(X_train_poly, y_train_complex)
y_train_pred_ridge = ridge.predict(X_train_poly)
y_test_pred_ridge = ridge.predict(X_test_poly)

# Lasso Regression (L1 regularization)
lasso = Lasso(alpha=0.01)  # Lasso can eliminate features (coefficients to zero)
lasso.fit(X_train_poly, y_train_complex)
y_train_pred_lasso = lasso.predict(X_train_poly)
y_test_pred_lasso = lasso.predict(X_test_poly)

# ElasticNet (Combination of L1 and L2)
elastic = ElasticNet(alpha=0.01, l1_ratio=0.5)  # l1_ratio is the mix of L1 vs L2
elastic.fit(X_train_poly, y_train_complex)
y_train_pred_elastic = elastic.predict(X_train_poly)
y_test_pred_elastic = elastic.predict(X_test_poly)

# Evaluate each model
def print_metrics(name, y_train, y_train_pred, y_test, y_test_pred):
    train_mse = mean_squared_error(y_train, y_train_pred)
    test_mse = mean_squared_error(y_test, y_test_pred)

    print(f"  Training R²: {train_r2:.4f}")
    print(f"  Test R²: {test_r2:.4f}")
    print()
    
    return train_mse, test_mse, train_r2, test_r2

results = {}

    'Linear Regression', 
    y_train_complex, y_train_pred_lr, 
    y_test_complex, y_test_pred_lr
)

    'Ridge Regression', 
    y_train_complex, y_train_pred_ridge, 
    y_test_complex, y_test_pred_ridge

    'Lasso Regression', 
    y_train_complex, y_train_pred_lasso,
```

## 10. Summary

Linear regression is a powerful and foundational algorithm in machine learning:

1. **Simplicity**: Linear regression is easy to understand and implement, making it a good starting point for machine learning.

2. **Interpretability**: The model's coefficients provide clear insights into the relationship between features and the target variable.

3. **Efficiency**: Linear regression is computationally efficient and scales well to large datasets.

4. **Foundation**: It forms the basis for more complex algorithms and techniques.

5. **Limitations**: It assumes a linear relationship between variables and is sensitive to outliers.

6. **Regularization**: Techniques like Ridge and Lasso regression help prevent overfitting and improve generalization.

7. **Evaluation**: Various metrics like MSE, RMSE, and R² help assess model performance.

8. **Assumptions**: Understanding and validating the assumptions of linear regression is crucial for reliable results.

Linear regression remains one of the most widely used algorithms in machine learning, particularly for predictive modeling tasks where the relationship between inputs and outputs can be approximated linearly. Despite its simplicity, it often performs surprisingly well in many real-world applications and serves as an excellent baseline model.

```python
def linear_regression_workflow(X, y, test_size=0.2, random_state=42):

    scaler = StandardScaler()
    X_train_scaled = scaler.fit_transform(X_train)
    X_test_scaled = scaler.transform(X_test)
    
    # 3. Try multiple models
    models = {
        'Linear Regression': LinearRegression(),
        'Ridge': Ridge(alpha=1.0),
        'Lasso': Lasso(alpha=0.1)
    }
    
    results = {}
    
    for name, model in models.items():
        # 4. Train the model
        model.fit(X_train_scaled, y_train)
        
        # 5. Make predictions
        y_train_pred = model.predict(X_train_scaled)
        y_test_pred = model.predict(X_test_scaled)
        
        # 6. Evaluate the model
        train_mse = mean_squared_error(y_train, y_train_pred)
        test_mse = mean_squared_error(y_test, y_test_pred)
        train_r2 = r2_score(y_train, y_train_pred)
        test_r2 = r2_score(y_test, y_test_pred)
        
        results[name] = {
            'model': model,
            'train_mse': train_mse,
            'test_mse': test_mse,
            'train_r2': train_r2,
            'test_r2': test_r2
        }
    
    # 7. Find the best model (based on test R²)
    best_model_name = max(results, key=lambda k: results[k]['test_r2'])
    best_model = results[best_model_name]
    
    print(f"Best Model: {best_model_name}")
    print(f"  Test MSE: {best_model['test_mse']:.4f}")
    print(f"  Test R²: {best_model['test_r2']:.4f}")
    
    return results, best_model_name, models[best_model_name]
```
