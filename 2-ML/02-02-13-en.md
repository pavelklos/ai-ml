## What is Deep Learning?

Deep Learning is a subset of machine learning that utilizes neural networks with multiple layers (hence "deep") to progressively extract higher-level features from raw input. Unlike traditional machine learning algorithms that require feature engineering, deep learning can automatically discover the representations needed for detection or classification from raw data.

Key concepts:
- Neural networks with multiple hidden layers
- Ability to automatically learn hierarchical features
- End-to-end learning from raw data to final output
- Specialized architectures for different data types (images, text, etc.)

## TensorFlow and Keras Overview

TensorFlow is an open-source machine learning framework developed by Google Brain that provides a comprehensive ecosystem of tools for deep learning. Keras is a high-level neural networks API that runs on top of TensorFlow, designed with a focus on enabling fast experimentation.

Key benefits:
- **TensorFlow**: Efficient numerical computation, distributed computing, and deployment options
- **Keras**: User-friendly, modular architecture, and easy model building

```python
# Keras is integrated into TensorFlow as tf.keras
# There are two main APIs in Keras:

# 1. Sequential API (simpler, linear stack of layers)
model_sequential = keras.Sequential([
    keras.layers.Dense(64, activation='relu', input_shape=(784,)),
    keras.layers.Dense(64, activation='relu'),
    keras.layers.Dense(10, activation='softmax')
])

# 2. Functional API (more flexible, allows complex model topologies)
inputs = keras.Input(shape=(784,))
x = keras.layers.Dense(64, activation='relu')(inputs)
x = keras.layers.Dense(64, activation='relu')(x)
outputs = keras.layers.Dense(10, activation='softmax')(x)
model_functional = keras.Model(inputs=inputs, outputs=outputs)

# Both APIs create equivalent models in this simple case
print("Sequential Model Summary:")
model_sequential.summary()

print("\nFunctional Model Summary:")
model_functional.summary()
```

## Neural Network Fundamentals

Neural networks are composed of layers of neurons that process input data to make predictions. Let's explore the fundamental building blocks:

```python
# Basic components of neural networks

# 1. Neurons and Activations
x = np.linspace(-10, 10, 100)
activations = {
    'Sigmoid': tf.keras.activations.sigmoid(x),
    'Tanh': tf.keras.activations.tanh(x),
    'ReLU': tf.keras.activations.relu(x),
    'Leaky ReLU': tf.keras.activations.relu(x, alpha=0.1),
    'Softmax': tf.keras.activations.softmax(np.array([x, -x])).T
}

plt.figure(figsize=(12, 8))
for i, (name, values) in enumerate(activations.items(), 1):
    if name == 'Softmax':
        plt.subplot(2, 3, i)
        plt.plot(x, values[:, 0], label='Class 1')
        plt.plot(x, values[:, 1], label='Class 2')
        plt.title(f'{name} Activation')
        plt.legend()
    else:
        plt.subplot(2, 3, i)
        plt.plot(x, values)
        plt.title(f'{name} Activation')
plt.tight_layout()
plt.show()

# 2. Layers
# Dense (Fully Connected) Layer
dense = keras.layers.Dense(32, activation='relu', input_shape=(10,))
print(f"Dense layer output shape: {dense.compute_output_shape((None, 10))}")

# Convolutional Layer
conv = keras.layers.Conv2D(32, kernel_size=(3, 3), activation='relu', input_shape=(28, 28, 1))
print(f"Conv2D layer output shape: {conv.compute_output_shape((None, 28, 28, 1))}")

# Pooling Layer
maxpool = keras.layers.MaxPooling2D(pool_size=(2, 2))
print(f"MaxPooling2D layer output shape: {maxpool.compute_output_shape((None, 28, 28, 32))}")

# Dropout Layer
dropout = keras.layers.Dropout(0.5)
print(f"Dropout preserves shape: {dropout.compute_output_shape((None, 10)) == (None, 10)}")

# 3. Loss Functions
losses = {
    'Binary Crossentropy': keras.losses.BinaryCrossentropy(),
    'Categorical Crossentropy': keras.losses.CategoricalCrossentropy(),
    'Mean Squared Error': keras.losses.MeanSquaredError()
}

# Example of loss calculation
y_true = tf.constant([[0, 1], [0, 0], [1, 1]])
y_pred = tf.constant([[0.2, 0.8], [0.1, 0.2], [0.9, 0.7]])

for name, loss_fn in losses.items():
    loss_value = loss_fn(y_true, y_pred)
    print(f"{name} loss: {loss_value.numpy():.4f}")
```

## Building Your First Neural Network

Let's create a simple neural network to classify the MNIST dataset of handwritten digits:

```python
# Load the MNIST dataset
(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()

# Preprocess the data
x_train = x_train.astype('float32') / 255.0  # Normalize pixel values to [0, 1]
x_test = x_test.astype('float32') / 255.0

# Reshape the data for the model
x_train = x_train.reshape(-1, 28*28)  # Flatten the 28x28 images
x_test = x_test.reshape(-1, 28*28)

# One-hot encode the labels
y_train = keras.utils.to_categorical(y_train, 10)
y_test = keras.utils.to_categorical(y_test, 10)

# Build a simple feedforward neural network
model = keras.Sequential([
    keras.layers.Dense(128, activation='relu', input_shape=(28*28,)),
    keras.layers.Dropout(0.2),  # Add dropout for regularization
    keras.layers.Dense(64, activation='relu'),
    keras.layers.Dropout(0.2),
    keras.layers.Dense(10, activation='softmax')  # 10 output classes (digits 0-9)
])

# Compile the model
model.compile(
    optimizer='adam',  # Adam optimizer
    loss='categorical_crossentropy',  # Loss function for multi-class classification
    metrics=['accuracy']  # Monitor accuracy during training
)

# Display the model architecture
model.summary()

# Visualize the model (optional)
if 'google.colab' in str(get_ipython()):
    from tensorflow.keras.utils import plot_model
    plot_model(model, show_shapes=True, show_layer_names=True)
```

## Saving and Loading Models

After training a model, it's essential to save it for later use or deployment:

```python

# Save only the weights

model_json = model.to_json()
with open("model_architecture.json", "w") as json_file:
    json_file.write(model_json)

# Load the model architecture

    loaded_model_json = json_file.read()

# Load the weights into the new model
new_model.load_weights("mnist_model_weights.h5")

# Compile the model (needed before using it)

    optimizer='adam',
    loss='categorical_crossentropy',
    metrics=['accuracy']
```

## Introduction to CNNs

Convolutional Neural Networks (CNNs) are specialized for processing grid-like data such as images:

```python
# Load MNIST dataset in image format
(x_train, y_train), (x_test, y_test) = keras.datasets.mnist.load_data()

# Reshape for CNN input (add channel dimension)
x_train = x_train.reshape(-1, 28, 28, 1).astype('float32') / 255.0
x_test = x_test.reshape(-1, 28, 28, 1).astype('float32') / 255.0

# One-hot encode the labels
y_train = keras.utils.to_categorical(y_train, 10)
y_test = keras.utils.to_categorical(y_test, 10)

# Build a simple CNN
cnn_model = keras.Sequential([
    # Convolutional layers
    keras.layers.Conv2D(32, kernel_size=(3, 3), activation='relu', input_shape=(28, 28, 1)),
    keras.layers.MaxPooling2D(pool_size=(2, 2)),
    
    keras.layers.Conv2D(64, kernel_size=(3, 3), activation='relu'),
    keras.layers.MaxPooling2D(pool_size=(2, 2)),
    
    # Flatten and dense layers
    keras.layers.Flatten(),
    keras.layers.Dense(128, activation='relu'),
    keras.layers.Dropout(0.5),
    keras.layers.Dense(10, activation='softmax')
])

# Compile the CNN model
cnn_model.compile(
    optimizer='adam',
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

# Display model architecture
cnn_model.summary()

# Train the CNN
cnn_history = cnn_model.fit(
    x_train, y_train,
    batch_size=128,
    epochs=5,
    validation_split=0.1
)

# Evaluate on test set
cnn_test_loss, cnn_test_acc = cnn_model.evaluate(x_test, y_test)
print(f"CNN Test accuracy: {cnn_test_acc:.4f}")

# Visualize feature maps (intermediate activations)
def visualize_feature_maps(model, img_index):
    img = x_test[img_index:img_index+1]  # Get single image
    
    # Create a model that outputs activations from intermediate layers
    layer_outputs = [layer.output for layer in model.layers if isinstance(layer, keras.layers.Conv2D)]
    activation_model = keras.Model(inputs=model.input, outputs=layer_outputs)
    
    # Get all activations for the image
    activations = activation_model.predict(img)
    
    # Visualize the first few feature maps from each convolutional layer
    plt.figure(figsize=(15, 8))
    
    # Display the input image
    plt.subplot(1, len(activations) + 1, 1)
    plt.title("Input Image")
    plt.imshow(img[0, :, :, 0], cmap='gray')
    plt.axis('off')
    
    # Display feature maps
    for i, layer_activations in enumerate(activations):
        n_features = min(8, layer_activations.shape[-1])  # Display up to 8 feature maps
        
        for j in range(n_features):
            plt.subplot(n_features, len(activations) + 1, (j * (len(activations) + 1)) + i + 2)
            plt.title(f"Layer {i+1}, Map {j+1}")
            plt.imshow(layer_activations[0, :, :, j], cmap='viridis')
            plt.axis('off')
    
    plt.tight_layout()
    plt.show()

# Visualize feature maps for a sample image
if np.argmax(y_test[0]) == 5:  # If first test image is digit 5
    visualize_feature_maps(cnn_model, 0)
else:
    # Find an example of digit 5
    for i in range(len(y_test)):
        if np.argmax(y_test[i]) == 5:
            visualize_feature_maps(cnn_model, i)
            break
```

## Working with Pre-trained Models

Transfer learning allows us to leverage pre-trained models to solve new problems more efficiently:

```python
# Import necessary libraries for transfer learning
from tensorflow.keras.applications import VGG16, ResNet50
from tensorflow.keras.preprocessing.image import ImageDataGenerator
from tensorflow.keras.layers import Dense, GlobalAveragePooling2D
from tensorflow.keras.models import Model

# Load a pre-trained model without the top classification layer
base_model = VGG16(weights='imagenet', include_top=False, input_shape=(224, 224, 3))

# Freeze base model layers
for layer in base_model.layers:
    layer.trainable = False

# Add custom layers on top
x = base_model.output
x = GlobalAveragePooling2D()(x)
x = Dense(256, activation='relu')(x)
predictions = Dense(10, activation='softmax')(x)  # 10 classes for example

# Create new model
transfer_model = Model(inputs=base_model.input, outputs=predictions)

# Compile the model
transfer_model.compile(
    optimizer='adam',
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

# Display model structure
print("Transfer learning model summary:")
print(f"Total layers: {len(transfer_model.layers)}")
print(f"Trainable layers: {sum(1 for layer in transfer_model.layers if layer.trainable)}")
print(f"Non-trainable layers: {sum(1 for layer in transfer_model.layers if not layer.trainable)}")

# Example of data preprocessing for pre-trained models
def preprocess_for_vgg(image):
    """Preprocess images for VGG16"""
    # Resize to 224x224 (if needed)
    # Convert to RGB (if needed)
    # Subtract ImageNet mean pixel values
    return tf.keras.applications.vgg16.preprocess_input(image)

# Example of data augmentation for training
train_datagen = ImageDataGenerator(
    preprocessing_function=preprocess_for_vgg,
    rotation_range=20,
    width_shift_range=0.2,
    height_shift_range=0.2,
    shear_range=0.2,
    zoom_range=0.2,
    horizontal_flip=True,
    fill_mode='nearest'
)

# Note: To complete the transfer learning process, you would typically:
# 1. Prepare your dataset (e.g., using flow_from_directory)
# 2. Train the model with your data
# 3. Fine-tune by unfreezing some layers and training with a small learning rate

# To demonstrate transfer learning on images of different sizes, let's also create a more flexible approach
def create_transfer_learning_model(input_shape, num_classes, base_model_name='resnet50'):
    """Create a transfer learning model with flexible input size"""
    if base_model_name.lower() == 'vgg16':
        base_model = VGG16(weights='imagenet', include_top=False, input_shape=input_shape)
        preprocess_func = tf.keras.applications.vgg16.preprocess_input
    elif base_model_name.lower() == 'resnet50':
        base_model = ResNet50(weights='imagenet', include_top=False, input_shape=input_shape)
        preprocess_func = tf.keras.applications.resnet50.preprocess_input
    else:
        raise ValueError("Unsupported base model")
    
    # Freeze base model layers
    for layer in base_model.layers:
        layer.trainable = False
    
    # Add classification head
    x = base_model.output
    x = GlobalAveragePooling2D()(x)
    x = Dense(512, activation='relu')(x)
    x = keras.layers.Dropout(0.5)(x)
    predictions = Dense(num_classes, activation='softmax')(x)
    
    model = Model(inputs=base_model.input, outputs=predictions)
    
    return model, preprocess_func

# Example usage of flexible transfer learning function
custom_model, preprocess_function = create_transfer_learning_model(
    input_shape=(150, 150, 3),  # Custom input shape
    num_classes=5,              # Number of target classes
    base_model_name='resnet50'  # Base model architecture
)

custom_model.compile(
    optimizer=keras.optimizers.Adam(1e-4),  # Lower learning rate for transfer learning
    loss='categorical_crossentropy',
    metrics=['accuracy']
)

print("Custom transfer learning model created!")
```
