## 1. Porozumění modulům v Pythonu

Moduly v Pythonu jsou soubory obsahující pythonový kód, které lze importovat a používat v jiných pythonových programech. Pomáhají organizovat kód do znovupoužitelných komponent, čímž jej činí udržitelnějším a škálovatelnějším. Modul může obsahovat funkce, třídy, proměnné a dokonce i spustitelný kód.

```python
# Příklad toho, co je modul koncepčně
# Toto by bylo v souboru "muj_modul.py"

# Proměnné v modulu
pi = 3.14159
pozdrav = "Ahoj, světe!"

# Funkce v modulu
def ctverec(x):
    """Vrátí druhou mocninu čísla."""
    return x * x

def krychle(x):
    """Vrátí třetí mocninu čísla."""
    return x * x * x

# Třída v modulu
class Kruh:
    def __init__(self, polomer):
        self.polomer = polomer
    
    def plocha(self):
        return pi * self.polomer ** 2
    
    def obvod(self):
        return 2 * pi * self.polomer

# Toto se spustí, když je modul spuštěn přímo
if __name__ == "__main__":
    print("Tento modul je spuštěn přímo.")
else:
    print("Tento modul byl importován.")
```

## 2. Importování modulů

Python poskytuje několik způsobů, jak importovat moduly do vašeho kódu. Můžete importovat celý modul, konkrétní atributy z modulu nebo přejmenovat moduly během importu.

```python
# Importování celého modulu
import math
print(math.sqrt(16))  # Výstup: 4.0

# Importování konkrétních funkcí z modulu
from math import sqrt, pi
print(sqrt(16))       # Výstup: 4.0
print(pi)             # Výstup: 3.141592653589793

# Importování s aliasem
import math as m
print(m.sqrt(16))     # Výstup: 4.0

# Importování všech názvů z modulu (nedoporučuje se v praxi)
from math import *
print(sqrt(16))       # Výstup: 4.0
print(cos(0))         # Výstup: 1.0
```

## 3. Cesta pro vyhledávání modulů

Když importujete modul, Python jej hledá na několika místech. Pochopení této vyhledávací cesty je zásadní při práci s moduly v různých adresářích.

```python
# Zobrazení cesty pro vyhledávání modulů
import sys
print(sys.path)  # Zobrazí seznam adresářů, kde Python hledá moduly

# Přidání adresáře do vyhledávací cesty
import sys
sys.path.append('/cesta/k/mym/modulům')

# Nyní můžete importovat moduly z přidaného adresáře
import muj_vlastni_modul

# Kontrola, zda je modul dostupný
try:
    import neexistujici_modul
except ImportError:
    print("Modul nebyl nalezen!")

# Zjištění, kde se modul nachází
import os
print(os.__file__)  # Zobrazí cestu k souboru modulu os
```

## 4. Vytváření vlastních modulů

Vytvoření vlastních modulů vám umožňuje organizovat kód a sdílet funkčnost mezi různými skripty a projekty.

```python
# Příklad: Vytvoření vlastního modulu
# Uložte to do souboru "kalkulacka.py"

def secti(a, b):
    """Sečte dvě čísla a vrátí výsledek."""
    return a + b

def odecti(a, b):
    """Odečte b od a a vrátí výsledek."""
    return a - b

def vynasob(a, b):
    """Vynásobí dvě čísla a vrátí výsledek."""
    return a * b

def vydel(a, b):
    """Vydělí a číslem b a vrátí výsledek."""
    if b == 0:
        raise ValueError("Nelze dělit nulou!")
    return a / b
```

```python
# V jiném skriptu importujte a použijte vlastní modul
import kalkulacka

vysledek1 = kalkulacka.secti(10, 5)
vysledek2 = kalkulacka.odecti(10, 5)
vysledek3 = kalkulacka.vynasob(10, 5)
vysledek4 = kalkulacka.vydel(10, 5)

print(f"Sčítání: {vysledek1}")      # Výstup: Sčítání: 15
print(f"Odčítání: {vysledek2}")     # Výstup: Odčítání: 5
print(f"Násobení: {vysledek3}")     # Výstup: Násobení: 50
print(f"Dělení: {vysledek4}")       # Výstup: Dělení: 2.0
```

## 5. Práce s balíčky

Balíčky jsou kolekce modulů uspořádaných v adresářích. Poskytují způsob, jak organizovat související moduly do hierarchické struktury.

```python
# Příklad struktury balíčku
# muj_balicek/
# ├── __init__.py
# ├── modul1.py
# ├── modul2.py
# └── podbalicek/
#     ├── __init__.py
#     └── modul3.py

# Soubor __init__.py označuje adresář jako balíček
# Může být prázdný nebo obsahovat inicializační kód

# Importování z balíčku
import muj_balicek.modul1
muj_balicek.modul1.nejaka_funkce()

# Importování konkrétní funkce z modulu v balíčku
from muj_balicek.modul1 import nejaka_funkce
nejaka_funkce()

# Importování z podbalíčku
from muj_balicek.podbalicek import modul3
modul3.dalsi_funkce()

# Relativní importy v rámci balíčku (v modul2.py)
# from . import modul1  # Import ze stejného adresáře
# from .podbalicek import modul3  # Import z podadresáře
# from .. import jiny_balicek  # Import z rodičovského adresáře
```

## 6. Varianty importu

Python nabízí několik způsobů, jak řídit, co se stane během importů a jak jsou názvy dostupné ve vašem kódu.

```python
# Základní import
import math
print(math.pi)  # Je potřeba použít název modulu jako prefix

# Import s přejmenováním
import math as m
print(m.pi)  # Použití kratšího aliasu

# Import konkrétních položek
from math import pi, sqrt
print(pi)    # Použití přímo bez prefixu modulu
print(sqrt(16))

# Import s lokálním přejmenováním
from math import pi as PI, sqrt as odmocnina
print(PI)
print(odmocnina(16))

# Import všech názvů (nedoporučuje se ve většině případů)
from math import *
print(pi)
print(sqrt(16))

# Podmíněné importy
try:
    import matplotlib.pyplot as plt
    MA_MATPLOTLIB = True
except ImportError:
    MA_MATPLOTLIB = False
    print("Matplotlib není nainstalován, vizualizace jsou zakázány.")

# Líné importy (import pouze když je potřeba)
def vykresli_data(data):
    if not 'plt' in globals():  # Kontrola, zda již bylo importováno
        import matplotlib.pyplot as plt
    plt.plot(data)
    plt.show()
```

## 7. Moduly standardní knihovny

Python přichází s bohatou standardní knihovnou modulů, které poskytují různé funkcionality. Tyto moduly jsou dostupné bez instalace dalších balíčků.

```python
# Práce s operačním systémem
import os
aktualni_adresar = os.getcwd()
print(f"Aktuální adresář: {aktualni_adresar}")
os.makedirs("novy_adresar", exist_ok=True)  # Vytvoření adresáře

# Cesty k souborům
import os.path
soubor_existuje = os.path.isfile("data.txt")
print(f"Soubor existuje: {soubor_existuje}")

# Operace s datem a časem
import datetime
nyni = datetime.datetime.now()
print(f"Aktuální datum a čas: {nyni}")
print(f"Rok: {nyni.year}, Měsíc: {nyni.month}, Den: {nyni.day}")

# Náhodná čísla
import random
print(f"Náhodné číslo: {random.random()}")
print(f"Náhodné celé číslo mezi 1 a 10: {random.randint(1, 10)}")
print(f"Náhodný výběr ze seznamu: {random.choice(['jablko', 'banán', 'třešeň'])}")

# Matematické operace
import math
print(f"Druhá odmocnina z 16: {math.sqrt(16)}")
print(f"Hodnota pi: {math.pi}")
print(f"Sinus 0: {math.sin(0)}")

# Zpracování JSON dat
import json
data = {"jmeno": "Jan", "vek": 30, "mesto": "Praha"}
json_retezec = json.dumps(data)
print(f"JSON řetězec: {json_retezec}")
zpracovana_data = json.loads(json_retezec)
print(f"Zpracovaná data: {zpracovana_data}")
```

## 8. Práce s moduly třetích stran

Ekosystém Pythonu zahrnuje tisíce modulů třetích stran, které lze nainstalovat pomocí správců balíčků, jako je pip. Tyto moduly rozšiřují funkčnost Pythonu pro různé specializované úkoly.

```python
# Nejprve nainstalujte modul pomocí pip (spusťte v terminálu/příkazovém řádku)
# pip install requests

# Použití modulu třetích stran pro HTTP požadavky
import requests

# Provedení GET požadavku na webovou stránku
odpoved = requests.get('https://api.github.com')

# Kontrola, zda byl požadavek úspěšný
if odpoved.status_code == 200:
    print("Požadavek byl úspěšný!")
    # Zpracování JSON odpovědi
    data = odpoved.json()
    print(f"Verze API GitHub: {data.get('current_user_url', 'Nenalezeno')}")
else:
    print(f"Požadavek selhal se stavovým kódem: {odpoved.status_code}")

# Příklad s NumPy (potřeba nainstalovat pomocí: pip install numpy)
try:
    import numpy as np
    
    # Vytvoření pole NumPy
    pole = np.array([1, 2, 3, 4, 5])
    print(f"Pole NumPy: {pole}")
    print(f"Průměr pole: {np.mean(pole)}")
    print(f"Směrodatná odchylka pole: {np.std(pole)}")
except ImportError:
    print("NumPy není nainstalován. Nainstalujte jej pomocí: pip install numpy")
```
