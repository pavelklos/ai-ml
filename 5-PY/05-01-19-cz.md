## 1. Úvod do argumentů příkazového řádku a vstupu

Argumenty příkazového řádku a vstup jsou nezbytné pro vytváření flexibilních Python aplikací, které mohou přijímat data od uživatelů bez nutnosti měnit kód. Argumenty příkazového řádku jsou hodnoty poskytnuté při spuštění Python skriptu, zatímco vstup označuje data shromážděná během provádění programu.

## 2. Použití sys.argv pro základní zpracování argumentů

Seznam `sys.argv` obsahuje argumenty příkazového řádku předané Python skriptu. První položka (`sys.argv[0]`) je vždy název skriptu.

```python
import sys

# Základní zpracování argumentů pomocí sys.argv
def main():
    # Vytisknutí názvu skriptu
    print(f"Název skriptu: {sys.argv[0]}")
    
    # Kontrola, zda existují nějaké argumenty
    if len(sys.argv) > 1:
        print(f"Počet argumentů: {len(sys.argv) - 1}")
        # Vytisknutí všech argumentů
        for i, arg in enumerate(sys.argv[1:], start=1):
            print(f"Argument {i}: {arg}")
    else:
        print("Nebyly poskytnuty žádné argumenty")

# Příklad kalkulačky
def jednoduchá_kalkulačka():
    if len(sys.argv) != 4:
        print("Použití: python skript.py <číslo1> <operace> <číslo2>")
        print("Operace: add, subtract, multiply, divide")
        return
    
    try:
        číslo1 = float(sys.argv[1])
        operace = sys.argv[2]
        číslo2 = float(sys.argv[3])
        
        if operace == "add":
            výsledek = číslo1 + číslo2
        elif operace == "subtract":
            výsledek = číslo1 - číslo2
        elif operace == "multiply":
            výsledek = číslo1 * číslo2
        elif operace == "divide":
            if číslo2 == 0:
                print("Chyba: Dělení nulou")
                return
            výsledek = číslo1 / číslo2
        else:
            print("Neznámá operace")
            return
        
        print(f"Výsledek: {výsledek}")
    except ValueError:
        print("Chyba: Prosím zadejte platná čísla")

if __name__ == "__main__":
    # main()
    jednoduchá_kalkulačka()
```

## 3. Modul argparse pro pokročilé parsování příkazového řádku

Modul `argparse` poskytuje sofistikovanější způsob zpracování argumentů příkazového řádku s funkcemi jako automatické generování nápovědy, konverze typů a výchozí hodnoty.

```python
import argparse

def zpracování_pomocí_argparse():
    # Vytvoření objektu parseru
    parser = argparse.ArgumentParser(description="Zpracování celých čísel.")
    
    # Přidání pozičních argumentů
    parser.add_argument("integers", metavar="N", type=int, nargs="+",
                        help="Celé číslo pro akumulátor")
    
    # Přidání volitelných argumentů
    parser.add_argument("--sum", dest="accumulate", action="store_const",
                        const=sum, default=max,
                        help="Sečíst celá čísla (výchozí: najít maximum)")
    
    # Parsování argumentů
    args = parser.parse_args()
    
    # Použití argumentů
    result = args.accumulate(args.integers)
    print(f"Výsledek: {result}")

def vytvoření_procesoru_souborů():
    parser = argparse.ArgumentParser(description="Zpracování souboru s různými možnostmi.")
    
    # Povinný argument souboru
    parser.add_argument("file", help="Soubor ke zpracování")
    
    # Volitelné argumenty různých typů
    parser.add_argument("-c", "--count", action="store_true", 
                        help="Počítání řádků v souboru")
    parser.add_argument("-l", "--limit", type=int, default=0,
                        help="Omezit zpracování na prvních N řádků")
    parser.add_argument("-v", "--verbose", action="store_true",
                        help="Zvýšit podrobnost výstupu")
    parser.add_argument("-o", "--output", type=str,
                        help="Výstupní soubor (výchozí je stdout)")
    parser.add_argument("-m", "--mode", choices=["read", "count", "analyze"],
                        default="read", help="Režim zpracování")
    
    args = parser.parse_args()
    
    # Vytisknutí parsovaných argumentů
    print(f"Soubor ke zpracování: {args.file}")
    print(f"Počítat řádky: {args.count}")
    print(f"Limit řádků: {args.limit}")
    print(f"Podrobný režim: {args.verbose}")
    print(f"Výstupní soubor: {args.output}")
    print(f"Režim: {args.mode}")

if __name__ == "__main__":
    # zpracování_pomocí_argparse()
    vytvoření_procesoru_souborů()
```

## 5. Zpracování validace vstupů

Validace uživatelského vstupu je zásadní pro vytváření robustních aplikací.

```python
def validace_celočíselného_vstupu(výzva, min_hodnota=None, max_hodnota=None):
    """Získání a validace celočíselného vstupu v rámci volitelného rozsahu."""
    while True:
        try:
            hodnota = int(input(výzva))
            
            if min_hodnota is not None and hodnota < min_hodnota:
                print(f"Chyba: Vstup musí být alespoň {min_hodnota}")
                continue
                
            if max_hodnota is not None and hodnota > max_hodnota:
                print(f"Chyba: Vstup musí být nejvýše {max_hodnota}")
                continue
                
            return hodnota
        except ValueError:
            print("Chyba: Zadejte platné celé číslo")

def validace_výběru(výzva, platné_možnosti):
    """Získání a validace vstupu proti seznamu platných možností."""
    while True:
        hodnota = input(výzva).lower()
        if hodnota in platné_možnosti:
            return hodnota
        print(f"Chyba: Zadejte jednu z možností {', '.join(platné_možnosti)}")

def příklad_validace_vstupu():
    # Validace celého čísla v rozsahu
    věk = validace_celočíselného_vstupu("Zadejte svůj věk (18-120): ", 18, 120)
    print(f"Zadaný věk: {věk}")
    
    # Validace výběru z možností
    volba = validace_výběru("Zadejte svou oblíbenou barvu (červená, zelená, modrá): ", ["červená", "zelená", "modrá"])
    print(f"Vybraná barva: {volba}")
    
    # Validace hesla s požadavky
    while True:
        heslo = input("Vytvořte heslo (alespoň 8 znaků s 1 číslem): ")
        if len(heslo) < 8:
            print("Chyba: Heslo musí mít alespoň 8 znaků")
        elif not any(znak.isdigit() for znak in heslo):
            print("Chyba: Heslo musí obsahovat alespoň jedno číslo")
        else:
            break
    
    print("Heslo přijato!")

if __name__ == "__main__":
    příklad_validace_vstupu()
```

## 6. Práce se soubory jako vstupem

Aplikace pro příkazový řádek často zpracovávají soubory určené jako argumenty.

```python
import argparse
import sys

def zpracování_souboru(název_souboru, počítat_řádky=False, limit=0):
    """Zpracování souboru s možnostmi pro počítání řádků a omezení výstupu."""
    try:
        with open(název_souboru, 'r') as soubor:
            řádky = soubor.readlines()
            
            if počítat_řádky:
                print(f"Celkem řádků v {název_souboru}: {len(řádky)}")
                return
            
            # Aplikace limitu, pokud je specifikován
            if limit > 0:
                řádky = řádky[:limit]
            
            # Tisk řádků s čísly řádků
            for i, řádek in enumerate(řádky, start=1):
                print(f"{i}: {řádek.rstrip()}")
                
    except FileNotFoundError:
        print(f"Chyba: Soubor '{název_souboru}' nebyl nalezen")
        return
    except IOError as e:
        print(f"Chyba při čtení souboru: {e}")
        return

def procesor_souborů():
    parser = argparse.ArgumentParser(description="Zpracování textového souboru.")
    parser.add_argument("file", help="Soubor ke zpracování")
    parser.add_argument("-c", "--count", action="store_true", 
                        help="Počet řádků v souboru")
    parser.add_argument("-l", "--limit", type=int, default=0,
                        help="Omezit výstup na prvních N řádků")
    
    args = parser.parse_args()
    zpracování_souboru(args.file, args.count, args.limit)

# Příklad čtení ze stdin (pro přesměrovaný vstup)
def zpracování_stdin():
    print("Čtení ze stdin (napište text, stiskněte Ctrl+D pro dokončení):")
    řádky = sys.stdin.readlines()
    print(f"Přijato {len(řádky)} řádků:")
    for i, řádek in enumerate(řádky, start=1):
        print(f"{i}: {řádek.rstrip()}")

if __name__ == "__main__":
    # procesor_souborů()
    # Pro spuštění: python skript.py nazev_souboru.txt -c -l 10
    
    # Pro příklad stdin:
    # zpracování_stdin()
    # Pro spuštění s přesměrováním: cat nazev_souboru.txt | python skript.py
    
    # Příklad, jak zkontrolovat, zda stdin má data
    if not sys.stdin.isatty():
        print("Data jsou přesměrována")
        zpracování_stdin()
    else:
        print("Žádná přesměrovaná data, spouštím procesor souborů")
        procesor_souborů()
```

## 7. Použití knihovny Click pro elegantní rozhraní příkazového řádku

`Click` je Python balíček pro vytváření krásných rozhraní příkazového řádku s minimálním kódem.

```python
import click

@click.command()
@click.argument('jméno')
@click.option('--počet', default=1, help='Počet pozdravů.')
@click.option('--pozdrav', default='Ahoj', help='Pozdrav, který se má použít.')
@click.option('--velká-písmena', is_flag=True, help='Zobrazit velkými písmeny.')
def ahoj(jméno, počet, pozdrav, velká_písmena):
    """Jednoduchý program, který pozdraví JMÉNO celkem POČETkrát."""
    zpráva = f"{pozdrav}, {jméno}!"
    if velká_písmena:
        zpráva = zpráva.upper()
    for _ in range(počet):
        click.echo(zpráva)

@click.group()
def cli():
    """Nástroj příkazového řádku s podpříkazy."""
    pass

@cli.command()
@click.argument('název_souboru')
@click.option('--limit', default=0, help='Omezit na N řádků')
def čtení(název_souboru, limit):
    """Přečte soubor a vytiskne jeho obsah."""
    try:
        with open(název_souboru, 'r') as f:
            řádky = f.readlines()
            if limit > 0:
                řádky = řádky[:limit]
            for řádek in řádky:
                click.echo(řádek.rstrip())
    except FileNotFoundError:
        click.echo(f"Chyba: Soubor '{název_souboru}' nebyl nalezen.", err=True)
        
@cli.command()
@click.argument('název_souboru')
def počet(název_souboru):
    """Spočítá řádky v souboru."""
    try:
        with open(název_souboru, 'r') as f:
            počet = sum(1 for _ in f)
            click.echo(f"Počet řádků: {počet}")
    except FileNotFoundError:
        click.echo(f"Chyba: Soubor '{název_souboru}' nebyl nalezen.", err=True)

if __name__ == '__main__':
    # Pro spuštění prvního příkladu:
    # ahoj()
    
    # Pro spuštění příkladu skupiny příkazů:
    cli()
```

## 8. Zpracování chyb při vstupech z příkazového řádku

Správné zpracování chyb je zásadní pro uživatelsky přívětivé aplikace příkazového řádku.

```python
import argparse
import sys
import os

def nastavení_parseru():
    parser = argparse.ArgumentParser(
        description='Nástroj pro zpracování souborů',
        formatter_class=argparse.ArgumentDefaultsHelpFormatter
    )
    parser.add_argument('soubor', help='Soubor ke zpracování')
    parser.add_argument('--výstup', '-o', help='Výstupní soubor')
    parser.add_argument('--operace', choices=['počet', 'velká_písmena', 'malá_písmena'],
                        default='počet', help='Operace k provedení')
    return parser

def zpracování_souboru_s_ošetřením_chyb(args):
    # Validace vstupního souboru
    if not os.path.exists(args.soubor):
        sys.stderr.write(f"Chyba: Vstupní soubor '{args.soubor}' neexistuje\n")
        return 1
        
    if not os.path.isfile(args.soubor):
        sys.stderr.write(f"Chyba: '{args.soubor}' není běžný soubor\n")
        return 1
        
    # Validace výstupního souboru
    if args.výstup and os.path.exists(args.výstup):
        odpověď = input(f"Výstupní soubor '{args.výstup}' existuje. Přepsat? (a/n): ").lower()
        if odpověď != 'a':
            print("Operace zrušena")
            return 0
    
    try:
        # Otevření a čtení vstupního souboru
        with open(args.soubor, 'r') as vstup:
            obsah = vstup.read()
        
        # Provedení požadované operace
        if args.operace == 'počet':
            výsledek = f"Řádky: {obsah.count(os.linesep) + 1}\n"
            výsledek += f"Slova: {len(obsah.split())}\n"
            výsledek += f"Znaky: {len(obsah)}\n"
        elif args.operace == 'velká_písmena':
            výsledek = obsah.upper()
        elif args.operace == 'malá_písmena':
            výsledek = obsah.lower()
        
        # Zápis do výstupu nebo stdout
        if args.výstup:
            with open(args.výstup, 'w') as výstup:
                výstup.write(výsledek)
            print(f"Výsledek zapsán do {args.výstup}")
        else:
            print(výsledek)
            
        return 0
        
    except IOError as e:
        sys.stderr.write(f"I/O chyba: {str(e)}\n")
        return 1
    except Exception as e:
        sys.stderr.write(f"Neočekávaná chyba: {str(e)}\n")
        return 1

def main():
    parser = nastavení_parseru()
    args = parser.parse_args()
    
    try:
        return zpracování_souboru_s_ošetřením_chyb(args)
    except KeyboardInterrupt:
        sys.stderr.write("Operace zrušena uživatelem\n")
        return 130  # Standardní návratový kód pro SIGINT

if __name__ == "__main__":
    sys.exit(main())
```

## 10. Shrnutí

Argumenty příkazového řádku a uživatelský vstup jsou základními aspekty vytváření flexibilních a interaktivních Python aplikací. Klíčové poznatky zahrnují:

1. **Základní zpracování argumentů**: `sys.argv` poskytuje jednoduchý přístup k argumentům příkazového řádku.
2. **Pokročilé parsování**: `argparse` nabízí sofistikované parsování argumentů s generováním nápovědy, konverzí typů a dalšími funkcemi.
3. **Uživatelská interakce**: Funkce `input()` umožňuje Python programům získávat data od uživatelů během provádění.
4. **Validace vstupu**: Vždy validujte uživatelský vstup pro vytváření robustních aplikací.
5. **Zpracování souborů**: Nástroje příkazového řádku často zpracovávají soubory určené jako argumenty.
6. **Vylepšená CLI**: Knihovny jako `Click` mohou vytvářet elegantní rozhraní příkazového řádku s minimálním kódem.
7. **Zpracování chyb**: Správné zpracování chyb zlepšuje uživatelskou zkušenost.
8. **Interaktivní CLI**: Pro složité interakce zvažte vytvoření interaktivního systému nabídek.

Při navrhování aplikací pro příkazový řádek se zaměřte na vytváření uživatelsky přívětivých rozhraní s jasnými možnostmi, dobrou dokumentací a elegantním zpracováním chyb. Dodržování těchto principů učiní vaše Python nástroje přístupnějšími a efektivnějšími.
