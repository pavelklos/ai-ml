## 1. Úvod do výjimek a zpracování chyb

Zpracování chyb je nezbytné pro vytváření robustních programů v Pythonu. Když během provádění programu dojde k chybám, Python vyvolá výjimky - signály, že se stalo něco neočekávaného. Bez řádného zpracování chyb tyto výjimky způsobí pád programu. Konstrukce try/except vám umožňuje zachytit a zpracovat tyto výjimky, čímž je váš kód odolnější.

```python
# Bez zpracování chyb
def deleni(a, b):
    return a / b

# Toto způsobí pád programu s výjimkou ZeroDivisionError
# vysledek = deleni(10, 0)

# Základní ukázka toho, co se děje s výjimkami
try:
    print("Pokus o dělení nulou...")
    vysledek = 10 / 0
    print("Tento řádek se nikdy neprovede")
except:
    print("Došlo k chybě!")

print("Program pokračuje v běhu")

# Běžné výjimky, se kterými se můžete setkat:
# ZeroDivisionError: Při dělení nulou
# TypeError: Když jsou operace prováděny s nekompatibilními typy
# ValueError: Když funkce dostane argument správného typu, ale nevhodné hodnoty
# FileNotFoundError: Když se pokusíte přistoupit k souboru, který neexistuje
# IndexError: Když se pokusíte přistoupit k indexu, který je mimo rozsah
```

## 2. Základní struktura try/except

Základní struktura try/except zahrnuje kód, který může vyvolat výjimku v bloku try, s obsluhou těchto výjimek v bloku except.

```python
# Základní vzor try/except
try:
    # Kód, který může vyvolat výjimku
    uzivatelsky_vstup = input("Zadejte číslo: ")
    cislo = int(uzivatelsky_vstup)
    vysledek = 100 / cislo
    print(f"100 děleno {cislo} je {vysledek}")
except:
    # Toto zpracuje JAKOUKOLIV výjimku (nedoporučuje se pro produkční kód)
    print("Něco se pokazilo")

# Lepší praxe: specifické typy výjimek
try:
    uzivatelsky_vstup = input("Zadejte číslo: ")
    cislo = int(uzivatelsky_vstup)
    vysledek = 100 / cislo
    print(f"100 děleno {cislo} je {vysledek}")
except ValueError:
    print("To není platné číslo!")
except ZeroDivisionError:
    print("Nelze dělit nulou!")

# Přístup k objektu výjimky pomocí 'as'
try:
    with open("neexistujici_soubor.txt", "r") as soubor:
        obsah = soubor.read()
except FileNotFoundError as e:
    print(f"Chyba: {e}")
    print(f"Typ chyby: {type(e).__name__}")
```

## 3. Zpracování více typů výjimek

Python nabízí několik způsobů, jak zpracovat více typů výjimek, což umožňuje flexibilní strategie správy chyb.

```python
# Zpracování více typů výjimek jednotlivě
try:
    # Nějaká riziková operace
    cislo = int(input("Zadejte číslo: "))
    vysledek = 100 / cislo
    print(f"Výsledek: {vysledek}")
except ValueError:
    print("Zadejte platné číslo")
except ZeroDivisionError:
    print("Nelze dělit nulou")

# Zpracování více typů výjimek se stejnou obsluhou
try:
    soubor = open("dulezita_data.txt", "r")
    obsah = soubor.read()
    hodnota = int(obsah)
    vysledek = 100 / hodnota
except (FileNotFoundError, ValueError, ZeroDivisionError) as e:
    print(f"Došlo k chybě: {e}")

# Použití hierarchie výjimek (od konkrétnější k obecnější)
try:
    cisla = [1, 2, 3]
    print(cisla[5])  # IndexError
except IndexError:
    print("Konkrétní chyba: Index mimo rozsah")
except LookupError:  # Rodičovská třída IndexError
    print("Toto se neprovede pro IndexError")
except Exception:  # Zachytí většinu výjimek
    print("Toto je záloha pro jiné výjimky")
```

## 4. Klauzule else a finally

Příkaz try/except může zahrnovat volitelné klauzule else a finally pro zpracování složitějších scénářů chyb.

```python
# Klauzule else se provede, pokud nebyla vyvolána žádná výjimka
try:
    cislo = int(input("Zadejte kladné číslo: "))
    if cislo <= 0:
        raise ValueError("To není kladné číslo!")
except ValueError as e:
    print(f"Chyba: {e}")
else:
    # Toto se provede pouze pokud nebyly vyvolány žádné výjimky
    print(f"Zadali jste {cislo}, což je platné!")
    vysledek = 100 / cislo
    print(f"100 děleno {cislo} je {vysledek}")

# Klauzule finally se vždy provede, bez ohledu na výjimky
try:
    soubor = open("ukazka.txt", "r")
    obsah = soubor.read()
    print(obsah)
except FileNotFoundError:
    print("Soubor neexistuje")
finally:
    # Tento kód se vždy provede, i když dojde k výjimce
    # Užitečné pro úklid operací jako zavírání souborů nebo připojení
    print("Provedení dokončeno, provádí se úklid...")
    try:
        soubor.close()
        print("Soubor úspěšně zavřen")
    except:
        print("Žádný soubor k zavření")

# Kompletní příklad try/except/else/finally
try:
    soubor = open("data.txt", "r")
    obsah = soubor.read()
    hodnota = int(obsah)
except FileNotFoundError:
    print("Soubor nenalezen, vytváří se...")
    soubor = open("data.txt", "w")
    soubor.write("0")
    hodnota = 0
except ValueError:
    print("Soubor neobsahuje platné číslo")
    hodnota = 0
else:
    print(f"Úspěšně načtena hodnota: {hodnota}")
finally:
    try:
        soubor.close()
        print("Soubor zavřen")
    except:
        print("Žádný soubor k zavření")
```

## 5. Hierarchie výjimek a vestavěné výjimky

Python má bohatou hierarchii vestavěných výjimek. Porozumění této hierarchii vám pomůže zachytit výjimky na odpovídající úrovni specifičnosti.

```python
# Hierarchie výjimek Pythonu (částečná)
import sys

# Výpis části hierarchie výjimek
def vypis_hierarchii_vyjimek(trida_vyjimky, odsazeni=0):
    print(' ' * odsazeni + trida_vyjimky.__name__)
    for podtrida in trida_vyjimky.__subclasses__():
        vypis_hierarchii_vyjimek(podtrida, odsazeni + 4)

# Výpis částečné hierarchie začínající od BaseException
print("Běžná hierarchie výjimek:")
vypis_hierarchii_vyjimek(BaseException, 0)

# Příklad použití hierarchie pro zpracování výjimek
try:
    # Toto může vyvolat buď IndexError nebo KeyError
    polozky = {"a": 1, "b": 2}
    hodnota = polozky["c"]  # KeyError
    # NEBO
    # seznam_polozek = [1, 2, 3]
    # hodnota = seznam_polozek[10]  # IndexError
except LookupError as e:
    # Toto zachytí jak IndexError, tak KeyError
    print(f"Chyba vyhledávání: {type(e).__name__} - {e}")

# Běžné vestavěné výjimky a kdy nastávají
priklady = [
    "1 / 0",                       # ZeroDivisionError
    "int('abc')",                  # ValueError
    "'hello' + 5",                 # TypeError
    "import neexistujici_modul",   # ModuleNotFoundError
    "open('neexistujici.txt')",    # FileNotFoundError
    "[1, 2, 3][10]",               # IndexError
    "{}['klic']",                  # KeyError
    "nedefinovana_promenna",       # NameError
]

for kod in priklady:
    try:
        print(f"Zkouším: {kod}")
        exec(kod)
    except Exception as e:
        print(f"Zachyceno: {type(e).__name__} - {e}")
    print("-" * 40)
```

## 6. Vyvolávání výjimek

Někdy potřebujete vyvolat výjimky ve vlastním kódu k signalizaci chybových stavů nebo neplatných operací.

```python
# Základní příkaz raise
def nastav_vek(vek):
    if vek < 0:
        raise ValueError("Věk nemůže být záporný")
    if vek > 150:
        raise ValueError("Věk je nereálně vysoký")
    return vek

try:
    uzivatelsky_vek = nastav_vek(-5)
except ValueError as e:
    print(f"Neplatný věk: {e}")

# Vyvolávání výjimek s vlastními zprávami
def delen_cisel(a, b):
    if b == 0:
        raise ZeroDivisionError("Nelze dělit nulou")
    return a / b

try:
    vysledek = delen_cisel(10, 0)
except ZeroDivisionError as e:
    print(f"Chyba: {e}")

# Opětovné vyvolání výjimek po zpracování
def zpracuj_data(data):
    try:
        hodnota = int(data)
        return hodnota
    except ValueError:
        print("Převod dat selhal, neplatné číslo")
        raise  # Znovu vyvolá poslední výjimku

try:
    zpracuj_data("abc")
except ValueError:
    print("Zachycena opětovně vyvolaná výjimka")

# Vyvolávání výjimek s klauzulí from (řetězení výjimek)
def nacti_data(nazev_souboru):
    try:
        with open(nazev_souboru, "r") as soubor:
            return soubor.read()
    except FileNotFoundError as e:
        raise RuntimeError(f"Nelze zpracovat data ze souboru {nazev_souboru}") from e

try:
    data = nacti_data("chybejici_soubor.txt")
except RuntimeError as e:
    print(f"Hlavní chyba: {e}")
    print(f"Původní příčina: {e.__cause__}")
```

## 7. Vytváření vlastních výjimek

Pro specifičtější zpracování chyb můžete definovat vlastní třídy výjimek děděním z vestavěných tříd výjimek.

```python
# Základní vlastní výjimka
class NedostatekProstredkuError(Exception):
    """Vyvolána, když výběr překročí dostupný zůstatek"""
    pass

class BankovniUcet:
    def __init__(self, vlastnik, zustatek=0):
        self.vlastnik = vlastnik
        self.zustatek = zustatek
        
    def vklad(self, castka):
        if castka <= 0:
            raise ValueError("Částka vkladu musí být kladná")
        self.zustatek += castka
        return self.zustatek
    
    def vyber(self, castka):
        if castka <= 0:
            raise ValueError("Částka výběru musí být kladná")
        if castka > self.zustatek:
            raise NedostatekProstredkuError(f"Nelze vybrat ${castka}. K dispozici je pouze ${self.zustatek}.")
        self.zustatek -= castka
        return self.zustatek

# Použití vlastní výjimky
ucet = BankovniUcet("Alice", 100)
try:
    ucet.vyber(150)
except NedostatekProstredkuError as e:
    print(f"Transakce selhala: {e}")

# Vlastní výjimka s dalšími atributy
class ValidationError(Exception):
    """Vyvolána, když selže validace vstupu"""
    def __init__(self, zprava, pole=None, hodnota=None):
        self.zprava = zprava
        self.pole = pole
        self.hodnota = hodnota
        super().__init__(zprava)
    
    def __str__(self):
        if self.pole and self.hodnota:
            return f"{self.zprava} (pole: {self.pole}, hodnota: {self.hodnota})"
        return self.zprava

def validuj_uzivatele(data):
    if "jmeno" not in data:
        raise ValidationError("Jméno je povinné", "jmeno", None)
    
    if len(data.get("jmeno", "")) < 2:
        raise ValidationError("Jméno je příliš krátké", "jmeno", data["jmeno"])
    
    if data.get("vek", 0) < 18:
        raise ValidationError("Musí být alespoň 18 let", "vek", data["vek"])
    
    return data

# Použití vlastní výjimky s dalšími daty
try:
    uzivatelska_data = {"jmeno": "A", "vek": 15}
    validuj_uzivatele(uzivatelska_data)
except ValidationError as e:
    print(f"Validace selhala: {e}")
    print(f"Problémové pole: {e.pole}")
    print(f"Neplatná hodnota: {e.hodnota}")
```

## 8. Osvědčené postupy pro zpracování chyb

Dodržování osvědčených postupů zajišťuje, že vaše zpracování chyb je efektivní, udržovatelné a pomáhá při ladění.

```python
# Osvědčený postup 1: Nezachytávejte všechny výjimky bez identifikace
# Špatný přístup:
try:
    vysledek = rizikova_operace()
except:  # Zachytí všechny výjimky včetně KeyboardInterrupt, SystemExit
    print("Došlo k chybě")  # Nemáme tušení, co se stalo

# Dobrý přístup:
try:
    vysledek = rizikova_operace()
except Exception as e:  # Stále obecné, ale alespoň vylučuje systémové výjimky
    print(f"Došlo k chybě: {type(e).__name__} - {e}")

# Osvědčený postup 2: Zpracování výjimek na vhodné úrovni
def nacti_konfiguraci(nazev_souboru):
    """Funkce nižší úrovně, která čte konfiguraci"""
    try:
        with open(nazev_souboru, 'r') as soubor:
            return soubor.read()
    except FileNotFoundError as e:
        # Znovu vyvoláme s více kontextu
        raise FileNotFoundError(f"Konfigurační soubor {nazev_souboru} nenalezen") from e

def inicializuj_aplikaci():
    """Funkce vyšší úrovně, která může vhodně zpracovat výjimku"""
    try:
        konfigurace = nacti_konfiguraci("config.ini")
        # Zpracování konfigurace...
        return True
    except FileNotFoundError as e:
        print(f"Inicializace aplikace selhala: {e}")
        print("Vytváření výchozího konfiguračního souboru...")
        vytvor_vychozi_konfiguraci()
        return False

# Osvědčený postup 3: Použití kontextových manažerů pro operace úklidu
# Místo:
soubor = open("data.txt", "w")
try:
    soubor.write("Ahoj, světe!")
finally:
    soubor.close()

# Lépe:
try:
    with open("data.txt", "w") as soubor:  # Kontextový manažer se postará o zavření
        soubor.write("Ahoj, světe!")
except IOError as e:
    print(f"Nelze zapisovat do souboru: {e}")

# Osvědčený postup 4: Nepoužívejte výjimky pro řízení toku programu
# Špatný přístup:
def ziskej_uzivatele_podle_id(id_uzivatele):
    try:
        return databaze_uzivatelu[id_uzivatele]
    except KeyError:
        return None  # Použití výjimek pro očekávané podmínky

# Dobrý přístup:
def ziskej_uzivatele_podle_id(id_uzivatele):
    if id_uzivatele in databaze_uzivatelu:  # Nejdřív zkontrolujeme
        return databaze_uzivatelu[id_uzivatele]
    return None

# Osvědčený postup 5: Zahrnout informace pro ladění ve vlastních výjimkách
class DatabaseError(Exception):
    def __init__(self, zprava, dotaz=None, parametry=None):
        self.zprava = zprava
        self.dotaz = dotaz
        self.parametry = parametry
        super().__init__(zprava)
    
    def __str__(self):
        chybova_zprava = self.zprava
        if self.dotaz:
            chybova_zprava += f" (Dotaz: {self.dotaz})"
        return chybova_zprava

def proved_dotaz(dotaz, parametry=None):
    try:
        # Simulace databázové operace
        if "SELECT" not in dotaz:
            raise ValueError("Neplatná syntaxe dotazu")
        # Více kódu...
    except ValueError as e:
        raise DatabaseError("Provedení dotazu selhalo", dotaz, parametry) from e

# Použití výjimky s informacemi pro ladění
try:
    proved_dotaz("INSERT INTO uzivatele VALUES (?, ?)", [1, "Alice"])
except DatabaseError as e:
    print(f"Chyba: {e}")
    # Zaznamenat podrobné informace o chybě pro ladění
    print(f"Dotaz: {e.dotaz}")
    print(f"Parametry: {e.parametry}")
    print(f"Původní chyba: {e.__cause__}")
```
