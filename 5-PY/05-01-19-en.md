## 1. Introduction to Command Line Arguments and Input

Command line arguments and input are essential for creating flexible Python applications that can accept data from users without modifying the code. Command line arguments are values provided when launching a Python script, while input refers to data collected during program execution.


## 2. Using sys.argv for Basic Argument Handling

The `sys.argv` list contains command line arguments passed to a Python script. The first item (`sys.argv[0]`) is always the script name.

```python
import sys

# Basic argument handling with sys.argv
def main():
    # Print the script name
    print(f"Script name: {sys.argv[0]}")
    
    # Check if there are any arguments
    if len(sys.argv) > 1:
        print(f"Arguments: {len(sys.argv) - 1}")
        # Print all arguments
        for i, arg in enumerate(sys.argv[1:], start=1):
            print(f"Argument {i}: {arg}")
    else:
        print("No arguments provided")

# Example calculator
def simple_calculator():
    if len(sys.argv) != 4:
        print("Usage: python script.py <number1> <operation> <number2>")
        print("Operations: add, subtract, multiply, divide")
        return
    
    try:
        num1 = float(sys.argv[1])
        operation = sys.argv[2]
        num2 = float(sys.argv[3])
        
        if operation == "add":
            result = num1 + num2
        elif operation == "subtract":
            result = num1 - num2
        elif operation == "multiply":
            result = num1 * num2
        elif operation == "divide":
            if num2 == 0:
                print("Error: Division by zero")
                return
            result = num1 / num2
        else:
            print("Unknown operation")
            return
        
        print(f"Result: {result}")
    except ValueError:
        print("Error: Please provide valid numbers")

if __name__ == "__main__":
    # main()
    simple_calculator()
```

## 3. The argparse Module for Advanced Command Line Parsing

The `argparse` module provides a more sophisticated way to handle command line arguments with features like automatic help generation, type conversion, and default values.

```python
import argparse

def process_with_argparse():
    # Create a parser object
    parser = argparse.ArgumentParser(description="Process some integers.")
    
    # Add positional arguments
    parser.add_argument("integers", metavar="N", type=int, nargs="+",
                        help="An integer for the accumulator")
    
    # Add optional arguments
    parser.add_argument("--sum", dest="accumulate", action="store_const",
                        const=sum, default=max,
                        help="Sum the integers (default: find the max)")
    
    # Parse arguments
    args = parser.parse_args()
    
    # Use the arguments
    result = args.accumulate(args.integers)
    print(f"Result: {result}")

def create_file_processor():
    parser = argparse.ArgumentParser(description="Process a file with various options.")
    
    # Required file argument
    parser.add_argument("file", help="File to process")
    
    # Optional arguments with different types
    parser.add_argument("-c", "--count", action="store_true", 
                        help="Count lines in the file")
    parser.add_argument("-l", "--limit", type=int, default=0,
                        help="Limit processing to first N lines")
    parser.add_argument("-v", "--verbose", action="store_true",
                        help="Increase output verbosity")
    parser.add_argument("-o", "--output", type=str,
                        help="Output file (default is stdout)")
    parser.add_argument("-m", "--mode", choices=["read", "count", "analyze"],
                        default="read", help="Processing mode")
    
    args = parser.parse_args()
    
    # Print the parsed arguments
    print(f"File to process: {args.file}")
    print(f"Count lines: {args.count}")
    print(f"Line limit: {args.limit}")
    print(f"Verbose mode: {args.verbose}")
    print(f"Output file: {args.output}")
    print(f"Mode: {args.mode}")

if __name__ == "__main__":
    # process_with_argparse()
    create_file_processor()
```

## 5. Handling Input Validation

Validating user input is crucial for creating robust applications.

```python
def validate_integer_input(prompt, min_value=None, max_value=None):
    """Get and validate integer input within an optional range."""
    while True:
        try:
            value = int(input(prompt))
            
            if min_value is not None and value < min_value:
                print(f"Error: Input must be at least {min_value}")
                continue
                
            if max_value is not None and value > max_value:
                print(f"Error: Input must be at most {max_value}")
                continue
                
            return value
        except ValueError:
            print("Error: Please enter a valid integer")

def validate_choice(prompt, valid_choices):
    """Get and validate input against a list of valid choices."""
    while True:
        value = input(prompt).lower()
        if value in valid_choices:
            return value
        print(f"Error: Please enter one of {', '.join(valid_choices)}")

def input_validation_example():
    # Validate integer within range
    age = validate_integer_input("Enter your age (18-120): ", 18, 120)
    print(f"Age entered: {age}")
    
    # Validate choice from options
    choice = validate_choice("Enter your favorite color (red, green, blue): ", ["red", "green", "blue"])
    print(f"Color chosen: {choice}")
    
    # Password validation with requirements
    while True:
        password = input("Create a password (at least 8 chars with 1 number): ")
        if len(password) < 8:
            print("Error: Password must be at least 8 characters long")
        elif not any(char.isdigit() for char in password):
            print("Error: Password must contain at least one number")
        else:
            break
    
    print("Password accepted!")

if __name__ == "__main__":
    input_validation_example()
```

## 6. Working with Files as Input

Command line applications often process files specified as arguments.

```python
import argparse
import sys

def process_file(filename, count_lines=False, limit=0):
    """Process a file with options for counting lines and limiting output."""
    try:
        with open(filename, 'r') as file:
            lines = file.readlines()
            
            if count_lines:
                print(f"Total lines in {filename}: {len(lines)}")
                return
            
            # Apply limit if specified
            if limit > 0:
                lines = lines[:limit]
            
            # Print lines with line numbers
            for i, line in enumerate(lines, start=1):
                print(f"{i}: {line.rstrip()}")
                
    except FileNotFoundError:
        print(f"Error: File '{filename}' not found")
        return
    except IOError as e:
        print(f"Error reading file: {e}")
        return

def file_processor():
    parser = argparse.ArgumentParser(description="Process a text file.")
    parser.add_argument("file", help="File to process")
    parser.add_argument("-c", "--count", action="store_true", 
                        help="Count lines in the file")
    parser.add_argument("-l", "--limit", type=int, default=0,
                        help="Limit output to first N lines")
    
    args = parser.parse_args()
    process_file(args.file, args.count, args.limit)

# Example of reading from stdin (for piped input)
def process_stdin():
    print("Reading from stdin (type text, press Ctrl+D to finish):")
    lines = sys.stdin.readlines()
    print(f"Received {len(lines)} lines:")
    for i, line in enumerate(lines, start=1):
        print(f"{i}: {line.rstrip()}")

if __name__ == "__main__":
    # file_processor()
    # To run: python script.py filename.txt -c -l 10
    
    # For stdin example:
    # process_stdin()
    # To run with pipe: cat filename.txt | python script.py
    
    # Example of how to check if stdin has data
    if not sys.stdin.isatty():
        print("Data is being piped in")
        process_stdin()
    else:
        print("No piped data, run file processor")
        file_processor()
```

## 7. Using Click for Elegant Command Line Interfaces

`Click` is a Python package for creating beautiful command line interfaces with minimal code.

```python
import click

@click.command()
@click.argument('name')
@click.option('--count', default=1, help='Number of greetings.')
@click.option('--greeting', default='Hello', help='The greeting to use.')
@click.option('--uppercase', is_flag=True, help='Display in uppercase.')
def hello(name, count, greeting, uppercase):
    """Simple program that greets NAME for a total of COUNT times."""
    message = f"{greeting}, {name}!"
    if uppercase:
        message = message.upper()
    for _ in range(count):
        click.echo(message)

@click.group()
def cli():
    """Command line tool with subcommands."""
    pass

@cli.command()
@click.argument('filename')
@click.option('--limit', default=0, help='Limit to N lines')
def read(filename, limit):
    """Read a file and print its contents."""
    try:
        with open(filename, 'r') as f:
            lines = f.readlines()
            if limit > 0:
                lines = lines[:limit]
            for line in lines:
                click.echo(line.rstrip())
    except FileNotFoundError:
        click.echo(f"Error: File '{filename}' not found.", err=True)
        
@cli.command()
@click.argument('filename')
def count(filename):
    """Count lines in a file."""
    try:
        with open(filename, 'r') as f:
            count = sum(1 for _ in f)
            click.echo(f"Line count: {count}")
    except FileNotFoundError:
        click.echo(f"Error: File '{filename}' not found.", err=True)

if __name__ == '__main__':
    # To run the first example:
    # hello()
    
    # To run the command group example:
    cli()
```

## 8. Error Handling with Command Line Inputs

Proper error handling is crucial for user-friendly command line applications.

```python
import argparse
import sys
import os

def setup_parser():
    parser = argparse.ArgumentParser(
        description='File processing tool',
        formatter_class=argparse.ArgumentDefaultsHelpFormatter
    )
    parser.add_argument('file', help='File to process')
    parser.add_argument('--output', '-o', help='Output file')
    parser.add_argument('--operation', choices=['count', 'uppercase', 'lowercase'],
                        default='count', help='Operation to perform')
    return parser

def process_file_with_error_handling(args):
    # Validate input file
    if not os.path.exists(args.file):
        sys.stderr.write(f"Error: Input file '{args.file}' does not exist\n")
        return 1
        
    if not os.path.isfile(args.file):
        sys.stderr.write(f"Error: '{args.file}' is not a regular file\n")
        return 1
        
    # Validate output file
    if args.output and os.path.exists(args.output):
        response = input(f"Output file '{args.output}' exists. Overwrite? (y/n): ").lower()
        if response != 'y':
            print("Operation cancelled")
            return 0
    
    try:
        # Open and read the input file
        with open(args.file, 'r') as infile:
            content = infile.read()
        
        # Perform the requested operation
        if args.operation == 'count':
            result = f"Lines: {content.count(os.linesep) + 1}\n"
            result += f"Words: {len(content.split())}\n"
            result += f"Characters: {len(content)}\n"
        elif args.operation == 'uppercase':
            result = content.upper()
        elif args.operation == 'lowercase':
            result = content.lower()
        
        # Write to output or stdout
        if args.output:
            with open(args.output, 'w') as outfile:
                outfile.write(result)
            print(f"Result written to {args.output}")
        else:
            print(result)
            
        return 0
        
    except IOError as e:
        sys.stderr.write(f"I/O error: {str(e)}\n")
        return 1
    except Exception as e:
        sys.stderr.write(f"Unexpected error: {str(e)}\n")
        return 1

def main():
    parser = setup_parser()
    args = parser.parse_args()
    
    try:
        return process_file_with_error_handling(args)
    except KeyboardInterrupt:
        sys.stderr.write("Operation cancelled by user\n")
        return 130  # Standard exit code for SIGINT

if __name__ == "__main__":
    sys.exit(main())
```

## 9. Creating Interactive Command Line Applications

Interactive command line applications can create a more engaging user experience.

```python
def simple_interactive_menu():
    while True:
        print("\nMain Menu:")
        print("1. View Files")
        print("2. Create File")
        print("3. Delete File")
        print("4. Exit")
        
        choice = input("\nEnter your choice (1-4): ")
        
        if choice == '1':
            print("\nCurrently available files:")
            import os
            files = os.listdir('.')
            for i, file in enumerate(files, 1):
                print(f"{i}. {file}")
            
            input("\nPress Enter to continue...")
            
        elif choice == '2':
            filename = input("\nEnter filename to create: ")
            try:
                with open(filename, 'w') as f:
                    content = input("Enter content (press Enter when done):\n")
                    f.write(content)
                print(f"File '{filename}' created successfully")
            except Exception as e:
                print(f"Error creating file: {e}")
            
        elif choice == '3':
            filename = input("\nEnter filename to delete: ")
            try:
                import os
                if os.path.exists(filename):
                    confirm = input(f"Are you sure you want to delete '{filename}'? (y/n): ")
                    if confirm.lower() == 'y':
                        os.remove(filename)
                        print(f"File '{filename}' deleted")
                    else:
                        print("Deletion cancelled")
                else:
                    print(f"File '{filename}' not found")
            except Exception as e:
                print(f"Error deleting file: {e}")
            
        elif choice == '4':
            print("\nExiting program. Goodbye!")
            break
            
        else:
            print("\nInvalid choice. Please enter a number between 1 and 4.")

# For a more advanced interactive CLI, you can use libraries like prompt_toolkit

            # Implementation for other commands would go here
```
