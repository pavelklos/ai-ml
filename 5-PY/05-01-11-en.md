## 1. Introduction to Exceptions and Error Handling

Error handling is essential for creating robust Python programs. When errors occur during program execution, Python raises exceptions - signals that something unexpected has happened. Without proper error handling, these exceptions cause your program to crash. The try/except construct allows you to catch and handle these exceptions, making your code more resilient.

```python
# Without error handling
def divide(a, b):
    return a / b

# This will crash the program with ZeroDivisionError
# result = divide(10, 0)

# Basic demonstration of what happens with exceptions
try:
    print("Attempting to divide by zero...")
    result = 10 / 0
    print("This line never executes")
except:
    print("An error occurred!")

print("Program continues to run")

# Common exceptions you might encounter:
# ZeroDivisionError: When dividing by zero
# TypeError: When operations are performed on incompatible types
# ValueError: When a function receives an argument of correct type but inappropriate value
# FileNotFoundError: When trying to access a file that doesn't exist
# IndexError: When trying to access an index that's out of range
```

## 2. Basic try/except Structure

The basic structure of try/except includes code that might raise an exception in the try block, with handlers for those exceptions in the except block.

```python
# Basic try/except pattern
try:
    # Code that might raise an exception
    user_input = input("Enter a number: ")
    number = int(user_input)
    result = 100 / number
    print(f"100 divided by {number} is {result}")
except:
    # This handles ANY exception (not recommended for production code)
    print("Something went wrong")

# Better practice: specify the exception type
try:
    user_input = input("Enter a number: ")
    number = int(user_input)
    result = 100 / number
    print(f"100 divided by {number} is {result}")
except ValueError:
    print("That's not a valid number!")
except ZeroDivisionError:
    print("You can't divide by zero!")

# Access the exception object using 'as'
try:
    with open("nonexistent_file.txt", "r") as file:
        contents = file.read()
except FileNotFoundError as e:
    print(f"Error: {e}")
    print(f"Error type: {type(e).__name__}")
```

## 3. Handling Multiple Exception Types

Python offers several ways to handle multiple types of exceptions, allowing for flexible error management strategies.

```python
# Handling multiple exception types individually
try:
    # Some risky operation
    number = int(input("Enter a number: "))
    result = 100 / number
    print(f"Result: {result}")
except ValueError:
    print("Please enter a valid number")
except ZeroDivisionError:
    print("Cannot divide by zero")

# Handling multiple exception types with the same handler
try:
    file = open("important_data.txt", "r")
    content = file.read()
    value = int(content)
    result = 100 / value
except (FileNotFoundError, ValueError, ZeroDivisionError) as e:
    print(f"An error occurred: {e}")

# Using exception hierarchies (more specific to less specific)
try:
    numbers = [1, 2, 3]
    print(numbers[5])  # IndexError
except IndexError:
    print("Specific error: Index out of range")
except LookupError:  # Parent class of IndexError
    print("This won't execute for IndexError")
except Exception:  # Catches most exceptions
    print("This is a fallback for other exceptions")
```

## 4. The else and finally Clauses

The try/except statement can include optional else and finally clauses to handle more complex error scenarios.

```python
# The else clause executes if no exception is raised
try:
    number = int(input("Enter a positive number: "))
    if number <= 0:
        raise ValueError("That's not a positive number!")
except ValueError as e:
    print(f"Error: {e}")
else:
    # This runs only if no exceptions were raised
    print(f"You entered {number}, which is valid!")
    result = 100 / number
    print(f"100 divided by {number} is {result}")

# The finally clause always executes, regardless of exceptions
try:
    file = open("sample.txt", "r")
    content = file.read()
    print(content)
except FileNotFoundError:
    print("The file does not exist")
finally:
    # This code always runs, even if an exception occurs
    # Useful for cleanup operations like closing files or connections
    print("Execution completed, performing cleanup...")
    try:
        file.close()
        print("File closed successfully")
    except:
        print("No file to close")

# Complete try/except/else/finally example
try:
    file = open("data.txt", "r")
    content = file.read()
    value = int(content)
except FileNotFoundError:
    print("File not found, creating one...")
    file = open("data.txt", "w")
    file.write("0")
    value = 0
except ValueError:
    print("File doesn't contain a valid number")
    value = 0
else:
    print(f"Successfully read value: {value}")
finally:
    try:
        file.close()
        print("File closed")
    except:
        print("No file to close")
```

## 5. Exception Hierarchy and Built-in Exceptions

Python has a rich hierarchy of built-in exceptions. Understanding this hierarchy helps you catch exceptions at the appropriate level of specificity.

```python
# Python's exception hierarchy (partial)
import sys

# Printing a portion of the exception hierarchy
def print_exception_hierarchy(exception_class, indent=0):
    print(' ' * indent + exception_class.__name__)
    for subclass in exception_class.__subclasses__():
        print_exception_hierarchy(subclass, indent + 4)

# Print partial hierarchy starting from BaseException
print("Common Exception Hierarchy:")
print_exception_hierarchy(BaseException, 0)

# Example of using the hierarchy for handling exceptions
try:
    # This could raise either IndexError or KeyError
    items = {"a": 1, "b": 2}
    value = items["c"]  # KeyError
    # OR
    # list_items = [1, 2, 3]
    # value = list_items[10]  # IndexError
except LookupError as e:
    # This will catch both IndexError and KeyError
    print(f"Lookup error: {type(e).__name__} - {e}")

# Common built-in exceptions and when they occur
examples = [
    "1 / 0",                       # ZeroDivisionError
    "int('abc')",                  # ValueError
    "'hello' + 5",                 # TypeError
    "import nonexistent_module",   # ModuleNotFoundError
    "open('nonexistent.txt')",     # FileNotFoundError
    "[1, 2, 3][10]",               # IndexError
    "{}['key']",                   # KeyError
    "undefined_variable",          # NameError
]

for code in examples:
    try:
        print(f"Trying: {code}")
        exec(code)
    except Exception as e:
        print(f"Caught: {type(e).__name__} - {e}")
    print("-" * 40)
```

## 6. Raising Exceptions

Sometimes you need to raise exceptions in your own code to signal error conditions or invalid operations.

```python
# Basic raise statement
def set_age(age):
    if age < 0:
        raise ValueError("Age cannot be negative")
    if age > 150:
        raise ValueError("Age is unrealistically high")
    return age

try:
    user_age = set_age(-5)
except ValueError as e:
    print(f"Invalid age: {e}")

# Raising exceptions with custom messages
def divide_numbers(a, b):
    if b == 0:
        raise ZeroDivisionError("Cannot divide by zero")
    return a / b

try:
    result = divide_numbers(10, 0)
except ZeroDivisionError as e:
    print(f"Error: {e}")

# Re-raising exceptions after handling
def process_data(data):
    try:
        value = int(data)
        return value
    except ValueError:
        print("Data conversion failed, not a valid number")
        raise  # Re-raises the last exception

try:
    process_data("abc")
except ValueError:
    print("Caught re-raised exception")

# Raising exceptions with the from clause (exception chaining)
def fetch_data(filename):
    try:
        with open(filename, "r") as file:
            return file.read()
    except FileNotFoundError as e:
        raise RuntimeError(f"Could not process data from {filename}") from e

try:
    data = fetch_data("missing_file.txt")
except RuntimeError as e:
    print(f"Main error: {e}")
    print(f"Original cause: {e.__cause__}")
```

## 7. Creating Custom Exceptions

For more specific error handling, you can define your own exception classes by inheriting from built-in exception classes.

```python
# Basic custom exception
class InsufficientFundsError(Exception):
    """Raised when a withdrawal exceeds the available balance"""
    pass

class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self.balance = balance
        
    def deposit(self, amount):
        if amount <= 0:
            raise ValueError("Deposit amount must be positive")
        self.balance += amount
        return self.balance
    
    def withdraw(self, amount):
        if amount <= 0:
            raise ValueError("Withdrawal amount must be positive")
        if amount > self.balance:
            raise InsufficientFundsError(f"Cannot withdraw ${amount}. Only ${self.balance} available.")
        self.balance -= amount
        return self.balance

# Using the custom exception
account = BankAccount("Alice", 100)
try:
    account.withdraw(150)
except InsufficientFundsError as e:
    print(f"Transaction failed: {e}")

# Custom exception with additional attributes
class ValidationError(Exception):
    """Raised when input validation fails"""
    def __init__(self, message, field=None, value=None):
        self.message = message
        self.field = field
        self.value = value
        super().__init__(message)
    
    def __str__(self):
        if self.field and self.value:
            return f"{self.message} (field: {self.field}, value: {self.value})"
        return self.message

def validate_user_data(data):
    if "name" not in data:
        raise ValidationError("Name is required", "name", None)
    
    if len(data.get("name", "")) < 2:
        raise ValidationError("Name is too short", "name", data["name"])
    
    if data.get("age", 0) < 18:
        raise ValidationError("Must be at least 18 years old", "age", data["age"])
    
    return data

# Using the custom exception with extra data
try:
    user_data = {"name": "A", "age": 15}
    validate_user_data(user_data)
except ValidationError as e:
    print(f"Validation failed: {e}")
    print(f"Problem field: {e.field}")
    print(f"Invalid value: {e.value}")
```

## 8. Best Practices for Error Handling

Following best practices ensures your error handling is effective, maintainable, and helps in debugging.

```python
# Best Practice 1: Don't catch all exceptions without identification
# Bad:
try:
    result = risky_operation()
except:  # Catches all exceptions including KeyboardInterrupt, SystemExit
    print("An error occurred")  # No idea what happened

# Good:
try:
    result = risky_operation()
except Exception as e:  # Still broad but at least excludes system exceptions
    print(f"An error occurred: {type(e).__name__} - {e}")

# Best Practice 2: Handle exceptions at the appropriate level
def read_config(filename):
    """Lower-level function that reads configuration"""
    try:
        with open(filename, 'r') as file:
            return file.read()
    except FileNotFoundError as e:
        # Re-raise with more context
        raise FileNotFoundError(f"Config file {filename} not found") from e

def initialize_application():
    """Higher-level function that can handle the exception appropriately"""
    try:
        config = read_config("config.ini")
        # Process config...
        return True
    except FileNotFoundError as e:
        print(f"Application initialization failed: {e}")
        print("Creating default config file...")
        create_default_config()
        return False

# Best Practice 3: Use context managers for cleanup operations
# Instead of:
file = open("data.txt", "w")
try:
    file.write("Hello, World!")
finally:
    file.close()

# Better:
try:
    with open("data.txt", "w") as file:  # Context manager handles closing
        file.write("Hello, World!")
except IOError as e:
    print(f"Could not write to file: {e}")

# Best Practice 4: Don't use exceptions for flow control
# Bad:
def get_user_by_id(user_id):
    try:
        return users_database[user_id]
    except KeyError:
        return None  # Using exceptions for expected conditions

# Good:
def get_user_by_id(user_id):
    if user_id in users_database:  # Check first
        return users_database[user_id]
    return None

# Best Practice 5: Include debugging information in custom exceptions
class DatabaseError(Exception):
    def __init__(self, message, query=None, parameters=None):
        self.message = message
        self.query = query
        self.parameters = parameters
        super().__init__(message)
    
    def __str__(self):
        error_msg = self.message
        if self.query:
            error_msg += f" (Query: {self.query})"
        return error_msg

def execute_query(query, parameters=None):
    try:
        # Mock database operation
        if "SELECT" not in query:
            raise ValueError("Invalid query syntax")
        # More code...
    except ValueError as e:
        raise DatabaseError("Query execution failed", query, parameters) from e

# Using the exception with debugging info
try:
    execute_query("INSERT INTO users VALUES (?, ?)", [1, "Alice"])
except DatabaseError as e:
    print(f"Error: {e}")
    # Log detailed error information for debugging
    print(f"Query: {e.query}")
    print(f"Parameters: {e.parameters}")
    print(f"Original error: {e.__cause__}")
```
