## 1. Introduction to Date and Time in Python

Python offers several modules for working with dates and times. The primary ones are:
- `datetime`: For working with dates, times, and time intervals
- `time`: For time-related functions (timestamps, formatting, etc.)
- `calendar`: For calendar-related functions
- `dateutil`: A third-party library that extends functionality
- `pytz`: For working with time zones

```python
# Importing the basic date and time modules
import datetime
import time
import calendar
from datetime import datetime as dt  # Common alias

# Current date and time
current_datetime = dt.now()
print(f"Current date and time: {current_datetime}")

# Current date
current_date = dt.now().date()
print(f"Current date: {current_date}")

# Current time
current_time = dt.now().time()
print(f"Current time: {current_time}")

# Unix timestamp (seconds since January 1, 1970)
timestamp = time.time()
print(f"Current timestamp: {timestamp}")

# Calendar for the current month
current_month = datetime.datetime.now().month
current_year = datetime.datetime.now().year
print(calendar.month(current_year, current_month))
```

## 2. Working with datetime Module

The `datetime` module contains several classes for working with dates and times:
- `datetime.date`: For dates (year, month, day)
- `datetime.time`: For times (hour, minute, second, microsecond)
- `datetime.datetime`: Combined date and time
- `datetime.timedelta`: Duration or difference between dates/times
- `datetime.tzinfo`: Abstract base class for time zone information

```python
from datetime import date, time, datetime, timedelta

# Creating date objects
today = date.today()
specific_date = date(2023, 12, 31)

print(f"Today: {today}")
print(f"Specific date: {specific_date}")
print(f"Year: {today.year}, Month: {today.month}, Day: {today.day}")
print(f"Weekday: {today.weekday()}")  # Monday is 0, Sunday is 6

# Creating time objects
current_time = time(14, 30, 45)
midnight = time(0, 0, 0)

print(f"Current time: {current_time}")
print(f"Hour: {current_time.hour}, Minute: {current_time.minute}, Second: {current_time.second}")

# Creating datetime objects
now = datetime.now()
specific_datetime = datetime(2023, 12, 31, 23, 59, 59)

print(f"Now: {now}")
print(f"Specific datetime: {specific_datetime}")

# Converting between types
dt_from_date = datetime.combine(today, current_time)
print(f"Combined datetime: {dt_from_date}")

date_from_dt = now.date()
time_from_dt = now.time()
print(f"Extracted date: {date_from_dt}")
print(f"Extracted time: {time_from_dt}")
```

## 3. Date and Time Formatting

Python provides flexible ways to format dates and times as strings using format codes:

```python
from datetime import datetime

now = datetime.now()

# Basic formatting with strftime
print(f"ISO format: {now.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"US format: {now.strftime('%m/%d/%Y %I:%M %p')}")
print(f"European format: {now.strftime('%d.%m.%Y %H:%M')}")
print(f"Long format: {now.strftime('%A, %B %d, %Y %I:%M %p')}")

# Common format codes:
# %Y - Year with century (2023)
# %m - Month as a zero-padded number (01-12)
# %d - Day as a zero-padded number (01-31)
# %H - Hour (24-hour clock) as a zero-padded number (00-23)
# %M - Minute as a zero-padded number (00-59)
# %S - Second as a zero-padded number (00-59)
# %I - Hour (12-hour clock) as a zero-padded number (01-12)
# %p - AM/PM
# %A - Weekday full name (Sunday, Monday, ...)
# %a - Weekday short name (Sun, Mon, ...)
# %B - Month full name (January, February, ...)
# %b - Month short name (Jan, Feb, ...)
# %j - Day of the year as a zero-padded number (001-366)
# %U - Week number of the year (Sunday as the first day of the week)
# %W - Week number of the year (Monday as the first day of the week)

# ISO 8601 format
print(f"ISO 8601: {now.isoformat()}")

# Using f-strings with datetime attributes
print(f"Custom format: {now.year}/{now.month:02d}/{now.day:02d} {now.hour:02d}:{now.minute:02d}")
```

## 4. Date Arithmetic and Timedeltas

The `timedelta` class allows you to perform arithmetic operations with dates and times:

```python
from datetime import datetime, timedelta

now = datetime.now()

# Creating timedeltas
one_day = timedelta(days=1)
one_week = timedelta(weeks=1)
one_hour = timedelta(hours=1)
complex_delta = timedelta(days=2, hours=3, minutes=15, seconds=30)

# Adding time to a datetime
tomorrow = now + one_day
next_week = now + one_week
one_hour_later = now + one_hour

print(f"Now: {now}")
print(f"Tomorrow: {tomorrow}")
print(f"Next week: {next_week}")
print(f"One hour later: {one_hour_later}")

# Subtracting time
yesterday = now - one_day
print(f"Yesterday: {yesterday}")

# Difference between two dates
birthday = datetime(2023, 6, 15)
time_until_birthday = birthday - now if birthday > now else datetime(2024, 6, 15) - now
print(f"Time until next birthday: {time_until_birthday}")
print(f"Days until next birthday: {time_until_birthday.days}")

# Working with timedelta components
print(f"Total seconds in complex_delta: {complex_delta.total_seconds()}")
print(f"Days: {complex_delta.days}, Seconds: {complex_delta.seconds}")

# Finding a date from a weekday
def next_weekday(d, weekday):
    """Find the next specified weekday after date d (0 = Monday, 6 = Sunday)"""
    days_ahead = weekday - d.weekday()
    if days_ahead <= 0:  # Target day already happened this week
        days_ahead += 7
    return d + timedelta(days=days_ahead)

next_monday = next_weekday(now, 0)  # 0 is Monday
print(f"Next Monday: {next_monday}")
```

## 5. Working with Time Zones

By default, Python's `datetime` objects are "naive" (do not include time zone info). The `pytz` library helps work with time zones:

```python
from datetime import datetime
import pytz

# Current time in default time zone (naive)
naive_now = datetime.now()
print(f"Naive datetime: {naive_now}")

# Current time in UTC
utc_now = datetime.now(pytz.UTC)
print(f"UTC time: {utc_now}")

# List of common time zones
common_timezones = ['US/Eastern', 'US/Pacific', 'Europe/London', 
                    'Asia/Tokyo', 'Australia/Sydney']

# Converting UTC time to different time zones
for tz_name in common_timezones:
    tz = pytz.timezone(tz_name)
    local_time = utc_now.astimezone(tz)
    print(f"Time in {tz_name}: {local_time}")

# Creating a datetime with a specific time zone
eastern = pytz.timezone('US/Eastern')
eastern_time = eastern.localize(datetime(2023, 12, 31, 23, 59, 59))
print(f"New Year's Eve in New York: {eastern_time}")

# Converting between time zones
tokyo = pytz.timezone('Asia/Tokyo')
tokyo_time = eastern_time.astimezone(tokyo)
print(f"Same moment in Tokyo: {tokyo_time}")

# Getting current time in a specific time zone
def get_current_time(timezone_str):
    """Get current time in the specified timezone."""
    tz = pytz.timezone(timezone_str)
    return datetime.now(tz)

print(f"Current time in Paris: {get_current_time('Europe/Paris')}")

# Checking if a timezone has daylight saving
tz = pytz.timezone('US/Pacific')
dt = datetime(2023, 1, 1, tzinfo=tz)  # Winter
dt2 = datetime(2023, 7, 1, tzinfo=tz)  # Summer
print(f"Pacific Time DST in January: {dt.dst()}")
print(f"Pacific Time DST in July: {dt2.dst()}")
```

## 6. Parsing Dates from Strings

Converting string representations of dates into datetime objects:

```python
from datetime import datetime
import dateutil.parser

# Basic parsing with datetime.strptime
date_str = "2023-07-15 14:30:45"
parsed_date = datetime.strptime(date_str, "%Y-%m-%d %H:%M:%S")
print(f"Parsed date: {parsed_date}")

# Multiple format parsing
date_formats = ["%Y-%m-%d", "%m/%d/%Y", "%d.%m.%Y", "%B %d, %Y"]
date_strings = ["2023-07-15", "07/15/2023", "15.07.2023", "July 15, 2023"]

for date_str in date_strings:
    for date_format in date_formats:
        try:
            parsed_date = datetime.strptime(date_str, date_format)
            print(f"Successfully parsed '{date_str}' with format '{date_format}': {parsed_date}")
            break
        except ValueError:
            continue

# Using dateutil.parser for more flexible parsing
complex_dates = [
    "Jul 15, 2023",
    "2023-07-15T14:30:45-05:00",
    "15/Jul/2023",
    "Sat, 15 Jul 2023 14:30:45 -0500",
    "July 15th, 2023 at 2:30pm"
]

for date_str in complex_dates:
    try:
        parsed_date = dateutil.parser.parse(date_str)
        print(f"Parsed '{date_str}' to: {parsed_date}")
    except ValueError as e:
        print(f"Could not parse '{date_str}': {e}")

# Parsing date with timezone
date_with_tz = "2023-07-15T14:30:45+02:00"
parsed_date_with_tz = dateutil.parser.parse(date_with_tz)
print(f"Parsed date with timezone: {parsed_date_with_tz}")
print(f"Timezone info: {parsed_date_with_tz.tzinfo}")
```

## 7. Working with Timestamps

Unix timestamps represent time as seconds since January 1, 1970, and are useful for calculations and comparisons:

```python
import time
from datetime import datetime, timezone

# Current timestamp
current_timestamp = time.time()
print(f"Current Unix timestamp: {current_timestamp}")

# Converting timestamp to datetime
dt_from_timestamp = datetime.fromtimestamp(current_timestamp)
print(f"Local datetime from timestamp: {dt_from_timestamp}")

# Converting timestamp to UTC datetime
dt_utc_from_timestamp = datetime.fromtimestamp(current_timestamp, tz=timezone.utc)
print(f"UTC datetime from timestamp: {dt_utc_from_timestamp}")

# Converting datetime to timestamp
now = datetime.now()
timestamp_from_dt = now.timestamp()
print(f"Timestamp from datetime: {timestamp_from_dt}")

# Comparing dates using timestamps
date1 = datetime(2023, 7, 15)
date2 = datetime(2023, 7, 20)

ts1 = date1.timestamp()
ts2 = date2.timestamp()

diff_seconds = ts2 - ts1
print(f"Difference in seconds: {diff_seconds}")
print(f"Difference in days: {diff_seconds / (60*60*24)}")

# Performance measurements with time
def slow_function():
    """A function that takes some time to execute."""
    total = 0
    for i in range(10000000):
        total += i
    return total

start_time = time.time()
result = slow_function()
end_time = time.time()

execution_time = end_time - start_time
print(f"Function executed in {execution_time:.4f} seconds")

# Sleep for a specific duration
print("Waiting for 2 seconds...")
time.sleep(2)
print("Done waiting")

# High-resolution time for benchmarking
start = time.perf_counter()
# Some operation to measure
time.sleep(0.1)
end = time.perf_counter()
print(f"Operation took {end - start:.6f} seconds")
```

## 8. Time Series with Pandas

For data analysis involving dates and times, Pandas provides powerful time series functionality:

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

# Creating a time series with Pandas
date_rng = pd.date_range(start='2023-01-01', end='2023-01-10', freq='D')
print("Date range:")
print(date_rng)

# Creating a DataFrame with a date index
df = pd.DataFrame(date_rng, columns=['date'])
df['value'] = np.random.randn(len(date_rng))
print("\nDataFrame with date column:")
print(df.head())

# Setting date as index
df.set_index('date', inplace=True)
print("\nDataFrame with date index:")
print(df.head())

# Date ranges with different frequencies
# D - calendar day, B - business day, H - hour, T/min - minute, S - second
# M - month end, MS - month start, Y/A - year end, YS/AS - year start
frequencies = ['D', 'B', 'H', 'T', 'M', 'MS', 'Q', 'Y']
for freq in frequencies:
    date_range = pd.date_range(start='2023-01-01', periods=5, freq=freq)
    print(f"\nDate range with frequency '{freq}':")
    print(date_range)

# Time series operations
# Resampling to different time frequencies
ts = pd.Series(np.random.randn(500), 
               index=pd.date_range('2023-01-01', periods=500, freq='H'))

print("\nOriginal time series (hourly data):")
print(ts.head())

# Downsampling: higher frequency to lower frequency (e.g., hourly to daily)
daily_mean = ts.resample('D').mean()
print("\nDownsampled to daily mean:")
print(daily_mean.head())

# Upsampling: lower frequency to higher frequency (e.g., daily to hourly)
daily = pd.Series(np.random.randn(10), 
                 index=pd.date_range('2023-01-01', periods=10, freq='D'))
hourly = daily.resample('H').ffill()  # Forward fill missing values
print("\nUpsampled from daily to hourly (with forward-fill):")
print(hourly.head())

# Time-based indexing and slicing
print("\nData for January 5th:")
print(ts['2023-01-05'])

print("\nData from January 5th to January 10th:")
print(ts['2023-01-05':'2023-01-10'].head())

# Time series shifting
shifted = ts.shift(periods=24)  # Shift by 24 hours
print("\nShifted time series (by 24 hours):")
print(pd.concat([ts.head(), shifted.head()], axis=1, keys=['original', 'shifted']))

# Rolling window calculations
rolling_mean = ts.rolling(window=24).mean()  # 24-hour rolling mean
print("\n24-hour rolling mean:")
print(rolling_mean.head(30))

# Plotting a time series
plt.figure(figsize=(10, 6))
daily_mean.plot()
plt.title('Daily Mean Values')
plt.ylabel('Value')
plt.grid(True)

# Saving the plot instead of displaying (for Markdown)
plt.savefig('timeseries_plot.png')
print("\nPlot saved as 'timeseries_plot.png'")

# Working with time zones in pandas
ts_utc = pd.Series(np.random.randn(5), 
                  index=pd.date_range('2023-01-01', periods=5, freq='D', tz='UTC'))
print("\nTime series with UTC timezone:")
print(ts_utc)

# Converting to another time zone
ts_ny = ts_utc.tz_convert('America/New_York')
print("\nTime series converted to New York timezone:")
print(ts_ny)
```
