## 1. Úvod do datumů a časů v Pythonu

Python nabízí několik modulů pro práci s datumy a časy. Hlavní z nich jsou:
- `datetime`: Pro práci s datumy, časy a časovými intervaly
- `time`: Pro funkce související s časem (časová razítka, formátování atd.)
- `calendar`: Pro funkce související s kalendářem
- `dateutil`: Knihovna třetí strany, která rozšiřuje funkcionalitu
- `pytz`: Pro práci s časovými pásmy

```python
# Import základních modulů pro práci s datumy a časy
import datetime
import time
import calendar
from datetime import datetime as dt  # Běžný alias

# Aktuální datum a čas
aktualni_datetime = dt.now()
print(f"Aktuální datum a čas: {aktualni_datetime}")

# Aktuální datum
aktualni_datum = dt.now().date()
print(f"Aktuální datum: {aktualni_datum}")

# Aktuální čas
aktualni_cas = dt.now().time()
print(f"Aktuální čas: {aktualni_cas}")

# Unix timestamp (sekundy od 1. ledna 1970)
timestamp = time.time()
print(f"Aktuální timestamp: {timestamp}")

# Kalendář pro aktuální měsíc
aktualni_mesic = datetime.datetime.now().month
aktualni_rok = datetime.datetime.now().year
print(calendar.month(aktualni_rok, aktualni_mesic))
```

## 2. Práce s modulem datetime

Modul `datetime` obsahuje několik tříd pro práci s datumy a časy:
- `datetime.date`: Pro datumy (rok, měsíc, den)
- `datetime.time`: Pro časy (hodina, minuta, sekunda, mikrosekunda)
- `datetime.datetime`: Kombinované datum a čas
- `datetime.timedelta`: Doba trvání nebo rozdíl mezi datumy/časy
- `datetime.tzinfo`: Abstraktní základní třída pro informace o časovém pásmu

```python
from datetime import date, time, datetime, timedelta

# Vytváření objektů date
dnes = date.today()
konkretni_datum = date(2023, 12, 31)

print(f"Dnes: {dnes}")
print(f"Konkrétní datum: {konkretni_datum}")
print(f"Rok: {dnes.year}, Měsíc: {dnes.month}, Den: {dnes.day}")
print(f"Den v týdnu: {dnes.weekday()}")  # Pondělí je 0, Neděle je 6

# Vytváření objektů time
aktualni_cas = time(14, 30, 45)
pulnoc = time(0, 0, 0)

print(f"Aktuální čas: {aktualni_cas}")
print(f"Hodina: {aktualni_cas.hour}, Minuta: {aktualni_cas.minute}, Sekunda: {aktualni_cas.second}")

# Vytváření objektů datetime
ted = datetime.now()
konkretni_datetime = datetime(2023, 12, 31, 23, 59, 59)

print(f"Teď: {ted}")
print(f"Konkrétní datetime: {konkretni_datetime}")

# Konverze mezi typy
dt_z_date = datetime.combine(dnes, aktualni_cas)
print(f"Kombinovaný datetime: {dt_z_date}")

datum_z_dt = ted.date()
cas_z_dt = ted.time()
print(f"Extrahované datum: {datum_z_dt}")
print(f"Extrahovaný čas: {cas_z_dt}")
```

## 3. Formátování datumů a časů

Python poskytuje flexibilní způsoby formátování datumů a časů jako řetězce pomocí formátovacích kódů:

```python
from datetime import datetime

ted = datetime.now()

# Základní formátování s strftime
print(f"ISO formát: {ted.strftime('%Y-%m-%d %H:%M:%S')}")
print(f"US formát: {ted.strftime('%m/%d/%Y %I:%M %p')}")
print(f"Evropský formát: {ted.strftime('%d.%m.%Y %H:%M')}")
print(f"Dlouhý formát: {ted.strftime('%A, %d. %B %Y %H:%M')}")

# Běžné formátovací kódy:
# %Y - Rok s stoletím (2023)
# %m - Měsíc jako číslo s nulou na začátku (01-12)
# %d - Den jako číslo s nulou na začátku (01-31)
# %H - Hodina (24hodinový formát) jako číslo s nulou na začátku (00-23)
# %M - Minuta jako číslo s nulou na začátku (00-59)
# %S - Sekunda jako číslo s nulou na začátku (00-59)
# %I - Hodina (12hodinový formát) jako číslo s nulou na začátku (01-12)
# %p - AM/PM
# %A - Celý název dne v týdnu (Neděle, Pondělí, ...)
# %a - Zkrácený název dne v týdnu (Ne, Po, ...)
# %B - Celý název měsíce (Leden, Únor, ...)
# %b - Zkrácený název měsíce (Led, Úno, ...)
# %j - Den v roce jako číslo s nulou na začátku (001-366)
# %U - Číslo týdne v roce (Neděle jako první den v týdnu)
# %W - Číslo týdne v roce (Pondělí jako první den v týdnu)

# ISO 8601 formát
print(f"ISO 8601: {ted.isoformat()}")

# Použití f-strings s atributy datetime
print(f"Vlastní formát: {ted.year}/{ted.month:02d}/{ted.day:02d} {ted.hour:02d}:{ted.minute:02d}")
```

## 4. Aritmetika s daty a timedelty

Třída `timedelta` umožňuje provádět aritmetické operace s datumy a časy:

```python
from datetime import datetime, timedelta

ted = datetime.now()

# Vytváření timedeltas
jeden_den = timedelta(days=1)
jeden_tyden = timedelta(weeks=1)
jedna_hodina = timedelta(hours=1)
komplexni_delta = timedelta(days=2, hours=3, minutes=15, seconds=30)

# Přidání času k datetime
zitra = ted + jeden_den
pristi_tyden = ted + jeden_tyden
za_hodinu = ted + jedna_hodina

print(f"Teď: {ted}")
print(f"Zítra: {zitra}")
print(f"Příští týden: {pristi_tyden}")
print(f"Za hodinu: {za_hodinu}")

# Odčítání času
vcera = ted - jeden_den
print(f"Včera: {vcera}")

# Rozdíl mezi dvěma datumy
narozeniny = datetime(2023, 6, 15)
cas_do_narozenin = narozeniny - ted if narozeniny > ted else datetime(2024, 6, 15) - ted
print(f"Čas do příštích narozenin: {cas_do_narozenin}")
print(f"Dnů do příštích narozenin: {cas_do_narozenin.days}")

# Práce s komponenty timedelta
print(f"Celkem sekund v komplexni_delta: {komplexni_delta.total_seconds()}")
print(f"Dnů: {komplexni_delta.days}, Sekund: {komplexni_delta.seconds}")

# Hledání data od dne v týdnu
def dalsi_den_v_tydnu(d, den_v_tydnu):
    """Najde další určený den v týdnu po datu d (0 = Pondělí, 6 = Neděle)"""
    dni_napred = den_v_tydnu - d.weekday()
    if dni_napred <= 0:  # Cílový den už tento týden proběhl
        dni_napred += 7
    return d + timedelta(days=dni_napred)

pristi_pondeli = dalsi_den_v_tydnu(ted, 0)  # 0 je Pondělí
print(f"Příští pondělí: {pristi_pondeli}")
```

## 5. Práce s časovými pásmy

Ve výchozím nastavení jsou objekty `datetime` v Pythonu "naivní" (neobsahují informace o časovém pásmu). Knihovna `pytz` pomáhá pracovat s časovými pásmy:

```python
from datetime import datetime
import pytz

# Aktuální čas ve výchozím časovém pásmu (naivní)
naivni_ted = datetime.now()
print(f"Naivní datetime: {naivni_ted}")

# Aktuální čas v UTC
utc_ted = datetime.now(pytz.UTC)
print(f"UTC čas: {utc_ted}")

# Seznam běžných časových pásem
bezna_casova_pasma = ['US/Eastern', 'US/Pacific', 'Europe/London', 
                      'Europe/Prague', 'Asia/Tokyo', 'Australia/Sydney']

# Konverze UTC času do různých časových pásem
for pasmo_nazev in bezna_casova_pasma:
    pasmo = pytz.timezone(pasmo_nazev)
    mistni_cas = utc_ted.astimezone(pasmo)
    print(f"Čas v {pasmo_nazev}: {mistni_cas}")

# Vytvoření datetime s konkrétním časovým pásmem
vychodni = pytz.timezone('US/Eastern')
vychodni_cas = vychodni.localize(datetime(2023, 12, 31, 23, 59, 59))
print(f"Silvestr v New Yorku: {vychodni_cas}")

# Konverze mezi časovými pásmy
tokyo = pytz.timezone('Asia/Tokyo')
tokyo_cas = vychodni_cas.astimezone(tokyo)
print(f"Stejný okamžik v Tokiu: {tokyo_cas}")

# Získání aktuálního času v konkrétním časovém pásmu
def ziskej_aktualni_cas(pasmo_str):
    """Získá aktuální čas v určeném časovém pásmu."""
    pasmo = pytz.timezone(pasmo_str)
    return datetime.now(pasmo)

print(f"Aktuální čas v Paříži: {ziskej_aktualni_cas('Europe/Paris')}")

# Kontrola, zda má časové pásmo letní čas
pasmo = pytz.timezone('US/Pacific')
dt = datetime(2023, 1, 1, tzinfo=pasmo)  # Zima
dt2 = datetime(2023, 7, 1, tzinfo=pasmo)  # Léto
print(f"Pacific Time DST v lednu: {dt.dst()}")
print(f"Pacific Time DST v červenci: {dt2.dst()}")
```

## 6. Parsování datumů z řetězců

Převod řetězcových reprezentací datumů na objekty datetime:

```python
from datetime import datetime
import dateutil.parser

# Základní parsování s datetime.strptime
date_str = "2023-07-15 14:30:45"
parsed_date = datetime.strptime(date_str, "%Y-%m-%d %H:%M:%S")
print(f"Parsované datum: {parsed_date}")

# Parsování více formátů
formaty_data = ["%Y-%m-%d", "%m/%d/%Y", "%d.%m.%Y", "%B %d, %Y"]
retezce_data = ["2023-07-15", "07/15/2023", "15.07.2023", "July 15, 2023"]

for date_str in retezce_data:
    for format_data in formaty_data:
        try:
            parsed_date = datetime.strptime(date_str, format_data)
            print(f"Úspěšně parsováno '{date_str}' s formátem '{format_data}': {parsed_date}")
            break
        except ValueError:
            continue

# Použití dateutil.parser pro flexibilnější parsování
komplexni_data = [
    "Jul 15, 2023",
    "2023-07-15T14:30:45-05:00",
    "15/Jul/2023",
    "Sat, 15 Jul 2023 14:30:45 -0500",
    "July 15th, 2023 at 2:30pm"
]

for date_str in komplexni_data:
    try:
        parsed_date = dateutil.parser.parse(date_str)
        print(f"Parsováno '{date_str}' na: {parsed_date}")
    except ValueError as e:
        print(f"Nelze parsovat '{date_str}': {e}")

# Parsování data s časovým pásmem
date_with_tz = "2023-07-15T14:30:45+02:00"
parsed_date_with_tz = dateutil.parser.parse(date_with_tz)
print(f"Parsované datum s časovým pásmem: {parsed_date_with_tz}")
print(f"Informace o časovém pásmu: {parsed_date_with_tz.tzinfo}")
```

## 7. Práce s časovými razítky

Unix časová razítka reprezentují čas jako sekundy od 1. ledna 1970 a jsou užitečná pro výpočty a porovnání:

```python
import time
from datetime import datetime, timezone

# Aktuální časové razítko
aktualni_timestamp = time.time()
print(f"Aktuální Unix timestamp: {aktualni_timestamp}")

# Převod časového razítka na datetime
dt_z_timestamp = datetime.fromtimestamp(aktualni_timestamp)
print(f"Místní datetime z časového razítka: {dt_z_timestamp}")

# Převod časového razítka na UTC datetime
dt_utc_z_timestamp = datetime.fromtimestamp(aktualni_timestamp, tz=timezone.utc)
print(f"UTC datetime z časového razítka: {dt_utc_z_timestamp}")

# Převod datetime na časové razítko
ted = datetime.now()
timestamp_z_dt = ted.timestamp()
print(f"Časové razítko z datetime: {timestamp_z_dt}")

# Porovnávání datumů pomocí časových razítek
datum1 = datetime(2023, 7, 15)
datum2 = datetime(2023, 7, 20)

ts1 = datum1.timestamp()
ts2 = datum2.timestamp()

rozdil_sekund = ts2 - ts1
print(f"Rozdíl v sekundách: {rozdil_sekund}")
print(f"Rozdíl ve dnech: {rozdil_sekund / (60*60*24)}")

# Měření výkonu s time
def pomala_funkce():
    """Funkce, která trvá nějakou dobu."""
    soucet = 0
    for i in range(10000000):
        soucet += i
    return soucet

cas_zacatku = time.time()
vysledek = pomala_funkce()
cas_konce = time.time()

doba_provedeni = cas_konce - cas_zacatku
print(f"Funkce proběhla za {doba_provedeni:.4f} sekund")

# Uspání na určitou dobu
print("Čekání 2 sekundy...")
time.sleep(2)
print("Čekání dokončeno")

# Čas s vysokým rozlišením pro benchmarking
zacatek = time.perf_counter()
# Nějaká operace k měření
time.sleep(0.1)
konec = time.perf_counter()
print(f"Operace trvala {konec - zacatek:.6f} sekund")
```

## 8. Časové řady s Pandas

Pro datovou analýzu zahrnující datumy a časy poskytuje Pandas výkonnou funkcionalitu časových řad:

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

# Vytvoření časové řady s Pandas
rozsah_data = pd.date_range(start='2023-01-01', end='2023-01-10', freq='D')
print("Rozsah dat:")
print(rozsah_data)

# Vytvoření DataFrame s datovým sloupcem
df = pd.DataFrame(rozsah_data, columns=['datum'])
df['hodnota'] = np.random.randn(len(rozsah_data))
print("\nDataFrame s datovým sloupcem:")
print(df.head())

# Nastavení data jako indexu
df.set_index('datum', inplace=True)
print("\nDataFrame s datovým indexem:")
print(df.head())

# Datové rozsahy s různými frekvencemi
# D - kalendářní den, B - pracovní den, H - hodina, T/min - minuta, S - sekunda
# M - konec měsíce, MS - začátek měsíce, Y/A - konec roku, YS/AS - začátek roku
frekvence = ['D', 'B', 'H', 'T', 'M', 'MS', 'Q', 'Y']
for freq in frekvence:
    rozsah_data = pd.date_range(start='2023-01-01', periods=5, freq=freq)
    print(f"\nRozsah dat s frekvencí '{freq}':")
    print(rozsah_data)

# Operace s časovými řadami
# Převzorkování na různé časové frekvence
ts = pd.Series(np.random.randn(500), 
               index=pd.date_range('2023-01-01', periods=500, freq='H'))

print("\nOriginální časová řada (hodinová data):")
print(ts.head())

# Downsampling: vyšší frekvence na nižší frekvenci (např. z hodinové na denní)
denni_prumer = ts.resample('D').mean()
print("\nDownsamplovano na denní průměr:")
print(denni_prumer.head())

# Upsampling: nižší frekvence na vyšší frekvenci (např. z denní na hodinovou)
denni = pd.Series(np.random.randn(10), 
                  index=pd.date_range('2023-01-01', periods=10, freq='D'))
hodinova = denni.resample('H').ffill()  # Dopředné vyplnění chybějících hodnot
print("\nUpsamplovano z denní na hodinovou (s dopředným vyplněním):")
print(hodinova.head())

# Indexování a řezání na základě času
print("\nData pro 5. ledna:")
print(ts['2023-01-05'])

print("\nData od 5. ledna do 10. ledna:")
print(ts['2023-01-05':'2023-01-10'].head())

# Posuny časových řad
posunuta = ts.shift(periods=24)  # Posun o 24 hodin
print("\nPosunutá časová řada (o 24 hodin):")
print(pd.concat([ts.head(), posunuta.head()], axis=1, keys=['původní', 'posunutá']))

# Výpočty s klouzavým oknem
klouzavy_prumer = ts.rolling(window=24).mean()  # 24hodinový klouzavý průměr
print("\n24hodinový klouzavý průměr:")
print(klouzavy_prumer.head(30))

# Vykreslení časové řady
plt.figure(figsize=(10, 6))
denni_prumer.plot()
plt.title('Denní průměrné hodnoty')
plt.ylabel('Hodnota')
plt.grid(True)

# Uložení grafu místo zobrazení (pro Markdown)
plt.savefig('timeseries_plot.png')
print("\nGraf uložen jako 'timeseries_plot.png'")

# Práce s časovými pásmy v pandas
ts_utc = pd.Series(np.random.randn(5), 
                  index=pd.date_range('2023-01-01', periods=5, freq='D', tz='UTC'))
print("\nČasová řada s časovým pásmem UTC:")
print(ts_utc)

# Konverze do jiného časového pásma
ts_ny = ts_utc.tz_convert('America/New_York')
print("\nČasová řada převedena do časového pásma New York:")
print(ts_ny)
```
