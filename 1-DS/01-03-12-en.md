## Introduction to Advanced SQL in Data Science

SQL (Structured Query Language) is a powerful tool in a data scientist's toolkit. While basic SQL focuses on CRUD operations, advanced SQL enables complex data analysis directly within the database, which can be more efficient than extracting data to Python for analysis.

Key benefits of advanced SQL in data science:
- **Reduced data transfer**: Analyze large datasets where they live
- **Performance improvements**: Database engines are optimized for data operations
- **Standardization**: SQL is widely used across different data platforms
- **Data integrity**: Work with consistent, transaction-based data

## Window Functions for Advanced Analytics

Window functions perform calculations across a set of table rows related to the current row, similar to GROUP BY but without collapsing the rows.

```python
# Create a sales table with time dimensions for window function demo
sales = pd.DataFrame({
    'date': pd.date_range(start='2023-01-01', periods=20, freq='D'),
    'product_id': np.random.choice(range(1, 6), 20),
    'salesperson_id': np.random.choice(range(1, 5), 20),
    'quantity': np.random.randint(1, 10, 20),
    'revenue': np.random.randint(100, 1000, 20)
})

sales.to_sql('sales', engine, index=False)

# Example with window functions
window_query = """
SELECT
    date,
    product_id,
    revenue,
    SUM(revenue) OVER (PARTITION BY product_id ORDER BY date) AS running_total,
    AVG(revenue) OVER (PARTITION BY product_id) AS product_avg_revenue,
    RANK() OVER (PARTITION BY product_id ORDER BY revenue DESC) AS revenue_rank
FROM sales
ORDER BY product_id, date
"""

window_results = pd.read_sql(window_query, engine)
print("Window function example:")
print(window_results.head(10))

# Rolling averages using window functions
rolling_avg_query = """
SELECT
    date,
    product_id,
    revenue,
    AVG(revenue) OVER (
        PARTITION BY product_id
        ORDER BY date
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS moving_avg_3day
FROM sales
ORDER BY product_id, date
"""

rolling_avg = pd.read_sql(rolling_avg_query, engine)
print("\n3-day rolling average by product:")
print(rolling_avg.head(10))
```

## Common Table Expressions (CTEs)

CTEs provide a way to write more readable and maintainable SQL by creating named temporary result sets.

```python
# Basic CTE example
basic_cte = """
WITH OrderSummary AS (
    SELECT
        customer_id,
        COUNT(*) AS order_count,
        SUM(amount) AS total_amount
    FROM orders
    GROUP BY customer_id
)
SELECT
    c.customer_name,
    c.country,
    os.order_count,
    os.total_amount
FROM customers c
JOIN OrderSummary os ON c.customer_id = os.customer_id
ORDER BY os.total_amount DESC
"""

cte_results = pd.read_sql(basic_cte, engine)
print("Basic CTE example:")
print(cte_results)

# Recursive CTE example (creating a date range)
recursive_cte = """
WITH RECURSIVE DateSeries AS (
    SELECT date('2023-01-01') AS date
    UNION ALL
    SELECT date(date, '+1 day')
    FROM DateSeries
    WHERE date < '2023-01-10'
)
SELECT date FROM DateSeries
"""

date_series = pd.read_sql(recursive_cte, engine)
print("\nRecursive CTE example - Date series:")
print(date_series)
```

## Handling Time Series Data in SQL

Time series analysis is common in data science, and SQL provides tools for working with time-based data.

```python
# Create time series data for sales across different regions
dates = pd.date_range(start='2022-01-01', periods=365, freq='D')
regions = ['North', 'South', 'East', 'West']

records = []
for date in dates:
    for region in regions:
        # Create seasonal pattern with some randomness
        month = date.month
        # More sales in summer and December
        seasonal_factor = 1.0
        if month in [6, 7, 8]:  # Summer
            seasonal_factor = 1.3
        elif month == 12:  # December
            seasonal_factor = 1.5
        
        # Weekend effect
        weekend_factor = 1.3 if date.dayofweek >= 5 else 1.0
        
        # Base sales with some randomness
        base_sales = {
            'North': 1000, 'South': 1200, 
            'East': 900, 'West': 1100
        }[region]
        
        sales_amount = int(base_sales * seasonal_factor * weekend_factor * np.random.uniform(0.8, 1.2))
        
        records.append({
            'date': date,
            'region': region,
            'sales': sales_amount
        })

time_series_df = pd.DataFrame(records)
time_series_df.to_sql('regional_sales', engine, index=False)

# Date extraction and grouping
time_grouping = """
SELECT 
    strftime('%Y-%m', date) AS month,
    region,
    SUM(sales) AS monthly_sales
FROM regional_sales
GROUP BY month, region
ORDER BY month, region
"""

monthly_sales = pd.read_sql(time_grouping, engine)
print("Monthly sales by region:")
print(monthly_sales.head(8))

# Moving averages with window functions
moving_avg = """
SELECT
    date,
    region,
    sales,
    AVG(sales) OVER (
        PARTITION BY region
        ORDER BY date
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS moving_avg_7day
FROM regional_sales
WHERE region = 'North'
ORDER BY date
LIMIT 20
"""

try:
    north_moving_avg = pd.read_sql(moving_avg, engine)
    print("\n7-day moving average for North region:")
    print(north_moving_avg)
except Exception as e:
    print(f"\nError with window function (SQLite may have limited support): {e}")
    print("Using pandas to calculate the moving average instead:")
    
    north_sales = pd.read_sql(
        "SELECT date, sales FROM regional_sales WHERE region = 'North' ORDER BY date",
        engine
    )
    north_sales['moving_avg_7day'] = north_sales['sales'].rolling(window=7).mean()
    print(north_sales.head(20))

# Year-over-year comparison
yoy_comparison = """
SELECT
    strftime('%m', date) AS month,
    strftime('%Y', date) AS year,
    region,
    SUM(sales) AS monthly_sales
FROM regional_sales
GROUP BY year, month, region
ORDER BY region, month, year
"""

yoy_data = pd.read_sql(yoy_comparison, engine)
print("\nYear-over-year monthly sales:")
print(yoy_data.head(8))
```

## Query Optimization for Data Science

Optimizing SQL queries is crucial when working with large datasets.

```python
import time

# Create a larger dataset for performance testing
num_records = 50000
large_df = pd.DataFrame({
    'id': range(1, num_records + 1),
    'category': np.random.choice(['A', 'B', 'C', 'D', 'E'], num_records),
    'value1': np.random.randint(1, 100, num_records),
    'value2': np.random.uniform(0, 1000, num_records),
    'text_field': np.random.choice(['apple', 'banana', 'orange', 'grape', 'kiwi'], num_records),
    'date_field': pd.date_range(start='2020-01-01', periods=num_records, freq='H')
})

large_df.to_sql('large_table', engine, index=False, if_exists='replace')

# Create indexes to improve performance
engine.execute('CREATE INDEX idx_category ON large_table(category)')
engine.execute('CREATE INDEX idx_date ON large_table(date_field)')

# Function to time query execution
def time_query(query, description):
    start_time = time.time()
    result = pd.read_sql(query, engine)
    end_time = time.time()
    print(f"{description} - Execution time: {end_time - start_time:.4f} seconds")
    print(f"Result shape: {result.shape}")
    return result

# Unoptimized query with full table scan
unoptimized_query = """
SELECT *
FROM large_table
WHERE category = 'A' AND value1 > 50
"""

# Optimized query using indexed columns first
optimized_query = """
SELECT *
FROM large_table
WHERE category = 'A' AND value1 > 50
"""

# Compare performance
print("Comparing query performance:")
unopt_results = time_query(unoptimized_query, "Unoptimized query")
opt_results = time_query(optimized_query, "Optimized query (with index)")

# Using EXPLAIN to understand query execution plan
explain_query = "EXPLAIN QUERY PLAN " + optimized_query
explain_results = engine.execute(explain_query).fetchall()
print("\nQuery execution plan:")
for row in explain_results:
    print(row)

# Using LIMIT appropriately
limit_query = """
SELECT *
FROM large_table
ORDER BY value2 DESC
LIMIT 10
"""

top_values = time_query(limit_query, "Using LIMIT for top values")

# Avoiding SELECT * for large tables
optimized_select = """
SELECT id, category, value1
FROM large_table
WHERE category = 'A' AND value1 > 50
"""

specific_cols = time_query(optimized_select, "Selecting specific columns only")
```

## Summary and Best Practices

Advanced SQL skills enable data scientists to efficiently work with large datasets directly in the database, reducing the need for additional processing in Python. Here are the key takeaways:

1. **Keep processing close to the data**: Use SQL for filtering, aggregating, and transforming data before extracting to Python
2. **Utilize window functions**: They enable sophisticated analytics like running totals, moving averages, and rankings
3. **Write readable SQL**: Common Table Expressions (CTEs) improve readability and maintainability
4. **Leverage advanced joins**: Master various join types and subqueries to combine data from multiple sources
5. **Optimize query performance**: Create appropriate indexes, avoid SELECT *, and understand query execution plans
6. **Learn database-specific functions**: Different database systems offer specialized statistical and analytical functions
7. **Use parameterization**: Avoid SQL injection and improve reusability
8. **Consider sqlalchemy**: It provides a consistent interface across different database engines

```python
# Example of putting it all together: A data science SQL workflow
def data_science_sql_workflow(engine, date_range_start, date_range_end, min_order_count=5):
    """
    Example of a complete data science workflow combining advanced SQL techniques
    """
    # Parameterized query using best practices
    query = """
    WITH CustomerOrders AS (
        -- CTE for order aggregations
        SELECT
            c.customer_id,
            c.customer_name,
            c.country,
            COUNT(DISTINCT o.order_id) AS order_count,
            SUM(o.amount) AS total_spent,
            AVG(o.amount) AS avg_order_value,
            MIN(o.order_date) AS first_order,
            MAX(o.order_date) AS last_order
        FROM customers c
        JOIN orders o ON c.customer_id = o.customer_id
        WHERE o.order_date BETWEEN ? AND ?
        GROUP BY c.customer_id, c.customer_name, c.country
        HAVING COUNT(DISTINCT o.order_id) >= ?
    ),
    CustomerProductPreferences AS (
        -- CTE for product preferences
        SELECT
            o.customer_id,
            p.product_name,
            SUM(oi.quantity) AS total_quantity,
            -- Window function to rank products by customer preference
            RANK() OVER (
                PARTITION BY o.customer_id 
                ORDER BY SUM(oi.quantity) DESC
            ) AS product_rank
        FROM orders o
        JOIN order_items oi ON o.order_id = oi.order_id
        JOIN products p ON oi.product_id = p.product_id
        WHERE o.order_date BETWEEN ? AND ?
        GROUP BY o.customer_id, p.product_name
    ),
    CustomerMetrics AS (
        -- CTE for advanced customer metrics
        SELECT
            co.customer_id,
            co.customer_name,
            co.country,
            co.order_count,
            co.total_spent,
            co.avg_order_value,
            -- Calculate time between first and last order (days)
            JULIANDAY(co.last_order) - JULIANDAY(co.first_order) AS customer_lifetime_days,
            -- Calculate spend velocity (amount per day)
            CASE
                WHEN JULIANDAY(co.last_order) = JULIANDAY(co.first_order) THEN co.total_spent
                ELSE co.total_spent / (JULIANDAY(co.last_order) - JULIANDAY(co.first_order) + 1)
            END AS spend_per_day
        FROM CustomerOrders co
    )
    
    -- Final query combining all CTEs
    SELECT
        cm.customer_id,
        cm.customer_name,
        cm.country,
        cm.order_count,
        cm.total_spent,
        cm.avg_order_value,
        cm.customer_lifetime_days,
        cm.spend_per_day,
        -- Get top product preference
        (
            SELECT product_name
            FROM CustomerProductPreferences cpp
            WHERE cpp.customer_id = cm.customer_id AND cpp.product_rank = 1
            LIMIT 1
        ) AS favorite_product
    FROM CustomerMetrics cm
    ORDER BY cm.total_spent DESC
    """
    
    # Execute parameterized query (each ? is replaced by a parameter)
    params = [date_range_start, date_range_end, min_order_count, date_range_start, date_range_end]
    try:
        result = pd.read_sql(query, engine, params=params)
        return result
    except Exception as e:
        print(f"Error executing SQL workflow: {e}")
        return None

# Example usage:
customer_analysis = data_science_sql_workflow(
    engine, 
    date_range_start='2023-01-01', 
    date_range_end='2023-12-31', 
    min_order_count=1
)

print("\nComplete customer analysis workflow result:")
if customer_analysis is not None:
    print(customer_analysis)
else:
    print("Workflow execution failed.")

# Clean up - close connection
engine.dispose()
```
