```python
# Vytvoření ukázkového DataFramu
def create_sample_df(n_rows=1000000):
    df = pd.DataFrame({
        'id': range(n_rows),
        'string_col': np.random.choice(['a', 'b', 'c', 'd', 'e'], n_rows),
        'float_col': np.random.random(n_rows),
        'int_col': np.random.randint(0, 100, n_rows),
        'datetime_col': pd.date_range(start='2020-01-01', periods=n_rows, freq='s'),
        'bool_col': np.random.choice([True, False], n_rows)
    })
    return df

# Vytvoření DataFramu
df = create_sample_df()

# Funkce pro získání využití paměti DataFramu
def get_memory_usage(df, detail=False):
    memory_usage = df.memory_usage(deep=True)
    memory_usage_mb = memory_usage.sum() / 1024**2
    if detail:
        print("Využití paměti podle sloupců:")
        for col in memory_usage.index:
            print(f"{col}: {memory_usage[col] / 1024**2:.2f} MB")
    return memory_usage_mb

# Kontrola počátečního využití paměti
initial_memory = get_memory_usage(df, detail=True)
print(f"\nCelkové využití paměti: {initial_memory:.2f} MB")
```

## Optimalizace datových typů

Jedním z nejefektivnějších způsobů, jak snížit využití paměti, je optimalizace datových typů.

```python
def optimize_dtypes(df):
    """Optimalizace datových typů pro snížení využití paměti."""
    df_optimized = df.copy()
    
    # Optimalizace celých čísel
    int_cols = df_optimized.select_dtypes(include=['int']).columns
    for col in int_cols:
        col_min = df_optimized[col].min()
        col_max = df_optimized[col].max()
        
        if col_min >= 0:  # Nezáporná celá čísla
            if col_max < 2**8:
                df_optimized[col] = df_optimized[col].astype(np.uint8)
            elif col_max < 2**16:
                df_optimized[col] = df_optimized[col].astype(np.uint16)
            elif col_max < 2**32:
                df_optimized[col] = df_optimized[col].astype(np.uint32)
        else:  # Kladná i záporná celá čísla
            if col_min > -2**7 and col_max < 2**7:
                df_optimized[col] = df_optimized[col].astype(np.int8)
            elif col_min > -2**15 and col_max < 2**15:
                df_optimized[col] = df_optimized[col].astype(np.int16)
            elif col_min > -2**31 and col_max < 2**31:
                df_optimized[col] = df_optimized[col].astype(np.int32)
    
    # Optimalizace desetinných čísel
    float_cols = df_optimized.select_dtypes(include=['float']).columns
    for col in float_cols:
        df_optimized[col] = df_optimized[col].astype(np.float32)
    
    return df_optimized

# Optimalizace našeho ukázkového DataFramu
df_optimized = optimize_dtypes(df)

# Kontrola využití paměti po optimalizaci
optimized_memory = get_memory_usage(df_optimized, detail=True)
print(f"\nCelkové využití paměti po optimalizaci: {optimized_memory:.2f} MB")
print(f"Snížení paměti: {(1 - optimized_memory/initial_memory)*100:.2f}%")

# Porovnání datových typů před a po optimalizaci
print("\nDatové typy před optimalizací:")
print(df.dtypes)
print("\nDatové typy po optimalizaci:")
print(df_optimized.dtypes)
```

```python
# Funkce pro zpracování dat po částech
def process_in_chunks(filename, chunksize=100000):
    # Nastavení úložiště výsledků
    results = []
    
    # Zpracování souboru po částech
    for i, chunk in enumerate(pd.read_csv(filename, chunksize=chunksize)):
        # Příklad zpracování: výpočet průměru číselných sloupců
        result = chunk.select_dtypes(include=[np.number]).mean()
        results.append(result)
        print(f"Zpracována část {i+1} s {len(chunk)} řádky")
    
    # Kombinace výsledků ze všech částí
    final_result = pd.concat(results, axis=1).mean(axis=1)
    return final_result
```

## Vytváření vlastních accessorů

Vlastní accessory umožňují rozšířit funkcionalitu pandas organizovaným, objektově orientovaným způsobem.

```python
@pd.api.extensions.register_dataframe_accessor("stats")
class StatsAccessor:
    def __init__(self, pandas_obj):
        self._obj = pandas_obj
    
    def describe_enhanced(self):
        """Rozšířená metoda describe s dodatečnými statistikami"""
        # Získání běžného describe
        desc = self._obj.describe()
        
        # Přidání dalších metrik
        numeric_cols = self._obj.select_dtypes(include=[np.number]).columns
        for col in numeric_cols:
            desc.loc['skew', col] = self._obj[col].skew()
            desc.loc['kurtosis', col] = self._obj[col].kurtosis()
            desc.loc['iqr', col] = self._obj[col].quantile(0.75) - self._obj[col].quantile(0.25)
            desc.loc['cv', col] = self._obj[col].std() / self._obj[col].mean() if self._obj[col].mean() != 0 else np.nan
        
        return desc
    
    def outliers(self, column, method='iqr', threshold=1.5):
        """Identifikace odlehlých hodnot ve sloupci
        
        Parametry:
        -----------
        column : str
            Název sloupce pro kontrolu odlehlých hodnot
        method : str, default 'iqr'
            Metoda použití ('iqr' nebo 'zscore')
        threshold : float, default 1.5
            Práh pro detekci odlehlých hodnot (pro metodu IQR)
        
        Vrací:
        --------
        DataFrame s odlehlými hodnotami
        """
        if method == 'iqr':
            Q1 = self._obj[column].quantile(0.25)
            Q3 = self._obj[column].quantile(0.75)
            IQR = Q3 - Q1
            
            lower_bound = Q1 - threshold * IQR
            upper_bound = Q3 + threshold * IQR
            
            outliers = self._obj[(self._obj[column] < lower_bound) | (self._obj[column] > upper_bound)]
            return outliers
        
        elif method == 'zscore':
            from scipy import stats
            z_scores = stats.zscore(self._obj[column])
            outliers = self._obj[abs(z_scores) > threshold]
            return outliers
        
        else:
            raise ValueError("Metoda musí být buď 'iqr' nebo 'zscore'")

@pd.api.extensions.register_dataframe_accessor("finance")
class FinanceAccessor:
    def __init__(self, pandas_obj):
        self._obj = pandas_obj
    
    def pct_change_enhanced(self, periods=1):
        """Výpočet procentuální změny s dodatečnými metrikami"""
        # Předpokládáme, že máme finanční DataFrame se sloupci jako 'close', 'open' atd.
        pct_change = self._obj.pct_change(periods)
        return pct_change
    
    def moving_average(self, column, window=14, type='simple'):
        """Výpočet klouzavého průměru
        
        Parametry:
        -----------
        column : str
            Sloupec pro výpočet klouzavého průměru
        window : int, default 14
            Velikost okna pro klouzavý průměr
        type : str, default 'simple'
            Typ klouzavého průměru ('simple', 'exponential', nebo 'weighted')
        """
        if type == 'simple':
            return self._obj[column].rolling(window=window).mean()
        elif type == 'exponential':
            return self._obj[column].ewm(span=window).mean()
        elif type == 'weighted':
            weights = np.arange(1, window + 1)
            return self._obj[column].rolling(window=window).apply(
                lambda x: np.sum(weights * x) / weights.sum(), raw=True)
        else:
            raise ValueError("Typ musí být jeden z 'simple', 'exponential', nebo 'weighted'")
```

```python
# Vytvoření většího ukázkového DataFramu pro pokročilé operace
def create_advanced_df(n_rows=500000):
    df = pd.DataFrame({
        'id': range(n_rows),
        'group': np.random.choice(['A', 'B', 'C', 'D', 'E'], n_rows),
        'subgroup': np.random.choice(['X', 'Y', 'Z'], n_rows),
        'value1': np.random.random(n_rows),
        'value2': np.random.random(n_rows) * 100,
        'date': pd.date_range(start='2020-01-01', periods=n_rows, freq='min')
    })
    return df

adv_df = create_advanced_df()

# Paměťově efektivní operace groupby
# Místo ukládání všech mezivýsledků můžeme použít agg přímo
t1 = pd.Timestamp.now()
result = adv_df.groupby(['group', 'subgroup']).agg({
    'value1': ['mean', 'std', 'min', 'max'],
    'value2': ['mean', 'std', 'min', 'max']
})
t2 = pd.Timestamp.now()
print(f"Čas agregace: {(t2-t1).total_seconds():.2f} sekund")
print(f"Tvar výsledku agregace: {result.shape}")

# Optimalizované operace apply pomocí knihovny numba
try:
    from numba import jit
    
    # Definice funkce, která bude optimalizována pomocí numba
    @jit(nopython=True)
    def numba_function(x, y):
        result = 0
        for i in range(len(x)):
            result += x[i] * y[i]
        return result / len(x)
    
    # Aplikace optimalizované funkce
    t1 = pd.Timestamp.now()
    adv_df['computed'] = numba_function(
        adv_df['value1'].values, 
        adv_df['value2'].values
    )
    t2 = pd.Timestamp.now()
    print(f"\nČas výpočtu optimalizovaného pomocí numba: {(t2-t1).total_seconds():.2f} sekund")
    
except ImportError:
    print("\nNumba není nainstalována, přeskakuji příklad optimalizovaného výpočtu")
    print("Nainstalujte pomocí: pip install numba")

# Použití eval pro efektivní výpočty
t1 = pd.Timestamp.now()
```

```python
        'total_amount': np.zeros(n_rows),  # Spočítáme to
        'store_and_fwd_flag': np.random.choice(['Y', 'N'], n_rows, p=[0.05, 0.95])
    })
    
    # Výpočet total_amount na základě ostatních polí

# Vytvoření taxi datasetu
taxi_df = create_taxi_dataset()

# Kontrola počátečního využití paměti
print("Počáteční taxi dataset:")
print(f"Tvar: {taxi_df.shape}")
init_memory = get_memory_usage(taxi_df, detail=True)

# Optimalizujme tento dataset
def optimize_taxi_dataset(df):
    """Aplikace více optimalizačních technik na taxi dataset"""
    df_opt = df.copy()
    
    # 1. Převod textových sloupců na kategorické
    for col in ['payment_type', 'store_and_fwd_flag']:
        df_opt[col] = df_opt[col].astype('category')
    
    # 2. Optimalizace číselných typů
    # Sloupce s celými čísly
    for col in ['passenger_count']:
        df_opt[col] = pd.to_numeric(df_opt[col], downcast='integer')
    
    # Sloupce s desetinnými čísly
    float_cols = ['trip_distance', 'fare_amount', 'extra', 'mta_tax', 
                 'tip_amount', 'tolls_amount', 'total_amount']
    for col in float_cols:
        df_opt[col] = pd.to_numeric(df_opt[col], downcast='float')
    
    # 3. Optimalizace dat o poloze
    # Převod na 32bitová desetinná čísla je obvykle dostačující pro GPS souřadnice
    for col in ['pickup_longitude', 'pickup_latitude',

        df_opt[col] = df_opt[col].astype(np.float32)
    
    # 4. Vytvoření datetime indexu pro efektivní dotazy založené na čase
    df_opt['pickup_datetime'] = pd.to_datetime(df_opt['pickup_datetime'])
    df_opt['dropoff_datetime'] = pd.to_datetime(df_opt['dropoff_datetime'])
    
    return df_opt
```
