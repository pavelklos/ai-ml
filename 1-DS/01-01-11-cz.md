## 1. Úvod do vizualizace dat s knihovnou Matplotlib

Vizualizace dat je klíčovou součástí datové vědy, neboť nám pomáhá pochopit vzory, trendy a odlehlé hodnoty v našich datech. Matplotlib je jednou z nejpopulárnějších knihoven pro tvorbu grafů v Pythonu, která poskytuje širokou škálu nástrojů pro vytváření statických, animovaných a interaktivních vizualizací.

```python
# Import knihoven
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

# Pro zajištění zobrazení grafů v Jupyter notebooku
%matplotlib inline

# Pro lepší vzhled grafů
plt.style.use('seaborn-v0_8-whitegrid')
```

## 2. Nastavení knihovny Matplotlib

Matplotlib nabízí dva hlavní způsoby použití:
- Rozhraní podobné MATLABu založené na stavovém řízení prostřednictvím `pyplot`
- Objektově orientované rozhraní prostřednictvím objektů Figure a Axes

```python
# Základní rozhraní pyplot
plt.figure(figsize=(10, 6))  # Vytvoření obrázku se zadanou velikostí
plt.plot([1, 2, 3, 4], [1, 4, 9, 16])
plt.title('Jednoduchý graf')
plt.xlabel('Osa X')
plt.ylabel('Osa Y')
plt.show()

# Objektově orientované rozhraní
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot([1, 2, 3, 4], [1, 4, 9, 16])
ax.set_title('Objektově orientovaný graf')
ax.set_xlabel('Osa X')
ax.set_ylabel('Osa Y')
plt.show()
```

## 3. Základní typy grafů

Matplotlib umožňuje vytvářet různé typy grafů pro vizualizaci různých aspektů vašich dat.

### Čárové grafy

```python
# Generování dat
x = np.linspace(0, 10, 100)
y1 = np.sin(x)
y2 = np.cos(x)

# Vytvoření čárového grafu
plt.figure(figsize=(10, 6))
plt.plot(x, y1, label='sin(x)', color='blue', linestyle='-')
plt.plot(x, y2, label='cos(x)', color='red', linestyle='--')
plt.title('Funkce sinus a kosinus')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()
plt.grid(True)
plt.show()
```

### Sloupcové grafy

```python
# Vytvoření sloupcového grafu
categories = ['A', 'B', 'C', 'D', 'E']
values = [25, 40, 30, 55, 15]

plt.figure(figsize=(10, 6))
plt.bar(categories, values, color='skyblue', edgecolor='black')
plt.title('Příklad sloupcového grafu')
plt.xlabel('Kategorie')
plt.ylabel('Hodnoty')
plt.show()

# Horizontální sloupcový graf
plt.figure(figsize=(10, 6))
plt.barh(categories, values, color='lightgreen', edgecolor='black')
plt.title('Horizontální sloupcový graf')
plt.xlabel('Hodnoty')
plt.ylabel('Kategorie')
plt.show()
```

### Bodové grafy

```python
# Generování náhodných dat
np.random.seed(42)
x = np.random.rand(50)
y = np.random.rand(50)
colors = np.random.rand(50)
sizes = 1000 * np.random.rand(50)

# Vytvoření bodového grafu
plt.figure(figsize=(10, 6))
plt.scatter(x, y, c=colors, s=sizes, alpha=0.6, cmap='viridis')
plt.title('Příklad bodového grafu')
plt.xlabel('Hodnoty X')
plt.ylabel('Hodnoty Y')
plt.colorbar(label='Barva')
plt.show()
```

### Histogramy

```python
# Generování dat s normálním rozdělením
data = np.random.normal(0, 1, 1000)

# Vytvoření histogramu
plt.figure(figsize=(10, 6))
plt.hist(data, bins=30, color='purple', alpha=0.7, edgecolor='black')
plt.title('Příklad histogramu')
plt.xlabel('Hodnota')
plt.ylabel('Četnost')
plt.grid(True, alpha=0.3)
plt.show()
```

### Koláčové grafy

```python
# Data pro koláčový graf
labels = ['A', 'B', 'C', 'D', 'E']
sizes = [25, 20, 30, 15, 10]
explode = (0, 0.1, 0, 0, 0)  # Vysunout 2. výseč

# Vytvoření koláčového grafu
plt.figure(figsize=(10, 6))
plt.pie(sizes, explode=explode, labels=labels, autopct='%1.1f%%',
        shadow=True, startangle=90)
plt.axis('equal')  # Stejný poměr stran zajistí, že koláč je vykreslen jako kruh
plt.title('Příklad koláčového grafu')
plt.show()
```

## 4. Přizpůsobení grafů

Matplotlib poskytuje rozsáhlé možnosti pro přizpůsobení vašich vizualizací.

```python
# Nastavení barev, značek a stylů čar
plt.figure(figsize=(10, 6))
plt.plot([1, 2, 3, 4], [1, 4, 9, 16], 'ro-', linewidth=2, markersize=8)
plt.plot([1, 2, 3, 4], [1, 2, 3, 4], 'bs--', linewidth=2, markersize=8)
plt.title('Přizpůsobený čárový graf')
plt.xlabel('Osa X')
plt.ylabel('Osa Y')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Nastavení limitů os
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.xlim(0, 5)
plt.ylim(-1.5, 1.5)
plt.title('Graf s vlastními limity os')
plt.grid(True)
plt.show()

# Použití různých barevných map
plt.figure(figsize=(10, 6))
plt.scatter(x, y, c=colors, s=sizes, alpha=0.6, cmap='plasma')
plt.colorbar(label='Hodnota barvy')
plt.title('Bodový graf s vlastní barevnou mapou')
plt.show()

# Přidání barvy pozadí
plt.figure(figsize=(10, 6), facecolor='#f0f0f0')
plt.plot(x, y1, color='navy')
plt.title('Graf s vlastním pozadím')
plt.xlabel('Osa X')
plt.ylabel('Osa Y')
plt.grid(True, linestyle='--', alpha=0.7)
plt.tight_layout()
plt.show()
```

## 5. Více grafů a podgrafy

Pro složitější analýzu můžete vytvořit více grafů v jednom obrázku.

```python
# Vytvoření jednoduché mřížky podgrafů
fig, axs = plt.subplots(2, 2, figsize=(12, 10))

# První podgraf
axs[0, 0].plot(x, y1)
axs[0, 0].set_title('Funkce sinus')
axs[0, 0].set_xlabel('x')
axs[0, 0].set_ylabel('sin(x)')
axs[0, 0].grid(True)

# Druhý podgraf
axs[0, 1].plot(x, y2, color='red')
axs[0, 1].set_title('Funkce kosinus')
axs[0, 1].set_xlabel('x')
axs[0, 1].set_ylabel('cos(x)')
axs[0, 1].grid(True)

# Třetí podgraf
axs[1, 0].scatter(np.random.rand(50), np.random.rand(50))
axs[1, 0].set_title('Náhodný bodový graf')
axs[1, 0].set_xlabel('X')
axs[1, 0].set_ylabel('Y')

# Čtvrtý podgraf
axs[1, 1].bar(categories, values)
axs[1, 1].set_title('Sloupcový graf')
axs[1, 1].set_xlabel('Kategorie')
axs[1, 1].set_ylabel('Hodnoty')

plt.tight_layout()
plt.show()

# Vytvoření podgrafů s různými velikostmi (GridSpec)
import matplotlib.gridspec as gridspec

fig = plt.figure(figsize=(12, 8))
gs = gridspec.GridSpec(2, 3)

# Hlavní graf zabírá celý první řádek
ax1 = fig.add_subplot(gs[0, :])
ax1.plot(x, y1)
ax1.set_title('Hlavní graf (celý řádek)')

# Tři menší grafy ve druhém řádku
ax2 = fig.add_subplot(gs[1, 0])
ax2.plot(x, y2)
ax2.set_title('Graf 1')

ax3 = fig.add_subplot(gs[1, 1])
ax3.scatter(x[:20], y1[:20])
ax3.set_title('Graf 2')

ax4 = fig.add_subplot(gs[1, 2])
ax4.bar(categories, values)
ax4.set_title('Graf 3')

plt.tight_layout()
plt.show()
```

## 6. Přidávání popisků

Popisky mohou pomoci zvýraznit důležité části vašich vizualizací.

```python
# Vytvoření grafu s popisky
plt.figure(figsize=(10, 6))
plt.plot(x, y1)

# Přidání bodu a jeho popis
max_x = x[np.argmax(y1)]
max_y = np.max(y1)
plt.plot(max_x, max_y, 'ro')
plt.annotate('Maximální hodnota', 
             xy=(max_x, max_y),
             xytext=(max_x-1, max_y+0.1),
             arrowprops=dict(facecolor='black', shrink=0.05, width=1.5),
             fontsize=12)

# Přidání textu do grafu
plt.text(5, 0, "sin(x) prochází nulou vícekrát", 
         fontsize=12, ha='center', bbox=dict(facecolor='yellow', alpha=0.3))

plt.title('Funkce sinus s popisky')
plt.xlabel('x')
plt.ylabel('sin(x)')
plt.grid(True)
plt.show()

# Přidání vertikální a horizontální čáry
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.axhline(y=0, color='r', linestyle='-', alpha=0.3)
plt.axvline(x=5, color='g', linestyle='--', alpha=0.3)
plt.title('Graf s referenčními čarami')
plt.grid(True)
plt.show()
```

## 8. Statistická vizualizace

Matplotlib lze použít k vytvoření běžných statistických vizualizací.

```python
# Generování dat
np.random.seed(42)
data1 = np.random.normal(0, 1, 1000)
data2 = np.random.normal(2, 1.5, 1000)

# Krabicový graf
plt.figure(figsize=(10, 6))
plt.boxplot([data1, data2], labels=['Rozdělení 1', 'Rozdělení 2'])
plt.title('Porovnání krabicových grafů')
plt.ylabel('Hodnoty')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Houslový graf
plt.figure(figsize=(10, 6))
plt.violinplot([data1, data2])
plt.title('Porovnání houslových grafů')
plt.xticks([1, 2], ['Rozdělení 1', 'Rozdělení 2'])
plt.ylabel('Hodnoty')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Chybové úsečky
x_data = np.array([1, 2, 3, 4, 5])
y_data = np.array([1, 2, 4, 8, 16])
y_error = np.array([0.1, 0.2, 0.4, 0.8, 1.6])

plt.figure(figsize=(10, 6))
plt.errorbar(x_data, y_data, yerr=y_error, fmt='o-', capsize=5, ecolor='red', markeredgecolor='blue')
plt.title('Graf s chybovými úsečkami')
plt.xlabel('X')
plt.ylabel('Y')
plt.grid(True)
plt.show()

# Použití integrace pandas s matplotlib pro statistickou vizualizaci
df = pd.DataFrame({
    'Skupina A': np.random.normal(0, 1, 100),
    'Skupina B': np.random.normal(2, 1, 100),
    'Skupina C': np.random.normal(-1, 1.5, 100)
})

# Histogram více rozdělení
plt.figure(figsize=(10, 6))
plt.hist(df, bins=20, alpha=0.7)
plt.title('Histogram více rozdělení')
plt.xlabel('Hodnota')
plt.ylabel('Četnost')
plt.legend(df.columns)
plt.grid(True, alpha=0.3)
plt.show()
```

## 9. Osvědčené postupy pro vizualizaci dat

Vytváření efektivních vizualizací vyžaduje dodržování některých osvědčených postupů:

```python
# 1. Používejte jasné názvy a popisky
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.title('Sinusová vlna v průběhu času', fontsize=14)
plt.xlabel('Čas (sekundy)', fontsize=12)
plt.ylabel('Amplituda', fontsize=12)
plt.grid(True)
plt.show()

# 2. Vybírejte vhodné barvy, zejména pro publikum s barvoslepostí
from matplotlib.cm import viridis, plasma, inferno, magma, cividis

plt.figure(figsize=(15, 8))
for i, cmap in enumerate([viridis, plasma, inferno, magma, cividis]):
    plt.subplot(1, 5, i+1)
    plt.scatter(np.random.rand(50), np.random.rand(50), c=np.random.rand(50), cmap=cmap)
    plt.title(cmap.name)
plt.tight_layout()
plt.show()

# 3. Vyhýbejte se zbytečným prvkům v grafu (chart junk)
# Špatný příklad s příliš mnoha prvky
plt.figure(figsize=(10, 6))
plt.plot(x, y1, 'b-')
plt.title('Přeplněný graf (špatný postup)')
plt.grid(True, which='both', axis='both')
plt.minorticks_on()
plt.grid(True, which='minor', alpha=0.3)
for i in range(0, 11, 1):
    plt.axvline(x=i, color='red', alpha=0.1)
    plt.axhline(y=i/10-0.5, color='green', alpha=0.1)
plt.show()

# Dobrý příklad s čistým designem
plt.figure(figsize=(10, 6))
plt.plot(x, y1, 'b-')
plt.title('Čistý graf (dobrý postup)')
plt.grid(True, alpha=0.3)
plt.show()

# 4. Používejte vhodné měřítko
# Lineární měřítko
plt.figure(figsize=(10, 6))
plt.subplot(1, 2, 1)
plt.plot(np.arange(1, 1001), np.arange(1, 1001))
plt.title('Lineární měřítko')
plt.xlabel('X')
plt.ylabel('Y')
plt.grid(True)

# Logaritmické měřítko pro stejná data
plt.subplot(1, 2, 2)
plt.loglog(np.arange(1, 1001), np.arange(1, 1001))
plt.title('Logaritmické měřítko')
plt.xlabel('X (log)')
plt.ylabel('Y (log)')
plt.grid(True, which='both')
plt.tight_layout()
plt.show()

# 5. Dbejte na přístupnost (adekvátní velikosti písma, kontrast)
plt.figure(figsize=(10, 6))
plt.plot(x, y1, linewidth=2.5, color='navy')
plt.title('Přístupný graf', fontsize=16)
plt.xlabel('Osa X', fontsize=14)
plt.ylabel('Osa Y', fontsize=14)
plt.xticks(fontsize=12)
plt.yticks(fontsize=12)
plt.grid(True, alpha=0.4)
plt.show()
```
