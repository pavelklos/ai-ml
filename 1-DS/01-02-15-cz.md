## Úvod

Kombinování dat z různých zdrojů je základní operací při analýze dat. Pandas poskytuje několik metod pro kombinování DataFramů: konkatenaci, slučování a připojování. Pochopení těchto metod je klíčové pro efektivní manipulaci s daty.

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
```

## Konkatenace DataFramů

Konkatenace je podobná skládání nebo připojování DataFramů buď vertikálně (podle řádků) nebo horizontálně (podle sloupců). Je užitečná, když máte datasety se stejnou strukturou.

### Vertikální konkatenace

```python
# Vytvoření ukázkových DataFramů
df1 = pd.DataFrame({
    'A': ['A0', 'A1', 'A2', 'A3'],
    'B': ['B0', 'B1', 'B2', 'B3'],
    'C': ['C0', 'C1', 'C2', 'C3']
}, index=[0, 1, 2, 3])

df2 = pd.DataFrame({
    'A': ['A4', 'A5', 'A6', 'A7'],
    'B': ['B4', 'B5', 'B6', 'B7'],
    'C': ['C4', 'C5', 'C6', 'C7']
}, index=[4, 5, 6, 7])

print("DataFrame 1:")
print(df1)
print("\nDataFrame 2:")
print(df2)

# Konkatenace DataFramů vertikálně (podle řádků)
result = pd.concat([df1, df2])
print("\nVýsledek vertikální konkatenace:")
print(result)
```

### Horizontální konkatenace

```python
# Vytvoření ukázkových DataFramů s různými sloupci
df3 = pd.DataFrame({
    'A': ['A0', 'A1', 'A2', 'A3'],
    'B': ['B0', 'B1', 'B2', 'B3']
}, index=[0, 1, 2, 3])

df4 = pd.DataFrame({
    'C': ['C0', 'C1', 'C2', 'C3'],
    'D': ['D0', 'D1', 'D2', 'D3']
}, index=[0, 1, 2, 3])

print("DataFrame 3:")
print(df3)
print("\nDataFrame 4:")
print(df4)

# Konkatenace DataFramů horizontálně (podle sloupců)
result_horizontal = pd.concat([df3, df4], axis=1)
print("\nVýsledek horizontální konkatenace:")
print(result_horizontal)
```

### Zpracování indexů při konkatenaci

```python
# Reset indexu během konkatenace
result_reset_index = pd.concat([df1, df2], ignore_index=True)
print("Konkatenace s resetováním indexu:")
print(result_reset_index)

# Konkatenace s klíči pro hierarchické indexování
result_with_keys = pd.concat([df1, df2], keys=['df1', 'df2'])
print("\nKonkatenace s klíči:")
print(result_with_keys)
```

## Slučování DataFramů

Slučování kombinuje DataFramy na základě společných sloupců nebo indexů, podobně jako SQL joiny. Je užitečné, když potřebujete kombinovat DataFramy, které sdílejí klíčové hodnoty.

```python
# Vytvoření ukázkových DataFramů pro slučování
left = pd.DataFrame({
    'key': ['K0', 'K1', 'K2', 'K3'],
    'A': ['A0', 'A1', 'A2', 'A3'],
    'B': ['B0', 'B1', 'B2', 'B3']
})

right = pd.DataFrame({
    'key': ['K0', 'K1', 'K2', 'K4'],
    'C': ['C0', 'C1', 'C2', 'C4'],
    'D': ['D0', 'D1', 'D2', 'D4']
})

print("Levý DataFrame:")
print(left)
print("\nPravý DataFrame:")
print(right)

# Inner merge (ponechává pouze řádky s odpovídajícími klíči)
inner_merge = pd.merge(left, right, on='key', how='inner')
print("\nVýsledek Inner Merge:")
print(inner_merge)

# Outer merge (ponechá všechny řádky z obou DataFramů)
outer_merge = pd.merge(left, right, on='key', how='outer')
print("\nVýsledek Outer Merge:")
print(outer_merge)

# Left merge (ponechá všechny řádky z levého DataFramu)
left_merge = pd.merge(left, right, on='key', how='left')
print("\nVýsledek Left Merge:")
print(left_merge)

# Right merge (ponechá všechny řádky z pravého DataFramu)
right_merge = pd.merge(left, right, on='key', how='right')
print("\nVýsledek Right Merge:")
print(right_merge)
```

### Vizualizace různých typů spojení

```python
# Vytvoření jednoduchých DataFramů pro vizualizaci spojení
df_A = pd.DataFrame({'id': [1, 2, 3, 4], 'value_A': ['A1', 'A2', 'A3', 'A4']})
df_B = pd.DataFrame({'id': [2, 3, 5, 6], 'value_B': ['B2', 'B3', 'B5', 'B6']})

print("DataFrame A:")
print(df_A)
print("\nDataFrame B:")
print(df_B)

# Vytvoření vizualizace různých typů spojení
plt.figure(figsize=(12, 8))

# Inner join
plt.subplot(2, 2, 1)
inner = pd.merge(df_A, df_B, on='id', how='inner')
sns.heatmap(inner.notna(), cmap='viridis', cbar=False, yticklabels=inner.index)
plt.title('Inner Join (řádků: {})'.format(len(inner)))

# Left join
plt.subplot(2, 2, 2)
left = pd.merge(df_A, df_B, on='id', how='left')
sns.heatmap(left.notna(), cmap='viridis', cbar=False, yticklabels=left.index)
plt.title('Left Join (řádků: {})'.format(len(left)))

# Right join
plt.subplot(2, 2, 3)
right = pd.merge(df_A, df_B, on='id', how='right')
sns.heatmap(right.notna(), cmap='viridis', cbar=False, yticklabels=right.index)
plt.title('Right Join (řádků: {})'.format(len(right)))

# Outer join
plt.subplot(2, 2, 4)
outer = pd.merge(df_A, df_B, on='id', how='outer')
sns.heatmap(outer.notna(), cmap='viridis', cbar=False, yticklabels=outer.index)
plt.title('Outer Join (řádků: {})'.format(len(outer)))

plt.tight_layout()
plt.suptitle('Vizualizace různých typů spojení', fontsize=16, y=1.05)
plt.show()
```

## Připojování DataFramů

Metoda `join()` je pohodlný způsob, jak spojit DataFramy pomocí jejich indexů. Je to zkratka pro merge se specifickými nastaveními.

```python
# Vytvoření ukázkových DataFramů pro připojování
df_left = pd.DataFrame({
    'A': ['A0', 'A1', 'A2'],
    'B': ['B0', 'B1', 'B2']
}, index=['K0', 'K1', 'K2'])

df_right = pd.DataFrame({
    'C': ['C0', 'C1', 'C2', 'C3'],
    'D': ['D0', 'D1', 'D2', 'D3']
}, index=['K0', 'K1', 'K2', 'K3'])

print("Levý DataFrame (s indexem):")
print(df_left)
print("\nPravý DataFrame (s indexem):")
print(df_right)

# Spojení DataFramů pomocí jejich indexů
joined = df_left.join(df_right)
print("\nVýsledek Join (výchozí: left join):")
print(joined)

# Join s metodou outer
outer_joined = df_left.join(df_right, how='outer')
print("\nVýsledek Outer Join:")
print(outer_joined)

# Join na konkrétních sloupcích místo indexu
df_a = pd.DataFrame({'key': ['K0', 'K1', 'K2'], 'A': ['A0', 'A1', 'A2']})
df_b = pd.DataFrame({'key': ['K0', 'K1', 'K3'], 'B': ['B0', 'B1', 'B3']})

print("\nPřipojení na sloupci 'key' místo indexu:")
print(df_a.join(df_b.set_index('key'), on='key'))
```

## Výběr správné metody

Pochopení, kdy použít kterou metodu, je klíčové pro efektivní manipulaci s daty:

```python
# Příklady scénářů s ukázkovými daty
# 1. Připojování podobně strukturovaných dat: Použijte concat
df_jan = pd.DataFrame({
    'date': pd.date_range(start='2023-01-01', periods=5),
    'sales': np.random.randint(100, 1000, 5)
})
df_feb = pd.DataFrame({
    'date': pd.date_range(start='2023-02-01', periods=5),
    'sales': np.random.randint(100, 1000, 5)
})

print("Data za leden:")
print(df_jan)
print("\nData za únor:")
print(df_feb)
print("\nKonkatenovaná měsíční data:")
print(pd.concat([df_jan, df_feb], ignore_index=True))

# 2. Kombinování dat se sdíleným klíčem: Použijte merge
customers = pd.DataFrame({
    'customer_id': [1, 2, 3, 4, 5],
    'name': ['Alice', 'Bob', 'Charlie', 'David', 'Eve']
})
orders = pd.DataFrame({
    'order_id': [101, 102, 103, 104],
    'customer_id': [1, 2, 2, 4],
    'amount': [150, 200, 300, 450]
})

print("\nZákazníci:")
print(customers)
print("\nObjednávky:")
print(orders)
print("\nZákazníci s daty objednávek:")
print(pd.merge(customers, orders, on='customer_id', how='left'))

# 3. Rozšiřování DataFramů horizontálně se sdíleným indexem: Použijte join
metrics_2022 = pd.DataFrame({
    'revenue': [10000, 15000, 12000],
    'expenses': [8000, 9000, 7500]
}, index=['Q1', 'Q2', 'Q3'])
metrics_2023 = pd.DataFrame({
    'revenue': [11000, 16000, 13500],
    'expenses': [8500, 9200, 8000]
}, index=['Q1', 'Q2', 'Q3'])

print("\nPorovnání metrik:")
print(metrics_2022.join(metrics_2023, lsuffix='_2022', rsuffix='_2023'))
```

## Pokročilé operace slučování

Tato část pokrývá složitější operace slučování, jako je slučování na základě více klíčů a používání různých názvů sloupců.

```python
# Slučování na více klíčích
customer_orders = pd.DataFrame({
    'customer_id': [1, 1, 2, 2, 3],
    'order_date': ['2023-01-15', '2023-02-10', '2023-01-20', '2023-03-05', '2023-02-25'],
    'product_id': [101, 102, 101, 103, 102],
    'quantity': [2, 1, 3, 2, 2]
})

products = pd.DataFrame({
    'product_id': [101, 102, 103, 104],
    'product_name': ['Widget', 'Gadget', 'Tool', 'Device'],
    'unit_price': [10.99, 24.99, 15.50, 49.99]
})

print("Objednávky zákazníků:")
print(customer_orders)
print("\nProdukty:")
print(products)

# Výpočet celkové hodnoty objednávky
orders_with_products = pd.merge(customer_orders, products, on='product_id')
orders_with_products['total_value'] = orders_with_products['quantity'] * orders_with_products['unit_price']
print("\nObjednávky s detaily produktů:")
print(orders_with_products)

# Slučování s různými názvy sloupců
employees = pd.DataFrame({
    'emp_id': [1, 2, 3, 4],
    'name': ['Jan', 'Sarah', 'Mike', 'Lisa'],
    'department_id': [10, 20, 10, 30]
})

departments = pd.DataFrame({
    'dept_id': [10, 20, 30, 40],
    'department_name': ['IT', 'HR', 'Finance', 'Marketing']
})

print("\nZaměstnanci:")
print(employees)
print("\nOddělení:")
print(departments)

# Použití left_on a right_on, když se názvy sloupců liší
merged_data = pd.merge(
    employees, departments,
    left_on='department_id', right_on='dept_id',
    how='left'
)
print("\nZaměstnanci s názvy oddělení:")
print(merged_data)
```

## Zpracování duplicitních sloupců

Při slučování DataFramů můžete narazit na duplicitní názvy sloupců. Zde je, jak je zpracovat:

```python
# Vytvoření DataFramů s překrývajícími se názvy sloupců
sales_data = pd.DataFrame({
    'date': pd.date_range(start='2023-01-01', periods=3),
    'store_id': [1, 2, 3],
    'sales': [1200, 1500, 1800],
    'region': ['Sever', 'Jih', 'Východ']
})

store_data = pd.DataFrame({
    'store_id': [1, 2, 3, 4],
    'store_name': ['Alpha', 'Beta', 'Gamma', 'Delta'],
    'region': ['Sever', 'Jih', 'Východ', 'Západ'],  # Duplicitní název sloupce
    'size_sqft': [1500, 2200, 1800, 2500]
})

print("Data prodejů:")
print(sales_data)
print("\nData obchodů:")
print(store_data)

# Sloučení s příponami pro zpracování duplicitních sloupců
merged_sales = pd.merge(sales_data, store_data, on='store_id', suffixes=('_prodej', '_obchod'))
print("\nSloučená data s příponami:")
print(merged_sales)

# Sloučení pouze specifických sloupců, aby se předešlo duplicitám
selected_merge = pd.merge(
    sales_data,
    store_data[['store_id', 'store_name', 'size_sqft']],
    on='store_id'
)
print("\nSloučená data s vybranými sloupci:")
print(selected_merge)

# Přejmenování sloupců před sloučením
store_data_renamed = store_data.rename(columns={'region': 'store_region'})
clean_merge = pd.merge(sales_data, store_data_renamed, on='store_id')
print("\nSloučená data s přejmenovanými sloupci:")
print(clean_merge)
```

## Praktické aplikace

Reálné scénáře, kde je kombinování DataFramů nezbytné:

```python
# Příklad: Analýza chování zákazníků s transakcemi a demografickými údaji
# Vytvoření ukázkových datasetů
customers = pd.DataFrame({
    'customer_id': range(1, 6),
    'name': ['Alice', 'Bob', 'Charlie', 'David', 'Eva'],
    'age': [34, 45, 29, 58, 22],
    'gender': ['Ž', 'M', 'M', 'M', 'Ž'],
    'signup_date': pd.date_range(start='2022-01-01', periods=5, freq='10D')
})

transactions = pd.DataFrame({
    'transaction_id': range(101, 111),
    'customer_id': [1, 2, 1, 3, 2, 5, 4, 1, 5, 3],
    'transaction_date': pd.date_range(start='2023-01-15', periods=10, freq='15D'),
    'amount': [120, 55, 300, 90, 75, 200, 150, 80, 45, 65]
})

products = pd.DataFrame({
    'transaction_id': range(101, 111),
    'product_category': ['Elektronika', 'Knihy', 'Oblečení', 'Domácnost', 'Elektronika', 
                         'Potraviny', 'Knihy', 'Domácnost', 'Oblečení', 'Elektronika'],
    'product_id': np.random.randint(1000, 9999, 10)
})

print("Data zákazníků:")
print(customers)
print("\nData transakcí:")
print(transactions)
print("\nData produktů:")
print(products)

# 1. Kombinace transakcí s informacemi o zákazníkovi
customer_transactions = pd.merge(transactions, customers, on='customer_id')
print("\nTransakce zákazníků:")
print(customer_transactions)

# 2. Přidání informací o produktech k transakcím
full_transactions = pd.merge(
    customer_transactions,
    products,
    on='transaction_id'
)
print("\nÚplné detaily transakcí:")
print(full_transactions[['transaction_id', 'customer_id', 'name', 'amount', 'product_category']])

# 3. Výpočet výdajů podle kategorie pro každého zákazníka
customer_spending = full_transactions.groupby(['customer_id', 'name', 'product_category'])['amount'].sum().reset_index()
print("\nVýdaje zákazníků podle kategorie:")
print(customer_spending)

# 4. Přetvarování dat pro vizualizaci (pivot tabulka)
spending_pivot = customer_spending.pivot_table(
    index='name',
    columns='product_category',
    values='amount',
    aggfunc='sum',
    fill_value=0
)
print("\nPivotní tabulka výdajů:")
print(spending_pivot)

# Vizualizace výdajů zákazníků podle kategorie
plt.figure(figsize=(10, 6))
spending_pivot.plot(kind='bar', stacked=True)
plt.title('Výdaje zákazníků podle kategorie produktů')
plt.xlabel('Zákazník')
plt.ylabel('Celkové výdaje (Kč)')
plt.legend(title='Kategorie produktů', bbox_to_anchor=(1.05, 1), loc='upper left')
plt.tight_layout()
plt.show()
```

## Výkonnostní aspekty

Při práci s velkými DataFramy se stává důležitým výkon:

```python
# Vytvoření větších DataFramů pro demonstraci vlivu na výkon
import time

def measure_performance(func, *args, **kwargs):
    start_time = time.time()
    result = func(*args, **kwargs)
    end_time = time.time()
    print(f"Operace trvala {end_time - start_time:.6f} sekund")
    return result

# Vytvoření větších DataFramů
n_rows = 100000
df_large1 = pd.DataFrame({
    'key': np.random.choice(['K0', 'K1', 'K2', 'K3', 'K4'], n_rows),

})

df_large2 = pd.DataFrame({
    'key': np.random.choice(['K0', 'K1', 'K2', 'K3', 'K4'], n_rows),
    'value_B': np.random.rand(n_rows)
})

print(f"DataFrame 1: {df_large1.shape}")
print(f"DataFrame 2: {df_large2.shape}")

# Benchmark různých metod slučování
print("\nSlučování bez seřazených klíčů:")
```

## Shrnutí

Klíčové body o slučování, spojování a konkatenaci DataFramů:

1. **Konkatenace (pd.concat())** je užitečná pro kombinování DataFramů s podobnými strukturami, ať už vertikálně nebo horizontálně.

2. **Slučování (pd.merge())** kombinuje DataFramy na základě společných sloupců, podobně jako SQL joiny, s možnostmi inner, outer, left a right join.

3. **Připojování (DataFrame.join())** je praktická metoda pro slučování podle indexu, která zjednodušuje některé operace.

4. Při výběru metody:
   - Použijte **concat**, když skládáte podobné DataFramy
   - Použijte **merge**, když kombinujete data na základě společných klíčů
   - Použijte **join**, když kombinujete podle hodnot indexu
