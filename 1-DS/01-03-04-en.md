## Introduction to Recommendation Systems

Recommendation systems are algorithms that suggest relevant items to users based on their preferences, past behavior, or similarity to other users. They're everywhere in our digital lives - from product recommendations on e-commerce sites to content suggestions on streaming platforms and social media.


The goal of recommendation systems can be framed as:
- Finding items that a user might like but hasn't discovered yet
- Predicting a user's rating for items they haven't interacted with
- Providing personalized experiences to increase engagement and satisfaction

## Types of Recommendation Systems

There are several main approaches to building recommendation systems, each with its own strengths and weaknesses.


The main types include:

1. **Collaborative Filtering**: Recommends items based on similar users' preferences
   - User-based: "Users who are similar to you liked..."
   - Item-based: "Users who liked this item also liked..."

2. **Content-Based Filtering**: Recommends items with attributes similar to items the user previously liked

3. **Knowledge-Based**: Makes recommendations based on explicit knowledge about users, items, and their relationships

4. **Hybrid Systems**: Combines multiple approaches to leverage their strengths

## Collaborative Filtering

Collaborative filtering makes recommendations by collecting and analyzing user behavior, activities, or preferences to predict what other users with similar tastes will like.

### User-Based Collaborative Filtering

```python
# Example of User-Based Collaborative Filtering
import pandas as pd
import numpy as np
from sklearn.metrics.pairwise import cosine_similarity

# Sample user-item rating matrix
data = {
    'Alice': {'Movie1': 5, 'Movie2': 3, 'Movie3': 4, 'Movie4': None, 'Movie5': 1},
    'Bob': {'Movie1': 3, 'Movie2': 1, 'Movie3': 2, 'Movie4': 3, 'Movie5': None},
    'Charlie': {'Movie1': 4, 'Movie2': 3, 'Movie3': 4, 'Movie4': 5, 'Movie5': None},
    'Dave': {'Movie1': 1, 'Movie2': 5, 'Movie3': 5, 'Movie4': 2, 'Movie5': None}
}

# Convert to DataFrame
ratings_df = pd.DataFrame(data)
print("User-Item Rating Matrix:")
print(ratings_df)

# Convert to user-user similarity matrix
user_similarity = cosine_similarity(ratings_df.fillna(0).T)
user_similarity_df = pd.DataFrame(user_similarity, index=ratings_df.columns, columns=ratings_df.columns)
print("\nUser-User Similarity Matrix:")
print(user_similarity_df)

# Function to recommend items to a user
def recommend_user_based(user, ratings, similarity_matrix, n_recommendations=2):
    # Get similarity scores for the user
    sim_scores = similarity_matrix[user].drop(user)
    
    # Get user's rated items
    user_ratings = ratings[user].dropna()
    
    # Get items the user hasn't rated
    unrated_items = ratings.index[ratings[user].isna()]
    
    # Calculate weighted average rating for each unrated item
    recommendations = {}
    
    for item in unrated_items:
        weighted_sum = 0
        similarity_sum = 0
        
        for other_user in sim_scores.index:
            if not pd.isna(ratings.loc[item, other_user]):
                weighted_sum += sim_scores[other_user] * ratings.loc[item, other_user]
                similarity_sum += sim_scores[other_user]
        
        if similarity_sum > 0:
            recommendations[item] = weighted_sum / similarity_sum
    
    # Sort and return top n recommendations
    top_recommendations = sorted(recommendations.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
    return top_recommendations

# Get recommendations for Alice
alice_recs = recommend_user_based('Alice', ratings_df, user_similarity_df)
print("\nRecommendations for Alice:")
for movie, score in alice_recs:
    print(f"{movie}: Predicted rating = {score:.2f}")
```

### Item-Based Collaborative Filtering

```python
# Example of Item-Based Collaborative Filtering
# Calculate item-item similarity matrix
item_similarity = cosine_similarity(ratings_df.fillna(0))
item_similarity_df = pd.DataFrame(item_similarity, index=ratings_df.index, columns=ratings_df.index)
print("Item-Item Similarity Matrix:")
print(item_similarity_df)

# Function to recommend items using item-based CF
def recommend_item_based(user, ratings, similarity_matrix, n_recommendations=2):
    # Get user's ratings
    user_ratings = ratings[user].dropna()
    
    # Calculate scores for unrated items
    scores = {}
    
    for item in ratings.index[ratings[user].isna()]:
        weighted_sum = 0
        similarity_sum = 0
        
        for rated_item, rating in user_ratings.items():
            sim = similarity_matrix.loc[item, rated_item]
            weighted_sum += sim * rating
            similarity_sum += abs(sim)
        
        if similarity_sum > 0:
            scores[item] = weighted_sum / similarity_sum
    
    # Return top n recommendations
    top_recommendations = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
    return top_recommendations

# Get item-based recommendations for Alice
alice_item_recs = recommend_item_based('Alice', ratings_df, item_similarity_df)
print("\nItem-Based Recommendations for Alice:")
for movie, score in alice_item_recs:
    print(f"{movie}: Predicted rating = {score:.2f}")
```

## Content-Based Filtering

Content-based filtering recommends items by comparing the content of items rather than user interactions or ratings. It uses item features and user preferences for those features.

```python
# Example of Content-Based Filtering
import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Sample movie data with descriptions
movies_data = {
    'Movie1': 'A sci-fi adventure in space with aliens and advanced technology',
    'Movie2': 'A romantic comedy about two people falling in love in New York',
    'Movie3': 'An action-packed thriller with spies and international intrigue',
    'Movie4': 'A sci-fi movie about time travel and changing the future',
    'Movie5': 'A comedy about friendship and growing up in high school'
}

# User's rated movies
user_ratings = {
    'Movie1': 5,  # User loved this sci-fi movie
    'Movie3': 4,  # User liked this action movie
    'Movie5': 2   # User didn't like this comedy much
}

# Convert movie descriptions to a DataFrame
movies_df = pd.DataFrame.from_dict(movies_data, orient='index', columns=['description'])
print("Movie Descriptions:")
print(movies_df)

# Create TF-IDF matrix of movie descriptions
tfidf = TfidfVectorizer(stop_words='english')
tfidf_matrix = tfidf.fit_transform(movies_df['description'])
print("\nTF-IDF Matrix shape:", tfidf_matrix.shape)

# Calculate cosine similarity between movies
cosine_sim = cosine_similarity(tfidf_matrix, tfidf_matrix)
cosine_sim_df = pd.DataFrame(cosine_sim, index=movies_df.index, columns=movies_df.index)
print("\nMovie Content Similarity Matrix:")
print(cosine_sim_df)

# Function to recommend movies based on content similarity and user preferences
def content_based_recommendations(user_ratings, similarity_matrix, n_recommendations=2):
    # Calculate a weighted score for each movie based on user ratings and content similarity
    scores = {}
    
    # Get movies not rated by the user
    unrated_movies = set(similarity_matrix.index) - set(user_ratings.keys())
    
    for movie in unrated_movies:
        weighted_score = 0
        
        # For each movie rated by the user, add weighted similarity
        for rated_movie, rating in user_ratings.items():
            # Normalize rating to be centered around 0 (assuming ratings 1-5)
            norm_rating = rating - 3
            # Add weighted similarity (positive for liked movies, negative for disliked)
            weighted_score += similarity_matrix.loc[movie, rated_movie] * norm_rating
        
        scores[movie] = weighted_score
    
    # Sort by score and return top n recommendations
    recommendations = sorted(scores.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
    return recommendations

# Get content-based recommendations
cb_recommendations = content_based_recommendations(user_ratings, cosine_sim_df)
print("\nContent-Based Recommendations:")
for movie, score in cb_recommendations:
    print(f"{movie}: Score = {score:.2f}")
    print(f"Description: {movies_data[movie]}")
```

## Hybrid Recommendation Systems

Hybrid systems combine multiple recommendation approaches to overcome the limitations of individual methods.

```python
# Simple example of a hybrid recommendation system combining collaborative and content-based filtering
def hybrid_recommendations(user, ratings_df, content_sim_matrix, item_sim_matrix, 
                          alpha=0.5, n_recommendations=2):
    """
    Generate hybrid recommendations by combining content-based and collaborative filtering.
    
    Parameters:
    - user: User ID
    - ratings_df: User-item ratings DataFrame
    - content_sim_matrix: Content-based similarity matrix
    - item_sim_matrix: Item-based collaborative similarity matrix
    - alpha: Weight for collaborative filtering (1-alpha for content-based)
    - n_recommendations: Number of recommendations to return
    """
    # Get user's ratings
    user_ratings = ratings_df[user].dropna()
    
    # Get unrated items
    unrated_items = ratings_df.index[ratings_df[user].isna()]
    
    # Calculate hybrid scores
    hybrid_scores = {}
    
    for item in unrated_items:
        # Calculate content-based score
        content_score = 0
        content_sim_sum = 0
        
        for rated_item, rating in user_ratings.items():
            norm_rating = rating - 3  # Normalize around 0
            content_sim = content_sim_matrix.loc[item, rated_item]
            content_score += content_sim * norm_rating
            content_sim_sum += abs(content_sim)
        
        if content_sim_sum > 0:
            content_score /= content_sim_sum
        
        # Calculate collaborative filtering score
        collab_score = 0
        collab_sim_sum = 0
        
        for rated_item, rating in user_ratings.items():
            collab_sim = item_sim_matrix.loc[item, rated_item]
            collab_score += collab_sim * rating
            collab_sim_sum += abs(collab_sim)
        
        if collab_sim_sum > 0:
            collab_score /= collab_sim_sum
            
        # Combine scores with weighting
        hybrid_scores[item] = alpha * collab_score + (1 - alpha) * content_score
    
    # Return top n recommendations
    top_recommendations = sorted(hybrid_scores.items(), key=lambda x: x[1], reverse=True)[:n_recommendations]
    return top_recommendations

# hybrid_recs = hybrid_recommendations('Alice', ratings_df, cosine_sim_df, item_similarity_df)
```

Common hybrid approaches include:
- Weighted: Combining scores from multiple recommenders
- Switching: Choosing between methods based on certain criteria
- Feature Combination: Using features from different recommender types
- Cascade: Using one recommender to refine the output of another

## Matrix Factorization Techniques

Matrix factorization approaches decompose the user-item interaction matrix into lower-dimensional matrices that capture latent factors.

```python
# Example of Singular Value Decomposition (SVD) using Surprise library
from surprise import Dataset, Reader, SVD
from surprise.model_selection import cross_validate

ratings_list = [

    ['User1', 'Item3', 4],

    ['User4', 'Item2', 2],

]

df = pd.DataFrame(ratings_list, columns=['userId', 'itemId', 'rating'])
print("Ratings DataFrame:")

reader = Reader(rating_scale=(1, 5))
data = Dataset.load_from_df(df[['userId', 'itemId', 'rating']], reader)
```

## Evaluating Recommendation Systems

Evaluating recommendation systems requires metrics that go beyond standard classification or regression metrics.

```python
# Example of recommendation system evaluation
from surprise import Dataset, Reader
from surprise.model_selection import train_test_split
from surprise import SVD, KNNBasic
from surprise import accuracy
import matplotlib.pyplot as plt

# Create a sample dataset
ratings_list = [
    ['User1', 'Item1', 5],
    ['User1', 'Item2', 3],
    ['User1', 'Item3', 4],
    ['User2', 'Item1', 3],
    ['User2', 'Item3', 1],
    ['User2', 'Item4', 5],
    ['User3', 'Item1', 4],
    ['User3', 'Item2', 4],
    ['User3', 'Item3', 5],
    ['User3', 'Item4', 3],
    ['User4', 'Item2', 2],
    ['User4', 'Item3', 4],
    ['User4', 'Item4', 4],
]

# Load data
reader = Reader(rating_scale=(1, 5))
data = Dataset.load_from_df(pd.DataFrame(ratings_list, columns=['userId', 'itemId', 'rating']), reader)

# Split into training and test sets
trainset, testset = train_test_split(data, test_size=0.25, random_state=42)

# Train models
models = {
    'SVD': SVD(random_state=42),
    'KNN': KNNBasic(sim_options={'name': 'cosine', 'user_based': False})
}

results = {}

for name, model in models.items():
    # Train model
    model.fit(trainset)
    
    # Make predictions
    predictions = model.test(testset)
    
    # Calculate RMSE and MAE
    rmse = accuracy.rmse(predictions)
    mae = accuracy.mae(predictions)
    
    results[name] = {'RMSE': rmse, 'MAE': mae}
    
    print(f"{name} - RMSE: {rmse:.4f}, MAE: {mae:.4f}")

# Visualize results
plt.figure(figsize=(10, 6))
metrics = ['RMSE', 'MAE']
x = np.arange(len(metrics))
width = 0.35

for i, (name, result) in enumerate(results.items()):
    plt.bar(x + i*width - width/2, [result[metric] for metric in metrics], width, label=name)

plt.ylabel('Error')
plt.title('Recommendation System Model Comparison')
plt.xticks(x, metrics)
plt.legend()
plt.tight_layout()
plt.show()
```

Common evaluation metrics include:
- Rating prediction metrics: RMSE, MAE
- Ranking metrics: Precision@k, Recall@k, nDCG (normalized discounted cumulative gain)
- Coverage, diversity, novelty
- A/B testing for business impact

## Building a Simple Recommendation System

Here we'll build a complete recommendation system using the MovieLens dataset.

```python
# Example of building a complete recommendation system using MovieLens dataset
from surprise import Dataset, Reader
from surprise.model_selection import train_test_split
from surprise import SVD, accuracy
import pandas as pd
import matplotlib.pyplot as plt

# Download the MovieLens 100k dataset (or load your local copy)
# This is a smaller version for demonstration purposes
# For the complete dataset, you would download it from: https://grouplens.org/datasets/movielens/

# Load the built-in MovieLens 100k dataset
data = Dataset.load_builtin('ml-100k')

# Split the data
trainset, testset = train_test_split(data, test_size=0.25)

# Define and train the model
model = SVD(n_factors=50, random_state=42)
model.fit(trainset)

# Test the model
predictions = model.test(testset)
rmse = accuracy.rmse(predictions)
print(f"Test RMSE: {rmse:.4f}")

# Load movie data
movies_df = pd.read_csv('ml-100k/u.item', 
                      sep='|', 
                      encoding='latin-1', 
                      header=None, 
                      names=['item_id', 'title', 'release_date', 'video_release_date', 
                             'IMDb_URL', 'unknown', 'Action', 'Adventure', 'Animation', 
                             'Children', 'Comedy', 'Crime', 'Documentary', 'Drama', 'Fantasy', 
                             'Film-Noir', 'Horror', 'Musical', 'Mystery', 'Romance', 'Sci-Fi', 
                             'Thriller', 'War', 'Western'])

# Function to get top N recommendations for a user
def get_top_n_recommendations(model, user_id, n=10):
    # Get a list of all movie IDs
    all_movie_ids = movies_df['item_id'].tolist()
    
    # Get movies the user has already rated
    user_data = data.raw_ratings
    user_rated_movies = [int(rating[1]) for rating in user_data if rating[0] == user_id]
    
    # Movies the user hasn't rated
    unseen_movies = [movie_id for movie_id in all_movie_ids if movie_id not in user_rated_movies]
    
    # Predict ratings for unseen movies
    predictions = [model.predict(str(user_id), str(movie_id)) for movie_id in unseen_movies]
    
    # Sort predictions by estimated rating
    predictions.sort(key=lambda x: x.est, reverse=True)
    
    # Get top N movies
    top_n = predictions[:n]
    top_movies = [(int(pred.iid), pred.est) for pred in top_n]
    
    # Add movie titles
    top_movies_with_titles = []
    for movie_id, est_rating in top_movies:
        title = movies_df[movies_df['item_id'] == movie_id]['title'].values[0]
        top_movies_with_titles.append((movie_id, title, est_rating))
    
    return top_movies_with_titles

# Get recommendations for user 196
user_id = '196'
top_10_recommendations = get_top_n_recommendations(model, user_id, 10)

print(f"\nTop 10 recommendations for user {user_id}:")
for i, (movie_id, title, est_rating) in enumerate(top_10_recommendations):
    print(f"{i+1}. {title} (Predicted rating: {est_rating:.2f})")

# Visualize the distribution of ratings
plt.figure(figsize=(12, 5))

# Plot 1: Distribution of actual ratings
plt.subplot(1, 2, 1)
ratings = [pred.r_ui for pred in predictions]
plt.hist(ratings, bins=9, alpha=0.7)
plt.title('Distribution of Actual Ratings')
plt.xlabel('Rating')
plt.ylabel('Count')

# Plot 2: Distribution of predicted ratings
plt.subplot(1, 2, 2)
pred_ratings = [pred.est for pred in predictions]
plt.hist(pred_ratings, bins=9, alpha=0.7)
plt.title('Distribution of Predicted Ratings')
plt.xlabel('Rating')
plt.ylabel('Count')

plt.tight_layout()
plt.show()
```

- The cold-start problem (recommendations for new users or items)
- The trade-off between exploration and exploitation
- The importance of diversity and serendipity in recommendations
- Ethical considerations like filter bubbles and privacy concerns
- Performance and scalability issues for large-scale deployments

As data availability increases and algorithms improve, recommendation systems continue to evolve, offering increasingly personalized experiences while addressing challenges like privacy, diversity, and explanation.
