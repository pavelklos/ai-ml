## Introduction to Feature Engineering

Feature engineering is the process of transforming raw data into meaningful features that better represent the underlying problem to predictive models, improving model performance. It's often considered the most important step in the machine learning pipeline.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler, OneHotEncoder, MinMaxScaler, LabelEncoder
from sklearn.feature_selection import SelectKBest, f_classif, RFE
from sklearn.decomposition import PCA
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split

# Sample dataset
# Let's create a simple dataset with different types of features
np.random.seed(42)
n_samples = 1000

# Creating a DataFrame with various feature types
df = pd.DataFrame({
    # Numeric features
    'age': np.random.normal(35, 10, n_samples).astype(int),
    'income': np.random.lognormal(10, 1, n_samples),
    'expense_ratio': np.random.uniform(0, 1, n_samples),
    
    # Categorical features
    'education': np.random.choice(['High School', 'Bachelor', 'Master', 'PhD'], n_samples),
    'marital_status': np.random.choice(['Single', 'Married', 'Divorced'], n_samples),
    
    # Temporal features
    'signup_date': pd.date_range(start='2020-01-01', periods=n_samples),
    
    # Text feature
    'feedback': np.random.choice(['Great service, very satisfied!', 
                                 'Good but could improve', 
                                 'Average experience, nothing special',
                                 'Poor service, need improvement',
                                 'Terrible experience, will not recommend'], n_samples)
})

# Creating a target variable
df['churn'] = (df['age'] < 25) | ((df['income'] < np.percentile(df['income'], 25)) & 
                                 (df['education'] == 'High School'))
df['churn'] = df['churn'].astype(int)

print("Dataset shape:", df.shape)
print("\nFirst few rows:")
print(df.head())
print("\nFeature types:")
print(df.dtypes)
print("\nTarget distribution:")
print(df['churn'].value_counts(normalize=True))
```

## Numeric Feature Transformations

Numeric features often require transformations to fit better with machine learning algorithms. Common transformations include:
- Log/Power transformations for skewed data
- Binning for converting continuous features into discrete
- Polynomial features for capturing non-linear relationships

```python
# 1. Log transformation for skewed data
df['income_log'] = np.log1p(df['income'])  # log1p avoids issues with zero values

# 2. Power transformation
df['age_squared'] = df['age'] ** 2

# 3. Binning (discretization)
df['age_group'] = pd.cut(df['age'], bins=[0, 18, 35, 50, 100], 
                         labels=['Teen', 'Young Adult', 'Middle Aged', 'Senior'])

# 4. Creating interaction features
df['income_per_age'] = df['income'] / df['age']

# 5. Polynomial features from sklearn
from sklearn.preprocessing import PolynomialFeatures
poly = PolynomialFeatures(degree=2, include_bias=False)
poly_features = poly.fit_transform(df[['age', 'expense_ratio']])
poly_df = pd.DataFrame(poly_features, 
                      columns=[f"{col}" for col in poly.get_feature_names_out(['age', 'expense_ratio'])])
print("Polynomial features:")
print(poly_df.head())

# Visualizing effect of transformations
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# Original distribution of income
sns.histplot(df['income'], kde=True, ax=axes[0, 0])
axes[0, 0].set_title('Original Income Distribution')

# Distribution after log transformation
sns.histplot(df['income_log'], kde=True, ax=axes[0, 1])
axes[0, 1].set_title('Log-transformed Income Distribution')

# Age distribution
sns.histplot(df['age'], kde=True, ax=axes[1, 0])
axes[1, 0].set_title('Age Distribution')

# Age group counts
sns.countplot(x='age_group', data=df, ax=axes[1, 1])
axes[1, 1].set_title('Age Group Distribution')

plt.tight_layout()
```

## Categorical Feature Encoding

Converting categorical variables into a numerical form that can be provided to machine learning algorithms.

```python
# Examine categorical variables
print("Education categories:", df['education'].unique())
print("Marital status categories:", df['marital_status'].unique())

# 1. One-Hot Encoding
education_encoded = pd.get_dummies(df['education'], prefix='edu')
print("\nOne-hot encoded education:")
print(education_encoded.head())

# 2. Label Encoding
le = LabelEncoder()
df['marital_status_encoded'] = le.fit_transform(df['marital_status'])
print("\nLabel encoded marital status:")
print(pd.DataFrame({'original': df['marital_status'].head(), 
                    'encoded': df['marital_status_encoded'].head()}))

# 3. Ordinal Encoding (manual)
education_order = {'High School': 0, 'Bachelor': 1, 'Master': 2, 'PhD': 3}
df['education_ordinal'] = df['education'].map(education_order)
print("\nOrdinal encoded education:")
print(pd.DataFrame({'original': df['education'].head(), 
                    'encoded': df['education_ordinal'].head()}))

# 4. Target encoding
target_means = df.groupby('education')['churn'].mean()
df['education_target_encoded'] = df['education'].map(target_means)
print("\nTarget encoded education:")
print(pd.DataFrame({'education': df['education'].head(),
                    'target_encoded': df['education_target_encoded'].head()}))

# 5. Count encoding
count_map = df['marital_status'].value_counts().to_dict()
df['marital_status_counts'] = df['marital_status'].map(count_map)
print("\nCount encoded marital status:")
print(pd.DataFrame({'marital_status': df['marital_status'].head(),
                    'count_encoded': df['marital_status_counts'].head()}))

# Visualizing the relationship between categorical features and target
plt.figure(figsize=(12, 6))
plt.subplot(1, 2, 1)
sns.barplot(x='education', y='churn', data=df)
plt.title('Churn Rate by Education')
plt.xticks(rotation=45)

plt.subplot(1, 2, 2)
sns.barplot(x='marital_status', y='churn', data=df)
plt.title('Churn Rate by Marital Status')
plt.tight_layout()
```

## Feature Scaling and Normalization

Normalizing features to a similar scale ensures that no single feature dominates the learning process.

```python
# Select numeric features for scaling
numeric_features = ['age', 'income', 'expense_ratio']
X_numeric = df[numeric_features].copy()

# 1. Standardization (z-score normalization)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_numeric)
X_scaled_df = pd.DataFrame(X_scaled, columns=[f"{col}_scaled" for col in numeric_features])
print("Standardized features:")
print(X_scaled_df.head())

# 2. Min-Max scaling
minmax = MinMaxScaler()
X_minmax = minmax.fit_transform(X_numeric)
X_minmax_df = pd.DataFrame(X_minmax, columns=[f"{col}_minmax" for col in numeric_features])
print("\nMin-Max scaled features:")
print(X_minmax_df.head())

# 3. Robust scaling (using sklearn.preprocessing.RobustScaler)
from sklearn.preprocessing import RobustScaler
robust_scaler = RobustScaler()
X_robust = robust_scaler.fit_transform(X_numeric)
X_robust_df = pd.DataFrame(X_robust, columns=[f"{col}_robust" for col in numeric_features])
print("\nRobustly scaled features:")
print(X_robust_df.head())

# 4. Normalization (scaling rows to unit norm)
from sklearn.preprocessing import Normalizer
normalizer = Normalizer()
X_normalized = normalizer.fit_transform(X_numeric)
X_normalized_df = pd.DataFrame(X_normalized, columns=[f"{col}_normalized" for col in numeric_features])
print("\nNormalized features:")
print(X_normalized_df.head())

# Visualizing the effect of different scaling techniques
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

sns.histplot(X_numeric['income'], kde=True, ax=axes[0, 0])
axes[0, 0].set_title('Original Income Distribution')

sns.histplot(X_scaled_df['income_scaled'], kde=True, ax=axes[0, 1])
axes[0, 1].set_title('Standardized Income')

sns.histplot(X_minmax_df['income_minmax'], kde=True, ax=axes[1, 0])
axes[1, 0].set_title('Min-Max Scaled Income')

sns.histplot(X_normalized_df['income_normalized'], kde=True, ax=axes[1, 1])
axes[1, 1].set_title('Normalized Income')

plt.tight_layout()
```

## Handling Temporal Features

Extracting meaningful features from date and time data.

```python
# Original datetime feature
print("Original datetime feature:")
print(df['signup_date'].head())

# 1. Extract basic components
df['signup_year'] = df['signup_date'].dt.year
df['signup_month'] = df['signup_date'].dt.month
df['signup_day'] = df['signup_date'].dt.day
df['signup_dayofweek'] = df['signup_date'].dt.dayofweek
df['signup_is_weekend'] = df['signup_date'].dt.dayofweek.isin([5, 6]).astype(int)
df['signup_quarter'] = df['signup_date'].dt.quarter

print("\nExtracted datetime components:")
print(df[['signup_date', 'signup_year', 'signup_month', 'signup_day', 
         'signup_dayofweek', 'signup_is_weekend', 'signup_quarter']].head())

# 2. Time-based features
# Days since the earliest date in the dataset
df['days_since_first'] = (df['signup_date'] - df['signup_date'].min()).dt.days

# 3. Cyclical encoding for month and day of week (to preserve cyclical nature)
df['signup_month_sin'] = np.sin(2 * np.pi * df['signup_month'] / 12)
df['signup_month_cos'] = np.cos(2 * np.pi * df['signup_month'] / 12)
df['signup_day_sin'] = np.sin(2 * np.pi * df['signup_dayofweek'] / 7)
df['signup_day_cos'] = np.cos(2 * np.pi * df['signup_dayofweek'] / 7)

print("\nCyclical encoding of temporal features:")
print(df[['signup_month', 'signup_month_sin', 'signup_month_cos']].head())

# 4. Time-based aggregation (example: calculate average churn rate by month)
monthly_churn = df.groupby('signup_month')['churn'].mean().reset_index()
print("\nAverage churn rate by signup month:")
print(monthly_churn)

# Visualizing temporal patterns
plt.figure(figsize=(12, 6))
plt.subplot(1, 2, 1)
sns.lineplot(x='signup_month', y='churn', data=monthly_churn)
plt.title('Churn Rate by Month')

plt.subplot(1, 2, 2)
weekday_churn = df.groupby('signup_dayofweek')['churn'].mean().reset_index()
sns.barplot(x='signup_dayofweek', y='churn', data=weekday_churn)
plt.title('Churn Rate by Day of Week (0=Monday, 6=Sunday)')
plt.tight_layout()
```

## Feature Selection Techniques

Techniques to select the most relevant features to improve model performance and interpretability.

```python
# Prepare data for feature selection
# Combine all numeric features we created
feature_cols = ['age', 'income', 'expense_ratio', 'income_log', 'age_squared',
                'income_per_age', 'education_ordinal', 'marital_status_encoded',
                'signup_month', 'signup_dayofweek', 'days_since_first',
                'feedback_length', 'word_count']

X = df[feature_cols]
y = df['churn']

# 1. Filter Method: Statistical tests (ANOVA F-test)
selector = SelectKBest(score_func=f_classif, k=5)
X_selected = selector.fit_transform(X, y)

# Get selected feature names and their scores
selected_indices = selector.get_support(indices=True)
selected_features = [feature_cols[i] for i in selected_indices]
selected_scores = selector.scores_[selected_indices]

print("Feature scores (ANOVA F-test):")
for feature, score in zip(selected_features, selected_scores):
    print(f"{feature}: {score:.4f}")

# 2. Wrapper Method: Recursive Feature Elimination (RFE)
model = RandomForestClassifier(n_estimators=100, random_state=42)
rfe = RFE(estimator=model, n_features_to_select=5)
X_rfe = rfe.fit_transform(X, y)

# Get selected feature names
rfe_selected_features = [feature for feature, selected in zip(feature_cols, rfe.support_) if selected]
print("\nFeatures selected by RFE:")
print(rfe_selected_features)

# 3. Embedded Method: Feature Importance from Random Forest
model.fit(X, y)
importances = model.feature_importances_
feature_importance = pd.DataFrame({
    'Feature': feature_cols,
    'Importance': importances
}).sort_values('Importance', ascending=False)

print("\nRandom Forest Feature Importance:")
print(feature_importance)

# 4. Dimensionality Reduction: PCA
pca = PCA(n_components=5)
X_pca = pca.fit_transform(StandardScaler().fit_transform(X))

print("\nPCA explained variance ratio:")
print(pca.explained_variance_ratio_)
print(f"Total explained variance: {sum(pca.explained_variance_ratio_):.4f}")

# Visualizing feature selection results
plt.figure(figsize=(12, 10))

# Plot feature importances
plt.subplot(2, 1, 1)
sns.barplot(x='Importance', y='Feature', data=feature_importance)
plt.title('Random Forest Feature Importance')

# Plot PCA components
plt.subplot(2, 1, 2)
plt.bar(range(1, len(pca.explained_variance_ratio_) + 1), pca.explained_variance_ratio_)
plt.xlabel('Principal Component')
plt.ylabel('Explained Variance Ratio')
plt.title('PCA Explained Variance')
plt.tight_layout()
```

## Automated Feature Engineering

Tools and techniques for automating the feature engineering process.

```python
# Automated feature engineering with featuretools
try:
    import featuretools as ft
    
    # Create an entity set for featuretools
    es = ft.EntitySet(id="customer_data")
    
    # Add dataframe as entity
    df_copy = df.copy()
    df_copy.index.name = 'customer_id'
    es = es.add_dataframe(
        dataframe_name="customers",
        dataframe=df_copy.reset_index(),
        index="customer_id",
        time_index="signup_date"
    )
    
    # Run deep feature synthesis to automatically create features
    feature_matrix, feature_defs = ft.dfs(
        entityset=es,
        target_dataframe_name="customers",
        trans_primitives=["add_numeric", "multiply_numeric"],
        verbose=True
    )
    
    print("\nAutomatically generated features (first 5):")
    for i, feat in enumerate(feature_defs[:5]):
        print(f"{i+1}. {feat}")
        
    print(f"\nTotal features generated: {len(feature_defs)}")
    print("\nFeature matrix sample:")
    print(feature_matrix.head(3))
    
except ImportError:
    print("\nfeaturetools is not installed. To use automated feature engineering, run: pip install featuretools")

# Alternative: Feature engineering with scikit-learn PolynomialFeatures
numeric_subset = X[['age', 'income', 'expense_ratio']]
```

## Feature Engineering Pipelines

Building robust and reusable pipelines for feature engineering to maintain consistency between training and prediction.

```python
# Creating a comprehensive feature engineering pipeline
numeric_features = ['age', 'income', 'expense_ratio']
categorical_features = ['education', 'marital_status']

# Numeric pipeline
numeric_transformer = Pipeline(steps=[
    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())
])

# Categorical pipeline
categorical_transformer = Pipeline(steps=[
    ('imputer', SimpleImputer(strategy='most_frequent')),
    ('onehot', OneHotEncoder(handle_unknown='ignore'))
])

# Combine transformations in a preprocessor
preprocessor = ColumnTransformer(
    transformers=[
        ('num', numeric_transformer, numeric_features),
        ('cat', categorical_transformer, categorical_features)
    ])

# Create full pipeline with model
full_pipeline = Pipeline(steps=[
    ('preprocessor', preprocessor),
    ('classifier', RandomForestClassifier(n_estimators=100, random_state=42))
])

# Split dataset
X_train, X_test, y_train, y_test = train_test_split(
    df[numeric_features + categorical_features], 
    df['churn'],
    test_size=0.2, 
    random_state=42
)

# Train the pipeline
full_pipeline.fit(X_train, y_train)

# Evaluate
from sklearn.metrics import classification_report
y_pred = full_pipeline.predict(X_test)
print("Classification report:")
print(classification_report(y_test, y_pred))

# Using the pipeline for new data
print("\nExample of using pipeline for prediction on new data:")
new_data = pd.DataFrame({
    'age': [25, 55],
    'income': [50000, 90000],
    'expense_ratio': [0.3, 0.2],
    'education': ['Bachelor', 'PhD'],
    'marital_status': ['Single', 'Married']
})
print("New data:")
print(new_data)

print("\nPredictions:")
print(full_pipeline.predict(new_data))
print("Prediction probabilities:")
print(full_pipeline.predict_proba(new_data))

# Extracting transformed features from pipeline
preprocessed_X_train = preprocessor.fit_transform(X_train)
feature_names = (
    numeric_features + 
    list(preprocessor.transformers_[1][1].named_steps['onehot']
        .get_feature_names_out(categorical_features))
)

preprocessed_df = pd.DataFrame(preprocessed_X_train, columns=feature_names)
print("\nPreprocessed features (first 5 rows):")
print(preprocessed_df.head())
```

In summary, effective feature engineering combines domain knowledge with technical skills to transform raw data into predictive features. Key techniques include:

1. **Numeric transformations** (log, polynomial, binning) to handle distributions and non-linear relationships
2. **Categorical encoding** to convert categories into model-compatible formats
3. **Feature scaling** to normalize different ranges
4. **Temporal feature extraction** to capture time-based patterns
5. **Text feature engineering** to quantify textual information
6. **Feature selection** to focus on the most predictive attributes
7. **Automated feature engineering** to discover complex relationships
8. **Feature engineering pipelines** for reproducible workflows

The quality of features often has a greater impact on model performance than the choice of algorithm, making feature engineering a critical skill for data scientists.
