## Introduction to SQL and pandas Integration

Integrating SQL with pandas allows data scientists to leverage the strengths of both tools: the robust database capabilities of SQL and the flexible data analysis features of pandas.

## Reading SQL Data into DataFrames

One of the most common operations is reading SQL data into pandas DataFrames for analysis.

```python
# First, let's create a sample table in SQLite for demonstration
sqlite_conn.execute('''
CREATE TABLE IF NOT EXISTS customers (
    id INTEGER PRIMARY KEY,
    name TEXT NOT NULL,
    age INTEGER,
    email TEXT,
    signup_date DATE,
    purchases INTEGER
)
''')

# Insert sample data
sample_data = [
    (1, 'Alice Smith', 28, 'alice@example.com', '2022-03-15', 12),
    (2, 'Bob Johnson', 35, 'bob@example.com', '2022-01-10', 5),
    (3, 'Charlie Brown', 42, 'charlie@example.com', '2021-11-20', 20),
    (4, 'Diana Miller', 31, 'diana@example.com', '2022-04-05', 8),
    (5, 'Edward Davis', 25, 'edward@example.com', '2022-02-28', 15)
]

sqlite_conn.executemany(
    "INSERT INTO customers VALUES (?, ?, ?, ?, ?, ?)",
    sample_data
)
sqlite_conn.commit()

# Method 1: Read SQL query results into a DataFrame
query = "SELECT * FROM customers WHERE age > 30"
df_query = pd.read_sql(query, sqlite_conn)
print("Customers over 30:")
print(df_query)

# Method 2: Read entire table into a DataFrame
df_table = pd.read_sql_table("customers", sqlite_engine)
print("\nAll customers (using read_sql_table):")
print(df_table)

# Method 3: Use the read_sql_query function for more explicit queries
df_detailed = pd.read_sql_query(
    "SELECT name, age, purchases FROM customers ORDER BY purchases DESC",
    sqlite_conn
)
print("\nCustomers sorted by purchases:")
print(df_detailed)

# Method 4: Parameterized queries for safety and flexibility
min_age = 25
max_age = 40
df_params = pd.read_sql_query(
    "SELECT * FROM customers WHERE age BETWEEN ? AND ?",
    sqlite_conn,
    params=(min_age, max_age)
)
print(f"\nCustomers between {min_age} and {max_age} years old:")
print(df_params)
```

## Writing DataFrames to SQL Databases

pandas makes it easy to write DataFrames to SQL databases, which is useful for storing processed data or analysis results.

```python
# Create a new DataFrame to write to the database
new_customers = pd.DataFrame({
    'id': [6, 7, 8],
    'name': ['Frank Wilson', 'Grace Lee', 'Henry Martin'],
    'age': [29, 34, 45],
    'email': ['frank@example.com', 'grace@example.com', 'henry@example.com'],
    'signup_date': ['2022-05-10', '2022-05-12', '2022-05-15'],
    'purchases': [3, 7, 2]
})

# Method 1: Write DataFrame to a new table
new_customers.to_sql('new_customers', sqlite_conn, index=False, if_exists='replace')

# Method 2: Append data to an existing table
new_customers.to_sql('customers', sqlite_conn, index=False, if_exists='append')

# Verify the data was written by reading it back
df_all = pd.read_sql("SELECT * FROM customers", sqlite_conn)
print("All customers after append:")
print(df_all)

# Method 3: Replace an existing table
modified_df = pd.DataFrame({
    'id': [1, 2, 3],
    'name': ['Alice Updated', 'Bob Updated', 'Charlie Updated'],
    'age': [29, 36, 43],
    'email': ['alice.new@example.com', 'bob.new@example.com', 'charlie.new@example.com'],
    'signup_date': ['2022-03-15', '2022-01-10', '2021-11-20'],
    'purchases': [15, 8, 25]
})

modified_df.to_sql('customers_v2', sqlite_conn, index=False, if_exists='replace')

# Other common options for if_exists:
# - 'fail': Raise a ValueError if the table exists (default)
# - 'replace': Drop the table before inserting new values
# - 'append': Insert new values to the existing table
```

## Using SQL Queries with pandas

You can use SQL queries directly with pandas to leverage SQL's powerful querying capabilities before bringing data into pandas.

```python
# Create a more complex database schema for this example
sqlite_conn.execute('''
CREATE TABLE IF NOT EXISTS orders (
    order_id INTEGER PRIMARY KEY,
    customer_id INTEGER,
    order_date DATE,
    total_amount REAL,
    FOREIGN KEY (customer_id) REFERENCES customers (id)
)
''')

# Sample order data
orders_data = [
    (101, 1, '2022-04-10', 120.50),
    (102, 2, '2022-03-22', 85.25),
    (103, 1, '2022-04-18', 220.00),
    (104, 3, '2022-02-15', 65.75),
    (105, 5, '2022-05-01', 160.30),
    (106, 3, '2022-03-10', 40.20),
    (107, 4, '2022-04-30', 95.60),
    (108, 1, '2022-05-05', 185.40)
]

sqlite_conn.executemany(
    "INSERT INTO orders VALUES (?, ?, ?, ?)",
    orders_data
)
sqlite_conn.commit()

# Join query between tables
join_query = """
SELECT 
    c.name, 
    c.email, 
    COUNT(o.order_id) as order_count, 
    SUM(o.total_amount) as total_spent,
    AVG(o.total_amount) as avg_order_value
FROM 
    customers c
JOIN 
    orders o ON c.id = o.customer_id
GROUP BY 
    c.id
ORDER BY 
    total_spent DESC
"""

customer_analysis = pd.read_sql_query(join_query, sqlite_conn)
print("Customer Order Analysis:")
print(customer_analysis)

# Complex SQL query with aggregations, filtering, and ordering
advanced_query = """
WITH monthly_stats AS (
    SELECT 
        customer_id,
        strftime('%Y-%m', order_date) as month,
        SUM(total_amount) as monthly_spent
    FROM 
        orders
    GROUP BY 
        customer_id, month
)
SELECT 
    c.name,
    COUNT(DISTINCT ms.month) as active_months,
    SUM(ms.monthly_spent) as total_spent,
    AVG(ms.monthly_spent) as avg_monthly_spent
FROM 
    customers c
JOIN 
    monthly_stats ms ON c.id = ms.customer_id
GROUP BY 
    c.id
HAVING 
    active_months > 1
ORDER BY 
    avg_monthly_spent DESC
"""

monthly_analysis = pd.read_sql_query(advanced_query, sqlite_conn)
print("\nMonthly Spending Analysis:")
print(monthly_analysis)
```

    print(f"\nChunk {i+1}:")

    "SELECT * FROM customers WHERE purchases > 10",

## SQLAlchemy Integration

SQLAlchemy provides a more robust and flexible way to work with databases in pandas.

```python
# Create a table through SQLAlchemy
from sqlalchemy import Column, Integer, String, Float, Date, MetaData, Table, ForeignKey

metadata = MetaData()

# Define a products table
products = Table(
    'products', 
    metadata,
    Column('product_id', Integer, primary_key=True),
    Column('name', String),
    Column('category', String),
    Column('price', Float),
    Column('stock', Integer)
)

# Create the table in the database
metadata.create_all(sqlite_engine)

# Create a DataFrame with product data
products_df = pd.DataFrame({
    'product_id': [1, 2, 3, 4, 5],
    'name': ['Laptop', 'Smartphone', 'Tablet', 'Headphones', 'Monitor'],
    'category': ['Electronics', 'Electronics', 'Electronics', 'Accessories', 'Electronics'],
    'price': [1200.00, 800.00, 350.00, 150.00, 300.00],
    'stock': [15, 25, 30, 40, 20]
})

# Write the DataFrame to the products table
products_df.to_sql('products', sqlite_engine, if_exists='append', index=False)

# Read the products table using SQLAlchemy
result_df = pd.read_sql_table('products', sqlite_engine)
print("Products table:")
print(result_df)

# Execute a more complex query using SQLAlchemy
from sqlalchemy import select, func

query = select([
    products.c.category,
    func.count(products.c.product_id).label('product_count'),
    func.avg(products.c.price).label('avg_price'),
    func.sum(products.c.stock).label('total_stock')
]).group_by(products.c.category)

result = pd.read_sql(query, sqlite_engine)
print("\nProduct category analysis:")
print(result)
```

## Database Schema Operations

pandas can also help with database schema inspection and management.

```python
# List all tables in the database
def list_tables(engine):
    inspector = sqlalchemy.inspect(engine)
    return inspector.get_table_names()

tables = list_tables(sqlite_engine)
print(f"Tables in the database: {tables}")

# Get schema information for a table
def get_schema(engine, table_name):
    inspector = sqlalchemy.inspect(engine)
    columns = inspector.get_columns(table_name)
    return pd.DataFrame([{
        'name': col['name'],
        'type': str(col['type']),
        'nullable': col['nullable'],
        'primary_key': col.get('primary_key', False)
    } for col in columns])

customers_schema = get_schema(sqlite_engine, 'customers')
print("\nCustomers table schema:")
print(customers_schema)

# Get foreign key information
def get_foreign_keys(engine, table_name):
    inspector = sqlalchemy.inspect(engine)
    foreign_keys = inspector.get_foreign_keys(table_name)
    return pd.DataFrame(foreign_keys) if foreign_keys else "No foreign keys"

orders_fk = get_foreign_keys(sqlite_engine, 'orders')
print("\nOrders table foreign keys:")
print(orders_fk)
```

## Best Practices and Performance Tips

Here are some best practices and performance tips for integrating SQL with pandas:

```python
# 1. Use appropriate dtypes when creating tables
from sqlalchemy import create_engine, Column, Integer, String, Float, Date
from sqlalchemy.ext.declarative import declarative_base

Base = declarative_base()

class OptimizedCustomer(Base):
    __tablename__ = 'optimized_customers'
    
    id = Column(Integer, primary_key=True)
    name = Column(String(100), nullable=False)  # Specify string length
    age = Column(Integer)
    email = Column(String(100), index=True)     # Add index for faster lookups
    signup_date = Column(Date)
    purchases = Column(Integer, default=0)

# Create optimized table
Base.metadata.create_all(sqlite_engine)

# 2. Use parameter binding for security and performance
def get_customers_by_age_range(min_age, max_age, connection):
    query = "SELECT * FROM customers WHERE age BETWEEN ? AND ?"
    return pd.read_sql_query(query, connection, params=(min_age, max_age))

filtered_customers = get_customers_by_age_range(30, 40, sqlite_conn)
print("\nCustomers between 30 and 40 (using parameter binding):")
print(filtered_customers)

# 3. Proper connection management
def safe_db_operation():
    # Create a new connection
    conn = sqlite3.connect('sample_database.db')
    try:
        # Perform database operations
        df = pd.read_sql("SELECT * FROM customers LIMIT 3", conn)
        print("\nSafely retrieved data:")
        print(df)
    finally:
        # Ensure connection is closed even if an error occurs
        conn.close()

safe_db_operation()

# 4. Use bulk operations for better performance
def bulk_insert_example(engine, data, table_name):
    # Convert list of tuples to DataFrame
    df = pd.DataFrame(data, columns=['id', 'name', 'age', 'email', 'signup_date', 'purchases'])
    
    # Use to_sql with method='multi' for bulk insert
    df.to_sql(table_name, engine, if_exists='append', index=False, method='multi')
    
    print(f"Bulk inserted {len(df)} rows into {table_name}")

new_data = [
    (9, 'Ian Johnson', 38, 'ian@example.com', '2022-05-20', 4),
    (10, 'Julia Smith', 27, 'julia@example.com', '2022-05-21', 9),
    (11, 'Kevin Lee', 33, 'kevin@example.com', '2022-05-22', 6)
]

bulk_insert_example(sqlite_engine, new_data, 'customers')

# 5. Use appropriate index when querying
# Create index on frequently queried column
sqlite_conn.execute("CREATE INDEX IF NOT EXISTS idx_customer_age ON customers(age)")
sqlite_conn.commit()

# Query now uses the index
indexed_query = pd.read_sql_query(
    "SELECT * FROM customers WHERE age > 30",
    sqlite_conn
)
print("\nQuery using index on age column:")
print(indexed_query)
```
