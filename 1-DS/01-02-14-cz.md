## Úvod do zpracování textových dat

Textová data jsou nestrukturovaná a vyžadují speciální techniky pro zpracování a analýzu. V datové vědě se textová data vyskytují všude - od uživatelských recenzí a příspěvků na sociálních sítích až po zpravodajské články a vědecké práce.

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
import re
import string
from collections import Counter

# Ukázková textová data
text_data = [
    "Data Science je zábava! Rád pracuji s Pythonem.",
    "Zpracování přirozeného jazyka (NLP) je podobor umělé inteligence.",
    "Regulární výrazy jsou mocným nástrojem pro manipulaci s textem.",
    "Textová data vyžadují předzpracování před analýzou.",
    "Python má skvělé knihovny pro NLP: NLTK, spaCy a další!"
]

# Vytvoření DataFrame s textovými daty
df = pd.DataFrame(text_data, columns=['text'])
print(df.head())

# Základní statistiky textu
df['char_count'] = df['text'].str.len()
df['word_count'] = df['text'].str.split().str.len()
df['avg_word_length'] = df['text'].apply(lambda x: sum(len(word) for word in x.split()) / len(x.split()))

print("\nStatistiky textu:")
print(df[['char_count', 'word_count', 'avg_word_length']].describe())
```

## Základy regulárních výrazů (Regex)

Regulární výrazy poskytují mocný způsob vyhledávání, extrakce a manipulace s textem na základě vzorů. Jsou zásadní pro čištění textu a extrakci příznaků.

```python
import re

# Základní vzory regulárních výrazů
sample_text = "Kontaktujte nás na podpora@priklad.cz nebo volejte 555-123-4567. Navštivte naše webové stránky: https://www.priklad.cz"

# Hledání všech e-mailových adres
email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
emails = re.findall(email_pattern, sample_text)
print(f"Nalezené e-maily: {emails}")

# Hledání telefonních čísel
phone_pattern = r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b'
phones = re.findall(phone_pattern, sample_text)
print(f"Nalezená telefonní čísla: {phones}")

# Hledání URL
url_pattern = r'https?://(?:[-\w.]|(?:%[\da-fA-F]{2}))+'
urls = re.findall(url_pattern, sample_text)
print(f"Nalezené URL: {urls}")

# Nahrazení textu
censored_text = re.sub(r'\d{3}-\d{3}-\d{4}', '[TELEFONNÍ ČÍSLO SKRYTO]', sample_text)
print(f"Cenzurovaný text: {censored_text}")

# Rozdělení textu
split_text = re.split(r'[.!?]', sample_text)
print(f"Rozdělené věty: {split_text}")

# Běžné vzory regulárních výrazů pro čištění dat
def clean_text(text):
    # Odstranění speciálních znaků a čísel
    text = re.sub(r'[^a-zA-Z\s]', '', text)
    # Odstranění nadbytečných mezer
    text = re.sub(r'\s+', ' ', text).strip()
    return text

df['cleaned_text'] = df['text'].apply(clean_text)
print("\nPříklady vyčištěného textu:")
print(df[['text', 'cleaned_text']].head())
```

## Techniky předzpracování textu

Před analýzou textových dat je důležité je předzpracovat odstraněním šumu a standardizací formátu. Běžné kroky předzpracování zahrnují převod na malá písmena, odstranění interpunkce, stop slov a stemming/lemmatizaci.

```python
# Instalace potřebných balíčků
# pip install nltk spacy
# python -m spacy download cs_core_news_sm

import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer, WordNetLemmatizer
import spacy

# Stažení potřebných NLTK zdrojů
nltk.download('punkt')
nltk.download('stopwords')
nltk.download('wordnet')

# Ukázkový text
text = "Datoví vědci potřebují předzpracovat textová data před analýzou. Zpracování zahrnuje odstranění stop slov a stemming."

# 1. Převod na malá písmena
text_lower = text.lower()
print(f"Malá písmena: {text_lower}")

# 2. Tokenizace
tokens = word_tokenize(text_lower)
print(f"Tokeny: {tokens}")

# 3. Odstranění interpunkce
tokens_no_punct = [token for token in tokens if token not in string.punctuation]
print(f"Tokeny bez interpunkce: {tokens_no_punct}")

# 4. Odstranění stop slov (použijeme anglická stop slova jako příklad, pro češtinu by bylo nutné použít český seznam)
stop_words = set(stopwords.words('english'))
tokens_no_stop = [token for token in tokens_no_punct if token not in stop_words]
print(f"Tokeny bez stop slov: {tokens_no_stop}")

# 5. Stemming (redukce slov na jejich kořen)
stemmer = PorterStemmer()
stemmed_tokens = [stemmer.stem(token) for token in tokens_no_stop]
print(f"Stemmované tokeny: {stemmed_tokens}")

# 6. Lemmatizace (redukce slov na jejich základní tvar, ale zajištění, že je to platné slovo)
lemmatizer = WordNetLemmatizer()
lemmatized_tokens = [lemmatizer.lemmatize(token) for token in tokens_no_stop]
print(f"Lemmatizované tokeny: {lemmatized_tokens}")

# Použití spaCy pro předzpracování (pro angličtinu)
nlp = spacy.load("en_core_web_sm")  # Pro češtinu použijte "cs_core_news_sm"
doc = nlp(text)

# Extrakce lemmat, odstranění stop slov a interpunkce
spacy_processed = [token.lemma_ for token in doc if not token.is_stop and not token.is_punct]
print(f"Text zpracovaný pomocí spaCy: {spacy_processed}")

# Funkce pro kompletní předzpracování textu
def preprocess_text(text):
    # Převod na malá písmena a tokenizace
    tokens = word_tokenize(text.lower())
    # Odstranění interpunkce
    tokens = [token for token in tokens if token not in string.punctuation]
    # Odstranění stop slov
    tokens = [token for token in tokens if token not in stop_words]
    # Lemmatizace
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    return tokens

# Aplikace předzpracování na DataFrame
df['preprocessed_tokens'] = df['text'].apply(preprocess_text)
print("\nPříklady předzpracovaného textu:")
for i, tokens in enumerate(df['preprocessed_tokens'].head()):
    print(f"{i+1}: {tokens}")
```

## Metody vektorizace textu

Převod textu na numerické vektory je nezbytný pro algoritmy strojového učení. Běžné metody vektorizace zahrnují Bag of Words, TF-IDF a word embeddings.

```python
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from gensim.models import Word2Vec
import numpy as np
import pandas as pd

# Ukázkový korpus
corpus = [
    "Obloha je modrá a krásná.",
    "Miluji tuto modrou a jasnou oblohu!",
    "Rychlá hnědá liška skáče přes líného psa.",
    "Králova snídaně obsahuje klobásy, šunku, slaninu, vejce, toast a fazole",
    "Miluji zelená vejce, šunku, klobásy a slaninu!",
    "Hnědá liška je rychlá a modrý pes je líný!",
    "Obloha je velmi modrá a obloha je dnes velmi krásná",
    "Pes je líný, ale hnědá liška je rychlá!"
]

# 1. Bag of Words (BoW)
print("Vektorizace Bag of Words:")
count_vectorizer = CountVectorizer()
bow_matrix = count_vectorizer.fit_transform(corpus)

bow_df = pd.DataFrame(
    bow_matrix.toarray(),
    columns=count_vectorizer.get_feature_names_out()
)
print(bow_df.shape)
print(bow_df.iloc[:3, :10])  # Vytiskne první 3 řádky, prvních 10 sloupců

# 2. TF-IDF Vektorizace
print("\nTF-IDF Vektorizace:")
tfidf_vectorizer = TfidfVectorizer()
tfidf_matrix = tfidf_vectorizer.fit_transform(corpus)

tfidf_df = pd.DataFrame(
    tfidf_matrix.toarray(),
    columns=tfidf_vectorizer.get_feature_names_out()
)
print(tfidf_df.shape)
print(tfidf_df.iloc[:3, :10])  # Vytiskne první 3 řádky, prvních 10 sloupců

# 3. Word Embeddings s Word2Vec
# Tokenizace korpusu
tokenized_corpus = [text.lower().split() for text in corpus]

# Trénování Word2Vec modelu
w2v_model = Word2Vec(
    tokenized_corpus,
    vector_size=100,   # Dimenze embeddingu
    window=5,          # Velikost kontextového okna
    min_count=1,       # Minimální frekvence slova
    workers=4          # Počet vláken
)

# Získání vektoru pro konkrétní slovo
print("\nWord2Vec Embeddings:")
word = "modrá"
if word in w2v_model.wv:
    print(f"Vektor pro '{word}': {w2v_model.wv[word][:5]}...")  # Ukázat prvních 5 dimenzí
    
    # Nalezení nejpodobnějších slov
    similar_words = w2v_model.wv.most_similar(word, topn=3)
    print(f"Slova nejpodobnější '{word}':")
    for similar_word, similarity in similar_words:
        print(f"- {similar_word}: {similarity:.4f}")

# 4. Document Embeddings (jednoduchý průměr word vektorů)
def document_vector(doc, model):
    words = [word for word in doc.lower().split() if word in model.wv]
    if len(words) == 0:
        return np.zeros(model.vector_size)
    return np.mean([model.wv[word] for word in words], axis=0)

# Generování dokumentových vektorů
doc_vectors = [document_vector(doc, w2v_model) for doc in corpus]
doc_vectors_df = pd.DataFrame(doc_vectors, columns=[f"dim_{i}" for i in range(5)])  # Ukázat prvních 5 dimenzí

print("\nDocument Embeddings (prvních 5 dimenzí):")
print(doc_vectors_df)
```

## Základní analýza sentimentu

Analýza sentimentu určuje emocionální tón textových dat, což je užitečné pro porozumění zpětné vazbě zákazníků, analýze sociálních médií a dalším aplikacím.

```python
from textblob import TextBlob
from nltk.sentiment import SentimentIntensityAnalyzer
import pandas as pd
import matplotlib.pyplot as plt

# Stažení NLTK zdrojů pro analýzu sentimentu
nltk.download('vader_lexicon')

# Ukázkové recenze
reviews = [
    "Tento produkt je úžasný! Naprosto ho miluji.",
    "Služba byla hrozná a personál byl nezdvořilý.",
    "Film byl průměrný, ne skvělý, ale ani ne hrozný.",
    "Vysoce doporučuji tuto knihu, je poučná a poutavá.",
    "Hotelový pokoj byl špinavý a postel byla nepohodlná.",
    "Jídlo bylo vynikající, ale trochu drahé.",
    "Nejsem si jistý, jestli se mi to líbí nebo ne, je to složité."
]

# Vytvoření DataFrame
reviews_df = pd.DataFrame(reviews, columns=['review'])

# 1. TextBlob Analýza sentimentu
reviews_df['textblob_polarity'] = reviews_df['review'].apply(lambda x: TextBlob(x).sentiment.polarity)
reviews_df['textblob_subjectivity'] = reviews_df['review'].apply(lambda x: TextBlob(x).sentiment.subjectivity)

# 2. VADER Analýza sentimentu
sia = SentimentIntensityAnalyzer()
reviews_df['vader_compound'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['compound'])
reviews_df['vader_pos'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['pos'])
reviews_df['vader_neg'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['neg'])
reviews_df['vader_neu'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['neu'])

# 3. Klasifikace sentimentu
def classify_sentiment(score):
    if score > 0.05:
        return 'Pozitivní'
    elif score < -0.05:
        return 'Negativní'
    else:
        return 'Neutrální'

reviews_df['textblob_sentiment'] = reviews_df['textblob_polarity'].apply(classify_sentiment)
reviews_df['vader_sentiment'] = reviews_df['vader_compound'].apply(classify_sentiment)

# Zobrazení výsledků
print("Výsledky analýzy sentimentu:")
print(reviews_df[['review', 'textblob_sentiment', 'vader_sentiment', 'textblob_polarity', 'vader_compound']])

# Vizualizace distribuce sentimentu
plt.figure(figsize=(10, 6))

plt.subplot(1, 2, 1)
reviews_df['textblob_sentiment'].value_counts().plot(kind='bar', color='skyblue')
plt.title('Distribuce sentimentu TextBlob')
plt.ylabel('Počet')

plt.subplot(1, 2, 2)
reviews_df['vader_sentiment'].value_counts().plot(kind='bar', color='lightgreen')
plt.title('Distribuce sentimentu VADER')
plt.ylabel('Počet')

plt.tight_layout()
plt.show()

# Bodový graf srovnávající TextBlob a VADER
plt.figure(figsize=(8, 6))
plt.scatter(reviews_df['textblob_polarity'], reviews_df['vader_compound'], alpha=0.7)
plt.xlabel('TextBlob Polarita')
plt.ylabel('VADER Compound Skóre')
plt.title('TextBlob vs. VADER Skóre sentimentu')
plt.grid(True, linestyle='--', alpha=0.7)

# Přidání oblastí sentimentu
plt.axhline(y=0.05, color='g', linestyle='-', alpha=0.3)
plt.axhline(y=-0.05, color='r', linestyle='-', alpha=0.3)
plt.axvline(x=0.05, color='g', linestyle='-', alpha=0.3)
plt.axvline(x=-0.05, color='r', linestyle='-', alpha=0.3)

# Přidání textových popisků pro každý bod
for i, review in enumerate(reviews):
    plt.annotate(f"Recenze {i+1}", 
                 (reviews_df['textblob_polarity'][i], reviews_df['vader_compound'][i]),
                 xytext=(5, 5), textcoords='offset points')

plt.tight_layout()
plt.show()

# Vytvoření jednoduchého klasifikátoru sentimentu pomocí scikit-learn
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report

# Použití VADER sentimentu jako "ground truth" pro tento příklad
X = reviews_df['review']
y = reviews_df['vader_sentiment']

# Vytvoření rozdělení na trénovací a testovací data
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

# Vektorizace textu
vectorizer = TfidfVectorizer(max_features=1000)
X_train_vec = vectorizer.fit_transform(X_train)
X_test_vec = vectorizer.transform(X_test)

# Trénování jednoduchého klasifikátoru
classifier = LogisticRegression()
classifier.fit(X_train_vec, y_train)

# Predikce
y_pred = classifier.predict(X_test_vec)

# Vyhodnocení
print("\nZpráva o klasifikaci sentimentu:")
print(classification_report(y_test, y_pred))
```

## Rozpoznávání pojmenovaných entit (NER)

NER identifikuje a klasifikuje pojmenované entity v textu jako osoby, místa, organizace a další.

```python
import spacy
import pandas as pd
import matplotlib.pyplot as plt
from collections import Counter

# Načtení spaCy modelu
nlp = spacy.load("en_core_web_sm")  # Pro češtinu použijte "cs_core_news_sm"

# Ukázkové texty s pojmenovanými entitami
texts = [
    "Apple Inc. plánuje otevřít novou pobočku v New Yorku příští měsíc.",
    "Amazon a Microsoft soutěží o zakázku Pentagonu na cloud za 10 miliard dolarů.",
    "Golden State Warriors porazili Toronto Raptors ve finále NBA.",
    "Prezident Joe Biden promluvil na Valném shromáždění OSN v září 2021.",
    "Elon Musk, CEO Tesly, oznámil plány na výstavbu gigafactory v Berlíně, Německu."
]

# Extrakce pojmenovaných entit
def extract_entities(text):
    doc = nlp(text)
    entities = [(ent.text, ent.label_) for ent in doc.ents]
    return entities

# Zpracování všech textů
all_entities = []
for text in texts:
    entities = extract_entities(text)
    all_entities.extend(entities)
    print(f"\nText: {text}")
    print("Entity:")
    for entity, label in entities:
        print(f"- {entity}: {label}")

# Počet typů entit
entity_types = [entity[1] for entity in all_entities]
entity_counts = Counter(entity_types)

# Vizualizace distribuce entit
plt.figure(figsize=(12, 6))
plt.bar(entity_counts.keys(), entity_counts.values(), color='skyblue')
plt.title('Distribuce typů pojmenovaných entit')
plt.xlabel('Typ entity')
plt.ylabel('Počet')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# Vlastní vizualizace entit pomocí displaCy od spaCy
from spacy import displacy
doc = nlp("Apple Inc. plánuje otevřít novou pobočku v New Yorku příští měsíc.")
# Odkomentujte pro zobrazení v Jupyter Notebooku
# displacy.render(doc, style='ent', jupyter=True)

# Vytvoření funkce pro extrakci entit do DataFrame
def add_entities_to_df(df, text_column):
    result_df = df.copy()
    
    # Extrakce entit podle typu
    result_df['persons'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'PERSON'])
    
    result_df['organizations'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'ORG'])
    
    result_df['locations'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ in ['GPE', 'LOC']])
    
    result_df['dates'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'DATE'])
    
    return result_df

# Vytvoření a zpracování DataFrame
texts_df = pd.DataFrame(texts, columns=['text'])
entities_df = add_entities_to_df(texts_df, 'text')

print("\nExtrahované entity podle typu:")
print(entities_df[['persons', 'organizations', 'locations', 'dates']])
```

import spacy
from spacy.lang.en.stop_words import STOP_WORDS
from string import punctuation
from heapq import nlargest

def text_summarizer(text, per=0.3):
    nlp = spacy.load("en_core_web_sm")
    doc = nlp(text)
    
    # Výpočet četnosti slov
    word_frequencies = {}
    for word in doc:
        if word.text.lower() not in STOP_WORDS and word.text.lower() not in punctuation:
            if word.text not in word_frequencies.keys():
                word_frequencies[word.text] = 1
            else:
                word_frequencies[word.text] += 1
    
    # Normalizace četnosti slov
    max_frequency = max(word_frequencies.values()) if word_frequencies else 1
    for word in word_frequencies.keys():
        word_frequencies[word] = word_frequencies[word] / max_frequency
    
    # Výpočet skóre vět na základě četnosti slov
    sentence_scores = {}
    for sent in doc.sents:
        for word in sent:
            if word.text.lower() in word_frequencies.keys():
                if sent not in sentence_scores.keys():
                    sentence_scores[sent] = word_frequencies[word.text.lower()]
                else:
                    sentence_scores[sent] += word_frequencies[word.text.lower()]
    
    # Získání sumarizace
    select_length = int(len(list(doc.sents)) * per)
    select_length = max(select_length, 1)  # Zajištění alespoň jedné věty
    summary = nlargest(select_length, sentence_scores, key=sentence_scores.get)
    
    return " ".join([sent.text for sent in summary])
