## Introduction to Graph Analysis

Graph analysis is a powerful approach to understanding complex relationships between entities. In data science, graphs provide a natural way to represent and analyze interconnected data, revealing patterns and insights that might be difficult to discover using traditional tabular methods.

A graph consists of:
- **Nodes** (or vertices): Representing entities
- **Edges**: Representing relationships between entities
- **Properties**: Attributes that can be attached to nodes and edges

## Understanding Graph Theory Fundamentals

Before diving into the tools, it's important to understand some fundamental graph theory concepts that form the basis of graph analysis.

### Types of Graphs

```python

G_weighted = nx.Graph()

```

### Important Graph Properties

- **Connectivity**: Whether all nodes are connected
- **Path**: Sequence of nodes where each adjacent pair is connected by an edge
- **Cycle**: Path that starts and ends at the same node
- **Degree**: Number of edges connected to a node
- **Centrality**: Measures of node importance in a network

```python
# Create a simple graph to demonstrate properties
G = nx.Graph()
edges = [('A', 'B'), ('B', 'C'), ('C', 'D'), ('D', 'A'), ('A', 'E')]

# Check connectivity

print(f"Is the graph connected? {is_connected}")
```

## Introduction to Neo4j

Neo4j is a graph database management system that stores data in nodes and relationships rather than tables. It's optimized for traversing relationships, making it ideal for highly connected data.

### Key Concepts

- **Nodes**: Entities with labels and properties
- **Relationships**: Connections between nodes with types and properties
- **Labels**: Group nodes into sets
- **Properties**: Key-value pairs stored in nodes and relationships
- **Cypher**: Declarative query language for Neo4j

```python
from neo4j import GraphDatabase

class Neo4jConnection:
    def __init__(self, uri, user, password):
        self.driver = GraphDatabase.driver(uri, auth=(user, password))
        
    def close(self):
        self.driver.close()
        
    def query(self, query, parameters=None):
        with self.driver.session() as session:
            result = session.run(query, parameters)
            return [record for record in result]
        
    def create_social_network(self):
```

### Converting between NetworkX and Neo4j

```python
def neo4j_to_networkx(conn):
    # Query all nodes and relationships
    nodes = conn.query("""
        MATCH (n) 
        RETURN id(n) as id, labels(n) as labels, properties(n) as props
    """)
    
    relationships = conn.query("""
        MATCH (n)-[r]->(m)
        RETURN id(n) as source, id(m) as target, type(r) as type, properties(r) as props
    """)
    
    # Create NetworkX graph
    G = nx.DiGraph()
    
    # Add nodes with attributes
    for node in nodes:
        G.add_node(node['id'], labels=node['labels'], **node['props'])
    
    # Add edges with attributes
    for rel in relationships:
        G.add_edge(rel['source'], rel['target'], type=rel['type'], **rel['props'])
    
    return G
```

## Graph Query Languages (Cypher)

Cypher is Neo4j's graph query language, designed to be expressive, efficient, and human-readable.

### Basic Cypher Syntax

```python
# Example Cypher queries

# Find all people
FIND_PEOPLE = """
MATCH (p:Person)
RETURN p.name as name, p.age as age
ORDER BY p.age DESC
"""

# Create a new person
CREATE_PERSON = """
CREATE (p:Person {name: $name, age: $age})
RETURN p
"""

# Create a relationship between two people
CREATE_RELATIONSHIP = """
MATCH (p1:Person {name: $name1})
MATCH (p2:Person {name: $name2})
CREATE (p1)-[:KNOWS {since: $since}]->(p2)
"""

# Find the shortest path between two people
SHORTEST_PATH = """
MATCH path = shortestPath((p1:Person {name: $name1})-[*]-(p2:Person {name: $name2}))
RETURN path
"""

FIND_COMMUNITIES = """
```

```python
# Function to find movie recommendations for a user
def recommend_movies(G, user, n=2):
    # Get movies the user has already watched
    watched_movies = [neighbor for neighbor in G.neighbors(user)
                     if G.nodes[neighbor]['type'] == 'movie']
    
    # Find users with similar tastes
    similar_users = []
    for movie in watched_movies:
        for other_user in G.neighbors(movie):
            if other_user != user and G.nodes[other_user]['type'] == 'user':
                similar_users.append(other_user)
    
    # Count frequency of similar users
    from collections import Counter
    user_counts = Counter(similar_users)
    most_similar = [user for user, count in user_counts.most_common(2)]
    
    # Find movies watched by similar users but not by the target user
    recommendations = set()
    for similar_user in most_similar:
        movies_by_similar = [m for m in G.neighbors(similar_user) 
                            if G.nodes[m]['type'] == 'movie']
        for movie in movies_by_similar:
            if movie not in watched_movies:
                # Get the rating from the similar user
                edge_data = G.get_edge_data(similar_user, movie)
                rating = edge_data.get('rating', 0)
                recommendations.add((movie, rating))
    
    # Return top N recommendations by rating
    top_recommendations = sorted(recommendations, key=lambda x: x[1], reverse=True)[:n]
    return top_recommendations

# Get recommendations for users
```

```python
normal_transactions = []
for i in range(30):
    source = random.choice(accounts)
    target = random.choice([a for a in accounts if a != source])
    amount = round(random.uniform(10, 1000), 2)
    timestamp = f"2023-{random.randint(1,12):02d}-{random.randint(1,28):02d}"
    normal_transactions.append((source, target, {'amount': amount, 'timestamp': timestamp, 'type': 'normal'}))

G_finance.add_edges_from(normal_transactions)

# Add suspicious pattern - circular transactions
suspicious_circle = [
    ('Account1', 'Account3', {'amount': 5000, 'timestamp': '2023-05-01', 'type': 'suspicious'}),
    ('Account3', 'Account7', {'amount': 4800, 'timestamp': '2023-05-01', 'type': 'suspicious'}),
    ('Account7', 'Account1', {'amount': 4700, 'timestamp': '2023-05-02', 'type': 'suspicious'})
]
G_finance.add_edges_from(suspicious_circle)

# Detect cycles as potential fraud indicators
cycles = list(nx.simple_cycles(G_finance))
print(f"Found {len(cycles)} cycles in the transaction network:")
for cycle in cycles:
    print(f"  Cycle: {' -> '.join(cycle)}")
    # Calculate total flow in the cycle
    total_amount = 0
    for i in range(len(cycle)):
        source = cycle[i]
        target = cycle[(i + 1) % len(cycle)]
        edge_data = G_finance.get_edge_data(source, target)
        if edge_data:
            total_amount += edge_data.get('amount', 0)
    print(f"  Total transaction amount in cycle: ${total_amount:.2f}")

# Calculate PageRank to identify important accounts
pagerank = nx.pagerank(G_finance)
top_pagerank = sorted(pagerank.items(), key=lambda x: x[1], reverse=True)[:3]
print("\nTop 3 accounts by PageRank:")
for account, score in top_pagerank:
    print(f"  {account}: {score:.4f}")

# Visualize the transaction network
plt.figure(figsize=(12, 10))
pos = nx.spring_layout(G_finance, seed=42)
```
