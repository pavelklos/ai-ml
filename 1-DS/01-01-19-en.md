## 1. Introduction to Summary Statistics

Summary statistics provide concise descriptions of datasets, helping to understand their main characteristics without examining every data point. Effective reporting communicates these insights clearly to stakeholders.

```python
# Import essential libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats

# Set visualization style
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_theme(style="whitegrid")

# Configure pandas display options
pd.set_option('display.max_columns', 50)
pd.set_option('display.precision', 3)
```

## 2. Essential Statistical Measures

Key statistical measures include:
- Central tendency (mean, median, mode)
- Dispersion (variance, standard deviation, range)
- Distribution shape (skewness, kurtosis)
- Relationship measures (correlation, covariance)

```python
# Load a sample dataset
# We'll use the classic iris dataset
from sklearn.datasets import load_iris

# Load and convert to pandas DataFrame
iris = load_iris()
df = pd.DataFrame(data=iris.data, columns=iris.feature_names)
df['species'] = pd.Categorical.from_codes(iris.target, iris.target_names)

print("First few rows of the dataset:")
print(df.head())

# Basic summary statistics
print("\nBasic summary statistics:")
print(df.describe())

# Additional statistics
print("\nSkewness (measure of distribution asymmetry):")
print(df.skew())

print("\nKurtosis (measure of 'tailedness'):")
print(df.kurtosis())

# Mode calculation
print("\nMode of each column:")
print(df.mode().iloc[0])  # Take first mode if multiple exist
```

## 3. Generating Descriptive Statistics

Pandas offers powerful functions for generating comprehensive descriptive statistics.

```python
# Comprehensive statistics with pandas
stats_summary = df.describe(include='all', percentiles=[.05, .25, .5, .75, .95])
print("Extended summary statistics with custom percentiles:")
print(stats_summary)

# Group-based statistics
group_stats = df.groupby('species').describe()
print("\nSummary statistics by species:")
print(group_stats)

# Specific aggregations
custom_stats = df.groupby('species').agg({
    'sepal length (cm)': ['min', 'max', 'mean', 'std', 'median'],
    'sepal width (cm)': ['min', 'max', 'mean', 'std', 'median'],
    'petal length (cm)': ['min', 'max', 'mean', 'std', 'median'],
    'petal width (cm)': ['min', 'max', 'mean', 'std', 'median']
})
print("\nCustom aggregated statistics:")
print(custom_stats)

# Correlation analysis
correlation = df.corr()
print("\nCorrelation matrix:")
print(correlation)

# Statistical significance tests
# Example: t-test for two groups
setosa = df[df['species'] == 'setosa']['sepal length (cm)']
versicolor = df[df['species'] == 'versicolor']['sepal length (cm)']
t_stat, p_value = stats.ttest_ind(setosa, versicolor)
print(f"\nt-test for sepal length between setosa and versicolor:")
print(f"t-statistic: {t_stat:.4f}, p-value: {p_value:.8f}")
```

## 5. Customized Statistical Reports

Creating custom reports tailored to specific analytical needs and audiences.

```python
# Customized statistical report function
def generate_statistical_report(dataframe, group_column=None):
    """Generate a comprehensive statistical report for a DataFrame."""
    
    report = {}
    
    # Overall statistics
    report['basic_stats'] = dataframe.describe()
    report['missing_values'] = dataframe.isnull().sum()
    report['data_types'] = dataframe.dtypes
    
    # Identify numeric columns
    numeric_columns = dataframe.select_dtypes(include=[np.number]).columns
    
    # Distribution statistics for numeric columns
    report['skewness'] = dataframe[numeric_columns].skew()
    report['kurtosis'] = dataframe[numeric_columns].kurtosis()
    
    # Correlation analysis
    if len(numeric_columns) >= 2:
        report['correlation'] = dataframe[numeric_columns].corr()
    
    # Grouped statistics if group column is provided
    if group_column and group_column in dataframe.columns:
        report['group_stats'] = dataframe.groupby(group_column)[numeric_columns].agg([
            'count', 'min', 'max', 'mean', 'std', 'median',
            lambda x: x.quantile(0.25),
            lambda x: x.quantile(0.75)
        ])
        report['group_stats'].columns = [
            'count', 'min', 'max', 'mean', 'std', 'median', '25%', '75%'
        ]
    
    return report

# Generate and display custom report
iris_report = generate_statistical_report(df, 'species')

print("Basic Statistics:")
print(iris_report['basic_stats'])

print("\nMissing Values:")
print(iris_report['missing_values'])

print("\nData Types:")
print(iris_report['data_types'])

print("\nSkewness:")
print(iris_report['skewness'])

print("\nGroup Statistics (first few rows):")
print(iris_report['group_stats'].head())
```

```python
# Custom automated reporting with matplotlib and pandas
def create_summary_dashboard(df, title="Statistical Summary Dashboard", save_path=None):
    """Generate a statistical dashboard for a DataFrame."""
    
    # Identify numeric and categorical columns
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    categorical_cols = df.select_dtypes(exclude=[np.number]).columns
    
    # Create a dashboard with subplots
    n_numeric = len(numeric_cols)
    n_categorical = len(categorical_cols)
    
    fig = plt.figure(figsize=(15, 10))
    fig.suptitle(title, fontsize=16)
    
    # Summary statistics
    plt.figtext(0.5, 0.95, f"Dataset shape: {df.shape}", ha='center')
    plt.figtext(0.5, 0.93, f"Missing values: {df.isnull().sum().sum()}", ha='center')
    
    # Create histograms for numeric columns
    for i, col in enumerate(numeric_cols):
        ax = fig.add_subplot(max(1, n_numeric // 2), 2, i+1)
        sns.histplot(df[col], kde=True, ax=ax)
        ax.axvline(df[col].mean(), color='r', linestyle='--')
        ax.axvline(df[col].median(), color='g', linestyle='-')
        ax.set_title(f'{col} Distribution')
    
    # Save if requested
    if save_path:
        plt.tight_layout(rect=[0, 0, 1, 0.92])
        plt.savefig(save_path)
        print(f"Dashboard saved to {save_path}")
    
    plt.tight_layout(rect=[0, 0, 1, 0.92])
    plt.show()

# Create and display the dashboard
create_summary_dashboard(df, "Iris Dataset Summary", "iris_dashboard.png")
```

## 8. Practical Applications

Applying summary statistics to real-world analytical scenarios.

```python
# Example 1: Quality Control Analysis
# Let's simulate some manufacturing data
np.random.seed(42)
n_samples = 200
manufacturing_data = pd.DataFrame({
    'batch': np.repeat(range(1, 21), 10),
    'temperature': np.random.normal(100, 5, n_samples),
    'pressure': np.random.normal(50, 3, n_samples),
    'defect_rate': np.random.uniform(0, 0.1, n_samples)
})

# Quality control report
qc_report = manufacturing_data.groupby('batch').agg({
    'temperature': ['mean', 'std', 'min', 'max'],
    'pressure': ['mean', 'std', 'min', 'max'],
    'defect_rate': ['mean', 'sum']
})

print("Quality Control Report:")
print(qc_report.head())

# Visualize quality metrics over time (batches)
plt.figure(figsize=(14, 8))

plt.subplot(3, 1, 1)
sns.lineplot(x='batch', y='temperature', data=manufacturing_data, ci='sd')
plt.title('Temperature by Batch with Standard Deviation')

plt.subplot(3, 1, 2)
sns.lineplot(x='batch', y='pressure', data=manufacturing_data, ci='sd')
plt.title('Pressure by Batch with Standard Deviation')

plt.subplot(3, 1, 3)
batch_defects = manufacturing_data.groupby('batch')['defect_rate'].sum().reset_index()
sns.barplot(x='batch', y='defect_rate', data=batch_defects)
plt.title('Total Defect Rate by Batch')
plt.axhline(batch_defects['defect_rate'].mean(), color='r', linestyle='--', label='Average')
plt.legend()

plt.tight_layout()
plt.show()

# Example 2: A/B Testing Analysis
# Simulate A/B test data for a website
np.random.seed(123)
n_users = 1000
ab_test_data = pd.DataFrame({
    'user_id': range(1, n_users + 1),
    'group': np.random.choice(['A', 'B'], n_users),
    'conversion': np.random.binomial(1, 0.1, n_users),  # base 10% conversion
    'time_spent': np.random.exponential(100, n_users)  # time in seconds
})

# Add effect: Group B has higher conversion and time spent
ab_test_data.loc[ab_test_data['group'] == 'B', 'conversion'] = np.random.binomial(
    1, 0.15, sum(ab_test_data['group'] == 'B'))  # 15% conversion for B
ab_test_data.loc[ab_test_data['group'] == 'B', 'time_spent'] *= 1.2  # 20% more time spent

# Generate A/B test report
ab_stats = ab_test_data.groupby('group').agg({
    'user_id': 'count',
    'conversion': ['mean', 'sum'],
    'time_spent': ['mean', 'median', 'std']
})
print("\nA/B Test Results:")
print(ab_stats)

# Statistical significance for conversion rates
from scipy.stats import chi2_contingency

# Create contingency table
contingency = pd.crosstab(ab_test_data['group'], ab_test_data['conversion'])
chi2, p, dof, expected = chi2_contingency(contingency)
print(f"\nChi-square test for conversion rates: chi2={chi2:.4f}, p-value={p:.6f}")

# Statistical significance for time spent
t_stat, p_value = stats.ttest_ind(
    ab_test_data[ab_test_data['group'] == 'A']['time_spent'],
    ab_test_data[ab_test_data['group'] == 'B']['time_spent']
)
print(f"T-test for time spent: t-stat={t_stat:.4f}, p-value={p:.6f}")

# Visualize A/B test results
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 6))

# Conversion rates
conv_data = ab_test_data.groupby('group')['conversion'].mean()
ax1.bar(conv_data.index, conv_data.values, yerr=[0, 0])
ax1.set_title('Conversion Rate by Group')
ax1.set_ylabel('Conversion Rate')
ax1.set_ylim(0, 0.2)
for i, v in enumerate(conv_data.values):
    ax1.text(i, v + 0.01, f"{v:.3f}", ha='center')

# Time spent
sns.boxplot(x='group', y='time_spent', data=ab_test_data, ax=ax2)
ax2.set_title('Time Spent by Group')
ax2.set_ylabel('Time Spent (seconds)')

plt.tight_layout()
plt.show()
```

## 9. Summary

Summary statistics and reporting are essential components of data science that:

1. **Condense Large Datasets**: Extract key insights from vast amounts of data
2. **Communicate Findings**: Present results in formats accessible to various stakeholders
3. **Guide Decision Making**: Provide quantitative evidence for data-driven decisions
4. **Identify Patterns**: Reveal trends, relationships, and anomalies within data
5. **Compare Groups**: Highlight differences between subpopulations or categories
6. **Track Changes**: Monitor metrics over time to observe changes

Effective statistical reporting combines:
- Clear numerical summaries
- Appropriate visualizations
- Statistical significance tests
- Custom formatting for the target audience
- Actionable insights derived from the data

With the tools and techniques covered in this module, you can create comprehensive statistical reports that transform raw data into valuable insights for informed decision-making.
