## Regular Expression (Regex) Basics

Regular expressions provide a powerful way to search, extract, and manipulate text based on patterns. They are essential for text cleaning and feature extraction.

```python
import re

# Basic regex patterns
sample_text = "Contact us at support@example.com or call 555-123-4567. Visit our website: https://www.example.com"

# Finding all email addresses
email_pattern = r'\b[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Z|a-z]{2,}\b'
emails = re.findall(email_pattern, sample_text)
print(f"Emails found: {emails}")

# Finding phone numbers
phone_pattern = r'\b\d{3}[-.]?\d{3}[-.]?\d{4}\b'
phones = re.findall(phone_pattern, sample_text)
print(f"Phone numbers found: {phones}")

# Finding URLs
url_pattern = r'https?://(?:[-\w.]|(?:%[\da-fA-F]{2}))+'
urls = re.findall(url_pattern, sample_text)
print(f"URLs found: {urls}")

# Substituting text
censored_text = re.sub(r'\d{3}-\d{3}-\d{4}', '[PHONE NUMBER REDACTED]', sample_text)
print(f"Censored text: {censored_text}")

# Splitting text
split_text = re.split(r'[.!?]', sample_text)
print(f"Split sentences: {split_text}")

# Common regex patterns for data cleaning
def clean_text(text):
    # Remove special characters and numbers
    text = re.sub(r'[^a-zA-Z\s]', '', text)
    # Remove extra whitespace
    text = re.sub(r'\s+', ' ', text).strip()
    return text

df['cleaned_text'] = df['text'].apply(clean_text)
print("\nCleaned text examples:")
print(df[['text', 'cleaned_text']].head())
```

## Text Preprocessing Techniques

Before analyzing text data, it's essential to preprocess it by removing noise and standardizing the format. Common preprocessing steps include lowercasing, removing punctuation, stopwords, and stemming/lemmatization.

```python
# Install necessary packages
# pip install nltk spacy
# python -m spacy download en_core_web_sm

import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer, WordNetLemmatizer
import spacy

# Download required NLTK resources
nltk.download('punkt')
nltk.download('stopwords')
nltk.download('wordnet')

# Sample text
text = "Data scientists need to preprocess text data before analysis. Processing includes removing stopwords and stemming."

# 1. Lowercasing
text_lower = text.lower()
print(f"Lowercase: {text_lower}")

# 2. Tokenization
tokens = word_tokenize(text_lower)
print(f"Tokens: {tokens}")

# 3. Removing punctuation
tokens_no_punct = [token for token in tokens if token not in string.punctuation]
print(f"Tokens without punctuation: {tokens_no_punct}")

# 4. Removing stopwords
stop_words = set(stopwords.words('english'))
tokens_no_stop = [token for token in tokens_no_punct if token not in stop_words]
print(f"Tokens without stopwords: {tokens_no_stop}")

# 5. Stemming (reducing words to their base/stem form)
stemmer = PorterStemmer()
stemmed_tokens = [stemmer.stem(token) for token in tokens_no_stop]
print(f"Stemmed tokens: {stemmed_tokens}")

# 6. Lemmatization (reducing words to their base form, but ensuring it's a valid word)
lemmatizer = WordNetLemmatizer()
lemmatized_tokens = [lemmatizer.lemmatize(token) for token in tokens_no_stop]
print(f"Lemmatized tokens: {lemmatized_tokens}")

# Using spaCy for preprocessing
nlp = spacy.load("en_core_web_sm")
doc = nlp(text)

# Extract lemmas, remove stopwords and punctuation
spacy_processed = [token.lemma_ for token in doc if not token.is_stop and not token.is_punct]
print(f"spaCy processed text: {spacy_processed}")

# Function for complete text preprocessing
def preprocess_text(text):
    # Lowercase and tokenize
    tokens = word_tokenize(text.lower())
    # Remove punctuation
    tokens = [token for token in tokens if token not in string.punctuation]
    # Remove stopwords
    tokens = [token for token in tokens if token not in stop_words]
    # Lemmatize
    tokens = [lemmatizer.lemmatize(token) for token in tokens]
    return tokens

# Apply preprocessing to the DataFrame
df['preprocessed_tokens'] = df['text'].apply(preprocess_text)
print("\nPreprocessed text examples:")
for i, tokens in enumerate(df['preprocessed_tokens'].head()):
    print(f"{i+1}: {tokens}")
```

## Text Vectorization Methods

Converting text into numerical vectors is essential for machine learning algorithms. Common vectorization methods include Bag of Words, TF-IDF, and word embeddings.

```python
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from gensim.models import Word2Vec
import numpy as np
import pandas as pd

# Sample corpus
corpus = [
    "The sky is blue and beautiful.",
    "Love this blue and clear sky!",
    "The quick brown fox jumps over the lazy dog.",
    "A king's breakfast has sausages, ham, bacon, eggs, toast and beans",
    "I love green eggs, ham, sausages and bacon!",
    "The brown fox is quick and the blue dog is lazy!",
    "The sky is very blue and the sky is very beautiful today",
    "The dog is lazy but the brown fox is quick!"
]

# 1. Bag of Words (BoW)
print("Bag of Words Vectorization:")
count_vectorizer = CountVectorizer()
bow_matrix = count_vectorizer.fit_transform(corpus)

bow_df = pd.DataFrame(
    bow_matrix.toarray(),
    columns=count_vectorizer.get_feature_names_out()
)
print(bow_df.shape)
print(bow_df.iloc[:3, :10])  # Print first 3 rows, first 10 columns

# 2. TF-IDF Vectorization
print("\nTF-IDF Vectorization:")
tfidf_vectorizer = TfidfVectorizer()
tfidf_matrix = tfidf_vectorizer.fit_transform(corpus)

tfidf_df = pd.DataFrame(
    tfidf_matrix.toarray(),
    columns=tfidf_vectorizer.get_feature_names_out()
)
print(tfidf_df.shape)
print(tfidf_df.iloc[:3, :10])  # Print first 3 rows, first 10 columns

# 3. Word Embeddings with Word2Vec
# Tokenize the corpus
tokenized_corpus = [text.lower().split() for text in corpus]

# Train Word2Vec model
w2v_model = Word2Vec(
    tokenized_corpus,
    vector_size=100,   # Embedding dimension
    window=5,          # Context window size
    min_count=1,       # Minimum word frequency
    workers=4          # Number of threads
)

# Get vector for specific word
print("\nWord2Vec Embeddings:")
word = "blue"
if word in w2v_model.wv:
    print(f"Vector for '{word}': {w2v_model.wv[word][:5]}...")  # Show first 5 dimensions
    
    # Find most similar words
    similar_words = w2v_model.wv.most_similar(word, topn=3)
    print(f"Words most similar to '{word}':")
    for similar_word, similarity in similar_words:
        print(f"- {similar_word}: {similarity:.4f}")

# 4. Document Embeddings (simple average of word vectors)
def document_vector(doc, model):
    words = [word for word in doc.lower().split() if word in model.wv]
    if len(words) == 0:
        return np.zeros(model.vector_size)
    return np.mean([model.wv[word] for word in words], axis=0)

# Generate document vectors
doc_vectors = [document_vector(doc, w2v_model) for doc in corpus]
doc_vectors_df = pd.DataFrame(doc_vectors, columns=[f"dim_{i}" for i in range(5)])  # Show first 5 dimensions

print("\nDocument Embeddings (first 5 dimensions):")
print(doc_vectors_df)
```

## Basic Sentiment Analysis

Sentiment analysis determines the emotional tone behind text data, which is useful for understanding customer feedback, social media analysis, and more.

```python
from textblob import TextBlob
from nltk.sentiment import SentimentIntensityAnalyzer
import pandas as pd
import matplotlib.pyplot as plt

# Download NLTK resources for sentiment analysis
nltk.download('vader_lexicon')

# Sample reviews
reviews = [
    "This product is amazing! I absolutely love it.",
    "The service was terrible and the staff was rude.",
    "The movie was okay, not great but not terrible either.",
    "I highly recommend this book, it's insightful and engaging.",
    "The hotel room was dirty and the bed was uncomfortable.",
    "The food was delicious but a bit expensive.",
    "I'm not sure if I like this or not, it's complicated."
]

# Create DataFrame
reviews_df = pd.DataFrame(reviews, columns=['review'])

# 1. TextBlob Sentiment Analysis
reviews_df['textblob_polarity'] = reviews_df['review'].apply(lambda x: TextBlob(x).sentiment.polarity)
reviews_df['textblob_subjectivity'] = reviews_df['review'].apply(lambda x: TextBlob(x).sentiment.subjectivity)

# 2. VADER Sentiment Analysis
sia = SentimentIntensityAnalyzer()
reviews_df['vader_compound'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['compound'])
reviews_df['vader_pos'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['pos'])
reviews_df['vader_neg'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['neg'])
reviews_df['vader_neu'] = reviews_df['review'].apply(lambda x: sia.polarity_scores(x)['neu'])

# 3. Classify sentiment
def classify_sentiment(score):
    if score > 0.05:
        return 'Positive'
    elif score < -0.05:
        return 'Negative'
    else:
        return 'Neutral'

reviews_df['textblob_sentiment'] = reviews_df['textblob_polarity'].apply(classify_sentiment)
reviews_df['vader_sentiment'] = reviews_df['vader_compound'].apply(classify_sentiment)

# Display results
print("Sentiment Analysis Results:")
print(reviews_df[['review', 'textblob_sentiment', 'vader_sentiment', 'textblob_polarity', 'vader_compound']])

# Visualize sentiment distribution
plt.figure(figsize=(10, 6))

plt.subplot(1, 2, 1)
reviews_df['textblob_sentiment'].value_counts().plot(kind='bar', color='skyblue')
plt.title('TextBlob Sentiment Distribution')
plt.ylabel('Count')

plt.subplot(1, 2, 2)
reviews_df['vader_sentiment'].value_counts().plot(kind='bar', color='lightgreen')
plt.title('VADER Sentiment Distribution')
plt.ylabel('Count')

plt.tight_layout()
plt.show()

# Scatter plot comparing TextBlob and VADER
plt.figure(figsize=(8, 6))
plt.scatter(reviews_df['textblob_polarity'], reviews_df['vader_compound'], alpha=0.7)
plt.xlabel('TextBlob Polarity')
plt.ylabel('VADER Compound Score')
plt.title('TextBlob vs. VADER Sentiment Scores')
plt.grid(True, linestyle='--', alpha=0.7)

# Add sentiment regions
plt.axhline(y=0.05, color='g', linestyle='-', alpha=0.3)
plt.axhline(y=-0.05, color='r', linestyle='-', alpha=0.3)
plt.axvline(x=0.05, color='g', linestyle='-', alpha=0.3)
plt.axvline(x=-0.05, color='r', linestyle='-', alpha=0.3)

# Add text labels for each point
for i, review in enumerate(reviews):
    plt.annotate(f"Review {i+1}", 
                 (reviews_df['textblob_polarity'][i], reviews_df['vader_compound'][i]),
                 xytext=(5, 5), textcoords='offset points')

plt.tight_layout()
plt.show()

# Build a simple sentiment classifier using scikit-learn
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import classification_report

# Use VADER sentiment as "ground truth" for this example
X = reviews_df['review']
y = reviews_df['vader_sentiment']

# Create train/test split
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.3, random_state=42)

# Vectorize text
vectorizer = TfidfVectorizer(max_features=1000)
X_train_vec = vectorizer.fit_transform(X_train)
X_test_vec = vectorizer.transform(X_test)

# Train a simple classifier
classifier = LogisticRegression()
classifier.fit(X_train_vec, y_train)

# Make predictions
y_pred = classifier.predict(X_test_vec)

# Evaluate
print("\nSentiment Classification Report:")
print(classification_report(y_test, y_pred))
```

## Named Entity Recognition (NER)

NER identifies and classifies named entities in text such as persons, locations, organizations, and more.

```python
import spacy
import pandas as pd
import matplotlib.pyplot as plt
from collections import Counter

# Load spaCy model
nlp = spacy.load("en_core_web_sm")

# Sample texts with named entities
texts = [
    "Apple Inc. is planning to open a new store in New York City next month.",
    "Amazon and Microsoft are competing for the $10 billion Pentagon cloud contract.",
    "The Golden State Warriors defeated the Toronto Raptors in the NBA finals.",
    "President Joe Biden addressed the United Nations General Assembly in September 2021.",
    "Tesla's CEO Elon Musk announced plans to build a gigafactory in Berlin, Germany."
]

# Extract named entities
def extract_entities(text):
    doc = nlp(text)
    entities = [(ent.text, ent.label_) for ent in doc.ents]
    return entities

# Process all texts
all_entities = []
for text in texts:
    entities = extract_entities(text)
    all_entities.extend(entities)
    print(f"\nText: {text}")
    print("Entities:")
    for entity, label in entities:
        print(f"- {entity}: {label}")

# Count entity types
entity_types = [entity[1] for entity in all_entities]
entity_counts = Counter(entity_types)

# Visualize entity distributions
plt.figure(figsize=(12, 6))
plt.bar(entity_counts.keys(), entity_counts.values(), color='skyblue')
plt.title('Named Entity Types Distribution')
plt.xlabel('Entity Type')
plt.ylabel('Count')
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

# Custom entity visualization using spaCy's displaCy
from spacy import displacy
doc = nlp("Apple Inc. is planning to open a new store in New York City next month.")
# Uncomment for Jupyter Notebook display
# displacy.render(doc, style='ent', jupyter=True)

# Create entity extraction function for DataFrames
def add_entities_to_df(df, text_column):
    result_df = df.copy()
    
    # Extract entities by type
    result_df['persons'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'PERSON'])
    
    result_df['organizations'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'ORG'])
    
    result_df['locations'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ in ['GPE', 'LOC']])
    
    result_df['dates'] = result_df[text_column].apply(
        lambda x: [ent.text for ent in nlp(x).ents if ent.label_ == 'DATE'])
    
    return result_df

# Create and process DataFrame
texts_df = pd.DataFrame(texts, columns=['text'])
entities_df = add_entities_to_df(texts_df, 'text')

print("\nExtracted entities by type:")
print(entities_df[['persons', 'organizations', 'locations', 'dates']])
```

## Topic Modeling Basics

Topic modeling helps discover abstract topics in text collections, useful for document categorization and content analysis.

```python

    "Climate change includes both global warming driven by human-induced emissions of greenhouse gases and the resulting large-scale shifts in weather patterns.",
    "Environmental science is an interdisciplinary academic field that integrates physical, biological and information sciences to the study of the environment and the solution of environmental problems.",
    "Renewable energy is energy that is collected from renewable resources, which are naturally replenished on a human timescale, such as sunlight, wind, rain, tides, waves, and geothermal heat."

# Preprocessing
def preprocess_text(text):
    # Tokenize, lowercase, and remove punctuation/stopwords

    # Remove stopwords

    corpus=corpus,

    passes=10,

    random_state=42
)

    return model_list, coherence_values
```

## Practical NLP Applications

This section demonstrates real-world applications of NLP techniques for tasks like text classification, summarization, and language generation.

```python
# Basic text classification (sentiment, spam, topic, etc.)
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import MultinomialNB
from sklearn.metrics import classification_report, confusion_matrix
import pandas as pd
import numpy as np
import seaborn as sns
import matplotlib.pyplot as plt

# Sample dataset - SMS spam classification
# (This is a small sample. In practice, use a larger dataset)
sms_data = [
    ("Free entry to concert! Just text YES to 12345", "spam"),
    ("Congratulations! You've won a free vacation. Call now!", "spam"),
    ("URGENT: Your bank account has been suspended", "spam"),
    ("Can you call me when you're free?", "ham"),
    ("I'll be home in 20 minutes", "ham"),
    ("Meeting is scheduled for tomorrow at 10am", "ham"),
    ("Want to grab lunch tomorrow?", "ham"),
    ("Claim your prize! Limited time offer!!!", "spam"),
    ("Don't forget to pick up milk on your way home", "ham"),
    ("Your package will be delivered today", "ham")
]

# Prepare data
sms_df = pd.DataFrame(sms_data, columns=['message', 'label'])

# Split data
X_train, X_test, y_train, y_test = train_test_split(
    sms_df['message'], sms_df['label'], test_size=0.3, random_state=42
)

# Vectorize
vectorizer = TfidfVectorizer(max_features=1000)
X_train_vec = vectorizer.fit_transform(X_train)
X_test_vec = vectorizer.transform(X_test)

# Train classifier
clf = MultinomialNB()
clf.fit(X_train_vec, y_train)

# Predict
y_pred = clf.predict(X_test_vec)

# Evaluate
print("Text Classification Results:")
print(classification_report(y_test, y_pred))

# Confusion matrix
cm = confusion_matrix(y_test, y_pred)
plt.figure(figsize=(8, 6))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
            xticklabels=['ham', 'spam'], yticklabels=['ham', 'spam'])
plt.xlabel('Predicted Label')
plt.ylabel('True Label')
plt.title('Confusion Matrix for SMS Classification')
plt.tight_layout()
plt.show()

# Text summarization with spaCy
import spacy
from spacy.lang.en.stop_words import STOP_WORDS
from string import punctuation
from heapq import nlargest

def text_summarizer(text, per=0.3):
    nlp = spacy.load("en_core_web_sm")
    doc = nlp(text)
    
    # Calculate word frequency
    word_frequencies = {}
    for word in doc:
        if word.text.lower() not in STOP_WORDS and word.text.lower() not in punctuation:
            if word.text not in word_frequencies.keys():
                word_frequencies[word.text] = 1
            else:
                word_frequencies[word.text] += 1
    
    # Normalize word frequencies
    max_frequency = max(word_frequencies.values()) if word_frequencies else 1
    for word in word_frequencies.keys():
        word_frequencies[word] = word_frequencies[word] / max_frequency
    
    # Calculate sentence scores based on word frequencies
    sentence_scores = {}
    for sent in doc.sents:
        for word in sent:
            if word.text.lower() in word_frequencies.keys():
                if sent not in sentence_scores.keys():
                    sentence_scores[sent] = word_frequencies[word.text.lower()]
                else:
                    sentence_scores[sent] += word_frequencies[word.text.lower()]
    
    # Get summary
    select_length = int(len(list(doc.sents)) * per)
    select_length = max(select_length, 1)  # Ensure at least one sentence
    summary = nlargest(select_length, sentence_scores, key=sentence_scores.get)
    
    return " ".join([sent.text for sent in summary])
```
