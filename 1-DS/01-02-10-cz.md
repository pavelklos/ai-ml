## Úvod do průzkumné analýzy dat

Průzkumná analýza dat (EDA) je kritickým prvním krokem v každém projektu datové vědy. Zahrnuje analýzu a sumarizaci datových sad za účelem získání poznatků, identifikace vzorů, detekce anomálií a testování hypotéz. EDA pomáhá datovým vědcům porozumět struktuře a charakteristikám dat před aplikací složitých modelovacích technik.

# Funkce pro výpočet a vizualizaci kategorických korelací pomocí Cramerova V
def categorical_correlations(df, categorical_features):
    """Výpočet Cramerova V korelace mezi kategorickými proměnnými"""
    from scipy.stats import chi2_contingency
    
    def cramers_v(x, y):
        confusion_matrix = pd.crosstab(x, y)
        chi2 = chi2_contingency(confusion_matrix)[0]
        n = confusion_matrix.sum().sum()
        phi2 = chi2 / n
        r, k = confusion_matrix.shape
        phi2corr = max(0, phi2 - ((k-1)*(r-1))/(n-1))
        r_corr = r - ((r-1)**2)/(n-1)
        k_corr = k - ((k-1)**2)/(n-1)
        return np.sqrt(phi2corr / min((k_corr-1), (r_corr-1)))
    
    corr_matrix = pd.DataFrame(np.zeros((len(categorical_features), len(categorical_features))), 
                              index=categorical_features, columns=categorical_features)
    
    for i, feat1 in enumerate(categorical_features):
        for j, feat2 in enumerate(categorical_features):
            if i > j:  # Výpočet pouze pro dolní trojúhelník
                corr_matrix.loc[feat1, feat2] = cramers_v(df[feat1], df[feat2])
    
    # Vytvoření symetrické matice
    for i, feat1 in enumerate(categorical_features):
        for j, feat2 in enumerate(categorical_features):
            if i < j:  # Vyplnění horního trojúhelníku
                corr_matrix.loc[feat1, feat2] = corr_matrix.loc[feat2, feat1]
            elif i == j:  # Nastavení diagonály na 1
                corr_matrix.loc[feat1, feat2] = 1.0
    
    return corr_matrix

# Získání pouze kategorických proměnných s 10 nebo méně unikátními hodnotami
suitable_cat_features = [col for col in categorical_features 
                        if titanic[col].nunique() <= 10 and titanic[col].isnull().sum() == 0]

if suitable_cat_features:
    # Výpočet kategorických korelací
    cat_corr = categorical_correlations(titanic, suitable_cat_features)
    
    # Vykreslení heat mapy kategorických korelací
    plt.figure(figsize=(10, 8))
    sns.heatmap(cat_corr, annot=True, cmap='coolwarm', vmin=0, vmax=1)
    plt.title("Korelace kategorických proměnných pomocí Cramerova V")
    plt.tight_layout()
    plt.show()

## Detekce a zpracování odlehlých hodnot

Identifikace a zpracování odlehlých hodnot je klíčové pro vytvoření robustních modelů.

```python
# Funkce pro detekci odlehlých hodnot pomocí metody IQR
def detect_outliers_iqr(df, feature):
    """Detekce odlehlých hodnot pomocí mezikvartilového rozpětí (IQR)"""
    Q1 = df[feature].quantile(0.25)
    Q3 = df[feature].quantile(0.75)
    IQR = Q3 - Q1
    
    lower_bound = Q1 - 1.5 * IQR
    upper_bound = Q3 + 1.5 * IQR
    
    outliers = df[(df[feature] < lower_bound) | (df[feature] > upper_bound)][feature]
    return outliers, lower_bound, upper_bound

# Vizualizace odlehlých hodnot pro numerickou proměnnou (např. 'fare')
feature = 'fare'
outliers, lower_bound, upper_bound = detect_outliers_iqr(titanic, feature)

print(f"Odlehlé hodnoty v '{feature}' pomocí metody IQR:")
print(f"Počet odlehlých hodnot: {len(outliers)}")
print(f"Procento odlehlých hodnot: {(len(outliers) / len(titanic)) * 100:.2f}%")
print(f"Dolní hranice: {lower_bound:.2f}")
print(f"Horní hranice: {upper_bound:.2f}")

# Krabicový graf se zvýrazněnými odlehlými hodnotami
plt.figure(figsize=(10, 6))
ax = sns.boxplot(x=titanic[feature])
plt.title(f'Krabicový graf proměnné {feature} s odlehlými hodnotami')
plt.tight_layout()
plt.show()

# Bodový graf věku vs. jízdné se zvýrazněnými potenciálními odlehlými hodnotami
plt.figure(figsize=(12, 8))
sns.scatterplot(x='age', y='fare', data=titanic, alpha=0.6)

# Zvýraznění odlehlých hodnot v proměnné jízdné
outlier_indices = titanic[titanic['fare'] > upper_bound].index
sns.scatterplot(x='age', y='fare', data=titanic.loc[outlier_indices], 
                color='red', s=100, label='Odlehlé hodnoty v jízdném')

plt.title('Bodový graf věku vs. jízdného se zvýrazněnými odlehlými hodnotami')
plt.legend()
plt.tight_layout()
plt.show()

# Zpracování odlehlých hodnot oříznutím
def cap_outliers(df, feature, lower_bound, upper_bound):
    """Oříznutí odlehlých hodnot na hranice"""
    df_capped = df.copy()
    df_capped[feature] = df_capped[feature].clip(lower=lower_bound, upper=upper_bound)
    return df_capped

# Oříznutí odlehlých hodnot v proměnné 'fare'
titanic_capped = cap_outliers(titanic, feature, lower_bound, upper_bound)

# Porovnání původní vs. oříznuté distribuce
plt.figure(figsize=(15, 6))

plt.subplot(1, 2, 1)
sns.histplot(titanic[feature], kde=True)
plt.title(f'Původní distribuce {feature}')

plt.subplot(1, 2, 2)
sns.histplot(titanic_capped[feature], kde=True)
plt.title(f'Oříznutá distribuce {feature}')

plt.tight_layout()
plt.show()
```

# Funkce pro vytvoření houslových grafů pro každou kombinaci
def plot_violin_grid(df, cat_vars, num_vars, n_cols=2):
    """Vytvoření mřížky houslových grafů pro kategorické vs. numerické proměnné"""
    n_plots = len(cat_vars) * len(num_vars)
    n_rows = (n_plots + n_cols - 1) // n_cols
    
    fig, axes = plt.subplots(n_rows, n_cols, figsize=(15, n_rows * 5))
    axes = axes.flatten()
    
    plot_idx = 0
    for cat in cat_vars:
        for num in num_vars:
            if plot_idx < len(axes):
                sns.violinplot(x=cat, y=num, data=df, ax=axes[plot_idx])
                axes[plot_idx].set_title(f'{num} podle {cat}')
                plot_idx += 1
    
    # Skrytí nepoužitých subplotů
    for j in range(plot_idx, len(axes)):
        fig.delaxes(axes[j])
    
    plt.tight_layout()
    plt.show()

plot_violin_grid(titanic, categorical_vars, numerical_vars)

# Skládané sloupcové grafy pro zobrazení proporcí
def plot_stacked_bars(df, x, y):
    """Vykreslení skládaného sloupcového grafu zobrazujícího proporce"""
    proportions = pd.crosstab(df[x], df[y], normalize='index')
    
    plt.figure(figsize=(10, 6))
    proportions.plot(kind='bar', stacked=True)
    plt.title(f'Proporce {y} podle {x}')
    plt.ylabel(f'Proporce {y}')
    plt.xlabel(x)
    plt.tight_layout()
    plt.show()

plot_stacked_bars(titanic, 'pclass', 'survived')
plot_stacked_bars(titanic, 'sex', 'survived')

try:
    apple_stock = yf.download("AAPL", start=start_date, end=end_date)
    
    # Kontrola, zda byla data úspěšně stažena
    if len(apple_stock) > 0:

        print(apple_stock.head())
        
        # Základní graf časové řady
        plt.figure(figsize=(15, 8))

        plt.title('Závěrečná cena akcií Apple - posledních 365 dní')

        plt.grid(True)
        plt.tight_layout()
        plt.show()
        
        # Převzorkování dat pro sledování měsíčních trendů
        monthly_data = apple_stock['Close'].resample('M').mean()
        
        plt.figure(figsize=(15, 6))

        plt.grid(True)
        plt.tight_layout()
        plt.show()
        
        # Výpočet denních výnosů
        apple_stock['Daily_Return'] = apple_stock['Close'].pct_change() * 100
        
        plt.figure(figsize=(15, 6))
        plt.plot(apple_stock.index, apple_stock['Daily_Return'], linewidth=1)
        plt.title('Denní výnosy (%)')
        plt.xlabel('Datum')
        plt.ylabel('Denní výnos (%)')
        plt.grid(True)
        plt.axhline(y=0, color='r', linestyle='--')
        plt.tight_layout()
        plt.show()
        
        # Distribuce denních výnosů
        plt.figure(figsize=(12, 6))
        sns.histplot(apple_stock['Daily_Return'].dropna(), kde=True)

        plt.xlabel('Denní výnos (%)')
        plt.ylabel('Četnost')

        plt.axvline(x=0, color='r', linestyle='--')
        plt.tight_layout()
        plt.show()
        
        # Klouzavé průměry
        apple_stock['MA50'] = apple_stock['Close'].rolling(window=50).mean()
        apple_stock['MA200'] = apple_stock['Close'].rolling(window=200).mean()
        
        plt.figure(figsize=(15, 8))
        plt.plot(apple_stock.index, apple_stock['Close'], label='Závěrečná cena', alpha=0.8)
        plt.plot(apple_stock.index, apple_stock['MA50'], label='50denní KP', linewidth=2)
        plt.plot(apple_stock.index, apple_stock['MA200'], label='200denní KP', linewidth=2)
        plt.title('Cena akcií Apple s klouzavými průměry')
        plt.xlabel('Datum')
        plt.ylabel('Cena ($)')

        plt.grid(True)
        plt.tight_layout()
        plt.show()

def quick_eda(df, target=None):
    """Provedení rychlé EDA na dataframe"""
    print("=" * 50)
    print("Přehled datasetu")
    print("=" * 50)
    print(f"Tvar: {df.shape}")
    print(f"\nSloupce: {df.columns.tolist()}")
    print("\nDatové typy:")
    print(df.dtypes)
    
    # Chybějící hodnoty
    missing = df.isnull().sum()
    missing_pct = (missing / len(df)) * 100
    missing_info = pd.DataFrame({
        'Chybějící hodnoty': missing,
        'Procento chybějících': missing_pct
    }).sort_values('Procento chybějících', ascending=False)
    
    print("\nChybějící hodnoty:")
    print(missing_info[missing_info['Chybějící hodnoty'] > 0])
    
    # Souhrnné statistiky
    print("\nSouhrnné statistiky:")
    print(df.describe().T)
    
    # Distribuce cílové proměnné, pokud je zadána
    if target and target in df.columns:
        print(f"\nDistribuce cílové proměnné ({target}):")
        print(df[target].value_counts())
        print(f"\nProporce cílové proměnné ({target}) (%):")
        print(df[target].value_counts(normalize=True) * 100)
    
    return {
        'shape': df.shape,
        'missing': missing_info,
        'summary_stats': df.describe(),
        'target_dist': df[target].value_counts() if target and target in df.columns else None
    }

## Shrnutí

Průzkumná analýza dat je klíčovou fází v pracovním procesu datové vědy, která pomáhá porozumět datům před aplikací složitých modelovacích technik. Klíčové aspekty efektivní EDA zahrnují:

1. **Úvodní přehled dat**: Porozumění základní struktuře, velikosti a obsahu datasetu pomocí souhrnných statistik a analýzy datových typů.

2. **Analýza distribuce**: Zkoumání distribucí jednotlivých proměnných k identifikaci vzorů, zkosení a potenciálních anomálií v datech.

3. **Korelační analýza**: Identifikace vztahů mezi proměnnými, což je klíčové pro výběr příznaků a pochopení potenciálních prediktivních faktorů.

4. **Detekce odlehlých hodnot**: Identifikace a případné zpracování neobvyklých datových bodů, které by mohly ovlivnit výkon modelu.

5. **Analýza chybějících hodnot**: Porozumění vzorům chybějících dat a implementace vhodných strategií imputace.

6. **Vztahy mezi příznaky**: Vizualizace interakcí mezi příznaky pro identifikaci komplexních vzorů a závislostí.

7. **Analýza časových řad**: Specializované techniky pro časově závislá data k identifikaci trendů, sezónnosti a dalších časových vzorů.

8. **Analýza kategorických dat**: Specifické přístupy k analýze a vizualizaci kategorických proměnných a jejich vztahů.

9. **Automatizační nástroje**: Využití automatizovaných nástrojů EDA pro urychlení procesu analýzy a zajištění konzistence.

10. **Iterativní proces**: EDA není jednorázovou aktivitou, ale iterativním procesem, který pokračuje v průběhu celého projektu datové vědy a informuje rozhodování v každé fázi.

Efektivní EDA kombinuje statistické techniky s vizualizací pro vytvoření komplexního porozumění datům, čímž usnadňuje lepší feature engineering, předzpracování dat a výběr modelu v následujících fázích projektu datové vědy.
