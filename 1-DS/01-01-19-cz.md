## 1. Úvod do souhrnných statistik

Souhrnné statistiky poskytují stručný popis datových sad, který pomáhá pochopit jejich hlavní charakteristiky bez nutnosti zkoumat každý datový bod. Efektivní reportování přenáší tyto poznatky jasně zúčastněným stranám.

```python
# Import základních knihoven
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from scipy import stats

# Nastavení stylu vizualizace
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_theme(style="whitegrid")

# Konfigurace možností zobrazení pandas
pd.set_option('display.max_columns', 50)
pd.set_option('display.precision', 3)
```

## 2. Základní statistická měření

Klíčová statistická měření zahrnují:
- Míry centrální tendence (průměr, medián, modus)
- Míry rozptylu (rozptyl, směrodatná odchylka, rozsah)
- Tvar distribuce (šikmost, špičatost)
- Míry vztahu (korelace, kovariance)

```python
# Načtení vzorového datasetu
# Použijeme klasický dataset iris
from sklearn.datasets import load_iris

# Načtení a převod do pandas DataFrame
iris = load_iris()
df = pd.DataFrame(data=iris.data, columns=iris.feature_names)
df['species'] = pd.Categorical.from_codes(iris.target, iris.target_names)

print("Prvních několik řádků datasetu:")
print(df.head())

# Základní souhrnné statistiky
print("\nZákladní souhrnné statistiky:")
print(df.describe())

# Další statistiky
print("\nŠikmost (míra asymetrie distribuce):")
print(df.skew())

print("\nŠpičatost (míra 'ocasatosti'):")
print(df.kurtosis())

# Výpočet modu
print("\nModus každého sloupce:")
print(df.mode().iloc[0])  # Vezme první modus, pokud jich existuje více
```

```python
custom_stats = df.groupby('species').agg({

    'petal length (cm)': ['min', 'max', 'mean', 'std', 'median'],
    'petal width (cm)': ['min', 'max', 'mean', 'std', 'median']
})
print("\nVlastní agregované statistiky:")

# Korelační analýza
correlation = df.corr()
print("\nKorelační matice:")

# Testy statistické významnosti

setosa = df[df['species'] == 'setosa']['sepal length (cm)']

t_stat, p_value = stats.ttest_ind(setosa, versicolor)

```


## 5. Přizpůsobené statistické reporty

Vytváření vlastních reportů přizpůsobených specifickým analytickým potřebám a cílovým skupinám.

```python
# Funkce pro přizpůsobený statistický report
def generate_statistical_report(dataframe, group_column=None):
    """Vygeneruje komplexní statistický report pro DataFrame."""
    
    report = {}
    
    # Celkové statistiky
    report['basic_stats'] = dataframe.describe()
    report['missing_values'] = dataframe.isnull().sum()
    report['data_types'] = dataframe.dtypes
    
    # Identifikace numerických sloupců
    numeric_columns = dataframe.select_dtypes(include=[np.number]).columns
    
    # Statistiky distribuce pro numerické sloupce
    report['skewness'] = dataframe[numeric_columns].skew()
    report['kurtosis'] = dataframe[numeric_columns].kurtosis()
    
    # Korelační analýza
    if len(numeric_columns) >= 2:
        report['correlation'] = dataframe[numeric_columns].corr()
    
    # Statistiky podle skupin, pokud je poskytnut skupinový sloupec
    if group_column and group_column in dataframe.columns:
        report['group_stats'] = dataframe.groupby(group_column)[numeric_columns].agg([
            'count', 'min', 'max', 'mean', 'std', 'median',
            lambda x: x.quantile(0.25),
            lambda x: x.quantile(0.75)
        ])
        report['group_stats'].columns = [
            'počet', 'min', 'max', 'průměr', 'směr. odchylka', 'medián', '25%', '75%'
        ]
    
    return report

# Generování a zobrazení vlastního reportu
iris_report = generate_statistical_report(df, 'species')

print("Základní statistiky:")
print(iris_report['basic_stats'])

print("\nChybějící hodnoty:")
print(iris_report['missing_values'])

print("\nDatové typy:")
print(iris_report['data_types'])

print("\nŠikmost:")
print(iris_report['skewness'])

print("\nStatistiky podle skupin (prvních několik řádků):")
print(iris_report['group_stats'].head())
```

```python
# Vlastní automatizované reportování s matplotlib a pandas
def create_summary_dashboard(df, title="Statistický přehledový dashboard", save_path=None):
    """Vygeneruje statistický dashboard pro DataFrame."""
    
    # Identifikace numerických a kategorických sloupců
    numeric_cols = df.select_dtypes(include=[np.number]).columns
    categorical_cols = df.select_dtypes(exclude=[np.number]).columns
    
    # Vytvoření dashboardu s podgrafy
    n_numeric = len(numeric_cols)
    n_categorical = len(categorical_cols)
    
    fig = plt.figure(figsize=(15, 10))
    fig.suptitle(title, fontsize=16)
    
    # Souhrnné statistiky
    plt.figtext(0.5, 0.95, f"Rozměry datasetu: {df.shape}", ha='center')
    plt.figtext(0.5, 0.93, f"Chybějící hodnoty: {df.isnull().sum().sum()}", ha='center')
    
    # Vytvoření histogramů pro numerické sloupce
    for i, col in enumerate(numeric_cols):
        ax = fig.add_subplot(max(1, n_numeric // 2), 2, i+1)
        sns.histplot(df[col], kde=True, ax=ax)
        ax.axvline(df[col].mean(), color='r', linestyle='--')
        ax.axvline(df[col].median(), color='g', linestyle='-')
        ax.set_title(f'Distribuce {col}')
    
    # Uložení, pokud je vyžádáno
    if save_path:
        plt.tight_layout(rect=[0, 0, 1, 0.92])
        plt.savefig(save_path)
        print(f"Dashboard uložen do {save_path}")
    
    plt.tight_layout(rect=[0, 0, 1, 0.92])
    plt.show()

# Vytvoření a zobrazení dashboardu
create_summary_dashboard(df, "Souhrnný přehled datasetu Iris", "iris_dashboard.png")
```

## 8. Praktické aplikace

Aplikace souhrnných statistik na reálné analytické scénáře.

```python
# Příklad 1: Analýza kontroly kvality
# Simulujme nějaká výrobní data
np.random.seed(42)
n_samples = 200
manufacturing_data = pd.DataFrame({
    'batch': np.repeat(range(1, 21), 10),
    'temperature': np.random.normal(100, 5, n_samples),
    'pressure': np.random.normal(50, 3, n_samples),
    'defect_rate': np.random.uniform(0, 0.1, n_samples)
})

# Report kontroly kvality
qc_report = manufacturing_data.groupby('batch').agg({
    'temperature': ['mean', 'std', 'min', 'max'],
    'pressure': ['mean', 'std', 'min', 'max'],
    'defect_rate': ['mean', 'sum']
})

print("Report kontroly kvality:")
print(qc_report.head())

# Vizualizace metrik kvality v čase (dávky)
plt.figure(figsize=(14, 8))

plt.subplot(3, 1, 1)
sns.lineplot(x='batch', y='temperature', data=manufacturing_data, ci='sd')
plt.title('Teplota podle dávky se směrodatnou odchylkou')

plt.subplot(3, 1, 2)
sns.lineplot(x='batch', y='pressure', data=manufacturing_data, ci='sd')
plt.title('Tlak podle dávky se směrodatnou odchylkou')

plt.subplot(3, 1, 3)
batch_defects = manufacturing_data.groupby('batch')['defect_rate'].sum().reset_index()
sns.barplot(x='batch', y='defect_rate', data=batch_defects)
plt.title('Celková míra defektů podle dávky')
plt.axhline(batch_defects['defect_rate'].mean(), color='r', linestyle='--', label='Průměr')
plt.legend()

plt.tight_layout()
plt.show()

# Příklad 2: Analýza A/B testování
# Simulace dat A/B testu pro webovou stránku
np.random.seed(123)
n_users = 1000
ab_test_data = pd.DataFrame({
    'user_id': range(1, n_users + 1),
    'group': np.random.choice(['A', 'B'], n_users),
    'conversion': np.random.binomial(1, 0.1, n_users),  # základní 10% konverze
    'time_spent': np.random.exponential(100, n_users)  # čas v sekundách
})

# Přidání efektu: Skupina B má vyšší konverzi a čas strávený na stránce
ab_test_data.loc[ab_test_data['group'] == 'B', 'conversion'] = np.random.binomial(
    1, 0.15, sum(ab_test_data['group'] == 'B'))  # 15% konverze pro B
ab_test_data.loc[ab_test_data['group'] == 'B', 'time_spent'] *= 1.2  # o 20% více stráveného času

# Generování reportu A/B testu
ab_stats = ab_test_data.groupby('group').agg({
    'user_id': 'count',
    'conversion': ['mean', 'sum'],
    'time_spent': ['mean', 'median', 'std']
})
print("\nVýsledky A/B testu:")
print(ab_stats)

# Statistická významnost pro míry konverze
from scipy.stats import chi2_contingency

# Vytvoření kontingenční tabulky
contingency = pd.crosstab(ab_test_data['group'], ab_test_data['conversion'])
chi2, p, dof, expected = chi2_contingency(contingency)
print(f"\nChí-kvadrát test pro míry konverze: chi2={chi2:.4f}, p-hodnota={p:.6f}")

# Statistická významnost pro čas strávený na stránce
t_stat, p_value = stats.ttest_ind(
    ab_test_data[ab_test_data['group'] == 'A']['time_spent'],
    ab_test_data[ab_test_data['group'] == 'B']['time_spent']
)
print(f"T-test pro čas strávený na stránce: t-stat={t_stat:.4f}, p-hodnota={p:.6f}")

# Vizualizace výsledků A/B testu
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(14, 6))

# Míry konverze
conv_data = ab_test_data.groupby('group')['conversion'].mean()
ax1.bar(conv_data.index, conv_data.values, yerr=[0, 0])
ax1.set_title('Míra konverze podle skupiny')
ax1.set_ylabel('Míra konverze')
ax1.set_ylim(0, 0.2)
for i, v in enumerate(conv_data.values):
    ax1.text(i, v + 0.01, f"{v:.3f}", ha='center')

# Čas strávený na stránce
sns.boxplot(x='group', y='time_spent', data=ab_test_data, ax=ax2)
ax2.set_title('Čas strávený podle skupiny')
ax2.set_ylabel('Čas strávený (sekundy)')

plt.tight_layout()
plt.show()
```

## 9. Shrnutí

Souhrnné statistiky a reportování jsou základními složkami datové vědy, které:

1. **Kondenzují velké datové sady**: Extrahují klíčové poznatky z obrovského množství dat
2. **Komunikují zjištění**: Prezentují výsledky ve formátech přístupných různým zúčastněným stranám
3. **Řídí rozhodování**: Poskytují kvantitativní důkazy pro rozhodnutí založená na datech
4. **Identifikují vzory**: Odhalují trendy, vztahy a anomálie v rámci dat
5. **Porovnávají skupiny**: Zvýrazňují rozdíly mezi subpopulacemi nebo kategoriemi
6. **Sledují změny**: Monitorují metriky v průběhu času pro sledování změn

Efektivní statistické reportování kombinuje:
- Jasné numerické shrnutí
- Vhodné vizualizace
- Testy statistické významnosti
- Formátování přizpůsobené cílové skupině
- Akční poznatky odvozené z dat

S nástroji a technikami popsanými v tomto modulu můžete vytvářet komplexní statistické reporty, které přeměňují surová data na cenné poznatky pro informované rozhodování.
