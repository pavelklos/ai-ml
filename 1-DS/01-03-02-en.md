## Data Type Optimization

One of the most effective ways to reduce memory usage is to optimize data types.

```python
def optimize_dtypes(df):
    """Optimize data types to reduce memory usage."""
    df_optimized = df.copy()
    
    # Optimize integers
    int_cols = df_optimized.select_dtypes(include=['int']).columns
    for col in int_cols:
        col_min = df_optimized[col].min()
        col_max = df_optimized[col].max()
        
        if col_min >= 0:  # Unsigned integers
            if col_max < 2**8:
                df_optimized[col] = df_optimized[col].astype(np.uint8)
            elif col_max < 2**16:
                df_optimized[col] = df_optimized[col].astype(np.uint16)
            elif col_max < 2**32:
                df_optimized[col] = df_optimized[col].astype(np.uint32)
        else:  # Signed integers
            if col_min > -2**7 and col_max < 2**7:
                df_optimized[col] = df_optimized[col].astype(np.int8)
            elif col_min > -2**15 and col_max < 2**15:
                df_optimized[col] = df_optimized[col].astype(np.int16)
            elif col_min > -2**31 and col_max < 2**31:
                df_optimized[col] = df_optimized[col].astype(np.int32)
    
    # Optimize floats
    float_cols = df_optimized.select_dtypes(include=['float']).columns
    for col in float_cols:
        df_optimized[col] = df_optimized[col].astype(np.float32)
    
    return df_optimized

# Optimize our sample DataFrame
df_optimized = optimize_dtypes(df)

# Check memory usage after optimization
optimized_memory = get_memory_usage(df_optimized, detail=True)
print(f"\nTotal memory usage after optimization: {optimized_memory:.2f} MB")
print(f"Memory reduction: {(1 - optimized_memory/initial_memory)*100:.2f}%")

# Compare data types before and after optimization
print("\nData types before optimization:")
print(df.dtypes)
print("\nData types after optimization:")
print(df_optimized.dtypes)
```

```python
def process_in_chunks(filename, chunksize=100000):
    # Set up result storage
    results = []
    
    # Process file in chunks
    for i, chunk in enumerate(pd.read_csv(filename, chunksize=chunksize)):
        # Example processing: calculate mean of numeric columns
        result = chunk.select_dtypes(include=[np.number]).mean()
        results.append(result)
        print(f"Processed chunk {i+1} with {len(chunk)} rows")
    
    # Combine results from all chunks
    final_result = pd.concat(results, axis=1).mean(axis=1)
    return final_result
```

## Using Sparse Data Structures

For data with many zeros or NaN values, sparse data structures can save significant memory.

```python
# Create a sparse DataFrame example (mostly zeros)
def create_sparse_example(n_rows=100000, n_cols=1000, sparsity=0.99):
    # Create a DataFrame with mostly zeros
    df = pd.DataFrame(np.random.random(size=(n_rows, n_cols)))
    # Set most values to zero to create sparsity
    mask = np.random.random(size=(n_rows, n_cols)) < sparsity
    df[mask] = 0
    return df

# Create a sparse example (smaller dimensions for demonstration)
sparse_df = create_sparse_example(10000, 100)
```

## Creating Custom Accessors

Custom accessors allow you to extend pandas functionality in an organized, object-oriented way.

```python
@pd.api.extensions.register_dataframe_accessor("stats")
class StatsAccessor:
    def __init__(self, pandas_obj):
        self._obj = pandas_obj
    
    def describe_enhanced(self):
        """Enhanced describe method with additional statistics"""
        # Get regular describe
        desc = self._obj.describe()
        
        # Add more metrics
        numeric_cols = self._obj.select_dtypes(include=[np.number]).columns
        for col in numeric_cols:
            desc.loc['skew', col] = self._obj[col].skew()
            desc.loc['kurtosis', col] = self._obj[col].kurtosis()
            desc.loc['iqr', col] = self._obj[col].quantile(0.75) - self._obj[col].quantile(0.25)
            desc.loc['cv', col] = self._obj[col].std() / self._obj[col].mean() if self._obj[col].mean() != 0 else np.nan
        
        return desc
    
    def outliers(self, column, method='iqr', threshold=1.5):
        """Identify outliers in a column
        
        Parameters:
        -----------
        column : str
            Column name to check for outliers
        method : str, default 'iqr'
            Method to use ('iqr' or 'zscore')
        threshold : float, default 1.5
            Threshold for outlier detection (for IQR method)
        
        Returns:
        --------
        DataFrame with outlier values
        """
        if method == 'iqr':
            Q1 = self._obj[column].quantile(0.25)
            Q3 = self._obj[column].quantile(0.75)
            IQR = Q3 - Q1
            
            lower_bound = Q1 - threshold * IQR
            upper_bound = Q3 + threshold * IQR
            
            outliers = self._obj[(self._obj[column] < lower_bound) | (self._obj[column] > upper_bound)]
            return outliers
        
        elif method == 'zscore':
            from scipy import stats
            z_scores = stats.zscore(self._obj[column])
            outliers = self._obj[abs(z_scores) > threshold]
            return outliers
        
        else:
            raise ValueError("Method must be one of 'iqr' or 'zscore'")

@pd.api.extensions.register_dataframe_accessor("finance")
class FinanceAccessor:
    def __init__(self, pandas_obj):
        self._obj = pandas_obj
    
    def pct_change_enhanced(self, periods=1):
        """Calculate percentage change with additional metrics"""
        # Assume we have a financial DataFrame with columns like 'close', 'open', etc.
        pct_change = self._obj.pct_change(periods)
        return pct_change
    
    def moving_average(self, column, window=14, type='simple'):
        """Calculate moving average
        
        Parameters:
        -----------
        column : str
            Column to calculate MA on
        window : int, default 14
            Window size for moving average
        type : str, default 'simple'
            Type of moving average ('simple', 'exponential', or 'weighted')
        """
        if type == 'simple':
            return self._obj[column].rolling(window=window).mean()
        elif type == 'exponential':
            return self._obj[column].ewm(span=window).mean()
        elif type == 'weighted':
            weights = np.arange(1, window + 1)
            return self._obj[column].rolling(window=window).apply(
                lambda x: np.sum(weights * x) / weights.sum(), raw=True)
        else:
            raise ValueError("Type must be one of 'simple', 'exponential', or 'weighted'")
```

```python
# Create a larger sample DataFrame for advanced operations
def create_advanced_df(n_rows=500000):
    df = pd.DataFrame({
        'id': range(n_rows),
        'group': np.random.choice(['A', 'B', 'C', 'D', 'E'], n_rows),

# Memory-efficient groupby operations
# Instead of storing all intermediate results, we can use agg directly

    'value1': ['mean', 'std', 'min', 'max'],
    'value2': ['mean', 'std', 'min', 'max']
})

# Optimized apply operations using numba
try:
    from numba import jit
    
    # Define a function to be optimized with numba
    @jit(nopython=True)
    def numba_function(x, y):
        result = 0
        for i in range(len(x)):
            result += x[i] * y[i]
        return result / len(x)
    
    # Apply the optimized function
    t1 = pd.Timestamp.now()
    adv_df['computed'] = numba_function(
        adv_df['value1'].values, 
        adv_df['value2'].values
    )
```

```python
def optimize_taxi_dataset(df):
```

## Summary

This tutorial covered advanced pandas techniques focusing on memory optimization and extending pandas functionality:

1. **Memory Optimization Techniques**:
   - Profiling memory usage with `memory_usage(deep=True)` and dedicated profilers
   - Optimizing data types using pandas' `to_numeric()` with 'downcast' and NumPy's smaller dtypes
   - Using categorical data for string columns with repetitive values
   - Processing large datasets in chunks to reduce memory footprint
   - Leveraging sparse data structures for datasets with many zeros or missing values

2. **Custom Accessors**:
   - Extending pandas functionality in an organized, object-oriented way
   - Creating domain-specific methods for common tasks
   - Building custom analysis tools that integrate seamlessly with pandas
   - Organizing related functionality into coherent namespaces

3. **Advanced Operations**:
   - Efficient groupby aggregations for large datasets
   - Using `eval()` for optimized computations
   - Optimizing time-based operations with DatetimeIndex
   - Accelerating loops and intensive calculations with numba

By implementing these techniques, you can handle larger datasets within memory constraints and build more maintainable code through custom extensions to pandas. These approaches are particularly valuable when working with data that pushes the limits of your hardware or when building domain-specific data analysis tools.

Remember to always profile your code before and after optimization to ensure you're achieving the expected benefits, as premature optimization can sometimes lead to more complex code without significant gains.
