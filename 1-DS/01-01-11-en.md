## 1. Introduction to Data Visualization with Matplotlib

Data visualization is a crucial part of data science as it helps us understand patterns, trends, and outliers in our data. Matplotlib is one of the most popular plotting libraries in Python that provides a wide range of tools for creating static, animated, and interactive visualizations.

```python
# Import libraries
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

# To ensure plots appear in Jupyter notebook
%matplotlib inline

# For better plot appearance
plt.style.use('seaborn-v0_8-whitegrid')
```

## 2. Setting up Matplotlib

Matplotlib offers two primary interfaces:
- The MATLAB-like state-based interface through `pyplot`
- The object-oriented interface through Figure and Axes objects

```python
# Basic pyplot interface
plt.figure(figsize=(10, 6))  # Create a figure with specified size
plt.plot([1, 2, 3, 4], [1, 4, 9, 16])
plt.title('Simple Plot')
plt.xlabel('X-axis')
plt.ylabel('Y-axis')
plt.show()

# Object-oriented interface
fig, ax = plt.subplots(figsize=(10, 6))
ax.plot([1, 2, 3, 4], [1, 4, 9, 16])
ax.set_title('Object-Oriented Plot')
ax.set_xlabel('X-axis')
ax.set_ylabel('Y-axis')
plt.show()
```

## 3. Basic Plot Types

Matplotlib can create various types of plots to visualize different aspects of your data.

### Line Plots

```python
# Generate data
x = np.linspace(0, 10, 100)
y1 = np.sin(x)
y2 = np.cos(x)

# Create a line plot
plt.figure(figsize=(10, 6))
plt.plot(x, y1, label='sin(x)', color='blue', linestyle='-')
plt.plot(x, y2, label='cos(x)', color='red', linestyle='--')
plt.title('Sine and Cosine Functions')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()
plt.grid(True)
plt.show()
```

### Bar Charts

```python
# Create a bar chart
categories = ['A', 'B', 'C', 'D', 'E']
values = [25, 40, 30, 55, 15]

plt.figure(figsize=(10, 6))
plt.bar(categories, values, color='skyblue', edgecolor='black')
plt.title('Bar Chart Example')
plt.xlabel('Categories')
plt.ylabel('Values')
plt.show()

# Horizontal bar chart
plt.figure(figsize=(10, 6))
plt.barh(categories, values, color='lightgreen', edgecolor='black')
plt.title('Horizontal Bar Chart')
plt.xlabel('Values')
plt.ylabel('Categories')
plt.show()
```

### Scatter Plots

```python
# Generate random data
np.random.seed(42)
x = np.random.rand(50)
y = np.random.rand(50)
colors = np.random.rand(50)
sizes = 1000 * np.random.rand(50)

# Create a scatter plot
plt.figure(figsize=(10, 6))
plt.scatter(x, y, c=colors, s=sizes, alpha=0.6, cmap='viridis')
plt.title('Scatter Plot Example')
plt.xlabel('X values')
plt.ylabel('Y values')
plt.colorbar(label='Color')
plt.show()
```

### Histograms

```python
# Generate normal distribution data
data = np.random.normal(0, 1, 1000)

# Create a histogram
plt.figure(figsize=(10, 6))
plt.hist(data, bins=30, color='purple', alpha=0.7, edgecolor='black')
plt.title('Histogram Example')
plt.xlabel('Value')
plt.ylabel('Frequency')
plt.grid(True, alpha=0.3)
plt.show()
```

### Pie Charts

```python
# Data for pie chart
labels = ['A', 'B', 'C', 'D', 'E']
sizes = [25, 20, 30, 15, 10]
explode = (0, 0.1, 0, 0, 0)  # Explode the 2nd slice

# Create a pie chart
plt.figure(figsize=(10, 6))
plt.pie(sizes, explode=explode, labels=labels, autopct='%1.1f%%',
        shadow=True, startangle=90)
plt.axis('equal')  # Equal aspect ratio ensures that pie is drawn as a circle
plt.title('Pie Chart Example')
plt.show()
```

## 4. Customizing Plots

Matplotlib provides extensive options for customizing your visualizations.

```python
# Setting colors, markers, and line styles
plt.figure(figsize=(10, 6))
plt.plot([1, 2, 3, 4], [1, 4, 9, 16], 'ro-', linewidth=2, markersize=8)
plt.plot([1, 2, 3, 4], [1, 2, 3, 4], 'bs--', linewidth=2, markersize=8)
plt.title('Customized Line Plot')
plt.xlabel('X-axis')
plt.ylabel('Y-axis')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Setting axis limits
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.xlim(0, 5)
plt.ylim(-1.5, 1.5)
plt.title('Plot with Custom Axis Limits')
plt.grid(True)
plt.show()

# Using different color maps
plt.figure(figsize=(10, 6))
plt.scatter(x, y, c=colors, s=sizes, alpha=0.6, cmap='plasma')
plt.colorbar(label='Color Value')
plt.title('Scatter Plot with Custom Colormap')
plt.show()

# Adding a background color
plt.figure(figsize=(10, 6), facecolor='#f0f0f0')
plt.plot(x, y1, color='navy')
plt.title('Plot with Custom Background')
plt.xlabel('X-axis')
plt.ylabel('Y-axis')
plt.grid(True, linestyle='--', alpha=0.7)
plt.tight_layout()
plt.show()
```

## 5. Multiple Plots and Subplots

For more complex analysis, you can create multiple plots in a single figure.

```python
# Creating a simple grid of subplots
fig, axs = plt.subplots(2, 2, figsize=(12, 10))

# First subplot
axs[0, 0].plot(x, y1)
axs[0, 0].set_title('Sine Function')
axs[0, 0].set_xlabel('x')
axs[0, 0].set_ylabel('sin(x)')
axs[0, 0].grid(True)

# Second subplot
axs[0, 1].plot(x, y2, color='red')
axs[0, 1].set_title('Cosine Function')
axs[0, 1].set_xlabel('x')
axs[0, 1].set_ylabel('cos(x)')
axs[0, 1].grid(True)

# Third subplot
axs[1, 0].scatter(np.random.rand(50), np.random.rand(50))
axs[1, 0].set_title('Random Scatter')
axs[1, 0].set_xlabel('X')
axs[1, 0].set_ylabel('Y')

# Fourth subplot
axs[1, 1].bar(categories, values)
axs[1, 1].set_title('Bar Chart')
axs[1, 1].set_xlabel('Categories')
axs[1, 1].set_ylabel('Values')

plt.tight_layout()
plt.show()

# Creating subplots with different sizes (GridSpec)
import matplotlib.gridspec as gridspec

fig = plt.figure(figsize=(12, 8))
gs = gridspec.GridSpec(2, 3)

# Main plot takes up first row
ax1 = fig.add_subplot(gs[0, :])
ax1.plot(x, y1)
ax1.set_title('Main Plot (Full Row)')

# Three smaller plots in second row
ax2 = fig.add_subplot(gs[1, 0])
ax2.plot(x, y2)
ax2.set_title('Plot 1')

ax3 = fig.add_subplot(gs[1, 1])
ax3.scatter(x[:20], y1[:20])
ax3.set_title('Plot 2')

ax4 = fig.add_subplot(gs[1, 2])
ax4.bar(categories, values)
ax4.set_title('Plot 3')

plt.tight_layout()
plt.show()
```

## 6. Adding Annotations

Annotations can help highlight important parts of your visualizations.

```python
# Creating a plot with annotations
plt.figure(figsize=(10, 6))
plt.plot(x, y1)

# Add a point and annotate it
max_x = x[np.argmax(y1)]
max_y = np.max(y1)
plt.plot(max_x, max_y, 'ro')
plt.annotate('Maximum value', 
             xy=(max_x, max_y),
             xytext=(max_x-1, max_y+0.1),
             arrowprops=dict(facecolor='black', shrink=0.05, width=1.5),
             fontsize=12)

# Adding text to the plot
plt.text(5, 0, "sin(x) crosses zero multiple times", 
         fontsize=12, ha='center', bbox=dict(facecolor='yellow', alpha=0.3))

plt.title('Sine Function with Annotations')
plt.xlabel('x')
plt.ylabel('sin(x)')
plt.grid(True)
plt.show()

# Adding a vertical and horizontal line
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.axhline(y=0, color='r', linestyle='-', alpha=0.3)
plt.axvline(x=5, color='g', linestyle='--', alpha=0.3)
plt.title('Plot with Reference Lines')
plt.grid(True)
plt.show()
```

## 7. Saving and Displaying Visualizations

Matplotlib allows you to save your visualizations in various formats.

```python
# Create a simple plot
plt.figure(figsize=(10, 6))
plt.plot(x, y1, label='sin(x)')
plt.plot(x, y2, label='cos(x)')
plt.title('Trigonometric Functions')
plt.xlabel('x')
plt.ylabel('y')
plt.legend()
plt.grid(True)

# Save in different formats
plt.savefig('trig_functions.png', dpi=300, bbox_inches='tight')
plt.savefig('trig_functions.pdf', bbox_inches='tight')
plt.savefig('trig_functions.svg', bbox_inches='tight')

# Display the plot
plt.show()
```

## 8. Statistical Visualization

Matplotlib can be used to create common statistical visualizations.

```python
# Generate some data
np.random.seed(42)
data1 = np.random.normal(0, 1, 1000)
data2 = np.random.normal(2, 1.5, 1000)

# Box Plot
plt.figure(figsize=(10, 6))
plt.boxplot([data1, data2], labels=['Distribution 1', 'Distribution 2'])
plt.title('Box Plot Comparison')
plt.ylabel('Values')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Violin Plot
plt.figure(figsize=(10, 6))
plt.violinplot([data1, data2])
plt.title('Violin Plot Comparison')
plt.xticks([1, 2], ['Distribution 1', 'Distribution 2'])
plt.ylabel('Values')
plt.grid(True, linestyle='--', alpha=0.7)
plt.show()

# Error Bars
x_data = np.array([1, 2, 3, 4, 5])
y_data = np.array([1, 2, 4, 8, 16])
y_error = np.array([0.1, 0.2, 0.4, 0.8, 1.6])

plt.figure(figsize=(10, 6))
plt.errorbar(x_data, y_data, yerr=y_error, fmt='o-', capsize=5, ecolor='red', markeredgecolor='blue')
plt.title('Plot with Error Bars')
plt.xlabel('X')
plt.ylabel('Y')
plt.grid(True)
plt.show()

# Using pandas integration with matplotlib for statistical visualization
df = pd.DataFrame({
    'Group A': np.random.normal(0, 1, 100),
    'Group B': np.random.normal(2, 1, 100),
    'Group C': np.random.normal(-1, 1.5, 100)
})

# Histogram of multiple distributions
plt.figure(figsize=(10, 6))
plt.hist(df, bins=20, alpha=0.7)
plt.title('Histogram of Multiple Distributions')
plt.xlabel('Value')
plt.ylabel('Frequency')
plt.legend(df.columns)
plt.grid(True, alpha=0.3)
plt.show()
```

## 9. Best Practices for Data Visualization

Creating effective visualizations requires following some best practices:

```python
# 1. Use clear titles and labels
plt.figure(figsize=(10, 6))
plt.plot(x, y1)
plt.title('Sine Wave Function Over Time', fontsize=14)
plt.xlabel('Time (seconds)', fontsize=12)
plt.ylabel('Amplitude', fontsize=12)
plt.grid(True)
plt.show()

# 2. Choose appropriate colors, especially for colorblind audiences
from matplotlib.cm import viridis, plasma, inferno, magma, cividis

plt.figure(figsize=(15, 8))
for i, cmap in enumerate([viridis, plasma, inferno, magma, cividis]):
    plt.subplot(1, 5, i+1)
    plt.scatter(np.random.rand(50), np.random.rand(50), c=np.random.rand(50), cmap=cmap)
    plt.title(cmap.name)
plt.tight_layout()
plt.show()

# 3. Avoid chart junk (unnecessary elements)
# Bad example with too many elements
plt.figure(figsize=(10, 6))
plt.plot(x, y1, 'b-')
plt.title('Cluttered Plot (Bad Practice)')
plt.grid(True, which='both', axis='both')
plt.minorticks_on()
plt.grid(True, which='minor', alpha=0.3)
for i in range(0, 11, 1):
    plt.axvline(x=i, color='red', alpha=0.1)
    plt.axhline(y=i/10-0.5, color='green', alpha=0.1)
plt.show()

# Good example with clean design
plt.figure(figsize=(10, 6))
plt.plot(x, y1, 'b-')
plt.title('Clean Plot (Good Practice)')
plt.grid(True, alpha=0.3)
plt.show()

# 4. Use appropriate scales
# Linear scale
plt.figure(figsize=(10, 6))
plt.subplot(1, 2, 1)
plt.plot(np.arange(1, 1001), np.arange(1, 1001))
plt.title('Linear Scale')
plt.xlabel('X')
plt.ylabel('Y')
plt.grid(True)

# Log scale for the same data
plt.subplot(1, 2, 2)
plt.loglog(np.arange(1, 1001), np.arange(1, 1001))
plt.title('Log-Log Scale')
plt.xlabel('X (log)')
plt.ylabel('Y (log)')
plt.grid(True, which='both')
plt.tight_layout()
plt.show()

# 5. Make it accessible (adequate font sizes, contrast)
plt.figure(figsize=(10, 6))
plt.plot(x, y1, linewidth=2.5, color='navy')
plt.title('Accessible Plot', fontsize=16)
plt.xlabel('X-axis', fontsize=14)
plt.ylabel('Y-axis', fontsize=14)
plt.xticks(fontsize=12)
plt.yticks(fontsize=12)
plt.grid(True, alpha=0.4)
plt.show()
```
