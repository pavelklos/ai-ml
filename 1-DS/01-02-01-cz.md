## Úvod do pokročilých možností Pandas

Pandas je mocná knihovna pro manipulaci s daty, ale mnoho uživatelů využívá jen zlomek jejích možností. Tento průvodce zkoumá pokročilé techniky, které vám pomohou psát efektivnější a výstižnější kód pro analýzu dat.

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import time
%matplotlib inline

# Vytvoření ukázkového datasetu pro naše příklady
np.random.seed(42)
dates = pd.date_range('20220101', periods=1000)
df = pd.DataFrame({
    'date': dates,
    'category': np.random.choice(['A', 'B', 'C', 'D'], size=1000),
    'subcategory': np.random.choice(['X', 'Y', 'Z'], size=1000),
    'value': np.random.randn(1000) * 100,
    'quantity': np.random.randint(1, 100, size=1000),
    'is_active': np.random.choice([True, False], size=1000),
    'id': np.arange(1000)
})

df['date_only'] = df['date'].dt.date
print(df.head())
```

## Řetězení metod pro čistší kód

Řetězení metod umožňuje aplikovat více operací postupně, což činí váš kód čitelnějším a stručnějším. Tento vzor snižuje potřebu dočasných proměnných.

```python
# Tradiční přístup s dočasnými proměnnými
df_filtered = df[df['value'] > 0]
df_grouped = df_filtered.groupby('category')
df_mean = df_grouped['value'].mean()
df_sorted = df_mean.sort_values(ascending=False)
print("Výsledek tradičního přístupu:")
print(df_sorted)

# Přístup s řetězením metod
result = (df[df['value'] > 0]
          .groupby('category')['value']
          .mean()
          .sort_values(ascending=False)
         )
print("\nVýsledek řetězení metod:")
print(result)

# Složitější příklad s více operacemi
advanced_result = (df
                  .assign(value_squared=lambda x: x['value']**2)
                  .query('value > 0 & quantity > 50')
                  .groupby(['category', 'subcategory'])
                  .agg({'value': ['mean', 'sum'], 
                        'value_squared': 'mean', 
                        'quantity': 'sum'})
                  .round(2)
                  .sort_values(('value', 'sum'), ascending=False)
                 )
print("\nPříklad složitého řetězení metod:")
print(advanced_result.head())
```

## Pokročilé operace GroupBy

Funkce GroupBy v Pandas je extrémně výkonná. Zde prozkoumáme pokročilé techniky seskupování, vlastní agregace a transformační operace.

```python
# Základní groupby s více agregačními funkcemi
result = df.groupby('category').agg({
    'value': ['min', 'max', 'mean', 'std'],
    'quantity': ['sum', 'count'],
    'is_active': 'sum'  # Počítá hodnoty True
})
print("Vícenásobné agregace:")
print(result.head())

# Pojmenované agregace pro přehlednější výstup
result = df.groupby('category').agg(
    min_value=('value', 'min'),
    max_value=('value', 'max'),
    avg_value=('value', 'mean'),
    total_quantity=('quantity', 'sum'),
    active_count=('is_active', 'sum')
)
print("\nPojmenované agregace:")
print(result.head())

# Vlastní agregační funkce
def q95(x):
    return x.quantile(0.95)

result = df.groupby(['category', 'subcategory']).agg({
    'value': [q95, lambda x: x.quantile(0.75) - x.quantile(0.25)],  # Vlastní funkce
    'quantity': 'sum'
})
print("\nVlastní agregace:")
print(result.head())

# GroupBy s transformací pro zarovnání s původními daty
df['value_centered'] = df.groupby('category')['value'].transform(
    lambda x: x - x.mean()
)
print("\nTransformovaná data (prvních 5 řádků):")
print(df[['category', 'value', 'value_centered']].head())

# Filtrování skupin na základě podmínky
filtered = df.groupby('category').filter(lambda x: x['value'].mean() > 0)
print(f"\nFiltrovaná data, tvar: {filtered.shape} (z původních {df.shape})")
```

## Okenní funkce a klouzavé výpočty

Okenní funkce umožňují provádět výpočty přes klouzavé okno dat, což je zvláště užitečné pro analýzu časových řad.

```python
# Nastavení data jako indexu pro operace s časovými řadami
ts_df = df.set_index('date').sort_index()

# Klouzavý průměr s oknem 7 dnů
rolling_7d = ts_df.groupby('category')['value'].rolling('7D').mean()
print("7denní klouzavý průměr podle kategorie:")
print(rolling_7d.head(10))

# Výpočty s rostoucím oknem (kumulativní)
ts_df['cumulative_sum'] = ts_df.groupby('category')['value'].expanding().sum().reset_index(level=0, drop=True)
print("\nKumulativní součet podle kategorie:")
print(ts_df[['category', 'value', 'cumulative_sum']].head())

# Klouzavé okno s vlastní funkcí
ts_df['rolling_efficiency'] = (ts_df.groupby('category')['value']
                              .rolling(10, min_periods=5)
                              .apply(lambda x: x.max() - x.min())
                              .reset_index(level=0, drop=True))

# Posouvání dat pro analýzu lead/lag
ts_df['prev_day_value'] = ts_df.groupby('category')['value'].shift(1)
ts_df['value_change'] = ts_df['value'] - ts_df['prev_day_value']
print("\nZměna hodnoty den po dni:")
print(ts_df[['category', 'value', 'prev_day_value', 'value_change']].head(10))
```

## Pokročilé techniky transformace dat

Tyto techniky pomáhají přetvarovat a restrukturalizovat data pro různé analytické potřeby.

```python
# Kontingenční tabulky
pivot_table = pd.pivot_table(df, 
                           values='value', 
                           index='category',
                           columns='subcategory',
                           aggfunc=['mean', 'sum', 'count'],
                           fill_value=0)
print("Příklad kontingenční tabulky:")
print(pivot_table.head())

# Roztavení dat pomocí melt
melted = pd.melt(df, 
                id_vars=['id', 'category'], 
                value_vars=['value', 'quantity'],
                var_name='metric',
                value_name='measurement')
print("\nRoztavená data:")
print(melted.head())

# Rozbalení seznamů do řádků
df_with_lists = pd.DataFrame({
    'id': [1, 2, 3],
    'values': [[1, 2, 3], [4, 5], [6, 7, 8, 9]]
})
exploded = df_with_lists.explode('values')
print("\nRozbalená data:")
print(exploded)

# Pokročilé stack a unstack
stacked = df.set_index(['category', 'subcategory'])[['value', 'quantity']]

print("\nOdstohovaná data:")

```

## Práce s víceúrovňovými indexy

Víceúrovňové indexy (hierarchické indexování) umožňují mít více úrovní indexů, což umožňuje sofistikovanější analýzu dat.

```python
# Vytvoření DataFrame s víceúrovňovým indexem
multi_idx = pd.MultiIndex.from_product([['A', 'B', 'C'], ['X', 'Y', 'Z']], 
                                       names=['category', 'subcategory'])
multi_df = pd.DataFrame(np.random.randn(9, 3), 
                       index=multi_idx, 
                       columns=['value1', 'value2', 'value3'])
print("DataFrame s víceúrovňovým indexem:")
print(multi_df)

# Přístup k datům v DataFrame s víceúrovňovým indexem
print("\nVýběr jedné úrovně:")
print(multi_df.loc['A'])
print("\nVýběr konkrétních prvků:")
print(multi_df.loc[('A', 'X')])

# Výběr průřezů pomocí xs
print("\nVýběr průřezu:")
print(multi_df.xs('Y', level='subcategory'))

# Záměna úrovní
print("\nZaměněné úrovně:")
print(multi_df.swaplevel('category', 'subcategory').head())

# Použití groupby s víceúrovňovým indexem
grouped = multi_df.groupby(level=['category']).mean()
print("\nSeskupeno podle úrovně:")
print(grouped)

# Odstohování dat
unstacked = multi_df.unstack(level='subcategory')
print("\nOdstohovaná data:")
print(unstacked.head())

# Stohování zpět
print("\nZnovu stohovaná data:")
print(unstacked.stack().head())
```

## Optimalizace výkonu Pandas

Pandas může být paměťově náročný s velkými datasety. Zde jsou techniky pro zlepšení výkonu.

```python
# Vytvoření většího datasetu pro demonstraci výkonu
large_df = pd.DataFrame(np.random.randn(100000, 10))

# Měření času operace
def time_operation(operation, df):
    start = time.time()
    result = operation(df)
    end = time.time()
    print(f"Operace trvala {end - start:.4f} sekund")
    return result

# 1. Použití správných datových typů
print("Využití paměti před optimalizací:", large_df.memory_usage().sum() / 1e6, "MB")
optimized_df = large_df.copy()
for col in optimized_df.columns:
    optimized_df[col] = pd.to_numeric(optimized_df[col], downcast='float')
print("Využití paměti po optimalizaci:", optimized_df.memory_usage().sum() / 1e6, "MB")

# 2. Vektorizované operace vs. apply
def slow_way(df):
    return df.apply(lambda x: x * 2)

def fast_way(df):
    return df * 2

print("\nPorovnání výkonu:")
_ = time_operation(slow_way, large_df.iloc[:1000])  # Omezeno na 1000 řádků pro apply
_ = time_operation(fast_way, large_df)

# 3. Použití query() pro filtrování
def filter_normal(df):
    return df[(df[0] > 0) & (df[1] < 0)]

def filter_query(df):
    return df.query('`0` > 0 and `1` < 0')

print("\nPorovnání filtrování:")
_ = time_operation(filter_normal, large_df)
_ = time_operation(filter_query, large_df)

# 4. Použití kategorií pro textová data
cat_df = pd.DataFrame({
    'id': range(100000),
    'category': np.random.choice(['A', 'B', 'C', 'D', 'E'], size=100000)
})

print("\nVyužití paměti s řetězci:", cat_df.memory_usage().sum() / 1e6, "MB")
cat_df['category'] = cat_df['category'].astype('category')
print("Využití paměti s kategoriemi:", cat_df.memory_usage().sum() / 1e6, "MB")
```

## Vlastní funkce s apply(), map() a applymap()

Tyto metody umožňují aplikovat vlastní funkce na data na různých úrovních.

```python
# Aplikace funkcí na sloupce nebo řádky
def column_stats(col):
    return pd.Series({
        'mean': col.mean(),
        'median': col.median(),
        'std': col.std(),
        'skew': col.skew()
    })

stats = df[['value', 'quantity']].apply(column_stats)
print("Statistiky sloupců pomocí apply():")
print(stats)

# Aplikace na řádky s axis=1
def row_processing(row):
    if row['category'] == 'A':
        return row['value'] * 2
    else:
        return row['value'] / 2

df['processed_value'] = df.apply(row_processing, axis=1)
print("\nZpracování řádků pomocí apply():")
print(df[['category', 'value', 'processed_value']].head())

# Map pro operace se sériemi
category_mapping = {'A': 'Skupina 1', 'B': 'Skupina 1', 'C': 'Skupina 2', 'D': 'Skupina 2'}
df['category_group'] = df['category'].map(category_mapping)
print("\nMapování kategorií:")
print(df[['category', 'category_group']].drop_duplicates())

# Použití applymap pro operace po prvcích na DataFrames
def format_value(x):
    if isinstance(x, (int, float)):
        return f"{x:.2f}" if x >= 0 else f"({abs(x):.2f})"
    return x

formatted = df[['value', 'quantity']].head().applymap(format_value)
print("\nFormátované hodnoty pomocí applymap():")
print(formatted)

# Vektorizované vlastní funkce pro lepší výkon
def vectorized_custom_func(s):
    return np.where(s > 0, s * 1.1, s * 0.9)

df['value_adjusted'] = vectorized_custom_func(df['value'])
print("\nVektorizovaná vlastní funkce:")
print(df[['value', 'value_adjusted']].head())
```

## Pokročilé techniky čištění dat

Tyto techniky pomáhají řešit komplexní problémy s daty nad rámec základního čištění.

```python
# Vytvoření datasetu s různými problémy s kvalitou dat
messy_df = pd.DataFrame({
    'col_A': ['Apple ', ' Banana', np.nan, 'Apple', 'ORANGE', 'orange ', None],
    'col_B': [1, 2, np.nan, '4', 5, 'Invalid', '7'],
    'col_C': pd.to_datetime(['2023-01-01', '2023-01-02', None, 'invalid date', 
                            '20230104', '2023/01/05', '01-06-2023']),
    'col_D': ['123-456-7890', '123.456.7891', np.nan, '(123) 456 7892', 
             '1234567893', 'invalid', '123-456-ABCD']
})
print("Neuspořádaný DataFrame:")
print(messy_df)

# 1. Standardizace textových dat
def clean_text(s):
    if pd.isna(s):
        return np.nan
    return s.strip().lower()

messy_df['col_A_clean'] = messy_df['col_A'].apply(clean_text)
print("\n1. Vyčištěný text:")
print(messy_df[['col_A', 'col_A_clean']])

# 2. Konverze smíšených typů
def convert_to_numeric(s):
    if pd.isna(s):
        return np.nan
    try:
        return float(s)
    except (ValueError, TypeError):
        return np.nan

messy_df['col_B_clean'] = messy_df['col_B'].apply(convert_to_numeric)
print("\n2. Převedeno na numerické hodnoty:")
print(messy_df[['col_B', 'col_B_clean']])

# 3. Vlastní parsování dat
def parse_dates(d):
    if pd.isna(d):
        return np.nan
    try:
        return pd.to_datetime(d, errors='raise')
    except:
        return np.nan

messy_df['col_C_clean'] = pd.to_datetime(messy_df['col_C'], errors='coerce')
print("\n3. Vyčištěná data:")
print(messy_df[['col_C', 'col_C_clean']])

# 4. Regulární výrazy pro čištění na základě vzorů
import re

def clean_phone(s):
    if pd.isna(s):
        return np.nan
    # Extrahuje pouze číslice
    digits = re.sub(r'[^0-9]', '', str(s))
    if len(digits) != 10:
        return np.nan
    return f"{digits[:3]}-{digits[3:6]}-{digits[6:]}"

messy_df['col_D_clean'] = messy_df['col_D'].apply(clean_phone)
print("\n4. Standardizovaná telefonní čísla:")
print(messy_df[['col_D', 'col_D_clean']])

# 5. Pokročilá detekce duplikátů s fuzzy porovnáváním
from fuzzywuzzy import process

def find_closest_match(value, choices, threshold=80):
    if pd.isna(value):
        return np.nan
    match, score = process.extractOne(value, choices)
    return match if score >= threshold else value

unique_fruits = ['apple', 'banana', 'orange']
messy_df['col_A_standardized'] = messy_df['col_A_clean'].apply(
    lambda x: find_closest_match(x, unique_fruits) if not pd.isna(x) else np.nan
)
print("\n5. Fuzzy porovnávání pro standardizaci:")
print(messy_df[['col_A', 'col_A_clean', 'col_A_standardized']])
```
