## Introduction to Advanced Pandas Capabilities

Pandas is a powerful library for data manipulation, but many users only scratch the surface of its capabilities. This guide explores advanced techniques that can help you write more efficient and expressive data analysis code.

```python
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import time
%matplotlib inline

# Sample dataset creation for our examples
np.random.seed(42)
dates = pd.date_range('20220101', periods=1000)
df = pd.DataFrame({
    'date': dates,
    'category': np.random.choice(['A', 'B', 'C', 'D'], size=1000),
    'subcategory': np.random.choice(['X', 'Y', 'Z'], size=1000),
    'value': np.random.randn(1000) * 100,
    'quantity': np.random.randint(1, 100, size=1000),

    'id': np.arange(1000)
})

df['date_only'] = df['date'].dt.date
print(df.head())
```

## Method Chaining for Cleaner Code

Method chaining allows you to apply multiple operations in sequence, making your code more readable and concise. This pattern reduces the need for intermediate variables.

```python
# Traditional approach with intermediate variables
df_filtered = df[df['value'] > 0]
df_grouped = df_filtered.groupby('category')
df_mean = df_grouped['value'].mean()
df_sorted = df_mean.sort_values(ascending=False)
print("Traditional approach result:")
print(df_sorted)

# Method chaining approach
result = (df[df['value'] > 0]
          .groupby('category')['value']
          .mean()
          .sort_values(ascending=False)
         )
print("\nMethod chaining result:")
print(result)

# More complex example with multiple operations
advanced_result = (df
                  .assign(value_squared=lambda x: x['value']**2)
                  .query('value > 0 & quantity > 50')
                  .groupby(['category', 'subcategory'])
                  .agg({'value': ['mean', 'sum'], 
                        'value_squared': 'mean', 
                        'quantity': 'sum'})
                  .round(2)
                  .sort_values(('value', 'sum'), ascending=False)
                 )
print("\nComplex method chaining example:")
print(advanced_result.head())
```

## Advanced GroupBy Operations

The GroupBy functionality in pandas is extremely powerful. Here we'll explore advanced grouping techniques, custom aggregations, and transformation operations.

```python
# Basic groupby with multiple aggregation functions
result = df.groupby('category').agg({
    'value': ['min', 'max', 'mean', 'std'],
    'quantity': ['sum', 'count'],
    'is_active': 'sum'  # Counts True values
})
print("Multiple aggregations:")
print(result.head())

# Named aggregations for clearer output
result = df.groupby('category').agg(
    min_value=('value', 'min'),
    max_value=('value', 'max'),
    avg_value=('value', 'mean'),
    total_quantity=('quantity', 'sum'),
    active_count=('is_active', 'sum')
)
print("\nNamed aggregations:")
print(result.head())

# Custom aggregation functions
def q95(x):
    return x.quantile(0.95)

result = df.groupby(['category', 'subcategory']).agg({
    'value': [q95, lambda x: x.quantile(0.75) - x.quantile(0.25)],  # Custom functions
    'quantity': 'sum'
})
print("\nCustom aggregation:")
print(result.head())

# GroupBy with transformation to align with original data
df['value_centered'] = df.groupby('category')['value'].transform(
    lambda x: x - x.mean()
)
print("\nTransformed data (first 5 rows):")
print(df[['category', 'value', 'value_centered']].head())

# Filter groups based on a condition
filtered = df.groupby('category').filter(lambda x: x['value'].mean() > 0)
print(f"\nFiltered data shape: {filtered.shape} (from original {df.shape})")
```

## Window Functions and Rolling Calculations

Window functions allow you to perform calculations across a sliding window of data, which is particularly useful for time-series analysis.

```python
# Set date as index for time-series operations
ts_df = df.set_index('date').sort_index()

# Rolling mean with a 7-day window
rolling_7d = ts_df.groupby('category')['value'].rolling('7D').mean()
print("7-day rolling average by category:")
print(rolling_7d.head(10))

# Expanding window calculations (cumulative)
ts_df['cumulative_sum'] = ts_df.groupby('category')['value'].expanding().sum().reset_index(level=0, drop=True)
print("\nCumulative sum by category:")
print(ts_df[['category', 'value', 'cumulative_sum']].head())

# Rolling window with custom function
ts_df['rolling_efficiency'] = (ts_df.groupby('category')['value']
                              .rolling(10, min_periods=5)
                              .apply(lambda x: x.max() - x.min())
                              .reset_index(level=0, drop=True))

# Shifting data for lead/lag analysis
ts_df['prev_day_value'] = ts_df.groupby('category')['value'].shift(1)
ts_df['value_change'] = ts_df['value'] - ts_df['prev_day_value']
print("\nDay-over-day value change:")
print(ts_df[['category', 'value', 'prev_day_value', 'value_change']].head(10))
```

## Working with MultiIndex

MultiIndex (hierarchical indexing) allows you to have multiple levels of indexes, enabling more sophisticated data analysis.

```python
# Creating a MultiIndex DataFrame
multi_idx = pd.MultiIndex.from_product([['A', 'B', 'C'], ['X', 'Y', 'Z']], 
                                       names=['category', 'subcategory'])
multi_df = pd.DataFrame(np.random.randn(9, 3), 
                       index=multi_idx, 
                       columns=['value1', 'value2', 'value3'])
print("MultiIndex DataFrame:")
print(multi_df)

# Accessing data in a MultiIndex DataFrame
print("\nSelecting one level:")
print(multi_df.loc['A'])
print("\nSelecting specific elements:")
print(multi_df.loc[('A', 'X')])

# Cross-section selection with xs
print("\nCross-section selection:")
print(multi_df.xs('Y', level='subcategory'))

# Swapping levels
print("\nSwapped levels:")
print(multi_df.swaplevel('category', 'subcategory').head())

# Using groupby with MultiIndex
grouped = multi_df.groupby(level=['category']).mean()
print("\nGrouped by level:")
print(grouped)

# Unstacking to pivot the data
unstacked = multi_df.unstack(level='subcategory')
print("\nUnstacked data:")
print(unstacked.head())

# Stacking it back
print("\nStacked back:")
print(unstacked.stack().head())
```

## Optimizing Pandas Performance

Pandas can be memory-intensive with large datasets. Here are techniques to improve performance.

```python
# Create a larger dataset to demonstrate performance
large_df = pd.DataFrame(np.random.randn(100000, 10))

# Measure operation time
def time_operation(operation, df):
    start = time.time()
    result = operation(df)
    end = time.time()
    print(f"Operation took {end - start:.4f} seconds")
    return result

# 1. Use proper data types
print("Memory usage before optimization:", large_df.memory_usage().sum() / 1e6, "MB")
optimized_df = large_df.copy()
for col in optimized_df.columns:
    optimized_df[col] = pd.to_numeric(optimized_df[col], downcast='float')
print("Memory usage after optimization:", optimized_df.memory_usage().sum() / 1e6, "MB")

# 2. Vectorized operations vs. apply
def slow_way(df):
    return df.apply(lambda x: x * 2)

def fast_way(df):
    return df * 2

print("\nComparing performance:")
_ = time_operation(slow_way, large_df.iloc[:1000])  # Limit to 1000 rows for apply
_ = time_operation(fast_way, large_df)

# 3. Use query() for filtering
def filter_normal(df):
    return df[(df[0] > 0) & (df[1] < 0)]

def filter_query(df):
    return df.query('`0` > 0 and `1` < 0')

print("\nFiltering comparison:")
_ = time_operation(filter_normal, large_df)
_ = time_operation(filter_query, large_df)

# 4. Use categoricals for string data
cat_df = pd.DataFrame({
    'id': range(100000),
    'category': np.random.choice(['A', 'B', 'C', 'D', 'E'], size=100000)
})

print("\nMemory usage with strings:", cat_df.memory_usage().sum() / 1e6, "MB")
cat_df['category'] = cat_df['category'].astype('category')
print("Memory usage with categories:", cat_df.memory_usage().sum() / 1e6, "MB")
```

## Custom Functions with apply(), map(), and applymap()

These methods let you apply custom functions to your data at different levels.

```python
# Apply functions to columns or rows
def column_stats(col):
    return pd.Series({
        'mean': col.mean(),
        'median': col.median(),
        'std': col.std(),
        'skew': col.skew()
    })

stats = df[['value', 'quantity']].apply(column_stats)
print("Column statistics using apply():")
print(stats)

# Apply to rows with axis=1
def row_processing(row):
    if row['category'] == 'A':
        return row['value'] * 2
    else:
        return row['value'] / 2

df['processed_value'] = df.apply(row_processing, axis=1)
print("\nRow processing with apply():")
print(df[['category', 'value', 'processed_value']].head())

# Map for Series operations
category_mapping = {'A': 'Group 1', 'B': 'Group 1', 'C': 'Group 2', 'D': 'Group 2'}
df['category_group'] = df['category'].map(category_mapping)
print("\nMapping categories:")
print(df[['category', 'category_group']].drop_duplicates())

# Using applymap for element-wise operations on DataFrames
def format_value(x):
    if isinstance(x, (int, float)):
        return f"{x:.2f}" if x >= 0 else f"({abs(x):.2f})"
    return x

formatted = df[['value', 'quantity']].head().applymap(format_value)
print("\nFormatted values with applymap():")
print(formatted)

# Vectorized custom functions for better performance
def vectorized_custom_func(s):
    return np.where(s > 0, s * 1.1, s * 0.9)

df['value_adjusted'] = vectorized_custom_func(df['value'])
print("\nVectorized custom function:")
print(df[['value', 'value_adjusted']].head())
```

## Advanced Data Cleaning Techniques

These techniques help handle complex data issues beyond basic cleaning.

```python
# Create a dataset with various data quality issues
messy_df = pd.DataFrame({
    'col_A': ['Apple ', ' Banana', np.nan, 'Apple', 'ORANGE', 'orange ', None],
    'col_B': [1, 2, np.nan, '4', 5, 'Invalid', '7'],
    'col_C': pd.to_datetime(['2023-01-01', '2023-01-02', None, 'invalid date', 
                            '20230104', '2023/01/05', '01-06-2023']),
    'col_D': ['123-456-7890', '123.456.7891', np.nan, '(123) 456 7892', 
             '1234567893', 'invalid', '123-456-ABCD']
})
print("Messy DataFrame:")
print(messy_df)

# 1. Standardizing text data
def clean_text(s):
    if pd.isna(s):
        return np.nan
    return s.strip().lower()

messy_df['col_A_clean'] = messy_df['col_A'].apply(clean_text)
print("\n1. Cleaned text:")
print(messy_df[['col_A', 'col_A_clean']])

# 2. Converting mixed types
def convert_to_numeric(s):
    if pd.isna(s):
        return np.nan
    try:
        return float(s)
    except (ValueError, TypeError):
        return np.nan

messy_df['col_B_clean'] = messy_df['col_B'].apply(convert_to_numeric)
print("\n2. Converted to numeric:")
print(messy_df[['col_B', 'col_B_clean']])

# 3. Custom date parsing
def parse_dates(d):
    if pd.isna(d):
        return np.nan
    try:
        return pd.to_datetime(d, errors='raise')
    except:
        return np.nan

messy_df['col_C_clean'] = pd.to_datetime(messy_df['col_C'], errors='coerce')
print("\n3. Cleaned dates:")
print(messy_df[['col_C', 'col_C_clean']])

# 4. Regex for pattern-based cleaning
import re

def clean_phone(s):
    if pd.isna(s):
        return np.nan
    # Extract digits only
    digits = re.sub(r'[^0-9]', '', str(s))
    if len(digits) != 10:
        return np.nan
    return f"{digits[:3]}-{digits[3:6]}-{digits[6:]}"

messy_df['col_D_clean'] = messy_df['col_D'].apply(clean_phone)
print("\n4. Standardized phone numbers:")
print(messy_df[['col_D', 'col_D_clean']])

# 5. Advanced duplicate detection with fuzzy matching
from fuzzywuzzy import process

def find_closest_match(value, choices, threshold=80):
    if pd.isna(value):
        return np.nan
    match, score = process.extractOne(value, choices)
    return match if score >= threshold else value

unique_fruits = ['apple', 'banana', 'orange']
messy_df['col_A_standardized'] = messy_df['col_A_clean'].apply(
    lambda x: find_closest_match(x, unique_fruits) if not pd.isna(x) else np.nan
)
print("\n5. Fuzzy matching for standardization:")
print(messy_df[['col_A', 'col_A_clean', 'col_A_standardized']])
```
