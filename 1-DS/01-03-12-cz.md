## Úvod do pokročilého SQL v datové vědě

SQL (Structured Query Language) je mocným nástrojem v sadě nástrojů datového vědce. Zatímco základní SQL se zaměřuje na operace CRUD, pokročilé SQL umožňuje komplexní analýzu dat přímo v databázi, což může být efektivnější než extrakce dat do Pythonu pro analýzu.

Klíčové výhody pokročilého SQL v datové vědě:
- **Snížený přenos dat**: Analyzujte velké datové sady přímo tam, kde jsou uložené
- **Zlepšení výkonu**: Databázové enginy jsou optimalizovány pro datové operace
- **Standardizace**: SQL je široce používané napříč různými datovými platformami
- **Integrita dat**: Práce s konzistentními daty založenými na transakcích

```python
customers = pd.DataFrame({

    'amount': [100, 200, 150, 300, 250, 350, 400, 500, 450, 550]

products = pd.DataFrame({
    'product_id': range(1, 6),
    'product_name': ['Laptop', 'Phone', 'Tablet', 'Monitor', 'Keyboard'],
    'price': [1200, 800, 500, 300, 100]

order_items = pd.DataFrame({
    'order_id': [101, 101, 102, 103, 104, 105, 106, 107, 107, 108, 109, 110],

orders.to_sql('orders', engine, index=False)
products.to_sql('products', engine, index=False)

# Příklad komplexního spojení s agregací
complex_query = """
SELECT 
    c.customer_name,
    c.country,
    COUNT(DISTINCT o.order_id) AS total_orders,
    SUM(o.amount) AS total_spent,
    AVG(o.amount) AS avg_order_value,
    MAX(o.order_date) AS last_order_date
FROM customers c
LEFT JOIN orders o ON c.customer_id = o.customer_id
GROUP BY c.customer_id, c.customer_name, c.country
ORDER BY total_spent DESC
"""

customer_summary = pd.read_sql(complex_query, engine)
print("Souhrn objednávek zákazníků:")
print(customer_summary)
```

## Okenní funkce pro pokročilou analýzu

Okenní funkce provádějí výpočty přes sadu řádků tabulky, které souvisejí s aktuálním řádkem, podobně jako GROUP BY, ale bez slučování řádků.

```python
# Vytvoření tabulky prodejů s časovými dimenzemi pro demonstraci okenních funkcí
sales = pd.DataFrame({
    'date': pd.date_range(start='2023-01-01', periods=20, freq='D'),
    'product_id': np.random.choice(range(1, 6), 20),
    'salesperson_id': np.random.choice(range(1, 5), 20),
    'quantity': np.random.randint(1, 10, 20),
    'revenue': np.random.randint(100, 1000, 20)
})

sales.to_sql('sales', engine, index=False)

# Příklad s okenními funkcemi
window_query = """
SELECT
    date,
    product_id,
    revenue,
    SUM(revenue) OVER (PARTITION BY product_id ORDER BY date) AS running_total,
    AVG(revenue) OVER (PARTITION BY product_id) AS product_avg_revenue,
    RANK() OVER (PARTITION BY product_id ORDER BY revenue DESC) AS revenue_rank
FROM sales
ORDER BY product_id, date
"""

window_results = pd.read_sql(window_query, engine)
print("Příklad okenní funkce:")
print(window_results.head(10))

# Klouzavé průměry pomocí okenních funkcí
rolling_avg_query = """
SELECT
    date,
    product_id,
    revenue,
    AVG(revenue) OVER (
        PARTITION BY product_id
        ORDER BY date
        ROWS BETWEEN 2 PRECEDING AND CURRENT ROW
    ) AS moving_avg_3day
FROM sales
ORDER BY product_id, date
"""

rolling_avg = pd.read_sql(rolling_avg_query, engine)
print("\n3denní klouzavý průměr podle produktu:")
print(rolling_avg.head(10))
```

## Common Table Expressions (CTEs)

CTEs poskytují způsob, jak psát čitelnější a udržovatelnější SQL vytvářením pojmenovaných dočasných výsledných sad.

```python
# Základní příklad CTE
basic_cte = """
WITH OrderSummary AS (
    SELECT
        customer_id,
        COUNT(*) AS order_count,
        SUM(amount) AS total_amount
    FROM orders
    GROUP BY customer_id
)
SELECT
    c.customer_name,
    c.country,
    os.order_count,
    os.total_amount
FROM customers c
JOIN OrderSummary os ON c.customer_id = os.customer_id
ORDER BY os.total_amount DESC
"""

cte_results = pd.read_sql(basic_cte, engine)
print("Základní příklad CTE:")
print(cte_results)

# Rekurzivní příklad CTE (vytvoření rozsahu dat)
recursive_cte = """
WITH RECURSIVE DateSeries AS (
    SELECT date('2023-01-01') AS date
    UNION ALL
    SELECT date(date, '+1 day')
    FROM DateSeries
    WHERE date < '2023-01-10'
)
SELECT date FROM DateSeries
"""

date_series = pd.read_sql(recursive_cte, engine)
print("\nPříklad rekurzivní CTE - Časová řada:")
print(date_series)
```

## Pokročilé agregace a statistické funkce

SQL nabízí různé statistické funkce, které mohou ušetřit čas prováděním výpočtů přímo v databázi.

```python
# Vytvoření podrobnější tabulky zaměstnanců pro statistické příklady
employees_extended = pd.DataFrame({
    'id': range(1, 21),
    'name': [f'Employee {i}' for i in range(1, 21)],
    'department': np.random.choice(['HR', 'IT', 'Marketing', 'Finance', 'Sales'], 20),
    'salary': np.random.randint(50000, 100000, 20),

    'performance_score': np.random.uniform(1, 5, 20).round(1)

employees_extended.to_sql('employees_extended', engine, index=False)

# Statistické agregace
stats_query = """
SELECT
    department,
    COUNT(*) AS employee_count,
    MIN(salary) AS min_salary,
    MAX(salary) AS max_salary,
    AVG(salary) AS avg_salary,
    SUM(salary) AS total_salary,
    ROUND(STDEV(salary), 2) AS salary_stddev,
    ROUND(AVG(performance_score), 2) AS avg_performance

"""

try:
    dept_stats = pd.read_sql(stats_query, engine)
    print("Statistiky oddělení:")
    print(dept_stats)
except Exception as e:
    # SQLite nepodporuje nativně STDEV, upravíme tedy pro SQLite
    stats_query_sqlite = """
    SELECT
        department,
        COUNT(*) AS employee_count,
        MIN(salary) AS min_salary,
        MAX(salary) AS max_salary,
        AVG(salary) AS avg_salary,
        SUM(salary) AS total_salary,
        ROUND(AVG(performance_score), 2) AS avg_performance
    FROM employees_extended
    GROUP BY department
    """
    dept_stats = pd.read_sql(stats_query_sqlite, engine)
    print("Statistiky oddělení (bez směrodatné odchylky pro SQLite):")
    print(dept_stats)

# Výpočty percentilů (pomocí Pythonu, protože SQLite nepodporuje PERCENTILE přímo)
salary_data = pd.read_sql("SELECT salary FROM employees_extended", engine)
percentiles = np.percentile(salary_data['salary'], [25, 50, 75, 90])

print("\nPercentily platu:")
print(f"25. percentil: ${percentiles[0]:.2f}")
print(f"Medián (50.): ${percentiles[1]:.2f}")
print(f"75. percentil: ${percentiles[2]:.2f}")
print(f"90. percentil: ${percentiles[3]:.2f}")

# Příklad binning/roztřídění
binning_query = """
SELECT 
    CASE

    AVG(performance_score) AS avg_performance

salary_brackets = pd.read_sql(binning_query, engine)

print(salary_brackets)
```

## Práce s časovými řadami v SQL

Analýza časových řad je běžná v datové vědě a SQL poskytuje nástroje pro práci s daty založenými na čase.

```python
# Vytvoření dat časové řady pro prodeje v různých regionech
dates = pd.date_range(start='2022-01-01', periods=365, freq='D')
regions = ['Sever', 'Jih', 'Východ', 'Západ']

records = []
for date in dates:
    for region in regions:
        # Vytvoření sezónního vzoru s určitou náhodností
        month = date.month
        # Více prodejů v létě a prosinci
        seasonal_factor = 1.0
        if month in [6, 7, 8]:  # Léto
            seasonal_factor = 1.3
        elif month == 12:  # Prosinec
            seasonal_factor = 1.5
        
        # Efekt víkendu
        weekend_factor = 1.3 if date.dayofweek >= 5 else 1.0
        
        # Základní prodeje s určitou náhodností
        base_sales = {
            'Sever': 1000, 'Jih': 1200, 
            'Východ': 900, 'Západ': 1100
        }[region]
        
        sales_amount = int(base_sales * seasonal_factor * weekend_factor * np.random.uniform(0.8, 1.2))
        
        records.append({
            'date': date,
            'region': region,
            'sales': sales_amount
        })

time_series_df = pd.DataFrame(records)
time_series_df.to_sql('regional_sales', engine, index=False)

# Extrakce dat a seskupení
time_grouping = """
SELECT 
    strftime('%Y-%m', date) AS month,
    region,
    SUM(sales) AS monthly_sales
FROM regional_sales
GROUP BY month, region
ORDER BY month, region
"""

monthly_sales = pd.read_sql(time_grouping, engine)
print("Měsíční prodeje podle regionů:")
print(monthly_sales.head(8))

# Klouzavé průměry s okenními funkcemi
moving_avg = """
SELECT
    date,
    region,
    sales,
    AVG(sales) OVER (
        PARTITION BY region
        ORDER BY date
        ROWS BETWEEN 6 PRECEDING AND CURRENT ROW
    ) AS moving_avg_7day
FROM regional_sales
WHERE region = 'Sever'
ORDER BY date
LIMIT 20
"""

try:
    north_moving_avg = pd.read_sql(moving_avg, engine)
    print("\n7denní klouzavý průměr pro region Sever:")
    print(north_moving_avg)
except Exception as e:
    print(f"\nChyba s okenní funkcí (SQLite může mít omezenou podporu): {e}")
    print("Použití pandas pro výpočet klouzavého průměru místo toho:")
    
    north_sales = pd.read_sql(
        "SELECT date, sales FROM regional_sales WHERE region = 'Sever' ORDER BY date",
        engine
    )
    north_sales['moving_avg_7day'] = north_sales['sales'].rolling(window=7).mean()
    print(north_sales.head(20))

# Srovnání rok po roce
yoy_comparison = """
SELECT
    strftime('%m', date) AS month,
    strftime('%Y', date) AS year,
    region,
    SUM(sales) AS monthly_sales
FROM regional_sales
GROUP BY year, month, region
ORDER BY region, month, year
"""

yoy_data = pd.read_sql(yoy_comparison, engine)
print("\nMěsíční prodeje meziroční srovnání:")
print(yoy_data.head(8))
```

## Optimalizace dotazů pro datovou vědu

Optimalizace SQL dotazů je klíčová při práci s velkými datovými sadami.

```python
import time

# Vytvoření větší datové sady pro testování výkonu
num_records = 50000
large_df = pd.DataFrame({
    'id': range(1, num_records + 1),
    'category': np.random.choice(['A', 'B', 'C', 'D', 'E'], num_records),
    'value1': np.random.randint(1, 100, num_records),
    'value2': np.random.uniform(0, 1000, num_records),
    'text_field': np.random.choice(['jablko', 'banán', 'pomeranč', 'hroznové víno', 'kiwi'], num_records),
    'date_field': pd.date_range(start='2020-01-01', periods=num_records, freq='H')
})

large_df.to_sql('large_table', engine, index=False, if_exists='replace')

# Vytvoření indexů pro zlepšení výkonu
engine.execute('CREATE INDEX idx_category ON large_table(category)')
engine.execute('CREATE INDEX idx_date ON large_table(date_field)')

# Funkce pro měření času vykonání dotazu
def time_query(query, description):
    start_time = time.time()
    result = pd.read_sql(query, engine)
    end_time = time.time()
    print(f"{description} - Doba vykonání: {end_time - start_time:.4f} sekund")
    print(f"Velikost výsledku: {result.shape}")
    return result

# Neoptimalizovaný dotaz s úplným prohledáváním tabulky
unoptimized_query = """
SELECT *
FROM large_table
WHERE category = 'A' AND value1 > 50
"""

# Optimalizovaný dotaz s použitím indexovaných sloupců nejdříve
optimized_query = """
SELECT *
FROM large_table
WHERE category = 'A' AND value1 > 50
"""

# Porovnání výkonu
print("Porovnání výkonu dotazů:")
unopt_results = time_query(unoptimized_query, "Neoptimalizovaný dotaz")
opt_results = time_query(optimized_query, "Optimalizovaný dotaz (s indexem)")

# Použití EXPLAIN pro pochopení plánu vykonání dotazu
explain_query = "EXPLAIN QUERY PLAN " + optimized_query
explain_results = engine.execute(explain_query).fetchall()
print("\nPlán vykonání dotazu:")
for row in explain_results:
    print(row)

# Vhodné použití LIMIT
limit_query = """
SELECT *
FROM large_table
ORDER BY value2 DESC
LIMIT 10
"""

top_values = time_query(limit_query, "Použití LIMIT pro top hodnoty")

# Vyvarování se SELECT * pro velké tabulky
optimized_select = """
SELECT id, category, value1
FROM large_table
WHERE category = 'A' AND value1 > 50
"""
```

## Shrnutí a osvědčené postupy

Pokročilé dovednosti v SQL umožňují datovým vědcům efektivně pracovat s velkými datovými sadami přímo v databázi, což snižuje potřebu dodatečného zpracování v Pythonu. Zde jsou klíčové poznatky:

1. **Udržujte zpracování blízko k datům**: Používejte SQL pro filtrování, agregaci a transformaci dat před extrakcí do Pythonu
2. **Využívejte okenní funkce**: Umožňují sofistikovanou analýzu jako průběžné součty, klouzavé průměry a řazení
3. **Pište čitelné SQL**: Common Table Expressions (CTEs) zlepšují čitelnost a udržovatelnost
4. **Využívejte pokročilá spojení**: Ovládněte různé typy spojení a vnořených dotazů pro kombinování dat z více zdrojů
5. **Optimalizujte výkon dotazů**: Vytvářejte vhodné indexy, vyhýbejte se SELECT * a pochopte plány vykonání dotazů
6. **Naučte se funkce specifické pro databáze**: Různé databázové systémy nabízejí specializované statistické a analytické funkce
7. **Používejte parametrizaci**: Vyhněte se SQL injection a zlepšete znovupoužitelnost
8. **Zvažte použití sqlalchemy**: Poskytuje konzistentní rozhraní napříč různými databázovými enginy

```python
# Příklad komplexního pracovního postupu: Datově-vědecký SQL workflow
def data_science_sql_workflow(engine, date_range_start, date_range_end, min_order_count=5):
    """
    Příklad kompletního datově-vědeckého workflow kombinujícího pokročilé SQL techniky
    """
    # Parametrizovaný dotaz s použitím osvědčených postupů
    query = """
    WITH CustomerOrders AS (
        -- CTE pro agregace objednávek
        SELECT
            c.customer_id,
            c.customer_name,
            c.country,
            COUNT(DISTINCT o.order_id) AS order_count,
            SUM(o.amount) AS total_spent,
            AVG(o.amount) AS avg_order_value,
            MIN(o.order_date) AS first_order,
            MAX(o.order_date) AS last_order
        FROM customers c
        JOIN orders o ON c.customer_id = o.customer_id
        WHERE o.order_date BETWEEN ? AND ?
        GROUP BY c.customer_id, c.customer_name, c.country
        HAVING COUNT(DISTINCT o.order_id) >= ?
    ),
    CustomerProductPreferences AS (
        -- CTE pro preference produktů
        SELECT
            o.customer_id,
            p.product_name,
            SUM(oi.quantity) AS total_quantity,
            -- Okenní funkce pro řazení produktů podle preferencí zákazníků
            RANK() OVER (
                PARTITION BY o.customer_id 
                ORDER BY SUM(oi.quantity) DESC
            ) AS product_rank
        FROM orders o
        JOIN order_items oi ON o.order_id = oi.order_id
        JOIN products p ON oi.product_id = p.product_id
        WHERE o.order_date BETWEEN ? AND ?
        GROUP BY o.customer_id, p.product_name
    ),
    CustomerMetrics AS (
        -- CTE pro pokročilé metriky zákazníků
        SELECT
            co.customer_id,
            co.customer_name,
            co.country,
            co.order_count,
            co.total_spent,
            co.avg_order_value,
            -- Výpočet času mezi první a poslední objednávkou (dny)
            JULIANDAY(co.last_order) - JULIANDAY(co.first_order) AS customer_lifetime_days,
            -- Výpočet rychlosti útrat (částka za den)
            CASE
                WHEN JULIANDAY(co.last_order) = JULIANDAY(co.first_order) THEN co.total_spent
                ELSE co.total_spent / (JULIANDAY(co.last_order) - JULIANDAY(co.first_order) + 1)
            END AS spend_per_day
        FROM CustomerOrders co
    )
    
    -- Finální dotaz kombinující všechny CTEs
    SELECT
        cm.customer_id,
        cm.customer_name,
        cm.country,
        cm.order_count,
        cm.total_spent,
        cm.avg_order_value,
        cm.customer_lifetime_days,
        cm.spend_per_day,
        -- Zjištění nejoblíbenějšího produktu
        (
            SELECT product_name
            FROM CustomerProductPreferences cpp
            WHERE cpp.customer_id = cm.customer_id AND cpp.product_rank = 1
            LIMIT 1
        ) AS favorite_product
    FROM CustomerMetrics cm
    ORDER BY cm.total_spent DESC
    """
    
    # Vykonání parametrizovaného dotazu (každý ? je nahrazen parametrem)
    params = [date_range_start, date_range_end, min_order_count, date_range_start, date_range_end]
    try:
        result = pd.read_sql(query, engine, params=params)
        return result
    except Exception as e:
        print(f"Chyba při vykonání SQL workflow: {e}")
        return None

# Příklad použití:
customer_analysis = data_science_sql_workflow(
    engine, 
    date_range_start='2023-01-01', 
    date_range_end='2023-12-31', 
    min_order_count=1
)

print("\nVýsledek kompletní analýzy zákazníků:")
if customer_analysis is not None:
    print(customer_analysis)
else:
    print("Vykonání workflow selhalo.")

# Úklid - zavření připojení
engine.dispose()
```
