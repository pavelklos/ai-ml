## Úvod do tvorby příznaků

Tvorba příznaků (feature engineering) je proces transformace surových dat na smysluplné příznaky, které lépe reprezentují základní problém pro prediktivní modely, čímž zlepšují výkon modelu. Často je považována za nejdůležitější krok v pipeline strojového učení.

```python
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler, OneHotEncoder, MinMaxScaler, LabelEncoder
from sklearn.feature_selection import SelectKBest, f_classif, RFE
from sklearn.decomposition import PCA
from sklearn.pipeline import Pipeline
from sklearn.compose import ColumnTransformer
from sklearn.ensemble import RandomForestClassifier
from sklearn.model_selection import train_test_split

# Vzorový dataset
# Vytvořme jednoduchý dataset s různými typy příznaků
np.random.seed(42)
n_samples = 1000

# Vytvoření DataFrame s různými typy příznaků
df = pd.DataFrame({
    # Numerické příznaky
    'age': np.random.normal(35, 10, n_samples).astype(int),
    'income': np.random.lognormal(10, 1, n_samples),
    'expense_ratio': np.random.uniform(0, 1, n_samples),
    
    # Kategorické příznaky
    'education': np.random.choice(['High School', 'Bachelor', 'Master', 'PhD'], n_samples),
    'marital_status': np.random.choice(['Single', 'Married', 'Divorced'], n_samples),
    
    # Časové příznaky
    'signup_date': pd.date_range(start='2020-01-01', periods=n_samples),
    
    # Textový příznak
    'feedback': np.random.choice(['Great service, very satisfied!', 
                                 'Good but could improve', 
                                 'Average experience, nothing special',
                                 'Poor service, need improvement',
                                 'Terrible experience, will not recommend'], n_samples)
})

# Vytvoření cílové proměnné
df['churn'] = (df['age'] < 25) | ((df['income'] < np.percentile(df['income'], 25)) & 
                                 (df['education'] == 'High School'))
df['churn'] = df['churn'].astype(int)

print("Tvar datasetu:", df.shape)
print("\nPrvních několik řádků:")
print(df.head())
print("\nTypy příznaků:")
print(df.dtypes)
print("\nRozdělení cílové proměnné:")
print(df['churn'].value_counts(normalize=True))
```

## Transformace numerických příznaků

Numerické příznaky často vyžadují transformace, aby lépe vyhovovaly algoritmům strojového učení. Běžné transformace zahrnují:
- Logaritmické/mocninné transformace pro zešikmená data
- Rozdělení do košů (binning) pro převod spojitých příznaků na diskrétní
- Polynomiální příznaky pro zachycení nelineárních vztahů

```python
# 1. Logaritmická transformace pro zešikmená data
df['income_log'] = np.log1p(df['income'])  # log1p zabraňuje problémům s nulovými hodnotami

# 2. Mocninná transformace
df['age_squared'] = df['age'] ** 2

# 3. Rozdělení do košů (diskretizace)
df['age_group'] = pd.cut(df['age'], bins=[0, 18, 35, 50, 100], 
                         labels=['Teen', 'Young Adult', 'Middle Aged', 'Senior'])

# 4. Vytvoření interakčních příznaků
df['income_per_age'] = df['income'] / df['age']

# 5. Polynomiální příznaky z sklearn
from sklearn.preprocessing import PolynomialFeatures
poly = PolynomialFeatures(degree=2, include_bias=False)
poly_features = poly.fit_transform(df[['age', 'expense_ratio']])
poly_df = pd.DataFrame(poly_features, 
                      columns=[f"{col}" for col in poly.get_feature_names_out(['age', 'expense_ratio'])])
print("Polynomiální příznaky:")
print(poly_df.head())

# Vizualizace efektu transformací
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# Původní rozdělení příjmu
sns.histplot(df['income'], kde=True, ax=axes[0, 0])
axes[0, 0].set_title('Původní rozdělení příjmu')

# Rozdělení po logaritmické transformaci
sns.histplot(df['income_log'], kde=True, ax=axes[0, 1])
axes[0, 1].set_title('Rozdělení příjmu po logaritmické transformaci')

# Rozdělení věku
sns.histplot(df['age'], kde=True, ax=axes[1, 0])
axes[1, 0].set_title('Rozdělení věku')

# Počty ve věkových skupinách
sns.countplot(x='age_group', data=df, ax=axes[1, 1])
axes[1, 1].set_title('Rozdělení věkových skupin')

plt.tight_layout()
```

## Kódování kategorických příznaků

Převod kategorických proměnných na numerickou formu, kterou lze poskytnout algoritmům strojového učení.

```python
# Prozkoumání kategorických proměnných
print("Kategorie vzdělání:", df['education'].unique())
print("Kategorie rodinného stavu:", df['marital_status'].unique())

# 1. One-Hot kódování
education_encoded = pd.get_dummies(df['education'], prefix='edu')
print("\nOne-hot kódování vzdělání:")
print(education_encoded.head())

# 2. Label kódování
le = LabelEncoder()
df['marital_status_encoded'] = le.fit_transform(df['marital_status'])
print("\nLabel kódování rodinného stavu:")
print(pd.DataFrame({'původní': df['marital_status'].head(), 
                    'kódované': df['marital_status_encoded'].head()}))

# 3. Ordinální kódování (ruční)
education_order = {'High School': 0, 'Bachelor': 1, 'Master': 2, 'PhD': 3}
df['education_ordinal'] = df['education'].map(education_order)
print("\nOrdinální kódování vzdělání:")
print(pd.DataFrame({'původní': df['education'].head(), 
                    'kódované': df['education_ordinal'].head()}))

# 4. Cílové kódování
target_means = df.groupby('education')['churn'].mean()
df['education_target_encoded'] = df['education'].map(target_means)
print("\nCílové kódování vzdělání:")
print(pd.DataFrame({'vzdělání': df['education'].head(),
                    'cílově_kódované': df['education_target_encoded'].head()}))

# 5. Count kódování
count_map = df['marital_status'].value_counts().to_dict()
df['marital_status_counts'] = df['marital_status'].map(count_map)
print("\nCount kódování rodinného stavu:")
print(pd.DataFrame({'rodinný_stav': df['marital_status'].head(),
                    'count_kódování': df['marital_status_counts'].head()}))

# Vizualizace vztahu mezi kategorickými příznaky a cílovou proměnnou
plt.figure(figsize=(12, 6))
plt.subplot(1, 2, 1)
sns.barplot(x='education', y='churn', data=df)
plt.title('Míra odchodu podle vzdělání')
plt.xticks(rotation=45)

plt.subplot(1, 2, 2)
sns.barplot(x='marital_status', y='churn', data=df)
plt.title('Míra odchodu podle rodinného stavu')
plt.tight_layout()
```

## Škálování a normalizace příznaků

Normalizace příznaků na podobnou škálu zajišťuje, že žádný jednotlivý příznak nedominuje v procesu učení.

```python
# Výběr numerických příznaků pro škálování
numeric_features = ['age', 'income', 'expense_ratio']
X_numeric = df[numeric_features].copy()

# 1. Standardizace (z-score normalizace)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X_numeric)
X_scaled_df = pd.DataFrame(X_scaled, columns=[f"{col}_scaled" for col in numeric_features])
print("Standardizované příznaky:")
print(X_scaled_df.head())

# 2. Min-Max škálování
minmax = MinMaxScaler()
X_minmax = minmax.fit_transform(X_numeric)
X_minmax_df = pd.DataFrame(X_minmax, columns=[f"{col}_minmax" for col in numeric_features])
print("\nMin-Max škálované příznaky:")
print(X_minmax_df.head())

# 3. Robustní škálování (pomocí sklearn.preprocessing.RobustScaler)
from sklearn.preprocessing import RobustScaler
robust_scaler = RobustScaler()
X_robust = robust_scaler.fit_transform(X_numeric)
X_robust_df = pd.DataFrame(X_robust, columns=[f"{col}_robust" for col in numeric_features])
print("\nRobustně škálované příznaky:")
print(X_robust_df.head())

# 4. Normalizace (škálování řádků na jednotkovou normu)
from sklearn.preprocessing import Normalizer
normalizer = Normalizer()
X_normalized = normalizer.fit_transform(X_numeric)
X_normalized_df = pd.DataFrame(X_normalized, columns=[f"{col}_normalized" for col in numeric_features])
print("\nNormalizované příznaky:")
print(X_normalized_df.head())

# Vizualizace efektu různých technik škálování
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

sns.histplot(X_numeric['income'], kde=True, ax=axes[0, 0])
axes[0, 0].set_title('Původní rozdělení příjmu')

sns.histplot(X_scaled_df['income_scaled'], kde=True, ax=axes[0, 1])
axes[0, 1].set_title('Standardizovaný příjem')

sns.histplot(X_minmax_df['income_minmax'], kde=True, ax=axes[1, 0])
axes[1, 0].set_title('Min-Max škálovaný příjem')

sns.histplot(X_normalized_df['income_normalized'], kde=True, ax=axes[1, 1])
axes[1, 1].set_title('Normalizovaný příjem')

plt.tight_layout()
```

## Zpracování časových příznaků

Extrakce smysluplných příznaků z dat obsahujících datum a čas.

```python
# Původní datetime příznak
print("Původní datetime příznak:")
print(df['signup_date'].head())

# 1. Extrakce základních komponent
df['signup_year'] = df['signup_date'].dt.year
df['signup_month'] = df['signup_date'].dt.month
df['signup_day'] = df['signup_date'].dt.day
df['signup_dayofweek'] = df['signup_date'].dt.dayofweek

print(df[['signup_date', 'signup_year', 'signup_month', 'signup_day', 
         'signup_dayofweek', 'signup_is_weekend', 'signup_quarter']].head())

# 2. Časové příznaky
# Počet dní od nejstaršího data v datasetu
df['days_since_first'] = (df['signup_date'] - df['signup_date'].min()).dt.days

# 3. Cyklické kódování pro měsíc a den v týdnu (pro zachování cyklické povahy)

print("\nCyklické kódování časových příznaků:")
print(df[['signup_month', 'signup_month_sin', 'signup_month_cos']].head())

# 4. Časově založená agregace (příklad: výpočet průměrné míry odchodu podle měsíce)
monthly_churn = df.groupby('signup_month')['churn'].mean().reset_index()
```

## Techniky výběru příznaků

Techniky pro výběr nejrelevantnějších příznaků ke zlepšení výkonu modelu a interpretovatelnosti.

```python
# Příprava dat pro výběr příznaků
# Kombinace všech numerických příznaků, které jsme vytvořili
feature_cols = ['age', 'income', 'expense_ratio', 'income_log', 'age_squared',
                'income_per_age', 'education_ordinal', 'marital_status_encoded',
                'signup_month', 'signup_dayofweek', 'days_since_first',
                'feedback_length', 'word_count']

X = df[feature_cols]
y = df['churn']

# 1. Filtrační metoda: Statistické testy (ANOVA F-test)
selector = SelectKBest(score_func=f_classif, k=5)
X_selected = selector.fit_transform(X, y)

# Získání názvů vybraných příznaků a jejich skóre
selected_indices = selector.get_support(indices=True)
selected_features = [feature_cols[i] for i in selected_indices]
selected_scores = selector.scores_[selected_indices]

print("Skóre příznaků (ANOVA F-test):")
for feature, score in zip(selected_features, selected_scores):
    print(f"{feature}: {score:.4f}")

# 2. Wrapper metoda: Rekurzivní eliminace příznaků (RFE)
model = RandomForestClassifier(n_estimators=100, random_state=42)
rfe = RFE(estimator=model, n_features_to_select=5)
X_rfe = rfe.fit_transform(X, y)

# Získání názvů vybraných příznaků
rfe_selected_features = [feature for feature, selected in zip(feature_cols, rfe.support_) if selected]
print("\nPříznaky vybrané pomocí RFE:")
print(rfe_selected_features)

# 3. Vestavěná metoda: Důležitost příznaků z Random Forest
model.fit(X, y)
importances = model.feature_importances_
feature_importance = pd.DataFrame({
    'Příznak': feature_cols,
    'Důležitost': importances
}).sort_values('Důležitost', ascending=False)

print("\nDůležitost příznaků z Random Forest:")
print(feature_importance)

# 4. Redukce dimenzionality: PCA
pca = PCA(n_components=5)
X_pca = pca.fit_transform(StandardScaler().fit_transform(X))

print("\nPodíl vysvětlené variance PCA:")
print(pca.explained_variance_ratio_)
print(f"Celková vysvětlená variance: {sum(pca.explained_variance_ratio_):.4f}")

# Vizualizace výsledků výběru příznaků
plt.figure(figsize=(12, 10))

# Vykreslení důležitosti příznaků
plt.subplot(2, 1, 1)
sns.barplot(x='Důležitost', y='Příznak', data=feature_importance)
plt.title('Důležitost příznaků z Random Forest')

# Vykreslení PCA komponent
plt.subplot(2, 1, 2)
plt.bar(range(1, len(pca.explained_variance_ratio_) + 1), pca.explained_variance_ratio_)
plt.xlabel('Hlavní komponenta')
plt.ylabel('Podíl vysvětlené variance')
plt.title('Vysvětlená variance PCA')
plt.tight_layout()
```

## Automatizovaná tvorba příznaků

Nástroje a techniky pro automatizaci procesu tvorby příznaků.

```python
# Automatizovaná tvorba příznaků s featuretools
try:
    import featuretools as ft
    
    # Vytvoření entity set pro featuretools
    es = ft.EntitySet(id="customer_data")
    
    # Přidání dataframu jako entity
    df_copy = df.copy()

        dataframe_name="customers",
        dataframe=df_copy.reset_index(),
        index="customer_id",
        time_index="signup_date"
    )
    
    # Spuštění deep feature synthesis pro automatické vytváření příznaků
    feature_matrix, feature_defs = ft.dfs(
        entityset=es,
        target_dataframe_name="customers",
        trans_primitives=["add_numeric", "multiply_numeric"],
        verbose=True
    )
    
    print("\nAutomaticky vytvořené příznaky (prvních 5):")
    for i, feat in enumerate(feature_defs[:5]):
        print(f"{i+1}. {feat}")
        
    print(f"\nCelkem vytvořených příznaků: {len(feature_defs)}")
    print("\nUkázka matice příznaků:")
    print(feature_matrix.head(3))
    
except ImportError:
    print("\nfeaturetools není nainstalován. Pro automatizovanou tvorbu příznaků spusťte: pip install featuretools")
```

## Pipeline pro tvorbu příznaků

Budování robustních a opakovaně použitelných pipeline pro tvorbu příznaků s cílem zachovat konzistenci mezi trénovacími a predikčními daty.

```python
# Vytvoření komplexní pipeline pro tvorbu příznaků
numeric_features = ['age', 'income', 'expense_ratio']
categorical_features = ['education', 'marital_status']

# Pipeline pro numerické příznaky
numeric_transformer = Pipeline(steps=[
    ('imputer', SimpleImputer(strategy='median')),
    ('scaler', StandardScaler())
])

# Pipeline pro kategorické příznaky
categorical_transformer = Pipeline(steps=[
    ('imputer', SimpleImputer(strategy='most_frequent')),
    ('onehot', OneHotEncoder(handle_unknown='ignore'))
])

# Kombinace transformací v preprocessoru
preprocessor = ColumnTransformer(
    transformers=[
        ('num', numeric_transformer, numeric_features),
        ('cat', categorical_transformer, categorical_features)
    ])

# Vytvoření kompletní pipeline s modelem
full_pipeline = Pipeline(steps=[
    ('preprocessor', preprocessor),
    ('classifier', RandomForestClassifier(n_estimators=100, random_state=42))
])

# Rozdělení datasetu
X_train, X_test, y_train, y_test = train_test_split(
    df[numeric_features + categorical_features], 
    df['churn'],
    test_size=0.2, 
    random_state=42
)

# Trénování pipeline
full_pipeline.fit(X_train, y_train)

# Vyhodnocení
from sklearn.metrics import classification_report
y_pred = full_pipeline.predict(X_test)
print("Klasifikační report:")
print(classification_report(y_test, y_pred))

# Použití pipeline pro nová data
print("\nPříklad použití pipeline pro predikci na nových datech:")
new_data = pd.DataFrame({
    'age': [25, 55],
    'income': [50000, 90000],
    'expense_ratio': [0.3, 0.2],
    'education': ['Bachelor', 'PhD'],
    'marital_status': ['Single', 'Married']
})
print("Nová data:")
print(new_data)

print("\nPredikce:")
print(full_pipeline.predict(new_data))
print("Pravděpodobnosti predikce:")
print(full_pipeline.predict_proba(new_data))

# Extrakce transformovaných příznaků z pipeline
preprocessed_X_train = preprocessor.fit_transform(X_train)
feature_names = (
    numeric_features + 
    list(preprocessor.transformers_[1][1].named_steps['onehot']
        .get_feature_names_out(categorical_features))
)

preprocessed_df = pd.DataFrame(preprocessed_X_train, columns=feature_names)
```
