## 1. Introduction to Correlation Analysis

Correlation analysis helps us understand relationships between variables in our data. Visualizing these relationships is crucial for:

- Identifying patterns and trends
- Detecting potential predictive relationships
- Discovering multicollinearity issues
- Supporting feature selection for machine learning models

Let's set up our environment with the necessary libraries:

```python
# Import essential libraries for correlation visualization
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import plotly.express as px
import plotly.graph_objects as go
from sklearn.datasets import load_iris, load_boston, load_diabetes

# Set visualization styles
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_theme(style="whitegrid")

# For reproducible results
np.random.seed(42)

# Display settings for pandas
pd.set_option('display.max_columns', 20)

# Load sample datasets for our examples
iris = load_iris()
iris_df = pd.DataFrame(iris.data, columns=iris.feature_names)
iris_df['species'] = iris.target_names[iris.target]

# Create a sample dataset with known correlations
n_samples = 200
x = np.random.normal(0, 1, n_samples)
# Create variables with different correlation strengths
strong_pos_corr = x * 0.9 + np.random.normal(0, 0.3, n_samples)  # Strong positive
medium_pos_corr = x * 0.5 + np.random.normal(0, 0.7, n_samples)  # Medium positive
weak_pos_corr = x * 0.2 + np.random.normal(0, 0.9, n_samples)    # Weak positive
weak_neg_corr = x * -0.2 + np.random.normal(0, 0.9, n_samples)   # Weak negative
strong_neg_corr = x * -0.9 + np.random.normal(0, 0.3, n_samples) # Strong negative
no_corr = np.random.normal(0, 1, n_samples)                      # No correlation

# Create a DataFrame with these variables
corr_df = pd.DataFrame({
    'x': x,
    'strong_pos': strong_pos_corr,
    'medium_pos': medium_pos_corr,
    'weak_pos': weak_pos_corr,
    'weak_neg': weak_neg_corr,
    'strong_neg': strong_neg_corr,
    'no_corr': no_corr
})

print("Sample correlation dataframe:")
print(corr_df.head())

# Calculate the correlation matrix
correlation_matrix = corr_df.corr()
print("\nCorrelation matrix:")
print(correlation_matrix['x'].sort_values(ascending=False))
```

## 2. Understanding Correlation Coefficients

Before visualizing correlations, it's important to understand what we're measuring:

- **Pearson correlation coefficient** (r): Measures linear relationships (-1 to +1)
- **Spearman's rank correlation**: Measures monotonic relationships (works with non-linear data)
- **Kendall's Tau**: Another rank correlation measure, robust to outliers

```python
# Calculate different types of correlation coefficients
pearson_corr = corr_df.corr(method='pearson')
spearman_corr = corr_df.corr(method='spearman')
kendall_corr = corr_df.corr(method='kendall')

# Create a function to display different correlation measures for comparison
def compare_correlation_methods(x_var, y_var, data):
    """Compare different correlation methods between two variables."""
    pearson = data[x_var].corr(data[y_var], method='pearson')
    spearman = data[x_var].corr(data[y_var], method='spearman')
    kendall = data[x_var].corr(data[y_var], method='kendall')
    
    print(f"Correlation between {x_var} and {y_var}:")
    print(f"  Pearson: {pearson:.4f}")
    print(f"  Spearman: {spearman:.4f}")
    print(f"  Kendall: {kendall:.4f}")
    
    # Create scatter plot with correlation info
    plt.figure(figsize=(8, 6))
    plt.scatter(data[x_var], data[y_var], alpha=0.7)
    plt.title(f"Correlation between {x_var} and {y_var}")
    plt.xlabel(x_var)
    plt.ylabel(y_var)
    plt.annotate(f"Pearson r: {pearson:.4f}\nSpearman ρ: {spearman:.4f}\nKendall τ: {kendall:.4f}",
                xy=(0.05, 0.95), xycoords='axes fraction',
                bbox=dict(boxstyle="round,pad=0.5", fc="white", alpha=0.8),
                va='top')
    plt.grid(True, alpha=0.3)
    plt.show()

# Compare correlation methods for different relationship strengths
compare_correlation_methods('x', 'strong_pos', corr_df)
compare_correlation_methods('x', 'medium_pos', corr_df)
compare_correlation_methods('x', 'no_corr', corr_df)

# Create a non-linear relationship to show the difference in correlation methods
n = 200
x_nonlinear = np.linspace(0, 10, n)
y_nonlinear = x_nonlinear**2 + np.random.normal(0, 10, n)
nonlinear_df = pd.DataFrame({'x': x_nonlinear, 'y': y_nonlinear})

# Compare correlation methods for non-linear data
compare_correlation_methods('x', 'y', nonlinear_df)
```

## 5. Correlation Matrices and Their Interpretation

A correlation matrix displays correlation coefficients between multiple variables:

```python
# Calculate correlation matrix for iris features
iris_corr = iris_df.drop(columns=['species']).corr()

print("Iris correlation matrix:")

# Create a function to show absolute correlation values
def get_absolute_corr(df):
    corr_matrix = df.corr().abs()
    # Set self-correlation (diagonal) to 0 to find strongest non-self correlations
    np.fill_diagonal(corr_matrix.values, 0)
    return corr_matrix

# Find strongest correlations
abs_corr = get_absolute_corr(iris_df.drop(columns=['species']))
strongest_pair = abs_corr.unstack().sort_values(ascending=False)[0:1]
print("\nStrongest correlation:")
print(strongest_pair)
```

## 7. Customizing Heatmaps for Better Insights

```python
# Function to highlight significant correlations
def highlight_significant_corr(corr_matrix, threshold=0.5):
    """
    Create a heatmap with significant correlations highlighted.
    
    Parameters:
    -----------
    corr_matrix : pandas.DataFrame
        Correlation matrix to visualize
    threshold : float
        Absolute correlation threshold for highlighting
    """
    plt.figure(figsize=(12, 10))
    
    # Create a mask for insignificant correlations
    insignificant = np.abs(corr_matrix) < threshold
    
    # Create heatmap with significant correlations only
    ax = sns.heatmap(corr_matrix, 
                     cmap='coolwarm',
                     annot=True,
                     fmt='.2f', 
                     linewidths=0.5,
                     vmin=-1, vmax=1,
                     mask=insignificant)
    
    plt.title(f'Correlation Heatmap (|r| ≥ {threshold})')
    return ax

# Apply our custom function

plt.show()
```

## 9. Interactive Correlation Visualization

Interactive visualizations can provide a better exploration experience, especially for large datasets:

```python
# Interactive heatmap with annotations using Plotly
def create_interactive_corr_heatmap(df, title='Interactive Correlation Heatmap'):
    """Create an interactive correlation heatmap with Plotly."""
    corr_matrix = df.corr().round(2)
    
    # Create mask for the upper triangle
    mask = np.zeros_like(corr_matrix, dtype=bool)
    mask[np.triu_indices_from(mask, k=1)] = True
    
    # Convert the masked correlation matrix to a tidy format
    tidy_corr = corr_matrix.stack().reset_index()
    tidy_corr.columns = ['Var1', 'Var2', 'Correlation']
    # Filter out upper triangle
    tidy_corr = tidy_corr[~mask.stack()]
    
    # Create the heatmap
    fig = px.scatter(tidy_corr, x='Var1', y='Var2', size='Correlation', 
                     color='Correlation', size_max=60,
                     color_continuous_scale='RdBu_r',
                     range_color=[-1, 1],
                     text='Correlation',
                     hover_data={'Correlation': ':.2f'})
    
    fig.update_traces(textposition='middle center', textfont=dict(color='white'))
    fig.update_layout(title=title,
                      xaxis=dict(title=''),
                      yaxis=dict(title=''))
    
    return fig

# Apply the function to our dataset
corr_fig = create_interactive_corr_heatmap(iris_df.drop(columns=['species']))
corr_fig.show()

# Interactive scatter plot matrix with correlation information
def correlation_scatter_matrix(df, numeric_cols=None):
    """
    Create an interactive scatter plot matrix with correlation information.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        Data to visualize
    numeric_cols : list of str, optional
        List of numeric columns to include. If None, use all numeric columns.
    """
    # Select numeric columns if not specified
    if numeric_cols is None:
        numeric_cols = df.select_dtypes(include=[np.number]).columns.tolist()
    
    # Create subplots
    fig = go.Figure()
    
    # Calculate correlation matrix
    corr_matrix = df[numeric_cols].corr().round(2)
    
    # Add scatter plot for each pair of variables
    for i, var1 in enumerate(numeric_cols):
        for j, var2 in enumerate(numeric_cols):
            if i > j:  # Lower triangle
                fig.add_trace(
                    go.Scatter(
                        x=df[var1],
                        y=df[var2],
                        mode='markers',
                        marker=dict(size=6, opacity=0.5),
                        name=f'{var1} vs {var2}',
                        xaxis=f'x{i+1}',
                        yaxis=f'y{j+1}',
                        text=[f"Correlation: {corr_matrix.loc[var1, var2]:.2f}"],
                        hoverinfo='text'
                    )
                )
                
                # Add correlation annotation
                fig.add_annotation(
                    x=0.5, y=0.5,
                    xref=f'x{i+1} domain',
                    yref=f'y{j+1} domain',
                    text=f"r = {corr_matrix.loc[var1, var2]:.2f}",
                    showarrow=False,
                    font=dict(size=12, color='red')
                )
    
    # Create grid
    fig.update_layout(
        grid=dict(rows=len(numeric_cols), columns=len(numeric_cols), pattern='independent'),
        title='Interactive Correlation Scatter Matrix'
    )
    
    # Add axis labels
    for i, var in enumerate(numeric_cols):
        fig.update_xaxes(title_text=var, row=i+1, col=1)
        fig.update_yaxes(title_text=var, row=1, col=i+1)
    
    return fig

# Apply this function to Iris data
iris_numeric = iris_df.drop(columns=['species'])
matrix_fig = correlation_scatter_matrix(iris_numeric)
matrix_fig.show()
```

## 10. Summary and Best Practices

Correlation visualization is essential for understanding relationships in your data:

- **Scatter plots** are ideal for examining bivariate relationships
- **Heatmaps** provide an excellent overview of correlation patterns across multiple variables
- **Pair plots** allow you to explore many relationships simultaneously

### Key Best Practices:

1. **Always consider correlation type**: Pearson for linear relationships, Spearman or Kendall for non-linear relationships

2. **Be cautious with interpretation**:
   - Correlation does not imply causation
   - Strong correlation may be due to confounding variables
   - Outliers can significantly affect correlation values

3. **Enhance visualizations with additional dimensions**:
   - Use color to represent categories
   - Use size to represent another continuous variable
   - Add regression lines to show trends

4. **Use interactive visualizations** for large datasets to allow for deeper exploration

5. **Complement visualizations with statistical tests** when making decisions based on correlations

6. **Consider clustering** variables in heatmaps to identify groups of related variables

```python
# A complete example combining best practices
def correlation_analysis(df, categorical_var=None):
    """
    Perform a complete correlation analysis on a dataset.
    
    Parameters:
    -----------
    df : pandas.DataFrame
        The dataset to analyze
    categorical_var : str, optional
        Categorical variable for grouping, if applicable
    """
    # 1. Select only numeric columns for correlation
    numeric_df = df.select_dtypes(include=[np.number])
    
    # 2. Calculate correlation matrix
    corr_matrix = numeric_df.corr()
    
    # 3. Create a figure with multiple subplots
    fig = plt.figure(figsize=(20, 20))
    
    # 4. Basic correlation heatmap
    plt.subplot(2, 2, 1)
    sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', 
                vmin=-1, vmax=1, linewidths=0.5)
    plt.title('Correlation Matrix Heatmap')
    
    # 5. Clustered correlation heatmap
    plt.subplot(2, 2, 2)
    mask = np.triu(np.ones_like(corr_matrix, dtype=bool))
    sns.heatmap(corr_matrix, mask=mask, cmap='coolwarm',
               annot=True, center=0, square=True, linewidths=.5)
    plt.title('Clustered Correlation Heatmap (Lower Triangle)')
    
    # 6. Find strongest positive and negative correlations
    corr_pairs = corr_matrix.unstack()
    corr_pairs = corr_pairs[corr_pairs != 1.0]  # Remove self-correlations
    strongest_pos = corr_pairs.nlargest(3)
    strongest_neg = corr_pairs.nsmallest(3)
    
    # 7. Scatter plots of top correlations
    plt.subplot(2, 2, 3)
    if not strongest_pos.empty:
        var1, var2 = strongest_pos.index[0]
        sns.regplot(x=var1, y=var2, data=df, scatter_kws={'alpha': 0.5})
        plt.title(f'Strongest Positive Correlation: {var1} vs {var2}, r={strongest_pos.iloc[0]:.2f}')
    
    plt.subplot(2, 2, 4)
    if not strongest_neg.empty:
        var1, var2 = strongest_neg.index[0]
        sns.regplot(x=var1, y=var2, data=df, scatter_kws={'alpha': 0.5})
        plt.title(f'Strongest Negative Correlation: {var1} vs {var2}, r={strongest_neg.iloc[0]:.2f}')
    
    plt.tight_layout()
    plt.show()
    
    # 8. Generate pair plot if categorical variable is provided
    if categorical_var is not None and categorical_var in df.columns:
        plt.figure(figsize=(15, 15))
        sns.pairplot(df, hue=categorical_var, corner=True, diag_kind='kde')
        plt.suptitle(f'Pair Plot by {categorical_var}', y=1.02, fontsize=16)
        plt.show()
    
    # 9. Return summary of findings
    print("Correlation Analysis Summary:")
    print("-" * 40)
    print("Top 3 Positive Correlations:")
    for i, (idx, val) in enumerate(zip(strongest_pos.index, strongest_pos.values)):
        var1, var2 = idx
        print(f"{i+1}. {var1} and {var2}: r = {val:.4f}")
    
    print("\nTop 3 Negative Correlations:")
    for i, (idx, val) in enumerate(zip(strongest_neg.index, strongest_neg.values)):
        var1, var2 = idx
        print(f"{i+1}. {var1} and {var2}: r = {val:.4f}")

# Run the complete analysis on the iris dataset
correlation_analysis(iris_df, categorical_var='species')
```

Correlation visualization is a fundamental step in exploratory data analysis. By combining statistical methods with visual techniques, you can gain deep insights into the relationships within your data, helping you make better decisions for modeling and analysis.
