## Introduction

Advanced geospatial visualization goes beyond simple plotting of coordinates on a map. It involves creating rich, interactive visualizations that can effectively communicate complex spatial relationships, patterns, and insights from geographic data. This topic explores various advanced techniques and libraries in Python for creating sophisticated geospatial visualizations.

Geospatial visualization is essential in various fields:
- Urban planning and smart cities
- Environmental monitoring and climate change studies
- Transportation and logistics optimization
- Epidemiology and public health
- Market analysis and business intelligence
- Social and demographic studies

In this guide, we'll explore advanced techniques for visualizing geospatial data using Python's rich ecosystem of libraries.

## 3D Terrain Visualization

Creating 3D visualizations of terrain and elevation data:

```python
# This is a placeholder for 3D terrain visualization
# In a real implementation, you would load a DEM file
import numpy as np
from matplotlib import pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# Generate sample terrain data
def generate_terrain(size=100, roughness=0.5):
    noise = np.random.normal(0, 1, (size, size))
    terrain = np.zeros((size, size))
    
    step = size
    while step > 1:
        for i in range(0, size, step):
            for j in range(0, size, step):
                square_avg = noise[i:i+step, j:j+step].mean()
                noise[i:i+step, j:j+step] += square_avg * roughness
        step = step // 2
        roughness /= 2
        
    # Normalize to 0-1
    terrain = (noise - noise.min()) / (noise.max() - noise.min())
    return terrain

# Create a terrain surface
terrain = generate_terrain(size=100, roughness=0.7)

# Plot the terrain in 3D
fig = plt.figure(figsize=(12, 10))
ax = fig.add_subplot(111, projection='3d')

# Create grid for plotting
x = np.linspace(0, 1, terrain.shape[0])
y = np.linspace(0, 1, terrain.shape[1])
x, y = np.meshgrid(x, y)

# Plot the surface
surf = ax.plot_surface(x, y, terrain, 
                       cmap='terrain',
                       linewidth=0,
                       antialiased=True,
                       shade=True)

# Add color bar
fig.colorbar(surf, ax=ax, shrink=0.5, aspect=5, label='Elevation')

# Set labels and title
ax.set_xlabel('X')
ax.set_ylabel('Y')
ax.set_zlabel('Elevation')
```

## Animating Geospatial Data

Creating animations to show changes over time:

```python
# This is a placeholder for creating an animation of a weather system
# In practice, you would use real weather data
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.animation import FuncAnimation
from IPython.display import HTML
import matplotlib.colors as mcolors

# Generate synthetic weather data (temperature changing over time)
def generate_weather_data(width=50, height=25, frames=10):
    # Start with a base temperature distribution
    base = np.zeros((height, width))
    center_x, center_y = width // 2, height // 2
    
    # Create a distance matrix from center
    y, x = np.ogrid[:height, :width]
    dist = np.sqrt((x - center_x)**2 + (y - center_y)**2)
    
    # Create base temperature (warmer in center, cooler at edges)
    max_dist = np.sqrt(center_x**2 + center_y**2)
    base = 30 * (1 - dist / max_dist)  # Temperature in Celsius
    
    # Generate frames with a moving weather system
    frames_data = []
    
    for i in range(frames):
        # Create a moving weather front
        weather_front = np.zeros((height, width))
        front_center_x = int(width * 0.1) + int(width * 0.8 * i / frames)
        
        # Distance from the weather front
        front_dist = np.sqrt((x - front_center_x)**2 + (y - center_y)**2)
        front_effect = 15 * np.exp(-front_dist**2 / (2 * (width/5)**2))
        
        # Lower temperature due to weather front
        frame = base - front_effect
        frames_data.append(frame)
    
    return frames_data

# Generate weather data
weather_frames = generate_weather_data(width=80, height=40, frames=20)

# Create a custom colormap for temperature
colors = [(0, 'darkblue'), (0.2, 'blue'), (0.4, 'cyan'), 
          (0.6, 'lightgreen'), (0.8, 'yellow'), (1, 'red')]
cmap = mcolors.LinearSegmentedColormap.from_list('temperature', colors)

# Set up the figure and axis
fig, ax = plt.subplots(figsize=(12, 6))

# Initialize plot
img = ax.imshow(weather_frames[0], cmap=cmap, vmin=0, vmax=30)
cbar = fig.colorbar(img, ax=ax, label='Temperature (°C)')
ax.set_title('Weather System Movement Animation', fontsize=14)
ax.set_xlabel('Longitude')
ax.set_ylabel('Latitude')
ax.axis('off')

# Animation function
def update(frame):
    img.set_array(weather_frames[frame])
    ax.set_title(f'Weather System Animation - Day {frame+1}', fontsize=14)
    return [img]

# Create animation
ani = FuncAnimation(fig, update, frames=len(weather_frames), interval=200, blit=True)

# Display the animation
HTML(ani.to_jshtml())

# Save animation (optional)
```

```python
    east_london = Polygon([(-0.05, 51.50), (0.05, 51.50), (0.05, 51.55), (-0.05, 51.55)])
    south_london = Polygon([(-0.22, 51.45), (-0.05, 51.45), (-0.05, 51.50), (-0.22, 51.50)])
    west_london = Polygon([(-0.22, 51.50), (-0.05, 51.50), (-0.05, 51.55), (-0.22, 51.55)])
    
    uk = gpd.GeoDataFrame({
        'name': ['North London', 'East London', 'South London', 'West London'],
        'geometry': [north_london, east_london, south_london, west_london]
    }, crs="EPSG:4326")

# Perform a spatial join
joined_data = gpd.sjoin(points_gdf, uk, how='left', op='within')

# Count points in each region
point_counts = joined_data.groupby('name').size().reset_index(name='count')

# Merge counts back with regions
uk_with_counts = uk.merge(point_counts, on='name', how='left')
uk_with_counts['count'] = uk_with_counts['count'].fillna(0)

# Plot result
fig, ax = plt.subplots(figsize=(12, 8))

# Plot regions colored by point count
uk_with_counts.plot(column='count', cmap='YlOrRd', legend=True, ax=ax,
                  legend_kwds={'label': 'Number of Points'})

# Plot points
points_gdf.plot(ax=ax, color='blue', markersize=1, alpha=0.1)

ax.set_title('Spatial Distribution Analysis', fontsize=14)
ax.set_xlabel('Longitude')
```
