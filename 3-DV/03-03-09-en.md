## Introduction

Complex networks are systems of interconnected components that can be found in many domains including social relationships, biological systems, transportation routes, computer networks, and citation patterns. Visualizing these networks helps us understand structures, patterns, and relationships that might otherwise be hidden in the data.

In this guide, we'll explore techniques for visualizing complex networks in Python, from basic graph representations to interactive and large-scale network visualizations.

## Network Visualization Fundamentals

### What Makes a Network Visualization Effective

An effective network visualization should:
- Clearly show connections between nodes
- Reveal important structures (clusters, hubs, bridges)
- Minimize visual clutter
- Highlight key metrics or properties
- Be interpretable by the intended audience

### Common Network Types

- **Undirected networks**: Connections have no direction (friendship networks)
- **Directed networks**: Connections have direction (follower relationships)
- **Weighted networks**: Connections have varying strengths
- **Bipartite networks**: Nodes fall into two distinct groups
- **Temporal networks**: Connections change over time
- **Multilayer networks**: Multiple types of relationships exist simultaneously

## Interactive Network Visualization

Interactive visualizations allow for exploration of complex networks:

```python
# Using Plotly for interactive visualization
def create_interactive_network(G, pos=None, title="Interactive Network"):
    if pos is None:
        pos = nx.spring_layout(G, seed=42)
    
    # Create edge traces
    edge_x = []
    edge_y = []
    for edge in G.edges():
        x0, y0 = pos[edge[0]]
        x1, y1 = pos[edge[1]]
        edge_x.extend([x0, x1, None])
        edge_y.extend([y0, y1, None])
    
    edge_trace = go.Scatter(
        x=edge_x, y=edge_y,
        line=dict(width=0.5, color='#888'),
        hoverinfo='none',
        mode='lines')
    
    # Create node traces
    node_x = []
    node_y = []
    for node in G.nodes():
        x, y = pos[node]
        node_x.append(x)
        node_y.append(y)
    
    node_trace = go.Scatter(
        x=node_x, y=node_y,
        mode='markers',
        hoverinfo='text',
        marker=dict(
            showscale=True,
            colorscale='YlGnBu',
            reversescale=True,
            color=[],
            size=10,
            colorbar=dict(
                thickness=15,
                title='Node Connections',
                xanchor='left',
                titleside='right'
            ),
            line_width=2))
    
    # Color node points by the number of connections
    node_adjacencies = []
    node_text = []
    for node in G.nodes():
        adjacencies = list(G.neighbors(node))
        node_adjacencies.append(len(adjacencies))
        node_text.append(f'Node: {node}<br>Connections: {len(adjacencies)}')
    
    node_trace.marker.color = node_adjacencies
    node_trace.text = node_text
    
    # Create the figure
    fig = go.Figure(data=[edge_trace, node_trace],
                 layout=go.Layout(
                    title=title,
                    titlefont_size=16,
                    showlegend=False,
                    hovermode='closest',
                    margin=dict(b=20,l=5,r=5,t=40),
                    xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
                    yaxis=dict(showgrid=False, zeroline=False, showticklabels=False))
                    )
    return fig

# Create a moderate-sized random graph for demonstration
```

## Large-Scale Network Visualization

For larger networks, we need special techniques to make visualizations interpretable:

```python
# Generate a larger network
G_large = nx.barabasi_albert_graph(500, 3, seed=42)

# Calculate node degrees
degrees = dict(nx.degree(G_large))

# Function to filter to only show important nodes and connections
def filter_graph_by_degree(G, min_degree=5):
    important_nodes = [node for node, degree in dict(G.degree()).items() if degree >= min_degree]
    subgraph = G.subgraph(important_nodes)
    return subgraph

# Create a filtered graph
G_filtered = filter_graph_by_degree(G_large, min_degree=10)

print(f"Original network: {G_large.number_of_nodes()} nodes, {G_large.number_of_edges()} edges")
print(f"Filtered network: {G_filtered.number_of_nodes()} nodes, {G_filtered.number_of_edges()} edges")

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(20, 10))

pos_large = nx.spring_layout(G_large, seed=42)
nx.draw(G_large, pos=pos_large, ax=ax1, node_size=10, node_color='blue', alpha=0.6, width=0.1)
```

```python
# Create a simulated citation network
def create_citation_network(num_papers=100, citations_per_paper=5):
    G = nx.DiGraph()  # Directed graph for citations
    
    # Add papers as nodes with metadata
    topics = ["Machine Learning", "Networks", "Visualization", "Statistics"]
    years = list(range(2010, 2023))
    
    for i in range(num_papers):
        G.add_node(i, 
                  title=f"Paper {i}",
                  topic=random.choice(topics),
                  year=random.choice(years),
                  citations=0)
    
    # Add citations (papers can only cite older papers)
    for i in range(num_papers):
        year_i = G.nodes[i]['year']
        # Find potential papers to cite (older papers)
        candidates = [j for j in range(num_papers) if G.nodes[j]['year'] < year_i]
        if candidates:
            # Add citations
            num_citations = min(len(candidates), random.randint(1, citations_per_paper))
            for cited_paper in random.sample(candidates, num_citations):
                G.add_edge(i, cited_paper)
                G.nodes[cited_paper]['citations'] += 1
    
    return G

# Create the citation network
G_citations = create_citation_network(num_papers=80, citations_per_paper=4)

# Calculate metrics
in_degree_centrality = nx.in_degree_centrality(G_citations)  # Measures citation impact
```

## Case Study: Social Network Analysis

Let's analyze and visualize a simulated social network:

```python
# Create a simulated social network with communities
def create_social_network(num_users=100, num_communities=4, p_in=0.15, p_out=0.01):
    # Partition size for each community
    partition_sizes = [num_users // num_communities] * num_communities
    for i in range(num_users % num_communities):
        partition_sizes[i] += 1
    
    # Create a graph with community structure
    G = nx.random_partition_graph(partition_sizes, p_in, p_out, seed=42)
    
    # Add attributes to nodes
    age_groups = ["18-24", "25-34", "35-44", "45+"]
    interests = ["Sports", "Technology", "Arts", "Travel", "Food", "Music"]
    activity_levels = ["Low", "Medium", "High"]
    
    for i, node in enumerate(G.nodes()):
        community_id = i // (num_users // num_communities)  # Approximate community assignment
        G.nodes[node]['community'] = community_id
        G.nodes[node]['age_group'] = random.choice(age_groups)
        G.nodes[node]['interests'] = random.sample(interests, random.randint(1, 3))
        G.nodes[node]['activity'] = random.choice(activity_levels)
        G.nodes[node]['influence'] = random.uniform(0, 1)  # Random influence score
    
    # Calculate network metrics and add as attributes
    degree_cent = nx.degree_centrality(G)
    betweenness_cent = nx.betweenness_centrality(G, k=min(20, num_users))  # Approximate for larger networks
    eigenvector_cent = nx.eigenvector_centrality_numpy(G)
    
    for node in G.nodes():
        G.nodes[node]['degree_centrality'] = degree_cent[node]
        G.nodes[node]['betweenness_centrality'] = betweenness_cent[node]
        G.nodes[node]['eigenvector_centrality'] = eigenvector_cent[node]
    
    return G
```
