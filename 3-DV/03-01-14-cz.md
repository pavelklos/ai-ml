## 1. Úvod do korelační analýzy

Korelační analýza nám pomáhá porozumět vztahům mezi proměnnými v našich datech. Vizualizace těchto vztahů je klíčová pro:

- Identifikaci vzorů a trendů
- Detekci potenciálních prediktivních vztahů
- Objevení problémů s multikolinearitou
- Podporu výběru příznaků pro modely strojového učení

Nastavme si prostředí s potřebnými knihovnami:

```python
# Import základních knihoven pro vizualizaci korelací
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import plotly.express as px
import plotly.graph_objects as go
from sklearn.datasets import load_iris, load_boston, load_diabetes

# Nastavení stylů vizualizace
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_theme(style="whitegrid")

# Pro reprodukovatelné výsledky
np.random.seed(42)

# Nastavení zobrazení pro pandas
pd.set_option('display.max_columns', 20)

# Načtení vzorových datasetů pro naše příklady
iris = load_iris()
iris_df = pd.DataFrame(iris.data, columns=iris.feature_names)
iris_df['species'] = iris.target_names[iris.target]

# Vytvoření vzorového datasetu se známými korelacemi
n_samples = 200
x = np.random.normal(0, 1, n_samples)
# Vytvoření proměnných s různou silou korelace
strong_pos_corr = x * 0.9 + np.random.normal(0, 0.3, n_samples)  # Silná pozitivní
medium_pos_corr = x * 0.5 + np.random.normal(0, 0.7, n_samples)  # Střední pozitivní
weak_pos_corr = x * 0.2 + np.random.normal(0, 0.9, n_samples)    # Slabá pozitivní
weak_neg_corr = x * -0.2 + np.random.normal(0, 0.9, n_samples)   # Slabá negativní
strong_neg_corr = x * -0.9 + np.random.normal(0, 0.3, n_samples) # Silná negativní
no_corr = np.random.normal(0, 1, n_samples)                      # Žádná korelace

# Vytvoření DataFrame s těmito proměnnými
corr_df = pd.DataFrame({
    'x': x,
    'strong_pos': strong_pos_corr,
    'medium_pos': medium_pos_corr,
    'weak_pos': weak_pos_corr,
    'weak_neg': weak_neg_corr,
    'strong_neg': strong_neg_corr,
    'no_corr': no_corr
})

print("Ukázkový korelační datový rámec:")
print(corr_df.head())

# Výpočet korelační matice
correlation_matrix = corr_df.corr()
print("\nKorelační matice:")
print(correlation_matrix['x'].sort_values(ascending=False))
```

## 2. Porozumění korelačním koeficientům

Před vizualizací korelací je důležité pochopit, co vlastně měříme:

- **Pearsonův korelační koeficient** (r): Měří lineární vztahy (od -1 do +1)
- **Spearmanův pořadový koeficient korelace**: Měří monotónní vztahy (funguje s nelineárními daty)
- **Kendallovo Tau**: Další míra pořadové korelace, robustní vůči odlehlým hodnotám

```python
# Výpočet různých typů korelačních koeficientů
pearson_corr = corr_df.corr(method='pearson')
spearman_corr = corr_df.corr(method='spearman')
kendall_corr = corr_df.corr(method='kendall')

# Vytvoření funkce pro zobrazení různých korelačních měr pro porovnání
def compare_correlation_methods(x_var, y_var, data):
    """Porovnat různé metody korelace mezi dvěma proměnnými."""
    pearson = data[x_var].corr(data[y_var], method='pearson')
    spearman = data[x_var].corr(data[y_var], method='spearman')
    kendall = data[x_var].corr(data[y_var], method='kendall')
    
    print(f"Korelace mezi {x_var} a {y_var}:")
    print(f"  Pearson: {pearson:.4f}")
    print(f"  Spearman: {spearman:.4f}")
    print(f"  Kendall: {kendall:.4f}")
    
    # Vytvoření bodového grafu s informacemi o korelaci
    plt.figure(figsize=(8, 6))
    plt.scatter(data[x_var], data[y_var], alpha=0.7)
    plt.title(f"Korelace mezi {x_var} a {y_var}")
    plt.xlabel(x_var)
    plt.ylabel(y_var)
    plt.annotate(f"Pearson r: {pearson:.4f}\nSpearman ρ: {spearman:.4f}\nKendall τ: {kendall:.4f}",
                xy=(0.05, 0.95), xycoords='axes fraction',
                bbox=dict(boxstyle="round,pad=0.5", fc="white", alpha=0.8),
                va='top')
    plt.grid(True, alpha=0.3)
    plt.show()

# Porovnání korelačních metod pro různé síly vztahů
compare_correlation_methods('x', 'strong_pos', corr_df)
compare_correlation_methods('x', 'medium_pos', corr_df)
compare_correlation_methods('x', 'no_corr', corr_df)

# Vytvoření nelineárního vztahu pro ukázku rozdílu v korelačních metodách
n = 200
x_nonlinear = np.linspace(0, 10, n)
y_nonlinear = x_nonlinear**2 + np.random.normal(0, 10, n)
nonlinear_df = pd.DataFrame({'x': x_nonlinear, 'y': y_nonlinear})

# Porovnání korelačních metod pro nelineární data
compare_correlation_methods('x', 'y', nonlinear_df)
```

```python
# Výpočet korelační matice pro příznaky Iris
iris_corr = iris_df.drop(columns=['species']).corr()

# Vytvoření funkce pro zobrazení absolutních hodnot korelace
def get_absolute_corr(df):
    corr_matrix = df.corr().abs()
    # Nastavení vlastní korelace (diagonály) na 0 pro nalezení nejsilnějších ne-vlastních korelací
    np.fill_diagonal(corr_matrix.values, 0)
    return corr_matrix

# Nalezení nejsilnějších korelací
abs_corr = get_absolute_corr(iris_df.drop(columns=['species']))
strongest_pair = abs_corr.unstack().sort_values(ascending=False)[0:1]
print("\nNejsilnější korelace:")
print(strongest_pair)
```

```python
# Funkce pro zvýraznění významných korelací
def highlight_significant_corr(corr_matrix, threshold=0.5):
    """
    Vytvoří teplotní mapu se zvýrazněnými významnými korelacemi.
    
    Parametry:
    -----------
    corr_matrix : pandas.DataFrame
        Korelační matice k vizualizaci
    threshold : float
        Práh absolutní korelace pro zvýraznění
    """
    plt.figure(figsize=(12, 10))
    
    # Vytvoření masky pro nevýznamné korelace
    insignificant = np.abs(corr_matrix) < threshold
    
    # Vytvoření teplotní mapy pouze s významnými korelacemi
    ax = sns.heatmap(corr_matrix, 
                     cmap='coolwarm',
                     annot=True,
                     fmt='.2f', 
                     linewidths=0.5,
                     vmin=-1, vmax=1,
                     mask=insignificant)
    
    plt.title(f'Teplotní mapa korelací (|r| ≥ {threshold})')
    return ax

# Aplikace naší vlastní funkce

plt.show()

                               cmap='coolwarm',
                               annot=True,

                               vmin=-1, vmax=1,
                               figsize=(12, 12))

plt.title('Hierarchicky seskupená korelační matice')
```

## 9. Interaktivní vizualizace korelací

Interaktivní vizualizace mohou poskytnout lepší zážitek z průzkumu, zejména pro velké datasety:

```python
# Interaktivní teplotní mapa s anotacemi pomocí Plotly
def create_interactive_corr_heatmap(df, title='Interaktivní teplotní mapa korelací'):
    """Vytvoření interaktivní teplotní mapy korelací s Plotly."""
    corr_matrix = df.corr().round(2)
    
    # Vytvoření masky pro horní trojúhelník
    mask = np.zeros_like(corr_matrix, dtype=bool)
    mask[np.triu_indices_from(mask, k=1)] = True
    
    # Převod maskované korelační matice do přehledného formátu
    tidy_corr = corr_matrix.stack().reset_index()
    tidy_corr.columns = ['Var1', 'Var2', 'Correlation']
    # Filtrování horního trojúhelníku
    tidy_corr = tidy_corr[~mask.stack()]
    
    # Vytvoření teplotní mapy
    fig = px.scatter(tidy_corr, x='Var1', y='Var2', size='Correlation', 
                     color='Correlation', size_max=60,
                     color_continuous_scale='RdBu_r',
                     range_color=[-1, 1],
                     text='Correlation',
                     hover_data={'Correlation': ':.2f'})
    
    fig.update_traces(textposition='middle center', textfont=dict(color='white'))
    fig.update_layout(title=title,
                      xaxis=dict(title=''),
                      yaxis=dict(title=''))
    
    return fig

# Aplikace funkce na náš dataset
corr_fig = create_interactive_corr_heatmap(iris_df.drop(columns=['species']))
corr_fig.show()

# Interaktivní matice bodových grafů s informacemi o korelaci
def correlation_scatter_matrix(df, numeric_cols=None):
    """
    Vytvoří interaktivní matici bodových grafů s informacemi o korelaci.
    
    Parametry:
    -----------
    df : pandas.DataFrame
        Data k vizualizaci
    numeric_cols : list of str, optional
        Seznam číselných sloupců k zahrnutí. Pokud None, použijí se všechny číselné sloupce.
    """
    # Výběr číselných sloupců, pokud nejsou specifikovány
    if numeric_cols is None:
        numeric_cols = df.select_dtypes(include=[np.number]).columns.tolist()
    
    # Vytvoření podgrafů
    fig = go.Figure()
    
    # Výpočet korelační matice
    corr_matrix = df[numeric_cols].corr().round(2)
    
    # Přidání bodového grafu pro každou dvojici proměnných
    for i, var1 in enumerate(numeric_cols):
        for j, var2 in enumerate(numeric_cols):
            if i > j:  # Dolní trojúhelník
                fig.add_trace(
                    go.Scatter(
                        x=df[var1],
                        y=df[var2],
                        mode='markers',
                        marker=dict(size=6, opacity=0.5),
                        name=f'{var1} vs {var2}',
                        xaxis=f'x{i+1}',
                        yaxis=f'y{j+1}',
                        text=[f"Korelace: {corr_matrix.loc[var1, var2]:.2f}"],
                        hoverinfo='text'
                    )
                )
                
                # Přidání anotace korelace
                fig.add_annotation(
                    x=0.5, y=0.5,
                    xref=f'x{i+1} domain',
                    yref=f'y{j+1} domain',
                    text=f"r = {corr_matrix.loc[var1, var2]:.2f}",
                    showarrow=False,
                    font=dict(size=12, color='red')
                )
    
    # Vytvoření mřížky
    fig.update_layout(
        grid=dict(rows=len(numeric_cols), columns=len(numeric_cols), pattern='independent'),
        title='Interaktivní korelační matice bodových grafů'
    )
    
    # Přidání popisků os
    for i, var in enumerate(numeric_cols):
        fig.update_xaxes(title_text=var, row=i+1, col=1)
        fig.update_yaxes(title_text=var, row=1, col=i+1)
    
    return fig

# Aplikace této funkce na data Iris
iris_numeric = iris_df.drop(columns=['species'])
matrix_fig = correlation_scatter_matrix(iris_numeric)
matrix_fig.show()
```

## 10. Shrnutí a osvědčené postupy

Vizualizace korelací je nezbytná pro pochopení vztahů v datech:

- **Bodové grafy** jsou ideální pro zkoumání bivariačních vztahů
- **Teplotní mapy** poskytují vynikající přehled o korelačních vzorech napříč více proměnnými
- **Párové grafy** umožňují současně prozkoumat mnoho vztahů

### Klíčové osvědčené postupy:

1. **Vždy zvažujte typ korelace**: Pearson pro lineární vztahy, Spearman nebo Kendall pro nelineární vztahy

2. **Buďte opatrní při interpretaci**:
   - Korelace neimplikuje kauzalitu
   - Silná korelace může být způsobena zavádějícími proměnnými
   - Odlehlé hodnoty mohou výrazně ovlivnit hodnoty korelace

3. **Vylepšujte vizualizace dodatečnými dimenzemi**:
   - Použijte barvu k reprezentaci kategorií
   - Použijte velikost k reprezentaci další spojité proměnné
   - Přidejte regresní přímky pro znázornění trendů

4. **Používejte interaktivní vizualizace** pro velké datasety, které umožní hlubší průzkum

5. **Doplňujte vizualizace statistickými testy**, pokud činíte rozhodnutí na základě korelací

6. **Zvažte shlukování** proměnných v teplotních mapách pro identifikaci skupin souvisejících proměnných

```python
# Kompletní příklad kombinující osvědčené postupy
def correlation_analysis(df, categorical_var=None):
    """
    Proveďte kompletní korelační analýzu na datasetu.
    
    Parametry:
    -----------
    df : pandas.DataFrame
        Dataset k analýze
    categorical_var : str, optional
        Kategorická proměnná pro seskupení, pokud je použitelná
    """
    # 1. Vyberte pouze číselné sloupce pro korelaci
    numeric_df = df.select_dtypes(include=[np.number])
    
    # 2. Výpočet korelační matice
    corr_matrix = numeric_df.corr()
    
    # 3. Vytvoření obrazce s více podgrafy
    fig = plt.figure(figsize=(20, 20))
    
    # 4. Základní korelační teplotní mapa
    plt.subplot(2, 2, 1)
    sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', 
                vmin=-1, vmax=1, linewidths=0.5)
    plt.title('Teplotní mapa korelační matice')
    
    # 5. Seskupená korelační teplotní mapa
    plt.subplot(2, 2, 2)
    mask = np.triu(np.ones_like(corr_matrix, dtype=bool))
    sns.heatmap(corr_matrix, mask=mask, cmap='coolwarm',
               annot=True, center=0, square=True, linewidths=.5)
    plt.title('Seskupená teplotní mapa korelací (dolní trojúhelník)')
    
    # 6. Nalezení nejsilnějších pozitivních a negativních korelací
    corr_pairs = corr_matrix.unstack()
    corr_pairs = corr_pairs[corr_pairs != 1.0]  # Odstranění vlastních korelací
    strongest_pos = corr_pairs.nlargest(3)
    strongest_neg = corr_pairs.nsmallest(3)
    
    # 7. Bodové grafy top korelací
    plt.subplot(2, 2, 3)
    if not strongest_pos.empty:
        var1, var2 = strongest_pos.index[0]
        sns.regplot(x=var1, y=var2, data=df, scatter_kws={'alpha': 0.5})
        plt.title(f'Nejsilnější pozitivní korelace: {var1} vs {var2}, r={strongest_pos.iloc[0]:.2f}')
    
    plt.subplot(2, 2, 4)
    if not strongest_neg.empty:
        var1, var2 = strongest_neg.index[0]
        sns.regplot(x=var1, y=var2, data=df, scatter_kws={'alpha': 0.5})
        plt.title(f'Nejsilnější negativní korelace: {var1} vs {var2}, r={strongest_neg.iloc[0]:.2f}')
    
    plt.tight_layout()
    plt.show()
    
    # 8. Vygenerování párového grafu, pokud je poskytnuta kategorická proměnná
    if categorical_var is not None and categorical_var in df.columns:
        plt.figure(figsize=(15, 15))
        sns.pairplot(df, hue=categorical_var, corner=True, diag_kind='kde')
        plt.suptitle(f'Párový graf podle {categorical_var}', y=1.02, fontsize=16)
        plt.show()
    
    # 9. Vrácení souhrnu nálezů
    print("Souhrn korelační analýzy:")
    print("-" * 40)
    print("Top 3 pozitivní korelace:")
    for i, (idx, val) in enumerate(zip(strongest_pos.index, strongest_pos.values)):
        var1, var2 = idx
        print(f"{i+1}. {var1} a {var2}: r = {val:.4f}")
    
    print("\nTop 3 negativní korelace:")
    for i, (idx, val) in enumerate(zip(strongest_neg.index, strongest_neg.values)):
        var1, var2 = idx
        print(f"{i+1}. {var1} a {var2}: r = {val:.4f}")
```
