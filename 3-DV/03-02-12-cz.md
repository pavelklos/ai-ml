# Animované vizualizace v Pythonu

## Úvod

Animované vizualizace oživují data tím, že ukazují, jak se hodnoty mění v průběhu času nebo při různých podmínkách. Mohou odhalit vzory, přechody a časové vztahy, které by statické vizualizace mohly přehlédnout. Tento průvodce zkoumá, jak vytvořit působivé animované vizualizace v Pythonu pomocí různých knihoven, od jednoduchých animovaných grafů až po interaktivní vizualizace, které zaujmou vaše publikum.

Animace přidává mocnou dimenzi vyprávění příběhů pomocí dat tím, že vizualizuje:
- Vývoj časových řad
- Postupné procesy
- Srovnávací analýzu napříč různými podmínkami
- Chování dynamických systémů
- Komplexní prostorově-časové vzory

## Základní principy animace

Animace funguje zobrazením sekvence snímků rychlostí, která vytváří iluzi plynulého pohybu. Klíčové koncepty zahrnují:

- **Snímky**: Jednotlivé obrázky, které tvoří animaci
- **Snímková frekvence**: Rychlost, jakou jsou snímky zobrazovány (snímky za sekundu)
- **Přechody**: Jak se jeden snímek mění na další
- **Jemnost přechodu (easing)**: Řídí zrychlení a zpomalení přechodů

Vytvořme jednoduchou animaci pro ilustraci těchto konceptů:

```python
# Vytvoření jednoduché animace pro demonstraci základních principů
fig, ax = plt.subplots(figsize=(8, 6))

# Inicializace prázdné čáry
line, = ax.plot([], [], 'b-', linewidth=2)
ax.set_xlim(0, 2*np.pi)
ax.set_ylim(-1.2, 1.2)
ax.set_title('Animace sinusové vlny')
ax.grid(True)

# Funkce pro inicializaci grafu
def init():
    line.set_data([], [])
    return (line,)

# Funkce pro aktualizaci animace
def animate(i):
    x = np.linspace(0, 2*np.pi, 1000)
    y = np.sin(x - 0.1 * i)  # Pohybující se sinusová vlna
    line.set_data(x, y)
    return (line,)

# Vytvoření animace
ani = FuncAnimation(fig, animate, frames=100, init_func=init,
                    interval=50, blit=True)

# Zobrazení animace v notebooku
from IPython.display import HTML
HTML(ani.to_jshtml())
```

## Animace v Matplotlib

Animační API knihovny Matplotlib umožňuje vytvářet dynamické vizualizace s jemným ovládáním:

```python
# Vytvoření složitější animace s matplotlib

# Vygenerování dat náhodné procházky
np.random.seed(42)
n_walks = 5
n_steps = 100
walks = np.random.normal(0, 1, (n_walks, n_steps)).cumsum(axis=1)

# Nastavení grafu a os
fig, ax = plt.subplots(figsize=(10, 6))
ax.set_xlim(0, n_steps)
ax.set_ylim(walks.min() - 1, walks.max() + 1)
ax.set_title('Animace náhodné procházky')
ax.set_xlabel('Krok')
ax.set_ylabel('Pozice')

# Inicializace čar
lines = [ax.plot([], [], label=f"Chodec {i+1}", lw=2)[0] for i in range(n_walks)]
ax.legend()

# Funkce pro inicializaci animace
def init():
    for line in lines:
        line.set_data([], [])
    return lines

# Funkce pro aktualizaci animace
def animate(i):
    for j, line in enumerate(lines):
        x = list(range(i+1))
        y = walks[j, :i+1]
        line.set_data(x, y)
    return lines

# Vytvoření animace
ani = FuncAnimation(fig, animate, frames=n_steps, init_func=init,
                   interval=100, blit=True)

# Uložení animace jako GIF
ani.save('nahodne_prochazky.gif', writer='pillow', fps=15)

# Zobrazení v notebooku
HTML(ani.to_jshtml())
```

```python
fig.update_layout(
    xaxis_title='Kategorie',
```

## Vytváření GIFů pomocí ImageIO

Pro více vlastních animací můžeme vytvořit jednotlivé snímky a sestavit je pomocí ImageIO:

```python
# Vytváření animací s ImageIO

# Funkce pro vytvoření bodového grafu pro konkrétní snímek
def create_frame(frame_index, total_frames):
    # Vytvoření nových dat pro tento snímek
    n_points = 100
    x = np.random.normal(5 * np.cos(2 * np.pi * frame_index / total_frames), 0.5, n_points)
    y = np.random.normal(5 * np.sin(2 * np.pi * frame_index / total_frames), 0.5, n_points)
    colors = np.random.rand(n_points)
    
    # Vytvoření grafu
    fig, ax = plt.subplots(figsize=(8, 8))
    scatter = ax.scatter(x, y, c=colors, cmap='viridis', alpha=0.7)
    ax.set_xlim(-7, 7)
    ax.set_ylim(-7, 7)
    ax.grid(True, alpha=0.3)
    ax.set_title(f'Snímek {frame_index+1}/{total_frames}')
    
    # Uložení grafu do bufferu
    buf = io.BytesIO()
    plt.savefig(buf, format='png')
    plt.close(fig)
    buf.seek(0)
    
    # Vytvoření obrázku z bufferu
    img = Image.open(buf)
    return np.array(img)

# Vytvoření více snímků
n_frames = 24
frames = [create_frame(i, n_frames) for i in range(n_frames)]

# Uložení jako GIF
imageio.mimsave('animace_orbity.gif', frames, duration=0.1)

# Zobrazení GIFu (v notebooku)
with open('animace_orbity.gif', 'rb') as f:
    display(Image.open(io.BytesIO(f.read())))
```

```python
np.random.seed(42)

n_cities = 20
n_days = 10
city_locations = {
    'město': [f'Město {i}' for i in range(n_cities)],
    'lat': np.random.uniform(25, 50, n_cities),

temp_data = []
for day in range(1, n_days + 1):
    for i in range(n_cities):
        base_temp = np.random.normal(70, 10)  # Základní teplota ve Fahrenheitech
        daily_variation = np.sin(day/n_days * np.pi) * 15  # Teplota se mění s časem
        city_temp = base_temp + daily_variation + np.random.normal(0, 3)  # Přidání náhodnosti
        
        temp_data.append({
            'město': f'Město {i}',
            'den': day,
            'teplota': city_temp,
            'lat': city_locations['lat'][i],
            'lon': city_locations['lon'][i]

geo_df = pd.DataFrame(temp_data)

fig = px.scatter_geo(
    geo_df,
    lat='lat',
    lon='lon',
    color='teplota',
    size='teplota',
    animation_frame='den',
    projection='albers usa',
    scope='usa',
    title='Změny teploty napříč městy v USA',
    color_continuous_scale='Viridis'
)

fig.update_layout(
    height=600,
    width=900
)

fig.show()
```

## Animace dat v reálném čase

Pro vizualizaci dat v reálném čase můžeme dynamicky aktualizovat grafy:

```python
# Animace v reálném čase s Matplotlib

# Funkce pro generování streamovaných dat
def generate_data():
    return np.random.normal(0, 1, 100).cumsum()

fig, ax = plt.subplots(figsize=(10, 6))
ax.set_xlim(0, 100)
ax.set_ylim(-15, 15)
ax.set_title('Simulovaný datový proud v reálném čase')
ax.set_xlabel('Čas')
ax.set_ylabel('Hodnota')
ax.grid(True)

# Inicializace čáry
line, = ax.plot([], [], lw=2)

# Inicializace animace
def init():
    line.set_data([], [])
    return (line,)

# Buffer dat
data_buffer = []

# Funkce aktualizace animace
def animate(i):
    # V reálné aplikaci byste zde získávali nová data
    if i == 0:
        data_buffer.clear()
    
    # Simulace příchodu nových dat
    data_buffer.append(np.random.normal(0, 1))
    
    # Vytvoření kumulativního součtu pro simulaci náhodné procházky
    y = np.cumsum(data_buffer)
    x = list(range(len(y)))
    
    # Aktualizace čáry
    line.set_data(x, y)
    
    # Úprava limitů y podle potřeby
    if len(y) > 0:
        ax.set_ylim(min(min(y)-1, ax.get_ylim()[0]), max(max(y)+1, ax.get_ylim()[1]))
    
    # Úprava limitů x pro efekt posouvání, pokud data přesáhnou aktuální pohled
    if len(x) >= 100:
        ax.set_xlim(len(x) - 100, len(x))
    
    return (line,)

# Vytvoření animace
ani = FuncAnimation(fig, animate, init_func=init, frames=200,
                    interval=50, blit=True)

# Uložení jako GIF
ani.save('data_v_realnem_case.gif', writer='pillow', fps=20)

# Zobrazení v notebooku
HTML(ani.to_jshtml())
```

## Optimalizace výkonu

Animace mohou být náročné na zdroje. Zde jsou některé tipy pro lepší výkon:

```python
# Animace optimalizovaná pro výkon

# 1. Použití blittingu pro lepší výkon
def optimized_animation():
    fig, ax = plt.subplots(figsize=(8, 6))
    x = np.linspace(0, 2*np.pi, 100)
    line, = ax.plot(x, np.sin(x), 'b-', lw=2)
    ax.set_xlim(0, 2*np.pi)
    ax.set_ylim(-1.2, 1.2)
    
    def update(frame):
        # Aktualizace pouze toho, co se mění (nepřekreslování všeho)
        line.set_ydata(np.sin(x + frame/10))
        return (line,)
    
    ani = FuncAnimation(fig, update, frames=100, interval=30, blit=True)
    return ani

# 2. Snížení počtu datových bodů, pokud je to možné
def reduced_data_animation():
    # Použití méně datových bodů stále vypadá dobře, ale běží rychleji
    fig, ax = plt.subplots(figsize=(8, 6))
    x = np.linspace(0, 2*np.pi, 50)  # Sníženo ze 100 na 50 bodů
    line, = ax.plot(x, np.sin(x), 'b-', lw=2)
    ax.set_xlim(0, 2*np.pi)
    ax.set_ylim(-1.2, 1.2)
    
    def update(frame):
        line.set_ydata(np.sin(x + frame/10))
        return (line,)
    
    ani = FuncAnimation(fig, update, frames=100, interval=30, blit=True)
    return ani

# 3. Pokud možno, ukládejte výpočty do mezipaměti
def cached_computation_animation():
    fig, ax = plt.subplots(figsize=(8, 6))
    x = np.linspace(0, 2*np.pi, 100)
    line, = ax.plot(x, np.sin(x), 'b-', lw=2)
    ax.set_xlim(0, 2*np.pi)
    ax.set_ylim(-1.2, 1.2)
    
    # Předpočítání všech snímků
    frames = [np.sin(x + i/10) for i in range(100)]
    
    def update(frame):
        line.set_ydata(frames[frame])  # Použití předpočítaných dat
        return (line,)
    
    ani = FuncAnimation(fig, update, frames=100, interval=30, blit=True)
    return ani

# Zobrazení optimalizované animace
optimized_ani = optimized_animation()
HTML(optimized_ani.to_jshtml())
```

## Osvědčené postupy pro efektivní animace

```python
# Příklady demonstrující osvědčené postupy

# 1. Jasný účel: Ukázat stavy před a po
def animation_with_purpose():
    fig, ax = plt.subplots(figsize=(8, 6))
    
    # Generování vzorových dat
    x = np.linspace(0, 10, 100)
    y1 = np.sin(x)
    y2 = np.sin(x) * np.exp(x/10)
    
    # Inicializace čáry
    line, = ax.plot(x, y1, 'b-', lw=2)
    ax.set_xlim(0, 10)
    ax.set_ylim(-3, 15)
    ax.set_title('Efekt exponenciálního růstu na sinusovou vlnu')
    ax.grid(True)
    
    # Přidání anotace pro přehlednost
    annotation = ax.annotate("", xy=(0, 0), xytext=(1, 1),
                            arrowprops=dict(arrowstyle="->"))
    
    def update(frame):
        # Přechod mezi dvěma funkcemi
        t = frame / 50  # normalizovaný čas od 0 do 1
        y = (1-t) * y1 + t * y2  # lineární interpolace
        
        line.set_ydata(y)
        
        # Aktualizace anotace
        if frame == 0:
            annotation.set_text("Původní sinusová vlna")
            annotation.xy = (5, 0)
            annotation.xytext = (7, 2)
        elif frame == 50:
            annotation.set_text("S exponenciálním růstem")
            annotation.xy = (8, 10)
            annotation.xytext = (6, 7)
        
        return line, annotation
    
    ani = FuncAnimation(fig, update, frames=51, interval=100, blit=True)
    return ani

def animation_with_appropriate_speed():
```

```python
# Vizualizace chování řadícího algoritmu
def bubble_sort_animation():
    # Generování náhodných dat
    np.random.seed(42)
    data = np.random.randint(1, 100, 20)
    
    # Nastavení grafu
    fig, ax = plt.subplots(figsize=(12, 6))
    bar_container = ax.bar(range(len(data)), data, color='steelblue')
    ax.set_title('Vizualizace algoritmu Bubble Sort')
    ax.set_xlabel('Index')
    ax.set_ylabel('Hodnota')
    ax.set_ylim(0, max(data) * 1.1)
    
    # Text pro zobrazení počtu porovnání a výměn
    text_info = ax.text(0.02, 0.95, '', transform=ax.transAxes)
    
    # Kroky bubble sortu
    def generate_bubble_sort_frames():
        frames = []
        n_comparisons = 0
        n_swaps = 0
        arr = data.copy()
        
        # Uložení počátečního stavu
        frames.append({
            'array': arr.copy(),
            'comparisons': n_comparisons,
            'swaps': n_swaps,
            'i': -1,
            'j': -1
        })
        
        # Algoritmus bubble sort
        n = len(arr)
        for i in range(n):
            for j in range(0, n - i - 1):
                n_comparisons += 1
                if arr[j] > arr[j + 1]:
                    arr[j], arr[j + 1] = arr[j + 1], arr[j]
                    n_swaps += 1
                
                # Uložení stavu po tomto porovnání/výměně
                frames.append({
                    'array': arr.copy(),
                    'comparisons': n_comparisons,
                    'swaps': n_swaps,
                    'i': i,
                    'j': j
                })
        
        return frames
    
    sort_frames = generate_bubble_sort_frames()
    
    # Funkce aktualizace animace
    def update(frame_data):
        # Aktualizace výšek sloupců
        for i, height in enumerate(frame_data['array']):
            bar_container[i].set_height(height)
        
        # Aktualizace barev: modrá pro výchozí, červená pro porovnání, zelená pro seřazené
        colors = ['steelblue'] * len(data)
        
        # Porovnávané položky
        if frame_data['j'] >= 0:
            colors[frame_data['j']] = 'red'
            colors[frame_data['j'] + 1] = 'red'
        
        # Seřazené položky
        n = len(data)
        for i in range(n - 1, n - frame_data['i'] - 2, -1):
            if i >= 0:
                colors[i] = 'limegreen'
        
        # Aplikace barev
        for i, color in enumerate(colors):
            bar_container[i].set_color(color)
        
        # Aktualizace informačního textu
        text_info.set_text(f"Porovnání: {frame_data['comparisons']}, Výměny: {frame_data['swaps']}")
        
        return bar_container.patches + [text_info]
    
    ani = FuncAnimation(fig, update, frames=sort_frames,
                       interval=200, blit=True)
    return ani

# Spuštění a zobrazení animace řazení
sorting_ani = bubble_sort_animation()
HTML(sorting_ani.to_jshtml())
```

```python
# Vizualizace dynamiky populace
def population_dynamics_animation():
    # Časová období
    roky = list(range(1960, 2021))
    
    # Vygenerování syntetických dat o populaci
    np.random.seed(42)
    n_countries = 6
    base_populations = np.random.randint(10, 100, n_countries)
    growth_rates = np.random.uniform(0.005, 0.03, n_countries)
    
    # Názvy zemí
    zeme = ['Země A', 'Země B', 'Země C', 'Země D', 'Země E', 'Země F']
    
    # Generování dat o populaci s různými vzorci růstu
    populace = {}
    for i, zeme_nazev in enumerate(zeme):
        # Základní exponenciální růst s určitou náhodností
        pop_base = base_populations[i]
        growth_rate = growth_rates[i]
        
        # Přidání různých vzorců růstu
        if i % 3 == 0:
            # Exponenciální růst
            pop = pop_base * (1 + growth_rate) ** np.arange(len(roky))
        elif i % 3 == 1:
            # Logistický růst
            k = pop_base * 5  # kapacita prostředí
            r = growth_rate * 2
            t = np.arange(len(roky))
            pop = k / (1 + ((k - pop_base) / pop_base) * np.exp(-r * t))
        else:
            # Rychlý růst a pád
            t = np.arange(len(roky))
            pop = pop_base * (1 + growth_rate) ** np.minimum(t, 30) * \
                  np.where(t > 30, 1 - 0.01 * (t - 30), 1)
        
        # Přidání šumu
        pop = pop * (1 + np.random.normal(0, 0.01, len(roky)))
        populace[zeme_nazev] = pop.astype(int)
    
    # Vytvoření DataFrame
    data = []
    for zeme_nazev in zeme:
        for i, rok in enumerate(roky):
            data.append({
                'Rok': rok,
                'Země': zeme_nazev,
                'Populace': populace[zeme_nazev][i]
            })
    
    df = pd.DataFrame(data)
    
    # Vytvoření animovaného čárového grafu s plotly
    fig = px.line(
        df, 
        x='Rok', 
        y='Populace', 
        color='Země',
        animation_frame='Rok',
        range_y=[0, df['Populace'].max() * 1.1],
        title='Dynamika populace v průběhu času'
    )
    
    # Přidání nastavení animace
    fig.layout.updatemenus[0].buttons[0].args[1]['frame']['duration'] = 100
    fig.layout.updatemenus[0].buttons[0].args[1]['transition']['duration'] = 50
    
    return fig

# Vytvoření a zobrazení animace dynamiky populace
population_fig = population_dynamics_animation()
population_fig.show()
```

Animované vizualizace přidávají mocnou dimenzi k vyprávění příběhů pomocí dat tím, že oživují data a odhalují vzory, které by mohly být přehlédnuty ve statických vizualizacích. Klíčové poznatky z tohoto průvodce zahrnují:

1. **Několik knihoven nabízí možnosti animace** v Pythonu:

   - ImageIO umožňuje vlastní animace snímek po snímku
   - Bokeh poskytuje webové interaktivní animace
