## 2. Základní typy grafů a jejich aplikace

Pochopení správného typu grafu pro konkrétní data a analytické cíle je základem efektivní vizualizace. Tato sekce pokrývá základní typy grafů a kdy je použít.

```python
# Vytvoření komplexního datasetu pro demonstrace
np.random.seed(42)
n = 1000

# Generování komplexního datasetu pro různé typy grafů
data = {
    'numeric_a': np.random.normal(0, 1, n),
    'numeric_b': np.random.normal(2, 1.5, n),
    'category': np.random.choice(['A', 'B', 'C', 'D'], n),
    'ordinal': np.random.choice(['Nízký', 'Střední', 'Vysoký'], n),
    'binary': np.random.choice([0, 1], n),
    'count': np.random.randint(0, 100, n),
    'date': pd.date_range('20230101', periods=n),
    'x_coord': np.random.uniform(-1, 1, n),
    'y_coord': np.random.uniform(-1, 1, n)
}

plot_df = pd.DataFrame(data)
plot_df['ordinal'] = pd.Categorical(plot_df['ordinal'], categories=['Nízký', 'Střední', 'Vysoký'], ordered=True)

# 1. Čárový graf - Pro časové řady nebo trendy
plt.figure(figsize=(10, 6))
plt.plot(plot_df['date'][:100], plot_df['numeric_a'][:100], label='Řada A')
plt.plot(plot_df['date'][:100], plot_df['numeric_b'][:100], label='Řada B')
plt.title('Čárový graf: Zobrazení trendů v čase')
plt.xlabel('Datum')
plt.ylabel('Hodnota')
plt.legend()
plt.tight_layout()
plt.show()

# 2. Sloupcový graf - Pro porovnání kategorií
category_counts = plot_df['category'].value_counts()
plt.figure(figsize=(10, 6))
plt.bar(category_counts.index, category_counts.values)
plt.title('Sloupcový graf: Porovnání počtů napříč kategoriemi')
plt.xlabel('Kategorie')
plt.ylabel('Počet')
plt.tight_layout()
plt.show()

# 3. Bodový graf - Pro vztahy mezi proměnnými
plt.figure(figsize=(10, 6))
plt.scatter(plot_df['numeric_a'], plot_df['numeric_b'], alpha=0.5)
plt.title('Bodový graf: Vztah mezi dvěma proměnnými')
plt.xlabel('Proměnná A')
plt.ylabel('Proměnná B')
plt.tight_layout()
plt.show()

# 4. Histogram - Pro distribuce
plt.figure(figsize=(10, 6))
plt.hist(plot_df['numeric_a'], bins=30, alpha=0.7, label='A')
plt.hist(plot_df['numeric_b'], bins=30, alpha=0.7, label='B')
plt.title('Histogram: Distribuce hodnot')
plt.xlabel('Hodnota')
plt.ylabel('Četnost')
plt.legend()
plt.tight_layout()
plt.show()

# 5. Krabicový graf - Pro statistická shrnutí
plt.figure(figsize=(10, 6))
sns.boxplot(x='category', y='numeric_a', data=plot_df)
plt.title('Krabicový graf: Statistická distribuce podle kategorie')
plt.xlabel('Kategorie')
plt.ylabel('Hodnota')
plt.tight_layout()
plt.show()

# 6. Teplotní mapa - Pro korelační matice nebo 2D distribuce
corr = plot_df.select_dtypes(include=[np.number]).corr()
plt.figure(figsize=(10, 8))
sns.heatmap(corr, annot=True, cmap='coolwarm', vmin=-1, vmax=1)
plt.title('Teplotní mapa: Korelace mezi proměnnými')
plt.tight_layout()
plt.show()

# 7. Koláčový graf - Pro kompozici (používejte střídmě)
plt.figure(figsize=(8, 8))
plt.pie(category_counts.values, labels=category_counts.index, autopct='%1.1f%%')
plt.title('Koláčový graf: Proporce kategorií')
plt.axis('equal')  # Stejný poměr stran zajistí, že koláč je nakreslen jako kruh
plt.tight_layout()
plt.show()

# 8. Houslový graf - Pro porovnání distribuce
plt.figure(figsize=(10, 6))
sns.violinplot(x='category', y='numeric_a', data=plot_df)
plt.title('Houslový graf: Detailní distribuce podle kategorie')
plt.xlabel('Kategorie')
plt.ylabel('Hodnota')
plt.tight_layout()
plt.show()

# 9. Párový graf - Pro multivariační vztahy
sns.pairplot(plot_df[['numeric_a', 'numeric_b', 'count']], height=2.5)
plt.suptitle('Párový graf: Vztahy mezi více proměnnými', y=1.02)
plt.tight_layout()
plt.show()

# 10. Bublinový graf - Pro 3D vztahy (x, y, velikost)
plt.figure(figsize=(10, 6))
plt.scatter(plot_df['numeric_a'], plot_df['numeric_b'], s=plot_df['count'], alpha=0.5)
plt.title('Bublinový graf: Zobrazení 3 proměnných (x, y, velikost)')
plt.xlabel('Proměnná A')
plt.ylabel('Proměnná B')
plt.tight_layout()
plt.show()

# Průvodce výběrem typu grafu
plot_guide = {
    'Numerická vs. Čas': ['Čárový graf', 'Plošný graf'],
    'Porovnání kategorií': ['Sloupcový graf', 'Skupinový sloupcový graf', 'Skládaný sloupcový graf'],
    'Distribuce': ['Histogram', 'Hustotní graf', 'Krabicový graf', 'Houslový graf'],
    'Korelace': ['Bodový graf', 'Teplotní mapa', 'Párový graf'],
    'Kompozice': ['Koláčový graf', 'Skládaný plošný graf', 'Stromová mapa'],
    'Geoprostorová data': ['Choroplethová mapa', 'Bodová geo mapa'],
    'Části k celku': ['Skládaný sloupcový graf', 'Stromová mapa', 'Sunburst graf', 'Koláčový graf'],
    '3+ proměnné': ['Bublinový graf', 'Barevně kódovaný bodový graf', '3D grafy']
}

print("\nPrůvodce výběrem grafu:")
for data_type, plots in plot_guide.items():
    print(f"{data_type}: {', '.join(plots)}")
```

## 3. Techniky statistické vizualizace

Statistické vizualizace pomáhají v porozumění distribuci, vztahům a vzorům v datech, což je nezbytné pro analýzu dat a modelování.

```python
# Generování dat pro statistickou vizualizaci
np.random.seed(42)

# Vytvoření datasetů s různými distribucemi
normal_data = np.random.normal(0, 1, 1000)
skewed_data = np.random.exponential(2, 1000)
bimodal_data = np.concatenate([np.random.normal(-3, 1, 500), np.random.normal(3, 1, 500)])

# Vytvoření DataFrame s více funkcemi pro statistickou analýzu
stat_df = pd.DataFrame({
    'normal': normal_data,
    'skewed': skewed_data,
    'bimodal': bimodal_data,
    'category': np.random.choice(['Skupina 1', 'Skupina 2', 'Skupina 3'], 1000),
    'value': np.random.uniform(0, 100, 1000)
})

# 1. Distribuční grafy (Histogram, KDE, Rug Plot)
plt.figure(figsize=(12, 8))

# Histogram s KDE
plt.subplot(2, 2, 1)
sns.histplot(normal_data, kde=True, color='blue')
plt.title('Histogram s KDE (Normální distribuce)')
plt.xlabel('Hodnota')
plt.ylabel('Četnost')

# Porovnání více distribucí
plt.subplot(2, 2, 2)
sns.kdeplot(normal_data, label='Normální')
sns.kdeplot(skewed_data, label='Zešikmená')
sns.kdeplot(bimodal_data, label='Bimodální')
plt.title('Hustotní grafy pro různé distribuce')
plt.xlabel('Hodnota')
plt.ylabel('Hustota')
plt.legend()

# Rug Plot
plt.subplot(2, 2, 3)
sns.rugplot(normal_data, color='blue', height=0.3)
sns.rugplot(skewed_data, color='green', height=0.2)
plt.title('Rug Plot: Zobrazení jednotlivých pozorování')
plt.xlabel('Hodnota')

# ECDF (Empirická kumulativní distribuční funkce)
plt.subplot(2, 2, 4)
sns.ecdfplot(normal_data, label='Normální')
sns.ecdfplot(skewed_data, label='Zešikmená')
plt.title('ECDF: Porovnání distribucí')
plt.xlabel('Hodnota')
plt.ylabel('Podíl')
plt.legend()

plt.tight_layout()
plt.show()

# 2. Krabicové a houslové grafy
plt.figure(figsize=(14, 6))

# Krabicový graf
plt.subplot(1, 2, 1)
sns.boxplot(x='category', y='value', data=stat_df)
plt.title('Krabicový graf podle kategorie')
plt.xlabel('Kategorie')
plt.ylabel('Hodnota')

# Houslový graf
plt.subplot(1, 2, 2)
sns.violinplot(x='category', y='value', data=stat_df, inner='quartile')
plt.title('Houslový graf podle kategorie')
plt.xlabel('Kategorie')
plt.ylabel('Hodnota')

plt.tight_layout()
plt.show()

# 3. Grafy statistických vztahů
plt.figure(figsize=(16, 12))

# Bodový graf s regresní přímkou
plt.subplot(2, 2, 1)
sns.regplot(x='normal', y='value', data=stat_df, scatter_kws={'alpha':0.5})
plt.title('Bodový graf s regresní přímkou')
plt.xlabel('Normální distribuce')
plt.ylabel('Hodnota')

# Reziduální graf
plt.subplot(2, 2, 2)
sns.residplot(x='normal', y='value', data=stat_df)
plt.title('Reziduální graf')
plt.xlabel('Normální distribuce')
plt.ylabel('Rezidua')

# Joint Plot (Scatter + Histogramy)
plt.subplot(2, 2, 3)
# Poznámka: Joint plot vytváří svůj vlastní obrázek, takže bychom ho normálně použili samostatně
# Příkladový kód: sns.jointplot(x='normal', y='value', data=stat_df, kind='reg')
# Pro subplot budeme simulovat podobný efekt
plt.scatter(stat_df['normal'], stat_df['value'], alpha=0.5)
plt.title('Bodový graf (Joint Plot by zobrazil histogramy na okrajích)')
plt.xlabel('Normální distribuce')
plt.ylabel('Hodnota')

# Pair Plot příklad (samostatný obrázek)
sns.pairplot(stat_df[['normal', 'skewed', 'value']], corner=True)
plt.suptitle('Pair Plot: Vztahy mezi proměnnými', y=1.02)
plt.show()

# 4. Korelační analýza
correlation_matrix = stat_df[['normal', 'skewed', 'bimodal', 'value']].corr()

plt.figure(figsize=(10, 8))
sns.heatmap(correlation_matrix, annot=True, cmap='coolwarm', vmin=-1, vmax=1)
plt.title('Korelační teplotní mapa')
plt.tight_layout()
plt.show()

# 5. Kategorické grafy
plt.figure(figsize=(16, 8))

# Count Plot
plt.subplot(1, 3, 1)
sns.countplot(x='category', data=stat_df)
plt.title('Count Plot: Frekvence kategorií')
plt.xlabel('Kategorie')
plt.ylabel('Počet')

# Bar Plot
plt.subplot(1, 3, 2)
sns.barplot(x='category', y='value', data=stat_df, estimator=np.mean)
plt.title('Bar Plot: Průměrná hodnota podle kategorie')
plt.xlabel('Kategorie')
plt.ylabel('Průměrná hodnota')

# Point Plot
plt.subplot(1, 3, 3)
sns.pointplot(x='category', y='value', data=stat_df)
plt.title('Point Plot: Průměr s intervalem spolehlivosti')
plt.xlabel('Kategorie')
plt.ylabel('Hodnota')

plt.tight_layout()
plt.show()

# Statistické shrnutí pro různé distribuce
stats_summary = pd.DataFrame({
    'Distribuce': ['Normální', 'Zešikmená', 'Bimodální'],
    'Průměr': [normal_data.mean(), skewed_data.mean(), bimodal_data.mean()],
    'Medián': [np.median(normal_data), np.median(skewed_data), np.median(bimodal_data)],
    'Směrodatná odchylka': [normal_data.std(), skewed_data.std(), bimodal_data.std()],
    'Šikmost': [pd.Series(normal_data).skew(), pd.Series(skewed_data).skew(), pd.Series(bimodal_data).skew()],
    'Špičatost': [pd.Series(normal_data).kurt(), pd.Series(skewed_data).kurt(), pd.Series(bimodal_data).kurt()]
})

print("Statistické shrnutí:")
print(stats_summary.round(3))
```

## 4. Vizualizace geoprostorových dat

Geoprostorové vizualizace umožňují zobrazení dat na mapách, což je nezbytné pro analýzu a získávání poznatků na základě polohy.

```python
# Instalace potřebných balíčků
# pip install geopandas folium plotly

# Import knihoven
import matplotlib.pyplot as plt
import pandas as pd
import numpy as np
import geopandas as gpd
import folium
from folium.plugins import HeatMap
import plotly.express as px

# Ukázková data pro státy USA
# Vytvoření mock datasetu s informacemi o státech
states = ['Kalifornie', 'Texas', 'Florida', 'New York', 'Pensylvánie', 
          'Illinois', 'Ohio', 'Georgia', 'Severní Karolína', 'Michigan']
state_codes = ['CA', 'TX', 'FL', 'NY', 'PA', 'IL', 'OH', 'GA', 'NC', 'MI']

# Generování náhodných hodnot
np.random.seed(42)
values = np.random.uniform(10, 100, size=len(states))
populations = np.random.randint(1000000, 40000000, size=len(states))

# Vytvoření DataFrame
geo_df = pd.DataFrame({
    'state': states,
    'state_code': state_codes,
    'value': values,
    'population': populations,
    'latitude': [36.7, 31.1, 27.8, 42.2, 40.9, 40.0, 40.4, 33.0, 35.6, 44.3],
    'longitude': [-119.4, -100.0, -81.5, -74.0, -77.8, -89.0, -82.8, -83.1, -79.4, -85.4]
})

# 1. Statická mapa s Geopandas
# Načtení shapefile států USA
usa = gpd.read_file(gpd.datasets.get_path('naturalearth_lowres'))
usa = usa[(usa.continent == 'North America') & (usa.name != 'Canada') & (usa.name != 'Mexico')]

# Sloučení dat s geopandas dataframe
usa_data = usa.merge(geo_df, left_on='iso_a2', right_on='state_code', how='left')

# Vytvoření choroplethové mapy
fig, ax = plt.subplots(1, 1, figsize=(15, 10))
usa_data.plot(column='value', cmap='YlOrRd', linewidth=0.8, ax=ax, edgecolor='0.8')

# Přidání popisků států
for idx, row in usa_data.iterrows():
    if row['state_code'] in state_codes:
        plt.annotate(text=row['state_code'], xy=(row['geometry'].centroid.x, row['geometry'].centroid.y),
                    horizontalalignment='center', fontsize=8)

plt.title('Hodnota podle státu (Choroplethová mapa)')
plt.axis('off')
plt.colorbar(ax.get_children()[1], ax=ax, label='Hodnota')
plt.tight_layout()
plt.show()

# 2. Interaktivní mapa s Folium
# Vytvoření základní mapy soustředěné na USA
m = folium.Map(location=[37.0902, -95.7129], zoom_start=4, tiles="CartoDB positron")

# Přidání značek pro státy
for idx, row in geo_df.iterrows():
    folium.CircleMarker(
        location=[row['latitude'], row['longitude']],
        radius=row['value'] / 5,  # Škálování poloměru
        popup=f"{row['state']}<br>Hodnota: {row['value']:.2f}<br>Populace: {row['population']:,}",
        color='#3186cc',
        fill=True,
        fill_color='#3186cc'
    ).add_to(m)

# Uložení do HTML
m.save('us_states_map.html')
print("Interaktivní mapa uložena jako 'us_states_map.html'")

# 3. Teplotní mapa s Folium
# Vytvoření bodů pro teplotní mapu (více bodů v oblastech s vyššími hodnotami)
heat_data = []
for idx, row in geo_df.iterrows():
    # Generování více bodů pro státy s vyššími hodnotami
    points = int(row['value'] / 2)
    for _ in range(points):
        # Přidání náhodného šumu pro rozložení bodů
        lat = row['latitude'] + np.random.uniform(-1, 1)
        lon = row['longitude'] + np.random.uniform(-1, 1)
        heat_data.append([lat, lon])

# Vytvoření nové mapy pro teplotní mapu
heat_map = folium.Map(location=[37.0902, -95.7129], zoom_start=4, tiles="CartoDB dark_matter")

# Přidání vrstvy teplotní mapy
HeatMap(heat_data).add_to(heat_map)

# Uložení do HTML
heat_map.save('us_states_heatmap.html')
print("Teplotní mapa uložena jako 'us_states_heatmap.html'")

# 4. Interaktivní choropleth s Plotly
fig = px.choropleth(
    geo_df, 
    locations='state_code', 
    locationmode="USA-states", 
    scope="usa",
    color='value',
    color_continuous_scale="Viridis",
    hover_name='state',
    hover_data=['population', 'value'],
    title='Interaktivní choroplethová mapa států USA'
)

fig.update_layout(
    geo=dict(showlakes=True, lakecolor='rgb(255, 255, 255)'),
)

# Pro Jupyter notebooky byste zobrazili graf pomocí fig.show()
# Pro uložení do HTML
fig.write_html('us_states_plotly.html')
print("Interaktivní mapa Plotly uložena jako 'us_states_plotly.html'")

print("\nMožnosti vizualizace geoprostorových dat:")
print("1. Statické mapy: Vhodné pro zprávy a publikace")
print("2. Interaktivní mapy: Vynikající pro průzkum a webové aplikace")
print("3. Teplotní mapy: Nejlepší pro vizualizaci hustoty")
print("4. Choroplethové mapy: Ideální pro zobrazení statistik podle regionů")
print("5. Bublinové mapy: Vhodné pro zobrazení velikosti na konkrétních místech")
```

## 5. Vizualizace časových řad

Vizualizace časových řad jsou nezbytné pro analýzu temporálních dat, zobrazení trendů, sezónnosti a vzorů v čase.

```python
# Import knihoven
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
import seaborn as sns
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
import plotly.express as px
import plotly.graph_objects as go

# Vytvoření syntetického datasetu časových řad
np.random.seed(42)
dates = pd.date_range(start='2020-01-01', end='2022-12-31', freq='D')
n = len(dates)
```

```python
import pandas as pd
```

```python
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.decomposition import PCA
from sklearn.manifold import TSNE
import plotly.express as px
from mpl_toolkits.mplot3d import Axes3D
import umap.umap_ as umap

# Generování vícerozměrného datasetu
np.random.seed(42)
n_vzorků = 500
n_vlastností = 10

# Vytvoření shluků ve vysokorozměrném prostoru
shluk1 = np.random.normal(0, 1, size=(n_vzorků//3, n_vlastností))
shluk2 = np.random.normal(5, 1, size=(n_vzorků//3, n_vlastností))
shluk3 = np.random.normal(-5, 1, size=(n_vzorků//3, n_vlastností))

# Kombinace shluků
X = np.vstack([shluk1, shluk2, shluk3])

# Přidání korelace mezi dimenzemi
for i in range(1, n_vlastností):
    X[:, i] = X[:, i] + 0.3 * X[:, 0]

# Štítky pro shluky
štítky = np.array(['Shluk 1'] * (n_vzorků//3) + ['Shluk 2'] * (n_vzorků//3) + ['Shluk 3'] * (n_vzorků//3))

# Vytvoření DataFrame
high_dim_df = pd.DataFrame(X, columns=[f'dim_{i+1}' for i in range(n_vlastností)])
high_dim_df['štítek'] = štítky

# 1. Párový graf pro zkoumání vztahů mezi vlastnostmi
plt.figure(figsize=(12, 10))
subset_df = high_dim_df[['dim_1', 'dim_2', 'dim_3', 'dim_4', 'štítek']].copy()

plt.show()

mask = np.triu(np.ones_like(corr, dtype=bool))  # Vytvoření masky pro horní trojúhelník
sns.heatmap(corr, mask=mask, cmap='coolwarm', vmin=-1, vmax=1,

plt.title('Korelační teplotní mapa všech dimenzí')
plt.tight_layout()
plt.show()

# Funkce pro vytvoření paprskového grafu
def vytvor_paprskovy_graf(data, kategorie, titulek):
```

6. **Konzistentní stylování**: Udržujte jednotný styl napříč všemi obrázky v publikaci

```python
# Vytvoření znovupoužitelné šablony pro publikační obrázky
def vytvor_publikacni_obrazek(funkce_grafu, nazev_souboru, 
                              velikost=(8, 5), styl_casopisu='obecny'):
    """
    Šablona pro vytváření obrázků připravených pro publikaci.
    
    Parametry:
    -----------
    funkce_grafu : callable
        Funkce, která vytváří obsah grafu. Bere matplotlib osu jako argument.
    nazev_souboru : str
        Základní název souboru pro uložení obrázku (bez přípony)
    velikost : tuple
        Rozměry obrázku v palcích
    styl_casopisu : str
        Předvolba stylu ('obecny', 'nature', 'science', 'ieee')
    """
    # Nastavení stylu podle časopisu
    if styl_casopisu == 'nature':
        plt.style.use('seaborn-v0_8-whitegrid')
        rcParams['font.sans-serif'] = ['Arial']
        rcParams['font.family'] = 'sans-serif'
        rcParams['axes.linewidth'] = 0.5
    elif styl_casopisu == 'science':
        plt.style.use('seaborn-v0_8-whitegrid')
        rcParams['font.family'] = 'serif'
        rcParams['font.serif'] = ['Times New Roman']
    elif styl_casopisu == 'ieee':
        plt.style.use('seaborn-v0_8-whitegrid')
        rcParams['font.family'] = 'serif'
        rcParams['font.serif'] = ['Times New Roman']
        rcParams['font.size'] = 8
    else:  # obecný akademický styl
        plt.style.use('seaborn-v0_8-whitegrid')
        rcParams['font.family'] = 'serif'
    
    # Vytvoření obrázku
    fig, ax = plt.subplots(figsize=velikost, dpi=300)
    
    # Volání poskytnuté funkce vykreslení
    funkce_grafu(ax)
    
    # Doladění rozložení
    plt.tight_layout()
    
    # Uložení v různých formátech
    plt.savefig(f'{nazev_souboru}.png', dpi=300, bbox_inches='tight')
    plt.savefig(f'{nazev_souboru}.pdf', format='pdf', bbox_inches='tight')
```

```python
# Spodní třetí řada, poslední sloupec: teplotní mapa
ax6 = fig.add_subplot(gs[2, 3])
corr = np.corrcoef([y1, y2, y3])
sns.heatmap(corr, annot=True, cmap='coolwarm', vmin=-1, vmax=1, 
            xticklabels=['Sin', 'Cos', 'Sin*Cos'], 
            yticklabels=['Sin', 'Cos', 'Sin*Cos'], ax=ax6)
ax6.set_title('Korelační matice', fontsize=10)

# Úprava rozložení
plt.tight_layout()
plt.subplots_adjust(top=0.95, hspace=0.4, wspace=0.4)
fig.suptitle('Komplexní rozložení dashboardu s Matplotlib', fontsize=16, y=0.98)
plt.show()
```

## 10. Shrnutí a osvědčené postupy

Vytvoření efektivní vizualizace dat vyžaduje porozumění vašim datům, publiku a příběhu, který chcete vyprávět. Zde je shrnutí osvědčených postupů pro tvorbu vizualizací v Pythonu:

```python
# Příklad demonstrující osvědčené postupy
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd
import numpy as np

# Nastavení konzistentního stylu
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_context("notebook", font_scale=1.2)

# Vytvoření ukázkových dat s jasnou strukturou
np.random.seed(42)
kategorie = ['Produkt A', 'Produkt B', 'Produkt C', 'Produkt D']
regiony = ['Sever', 'Jih', 'Východ', 'Západ']

# Generování strukturovaných dat
data = []
for kat in kategorie:
    for region in regiony:
        prodeje = np.random.randint(100, 500) * 1000
        zisk = prodeje * np.random.uniform(0.1, 0.4)
        data.append([kat, region, prodeje, zisk])

# Vytvoření DataFrame s jasnými názvy sloupců
df = pd.DataFrame(data, columns=['Produkt', 'Region', 'Prodeje', 'Zisk'])

# Vytvoření obrázku s vhodnou velikostí
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

# Levý graf: Zaměření na hlavní srovnání
prodeje_podle_produktu = df.groupby('Produkt')['Prodeje'].sum() / 1000000  # Převod na miliony
sloupce = ax1.bar(prodeje_podle_produktu.index, prodeje_podle_produktu.values, color='skyblue')

# Přidání jasných popisků a titulku
ax1.set_title('Celkové prodeje podle produktu', fontsize=14)
ax1.set_xlabel('Produkt', fontsize=12)
ax1.set_ylabel('Prodeje (miliony)', fontsize=12)

# Přidání popisků hodnot pro přehlednost
for sloupec in sloupce:
    vyska = sloupec.get_height()
    ax1.text(sloupec.get_x() + sloupec.get_width()/2., vyska + 0.1,
            f'{vyska:.1f} mil. Kč', ha='center', va='bottom')

# Pravý graf: Zobrazení další dimenze dat
ziskova_marze = df.groupby('Produkt').apply(lambda x: (x['Zisk'].sum() / x['Prodeje'].sum()) * 100)
ziskova_marze = ziskova_marze.sort_values(ascending=False)

# Použití barev ke zvýraznění důležitých informací
barvy = ['#3498db', '#2ecc71', '#f39c12', '#e74c3c']
ax2.bar(ziskova_marze.index, ziskova_marze.values, color=barvy)
ax2.set_title('Zisková marže podle produktu', fontsize=14)
ax2.set_xlabel('Produkt', fontsize=12)
ax2.set_ylabel('Zisková marže (%)', fontsize=12)

# Přidání referenční čáry pro kontext
prumerna_marze = df['Zisk'].sum() / df['Prodeje'].sum() * 100
ax2.axhline(y=prumerna_marze, linestyle='--', color='red', 
           alpha=0.7, label=f'Průměr: {prumerna_marze:.1f}%')
ax2.legend()

# Přidání celkového nadpisu vysvětlujícího vizualizaci
plt.suptitle('Analýza výkonnosti produktů', fontsize=16, y=1.05)
fig.text(0.5, 0.01, 'Analýza ukazuje, že Produkt B má nejvyšší prodeje, zatímco Produkt C má nejlepší ziskovou marži.', 
         ha='center', fontsize=10, style='italic')

plt.tight_layout()
plt.show()
```

### Rámec pro rozhodování o vizualizaci

1. **Vyberte správný typ grafu pro svá data a cíl**:
   - Distribuce: Histogramy, KDE grafy, krabicové grafy, houslové grafy
   - Srovnání: Sloupcové grafy, lízátkové grafy, bodové grafy
   - Vztahy: Bodové grafy, bublinové grafy, teplotní mapy
   - Kompozice: Koláčové grafy, skládané sloupcové grafy, plošné grafy
   - Trendy: Čárové grafy, plošné grafy
   - Geoprostorová data: Choroplethové mapy, bublinové mapy

2. **Řiďte se těmito základními principy**:
   - **Jasnost**: Upřednostňujte jasnou komunikaci před dekorací
   - **Přesnost**: Reprezentujte data pravdivě bez zkreslení
   - **Efektivita**: Maximalizujte poměr datového inkoustu (Tufteho princip)
   - **Zaměření**: Směřujte pozornost k tomu, co je důležité
   - **Kontext**: Poskytněte nezbytné referenční body

3. **Technické osvědčené postupy**:
   - Používejte vhodné transformace dat, když je to potřeba (logaritmické měřítko, normalizace)
   - Vybírejte barevné palety vhodné pro osoby s poruchami barvocitu
   - Nastavujte informativní titulky, popisky os a legendy
   - Zahrnujte informace o zdroji a vysvětlující poznámky
   - Přizpůsobujte grafiku médiu, pro které je určena
   - Berte v úvahu technickou zdatnost vašeho publika

4. **Běžné chyby, kterým se vyhnout**:
   - Zavádějící osy (zkrácené měřítko, nevhodná nulová základna)
   - Grafický balast (zbytečné dekorace)
   - 3D grafy, které zkreslují vnímání
   - Překrývání, které zakrývá vzory v datech
   - Nevhodné volby barev, které vytvářejí falešné vzory
   - Příliš mnoho prvků soupeřících o pozornost

```python
# Příklad vyprávění datového příběhu pomocí sekvenčních vizualizací
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import pandas as pd

# Nastavení konzistentního stylu pro příběh
plt.style.use('seaborn-v0_8-whitegrid')
sns.set_context("notebook", font_scale=1.1)

# Vytvoření ukázkových dat pro analýzu odchodu zákazníků
np.random.seed(42)
n_zakazniku = 1000

# Generování syntetických dat
df = pd.DataFrame({
    'vek_zakaznika': np.random.normal(35, 15, n_zakazniku).clip(18, 90).astype(int),
    'doba_trvani_mesice': np.random.exponential(20, n_zakazniku).astype(int).clip(1, 120),
    'mesicni_poplatky': np.random.normal(65, 30, n_zakazniku).clip(20, 150).round(2),
    'celkove_poplatky': np.random.normal(1000, 800, n_zakazniku).clip(0, 8000).round(2),
    'odchod': np.random.choice([0, 1], n_zakazniku, p=[0.7, 0.3])
})

# Vytvoření věkových skupin pro lepší analýzu
df['vekova_skupina'] = pd.cut(df['vek_zakaznika'], bins=[18, 30, 45, 60, 100], 
                         labels=['18-30', '31-45', '46-60', '61+'])

# Vytvoření obrázku s více podgrafy pro vyprávění datového příběhu
fig = plt.figure(figsize=(15, 12))

# 1. Úvod k problému: Celková míra odchodu
ax1 = plt.subplot2grid((3, 3), (0, 0))
pocty_odchodu = df['odchod'].value_counts()
ax1.pie([pocty_odchodu[0], pocty_odchodu[1]], 
        labels=['Udržení', 'Odchod'], autopct='%1.1f%%',
        colors=['#2ecc71', '#e74c3c'], startangle=90)
ax1.set_title('Míra odchodu zákazníků', fontsize=14)

# 2. První poznatek: Odchod podle délky vztahu
ax2 = plt.subplot2grid((3, 3), (0, 1), colspan=2)
sns.boxplot(x='odchod', y='doba_trvani_mesice', data=df, 
           palette=['#2ecc71', '#e74c3c'], ax=ax2)
ax2.set_title('Doba trvání podle stavu odchodu', fontsize=14)
ax2.set_xticklabels(['Udržení', 'Odchod'])
ax2.set_ylabel('Měsíce jako zákazník')
ax2.set_xlabel('')

# 3. Druhý poznatek: Odchod podle měsíčních poplatků
ax3 = plt.subplot2grid((3, 3), (1, 0), colspan=2)
sns.histplot(data=df, x='mesicni_poplatky', hue='odchod', 
             palette=['#2ecc71', '#e74c3c'], 
             multiple='dodge', bins=10, ax=ax3)
ax3.set_title('Distribuce měsíčních poplatků podle odchodu', fontsize=14)
ax3.set_xlabel('Měsíční poplatky (Kč)')
ax3.legend(labels=['Udržení', 'Odchod'])

# 4. Třetí poznatek: Míra odchodu podle věkové skupiny
ax4 = plt.subplot2grid((3, 3), (1, 2))
odchod_podle_veku = df.groupby('vekova_skupina')['odchod'].mean() * 100
ax4.bar(odchod_podle_veku.index, odchod_podle_veku.values, color='#3498db')
ax4.set_title('Míra odchodu podle věkové skupiny', fontsize=14)
ax4.set_ylabel('Míra odchodu (%)')
ax4.set_ylim(0, 50)
for i, v in enumerate(odchod_podle_veku):
    ax4.text(i, v + 1, f"{v:.1f}%", ha='center')

# 5. Klíčový vztah: Doba trvání vs. Měsíční poplatky s odchodem
ax5 = plt.subplot2grid((3, 3), (2, 0), colspan=2)
scatter = ax5.scatter(df['doba_trvani_mesice'], df['mesicni_poplatky'], 
                     c=df['odchod'], cmap='RdYlGn_r', alpha=0.7)
ax5.set_title('Doba trvání vs. Měsíční poplatky', fontsize=14)
ax5.set_xlabel('Doba trvání (měsíce)')
ax5.set_ylabel('Měsíční poplatky (Kč)')
legend1 = ax5.legend(*scatter.legend_elements(), title="Odchod")
ax5.add_artist(legend1)

# 6. Závěrečný poznatek: Zvýraznění vysoce rizikového segmentu
ax6 = plt.subplot2grid((3, 3), (2, 2))
# Definice vysokého rizika jako noví zákazníci (< 12 měsíců) s vysokými poplatky (> 80)
df['vysoke_riziko'] = ((df['doba_trvani_mesice'] < 12) & (df['mesicni_poplatky'] > 80)).astype(int)
odchod_vysoke_riziko = df.groupby('vysoke_riziko')['odchod'].mean() * 100
ax6.bar(['Nízké riziko', 'Vysoké riziko'], odchod_vysoke_riziko, color=['#2ecc71', '#e74c3c'])
ax6.set_title('Míra odchodu podle rizikového segmentu', fontsize=14)
ax6.set_ylabel('Míra odchodu (%)')
ax6.set_ylim(0, 100)
for i, v in enumerate(odchod_vysoke_riziko):
    ax6.text(i, v + 2, f"{v:.1f}%", ha='center')

# Přidání celkového nadpisu a popisného podnadpisu
plt.suptitle('Analýza odchodu zákazníků', fontsize=20, y=0.98)
plt.figtext(0.5, 0.92, 
           'Analýza ukazuje vyšší odchod mezi novějšími zákazníky s vyššími měsíčními poplatky',
           ha='center', fontsize=12, style='italic')

# Přidání zápatí s informacemi o zdroji
plt.figtext(0.99, 0.01, 'Zdroj dat: Databáze zákazníků Q3 2023', 
           ha='right', fontsize=8)

plt.tight_layout()
plt.subplots_adjust(top=0.9, hspace=0.3, wspace=0.3)
plt.show()

print("Klíčová zjištění z vizualizačního příběhu:")
print("1. Celková míra odchodu je přibližně 30 % zákazníků")

print("3. Vyšší měsíční poplatky korelují se zvýšeným rizikem odchodu")
print("4. Zákazníci středního věku (31-45) mají nejvyšší míru odchodu")
```
