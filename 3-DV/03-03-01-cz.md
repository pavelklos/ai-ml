## Úvod do 3D vizualizace

3D vizualizace nám umožňuje zobrazit data v trojrozměrném prostoru, což přidává hloubku k naší datové analýze. To je obzvláště užitečné při analýze vztahů mezi třemi proměnnými nebo při vizualizaci složitých povrchů, objemů a vícerozměrných datasetů.

V tomto průvodci prozkoumáme, jak vytvořit efektivní 3D vizualizace pomocí dvou výkonných Python knihoven:

1. **Matplotlib**: Všestranná knihovna pro tvorbu grafů, která obsahuje 3D funkce prostřednictvím svého nástroje `mplot3d`
2. **Plotly**: Interaktivní vizualizační knihovna vynikající v tvorbě webových, interaktivních 3D grafů

## Matplotlib 3D vizualizace

### 3D bodové grafy

3D bodové grafy jsou užitečné pro vizualizaci vztahů mezi třemi proměnnými:

```python
# Generování vzorových dat
n = 100
x = np.random.standard_normal(n)
y = np.random.standard_normal(n)
z = np.random.standard_normal(n)
c = np.random.standard_normal(n)

# Vytvoření 3D bodového grafu
fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')

scatter = ax.scatter(x, y, z, c=c, cmap='viridis', alpha=0.7, s=50)

# Přidání popisků a titulku
ax.set_xlabel('Osa X')
ax.set_ylabel('Osa Y')
ax.set_zlabel('Osa Z')
ax.set_title('3D bodový graf pomocí Matplotlib')

# Přidání barevné stupnice
plt.colorbar(scatter, ax=ax, label='Hodnoty')

plt.show()
```

### 3D povrchové grafy

Povrchové grafy pomáhají vizualizovat funkce dvou proměnných nebo 3D povrchy:

```python
# Vytvoření dat pro povrchový graf
x = np.linspace(-5, 5, 50)
y = np.linspace(-5, 5, 50)
X, Y = np.meshgrid(x, y)
Z = np.sin(np.sqrt(X**2 + Y**2))

# Vytvoření 3D povrchového grafu
fig = plt.figure(figsize=(10, 8))
ax = fig.add_subplot(111, projection='3d')

surf = ax.plot_surface(X, Y, Z, cmap='coolwarm', 
                      linewidth=0, antialiased=True, alpha=0.8)

# Přidání popisků a titulku
ax.set_xlabel('Osa X')
ax.set_ylabel('Osa Y')
ax.set_zlabel('Osa Z')
ax.set_title('3D povrchový graf pomocí Matplotlib')

# Přidání barevné stupnice, která mapuje hodnoty na barvy
fig.colorbar(surf, ax=ax, shrink=0.5, aspect=5)

plt.show()
```

### Přizpůsobení 3D grafů v Matplotlib

Můžete přizpůsobit různé aspekty 3D grafů v Matplotlib:

```python
# Vytvoření dat
x = np.linspace(-5, 5, 50)
y = np.linspace(-5, 5, 50)
X, Y = np.meshgrid(x, y)
Z = np.sin(np.sqrt(X**2 + Y**2))

# Vytvoření figury a 3D osy
fig = plt.figure(figsize=(12, 10))
ax = fig.add_subplot(111, projection='3d')

# Vykreslení povrchu s přizpůsobeným vzhledem
surf = ax.plot_surface(X, Y, Z, cmap='plasma',
                       linewidth=0, antialiased=True, 
                       alpha=0.7, rstride=2, cstride=2)

# Přizpůsobení vlastností os
ax.set_xlabel('Osa X', fontsize=12, labelpad=10)
ax.set_ylabel('Osa Y', fontsize=12, labelpad=10)
ax.set_zlabel('Osa Z', fontsize=12, labelpad=10)
ax.set_title('Přizpůsobený 3D povrchový graf', fontsize=16, pad=20)

# Nastavení úhlu pohledu
ax.view_init(elev=30, azim=45)  # elevace a azimut

# Přidání mřížky
ax.grid(True, linestyle='-', alpha=0.6)

# Přidání barevné stupnice
fig.colorbar(surf, ax=ax, shrink=0.5, aspect=10, pad=0.1, label='Výška')

plt.tight_layout()
plt.show()
```

## Plotly 3D vizualizace

Plotly poskytuje interaktivní 3D vizualizace, které umožňují přiblížení, posouvání a otáčení.

### 3D bodové grafy s Plotly

```python
# Generování vzorových dat
n = 100
x = np.random.standard_normal(n)
y = np.random.standard_normal(n)
z = np.random.standard_normal(n)
size = np.abs(x) * 10 + 5
colors = np.random.standard_normal(n)

# Vytvoření 3D bodového grafu pomocí Plotly
fig = go.Figure(data=[go.Scatter3d(
    x=x, y=y, z=z,
    mode='markers',
    marker=dict(
        size=size,
        color=colors,
        colorscale='Viridis',
        opacity=0.8,
        colorbar=dict(title="Hodnoty"),
        showscale=True
    )
)])

# Aktualizace layoutu
fig.update_layout(
    title='3D bodový graf s Plotly',
    scene=dict(
        xaxis_title='Osa X',
        yaxis_title='Osa Y',
        zaxis_title='Osa Z',
    ),
    width=800,
    height=800,
)

fig.show()
```

u = np.linspace(0, 2*np.pi, 20)

fig = go.Figure(data=[go.Mesh3d(

### Interaktivní funkce v Plotly

Jednou z největších výhod Plotly jsou jeho interaktivní funkce:

```python
# Generování složitějšího datasetu pro demonstraci interaktivních funkcí
x = np.linspace(-5, 5, 100)
y = np.linspace(-5, 5, 100)
X, Y = np.meshgrid(x, y)

# Vytvoření více funkcí pro rozbalovací menu
Z1 = np.sin(np.sqrt(X**2 + Y**2))
Z2 = (X**2 + Y**2)/10

# Vytvoření figury s tlačítky pro přepínání mezi typy povrchů
fig = go.Figure()

# Přidání povrchů
fig.add_trace(
    go.Surface(z=Z1, x=X, y=Y, colorscale='Viridis', visible=True,
              name='Vlnky', showscale=True)
)

fig.add_trace(
    go.Surface(z=Z2, x=X, y=Y, colorscale='Plasma', visible=False,
              name='Paraboloid', showscale=True)
)

fig.add_trace(
    go.Surface(z=Z3, x=X, y=Y, colorscale='Turbo', visible=False,
              name='Sinusová vlna', showscale=True)

# Aktualizace layoutu s tlačítky a anotacemi
fig.update_layout(
    title='Interaktivní 3D povrchový graf s Plotly',
    scene=dict(
        xaxis_title='Osa X',
        yaxis_title='Osa Y',
        zaxis_title='Osa Z',
    ),
    updatemenus=[dict(
        type="buttons",
        direction="right",
        buttons=list([
            dict(label="Vlnky",
                 method="update",
                 args=[{"visible": [True, False, False]},
                       {"title": "Graf povrchu vlnek"}]),
            dict(label="Paraboloid",
                 method="update",
                 args=[{"visible": [False, True, False]},
                       {"title": "Graf povrchu paraboloidu"}]),
            dict(label="Sinusová vlna",
                 method="update",
                 args=[{"visible": [False, False, True]},
                       {"title": "Graf povrchu sinusové vlny"}]),
        ]),
        pad={"r": 10, "t": 10},
        showactive=True,
        x=0.1,
        xanchor="left",
        y=1.1,
        yanchor="top"
    )],
    width=800,
    height=800,
)

fig.show()
```

## Kombinování více 3D vizualizací

Můžete vytvářet složitější vizualizace kombinováním různých typů 3D grafů:

```python
# Vytvoření dat
x = np.linspace(-5, 5, 50)
y = np.linspace(-5, 5, 50)
X, Y = np.meshgrid(x, y)
Z = np.sin(np.sqrt(X**2 + Y**2))

# Vytvoření podgrafu se dvěma 3D grafy vedle sebe
fig = make_subplots(
    rows=1, cols=2,
    specs=[[{'type': 'surface'}, {'type': 'scene'}]],
    subplot_titles=('Povrchový graf', 'Bodový graf & vrstevnice')
)

# Přidání povrchového grafu do prvního podgrafu
fig.add_trace(
    go.Surface(z=Z, x=X, y=Y, colorscale='Viridis', showscale=False),
    row=1, col=1
)

# Výběr několika bodů z povrchu pro bodový graf
indices = np.random.choice(range(len(X.flatten())), 100)
x_sample = X.flatten()[indices]
y_sample = Y.flatten()[indices]
z_sample = Z.flatten()[indices]

# Přidání bodového grafu a vrstevnic do druhého podgrafu
fig.add_trace(
    go.Scatter3d(
        x=x_sample,
        y=y_sample,
        z=z_sample,
        mode='markers',
        marker=dict(
            size=5,
            color=z_sample,
            colorscale='Viridis',
            opacity=0.8
        )
    ),
    row=1, col=2
)

# Přidání vrstevnic na různých úrovních z
for z_level in [-0.75, 0, 0.75]:
    fig.add_trace(
        go.Scatter3d(
            x=X[Z>z_level].flatten(),
            y=Y[Z>z_level].flatten(),
            z=np.ones_like(X[Z>z_level].flatten()) * z_level,
            mode='markers',
            marker=dict(
                size=2,
                color=np.ones_like(X[Z>z_level].flatten()) * z_level,
                colorscale='Viridis',
                opacity=0.5
            ),
            name=f'z = {z_level}'
        ),
        row=1, col=2
    )

# Aktualizace layoutu pro oba podgrafy
fig.update_layout(
    title_text='Kombinované 3D vizualizace',
    height=600,
    width=1200
)

fig.show()
```

## Osvědčené postupy pro 3D vizualizaci

Při vytváření 3D vizualizací zvažte následující osvědčené postupy:

```python
# Příklad demonstrující osvědčené postupy
# Generování dat
np.random.seed(42)  # Nastavení seedu pro reprodukovatelnost
n_points = 100
x = np.random.uniform(-5, 5, n_points)
y = np.random.uniform(-5, 5, n_points)
z = 2*x - 3*y + np.random.normal(0, 2, n_points)  # Lineární vztah s šumem
categories = np.random.choice(['Skupina A', 'Skupina B', 'Skupina C'], n_points)

# Vytvoření barevné mapy pro kategorie
color_map = {'Skupina A': 'red', 'Skupina B': 'blue', 'Skupina C': 'green'}
colors = [color_map[cat] for cat in categories]

# Vytvoření přehledného 3D bodového grafu pomocí Plotly
fig = go.Figure()

# Přidání datových bodů s jasným seskupením
for cat in ['Skupina A', 'Skupina B', 'Skupina C']:
    mask = categories == cat
    fig.add_trace(go.Scatter3d(
        x=x[mask], y=y[mask], z=z[mask],
        mode='markers',
        marker=dict(
            size=6,
            color=color_map[cat],
            opacity=0.8
        ),
        name=cat
    ))

# Přidání roviny pro znázornění základního vztahu (z = 2x - 3y)
x_grid = np.linspace(-5, 5, 10)
y_grid = np.linspace(-5, 5, 10)
X_grid, Y_grid = np.meshgrid(x_grid, y_grid)
Z_grid = 2*X_grid - 3*Y_grid

fig.add_trace(go.Surface(
    x=X_grid, y=Y_grid, z=Z_grid,
    colorscale=[[0, 'rgba(230, 230, 230, 0.3)'], [1, 'rgba(230, 230, 230, 0.3)']],
    showscale=False,
    name='z = 2x - 3y'
))

# Aktualizace layoutu s jasnými popisky a dobrým úhlem pohledu
fig.update_layout(
    title='Vztah mezi proměnnými se seskupením',
    scene=dict(
        xaxis_title='Proměnná X',
        yaxis_title='Proměnná Y',
        zaxis_title='Odezva Z',
        # Nastavení konzistentního poměru stran
        aspectmode='cube',
        # Přidání mřížky pro lepší prostorové vnímání
        xaxis=dict(gridcolor='rgb(200, 200, 200)', showbackground=True),
        yaxis=dict(gridcolor='rgb(200, 200, 200)', showbackground=True),
        zaxis=dict(gridcolor='rgb(200, 200, 200)', showbackground=True),
        # Nastavení pozice kamery pro jasný pohled
        camera=dict(
            eye=dict(x=1.5, y=-1.5, z=1.2)
        )
    ),
    # Přidání legendy na vhodnou pozici
    legend=dict(
        yanchor="top",
        y=0.99,
        xanchor="right",
        x=0.99
    )
)

fig.show()
```

## Praktické aplikace

Podívejme se na praktickou aplikaci: vizualizace 3D modelu terénu:

```python
# Vytvoření vizualizace modelu terénu
from scipy.interpolate import griddata

# Generování náhodných dat terénu
np.random.seed(42)
# Simulace nasbíraných bodových dat (jako z GPS měření)
x_terrain = np.random.uniform(-10, 10, 500)
y_terrain = np.random.uniform(-10, 10, 500)

# Vytvoření funkce terénu s vrcholy a údolími
def terrain_function(x, y):
    return 2 * np.sin(x/2) * np.cos(y/2) + np.exp(-(x**2 + y**2)/40)

z_terrain = terrain_function(x_terrain, y_terrain) + np.random.normal(0, 0.1, 500)

# Vytvoření pravidelné mřížky
grid_x, grid_y = np.mgrid[-10:10:100j, -10:10:100j]

# Interpolace rozptýlených dat na pravidelnou mřížku
grid_z = griddata((x_terrain, y_terrain), z_terrain, (grid_x, grid_y), method='cubic')

# Vytvoření vizualizace terénu pomocí Plotly
fig = go.Figure()

# Přidání povrchu terénu
terrain_surface = go.Surface(
    z=grid_z,
    x=grid_x,
    y=grid_y,
    colorscale='Earth',
    lighting=dict(
        ambient=0.6,
        diffuse=0.5,
        fresnel=0.1,
        specular=0.8,
        roughness=0.5
    ),
    contours={
        "z": {"show": True, "start": -1, "end": 3, "size": 0.2, "color":"black"}
    }
)
fig.add_trace(terrain_surface)

# Přidání několika bodů pro zobrazení původních měření
fig.add_trace(go.Scatter3d(
    x=x_terrain[::10],  # Zobrazení pouze každého 10. bodu
    y=y_terrain[::10],
    z=z_terrain[::10],
    mode='markers',
    marker=dict(
        size=3,
        color='red',
        opacity=0.8
    ),
    name='Měřicí body'
))

# Aktualizace layoutu
fig.update_layout(
    title='Vizualizace 3D modelu terénu',
    scene=dict(
        xaxis_title='Vzdálenost X',
        yaxis_title='Vzdálenost Y',
        zaxis_title='Nadmořská výška',
        aspectmode='manual',
        aspectratio=dict(x=1, y=1, z=0.5)  # Vertikální převýšení
    ),
    width=900,
    height=700
)

fig.show()
```

## Shrnutí

V tomto průvodci jsme prozkoumali techniky 3D vizualizace pomocí knihoven Matplotlib a Plotly. Zde jsou hlavní poznatky:

1. **Matplotlib** nabízí solidní možnosti 3D vizualizace prostřednictvím svého nástroje `mplot3d`:
   - Vhodný pro statické 3D grafy publikační kvality
   - Podporuje bodové, povrchové, drátové a vrstevnicové 3D grafy
   - Méně interaktivní, ale vysoce přizpůsobitelný

2. **Plotly** poskytuje výkonné interaktivní 3D vizualizace:
   - Webové interaktivní grafy s možností přiblížení, posunutí a otáčení
   - Bohatá sada 3D typů grafů včetně bodových, povrchových, síťových a čárových grafů
   - Podporuje pokročilé funkce jako animace, podgrafy a uživatelské interakce

3. **Osvědčené postupy** pro efektivní 3D vizualizaci:
