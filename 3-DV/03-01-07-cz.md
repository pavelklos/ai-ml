## 1. Úvod do barev v datové vizualizaci

Barva je jedním z nejsilnějších nástrojů v datové vizualizaci. Při efektivním použití mohou barvy zlepšit porozumění, zaměřit pozornost na důležité informace, odhalit vzory a učinit vizualizace atraktivnějšími. Nevhodný výběr barev však může zmást, zavádět nebo způsobit, že vizualizace budou pro některé diváky nepřístupné.

```python
# Import základních knihoven
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns

# Nastavení tématu pro konzistentní styl
plt.style.use('seaborn-v0_8-whitegrid')

# Ukázková data
categories = ['Kategorie A', 'Kategorie B', 'Kategorie C', 'Kategorie D', 'Kategorie E']
values = [23, 45, 56, 78, 32]

# Jednoduchý sloupcový graf s výchozími barvami
plt.figure(figsize=(10, 6))
plt.bar(categories, values)
plt.title('Sloupcový graf s výchozími barvami', fontsize=16)
plt.xlabel('Kategorie', fontsize=14)
plt.ylabel('Hodnoty', fontsize=14)
plt.show()

# Stejný graf s pečlivě vybranou barevnou paletou
plt.figure(figsize=(10, 6))
plt.bar(categories, values, color=sns.color_palette("viridis", len(categories)))
plt.title('Sloupcový graf s paletou Viridis', fontsize=16)
plt.xlabel('Kategorie', fontsize=14)
plt.ylabel('Hodnoty', fontsize=14)
plt.show()
```

Klíčové aspekty při výběru barev pro vizualizace zahrnují:
- Typ vizualizovaných dat
- Účel vizualizace
- Vaše publikum a jejich možné poruchy barevného vidění
- Médium prezentace (obrazovka, tisk, projekce)
- Kulturní asociace a emocionální reakce na barvy

## 2. Základy teorie barev pro datovou vizualizaci

Pochopení základní teorie barev pomáhá při informovaném výběru barev pro datovou vizualizaci:

- **Odstín**: Čistá barva (jako červená, modrá, zelená)
- **Sytost**: Intenzita nebo čistota barvy
- **Hodnota/Jas**: Světlost nebo tmavost barvy
- **Barevná harmonie**: Jak barvy spolupracují

```python
import matplotlib.pyplot as plt
from matplotlib.colors import hsv_to_rgb
import numpy as np

# Vytvoření obrázku pro demonstraci HSV barevného prostoru
fig, axes = plt.subplots(1, 3, figsize=(18, 6))

# 1. Variace odstínu (při konstantní sytosti a hodnotě)
hues = np.linspace(0, 1, 10)
saturation = 0.8
value = 0.9
hsv_colors = np.zeros((10, 3))
hsv_colors[:, 0] = hues
hsv_colors[:, 1] = saturation
hsv_colors[:, 2] = value
rgb_colors = [hsv_to_rgb(hsv) for hsv in hsv_colors]

axes[0].bar(range(10), [1]*10, color=rgb_colors, width=0.8)
axes[0].set_title('Variace odstínu', fontsize=14)
axes[0].set_xticks([])
axes[0].set_yticks([])

# 2. Variace sytosti (při konstantním odstínu a hodnotě)
saturations = np.linspace(0.1, 1, 10)
hue = 0.6  # Modrá
hsv_colors = np.zeros((10, 3))
hsv_colors[:, 0] = hue
hsv_colors[:, 1] = saturations
hsv_colors[:, 2] = value
rgb_colors = [hsv_to_rgb(hsv) for hsv in hsv_colors]

axes[1].bar(range(10), [1]*10, color=rgb_colors, width=0.8)
axes[1].set_title('Variace sytosti', fontsize=14)
axes[1].set_xticks([])
axes[1].set_yticks([])

# 3. Variace hodnoty/jasu (při konstantním odstínu a sytosti)
values = np.linspace(0.2, 1, 10)
hsv_colors = np.zeros((10, 3))
hsv_colors[:, 0] = hue
hsv_colors[:, 1] = saturation
hsv_colors[:, 2] = values
rgb_colors = [hsv_to_rgb(hsv) for hsv in hsv_colors]

axes[2].bar(range(10), [1]*10, color=rgb_colors, width=0.8)
axes[2].set_title('Variace jasu/hodnoty', fontsize=14)
axes[2].set_xticks([])
axes[2].set_yticks([])

plt.tight_layout()
plt.show()
```

Vztahy barev ve vizualizaci:
- **Komplementární barvy** (protilehlé na barevném kruhu) vytvářejí silný kontrast
- **Analogické barvy** (sousední na barevném kruhu) vytvářejí harmonii
- **Monochromatické barvy** (variace jediné barvy) zobrazují gradaci

## 3. Vestavěné barevné palety v Python knihovnách

Vizualizační knihovny Pythonu nabízejí četné vestavěné barevné palety navržené pro specifické typy dat a vizualizační cíle:

```python
# Průzkum barevných palet v Matplotlib a Seaborn
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np

# Vytvoření obrázku s podgrafy
fig, axes = plt.subplots(3, 3, figsize=(15, 12))
axes = axes.flatten()

# Matplotlib barevné mapy
matplotlib_cmaps = ['viridis', 'plasma', 'inferno', 'magma', 'cividis']
for i, cmap in enumerate(matplotlib_cmaps):
    gradient = np.linspace(0, 1, 256)
    gradient = np.vstack((gradient, gradient))
    axes[i].imshow(gradient, aspect='auto', cmap=plt.get_cmap(cmap))
    axes[i].set_title(f'Matplotlib: {cmap}', fontsize=12)
    axes[i].set_xticks([])
    axes[i].set_yticks([])

# Seaborn barevné palety
seaborn_palettes = ['rocket', 'mako', 'flare', 'cubehelix']
for i, palette in enumerate(seaborn_palettes):
    data = np.random.rand(20, 20)
    axes[i+5].imshow(data, cmap=palette)
    axes[i+5].set_title(f'Seaborn: {palette}', fontsize=12)
    axes[i+5].set_xticks([])
    axes[i+5].set_yticks([])

plt.tight_layout()
plt.show()

# Ukázka kategoriálních palet ze Seabornu
plt.figure(figsize=(12, 10))
sns.set_style("white")

# Typy barevných palet Seabornu
palette_types = ['deep', 'muted', 'pastel', 'bright', 'dark', 'colorblind']
num_categories = 10

# Vytvoření grafu zobrazujícího různé palety
for i, palette in enumerate(palette_types):
    colors = sns.color_palette(palette, num_categories)
    for j, color in enumerate(colors):
        plt.bar(j + i*num_categories, 1, color=color, width=0.8)
    plt.text(i*num_categories + num_categories/2 - 1, 1.1, palette, 
             fontsize=12, ha='center')

plt.xlim(-0.5, len(palette_types)*num_categories - 0.5)
plt.ylim(0, 1.5)
plt.title('Kategoriální barevné palety Seabornu', fontsize=16)
plt.axis('off')
plt.tight_layout()
plt.show()
```

Klíčové vestavěné palety:
- **Matplotlib barevné mapy**: `viridis`, `plasma`, `inferno`, `magma`, atd.
- **Seaborn palety**: 
  - Typy barevných palet: `deep`, `muted`, `pastel`, `bright`, `dark`, `colorblind`
  - Sekvenční: `Blues`, `Greens`, `Oranges`, atd.
  - Divergentní: `RdBu`, `RdYlGn`, `coolwarm`, atd.

## 4. Vytváření vlastních barevných palet

I když jsou vestavěné barevné palety pohodlné, někdy potřebujete vytvořit vlastní palety, které odpovídají vaší značce, zdůrazňují určité datové body nebo dosahují specifického estetického vzhledu:

```python
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import matplotlib.colors as mcolors

# Ukázková data
categories = ['A', 'B', 'C', 'D', 'E']
values = [25, 40, 30, 55, 15]

# 1. Vytvoření vlastní barevné palety z RGB hodnot
rgb_custom = [(0.2, 0.4, 0.6), (0.8, 0.4, 0.2), 
             (0.3, 0.7, 0.5), (0.6, 0.3, 0.7),
             (0.8, 0.7, 0.2)]

# 2. Vytvoření vlastní barevné palety z hex hodnot
hex_custom = ['#3498db', '#e74c3c', '#2ecc71', '#f39c12', '#9b59b6']

# 3. Vytvoření vlastní sekvenční palety pomocí seaborn
seq_custom = sns.light_palette("seagreen", n_colors=5)

# 4. Vytvoření vlastní divergentní palety pomocí seaborn
div_custom = sns.diverging_palette(240, 10, n=5)

# Vytvoření obrázku s 2x2 podgrafy
fig, axes = plt.subplots(2, 2, figsize=(14, 10))

# Vykreslení každé vlastní palety
axes[0, 0].bar(categories, values, color=rgb_custom)
axes[0, 0].set_title('Vlastní RGB paleta', fontsize=14)
axes[0, 0].grid(axis='y', linestyle='--', alpha=0.7)

axes[0, 1].bar(categories, values, color=hex_custom)
axes[0, 1].set_title('Vlastní Hex barevná paleta', fontsize=14)
axes[0, 1].grid(axis='y', linestyle='--', alpha=0.7)

axes[1, 0].bar(categories, values, color=seq_custom)
axes[1, 0].set_title('Seaborn sekvenční paleta', fontsize=14)
axes[1, 0].grid(axis='y', linestyle='--', alpha=0.7)

axes[1, 1].bar(categories, values, color=div_custom)
axes[1, 1].set_title('Seaborn divergentní paleta', fontsize=14)
axes[1, 1].grid(axis='y', linestyle='--', alpha=0.7)

plt.tight_layout()
plt.show()

# Vytvoření vlastní barevné mapy pro spojitá data
gradient = np.linspace(0, 1, 256)
gradient = np.vstack((gradient, gradient))

custom_cmap = mcolors.LinearSegmentedColormap.from_list(
    "custom_cmap", ["#1a237e", "#4a148c", "#b71c1c", "#e65100", "#f9a825"]
)

plt.figure(figsize=(10, 4))
plt.imshow(gradient, aspect='auto', cmap=custom_cmap)
plt.title('Vlastní barevná mapa pro spojitá data', fontsize=16)
plt.xticks([])
plt.yticks([])
plt.tight_layout()
plt.show()
```

Metody vytváření vlastních barevných palet:
- Definování barev pomocí RGB n-tic (hodnoty mezi 0 a 1)
- Definování barev pomocí HEX barevných kódů
- Vytváření sekvenčních palet pomocí `sns.light_palette()` nebo `sns.dark_palette()`
- Vytváření divergentních palet pomocí `sns.diverging_palette()`
- Vytváření vlastní barevné mapy pomocí `matplotlib.colors.LinearSegmentedColormap.from_list()`

## 5. Barvoslepost a aspekty přístupnosti

Přibližně 8-10 % mužů a 0,5 % žen po celém světě má nějakou formu poruchy barevného vidění. Vytváření vizualizací, které jsou přístupné všem divákům, je zásadní:

```python
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
from colorspacious import cspace_converter

# Generování ukázkových dat
np.random.seed(42)
data = np.random.randn(10, 4)
categories = ['A', 'B', 'C', 'D']

# Různé barevné palety - některé dobré a některé špatné pro barvoslepost
palettes = {
    "Výchozí": ['red', 'green', 'blue', 'purple'],
    "Přátelská k barvosleposti": sns.color_palette("colorblind", 4),
    "Viridis": sns.color_palette("viridis", 4),
    "Červeno-zelená (Problematická)": ['red', 'green', 'darkgreen', 'lightgreen']
}

# Vytvoření obrázku s 2x2 podgrafy
fig, axes = plt.subplots(2, 2, figsize=(14, 10))
axes = axes.flatten()

# Vykreslení stejných dat s různými paletami
for i, (name, palette) in enumerate(palettes.items()):
    for j, color in enumerate(palette):
        axes[i].plot(data[:, j], color=color, linewidth=3, label=f'Řada {j+1}')
    axes[i].set_title(f"Paleta {name}", fontsize=14)
    axes[i].legend(loc='upper right')
    axes[i].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()

# Simulace různých typů barvosleposti
try:
    from colorspacious import cspace_converter
    
    def simulate_colorblindness(rgb, deficiency_type):
        """
        Převod RGB barev k simulaci barvosleposti.
        deficiency_type by měl být 'deuteranomaly', 'protanomaly', nebo 'tritanomaly'
        """
        return cspace_converter("sRGB1", f"sRGB1+{deficiency_type}")(rgb)
    
    # Vytvoření barevných vzorků pro paletu přátelskou k barvosleposti
    palette = sns.color_palette("colorblind", 8)
    
    # Typy poruch barevného vidění
    deficiency_types = [None, "deuteranomaly", "protanomaly", "tritanomaly"]
    deficiency_labels = ["Normální vidění", "Deuteranomalie", "Protanomalie", "Tritanomalie"]
    
    # Vytvoření obrázku s řádky pro různé poruchy barev
    fig, axes = plt.subplots(len(deficiency_types), 1, figsize=(10, 12))
    
    for i, (deficiency, label) in enumerate(zip(deficiency_types, deficiency_labels)):
        # Pokud jde o normální vidění, použijte původní barvy
        if deficiency is None:
            colors = palette
        else:
            # Simulace barvosleposti
            colors = [simulate_colorblindness(rgb, deficiency) for rgb in palette]
        
        # Vytvoření barevných vzorků
        for j, color in enumerate(colors):
            axes[i].bar(j, 1, color=color, width=0.8)
        
        axes[i].set_title(label, fontsize=14)
        axes[i].set_xlim(-0.5, len(palette) - 0.5)
        axes[i].set_yticks([])
        axes[i].set_xticks([])
    
    plt.tight_layout()
    plt.show()
    
except ImportError:
    print("Pro simulaci barvosleposti, prosím nainstalujte balíček colorspacious: pip install colorspacious")
```

Pokyny pro vizualizace přátelské k barvoslepým:
- Vyhněte se kombinacím červená-zelená (nejčastější forma barvosleposti)
- Používejte vzory, tvary nebo styly čar jako doplněk k barvám
- Zajistěte dostatečný kontrast mezi barvami
- Používejte palety přátelské k barvosleposti jako `viridis`, `plasma` nebo `colorblind` od Seabornu
- Zvažte použití nástrojů ke kontrole, jak vaše vizualizace vypadají pro lidi s poruchami barevného vidění

## 6. Používání barev k reprezentaci datových proměnných

Barvy mohou kódovat různé typy informací ve vaší vizualizaci. Pochopení, jak mapovat datové proměnné na barvy, je zásadní:

```python
import matplotlib.pyplot as plt
import seaborn as sns
import pandas as pd
import numpy as np

# Generování ukázkových dat
np.random.seed(42)
n = 100
data = pd.DataFrame({
    'x': np.random.uniform(-3, 3, n),
    'y': np.random.uniform(-3, 3, n),
    'size': np.random.uniform(20, 200, n),
    'category': np.random.choice(['Skupina A', 'Skupina B', 'Skupina C', 'Skupina D'], n),
    'value': np.random.randn(n)
})

# Přidání vypočítaných polí
data['distance'] = np.sqrt(data['x']**2 + data['y']**2)

# Nastavení obrázku s podgrafy
fig, axes = plt.subplots(2, 2, figsize=(16, 14))

# 1. Barva reprezentující kategorie (kvalitativní data)
sns.scatterplot(
    data=data, 
    x='x', y='y',
    hue='category',  # Barva podle kategoriální proměnné
    size='size',     # Velikost podle spojité proměnné
    sizes=(20, 200),
    palette='colorblind',  # Paleta přátelská k barvosleposti
    ax=axes[0, 0]
)
axes[0, 0].set_title('Barva reprezentující kategorie', fontsize=14)
axes[0, 0].grid(True, alpha=0.3)

# 2. Barva reprezentující spojité hodnoty (sekvenční paleta)
scatter = sns.scatterplot(
    data=data, 
    x='x', y='y',
    hue='value',     # Barva podle spojité proměnné
    size='size',
    sizes=(20, 200),
    palette='viridis',  # Sekvenční barevná mapa
    ax=axes[0, 1]
)
axes[0, 1].set_title('Barva reprezentující spojité hodnoty', fontsize=14)
plt.colorbar(scatter.collections[0], ax=axes[0, 1], label='Hodnota')
axes[0, 1].grid(True, alpha=0.3)

# 3. Barva reprezentující vzdálenost od počátku (sekvenční paleta)
scatter = sns.scatterplot(
    data=data,
    x='x', y='y',
    hue='distance',  # Barva podle vzdálenosti od počátku
    palette='YlOrRd',  # Žluto-oranžovo-červená sekvenční paleta
    ax=axes[1, 0]
)
axes[1, 0].set_title('Barva reprezentující vzdálenost (sekvenční)', fontsize=14)
plt.colorbar(scatter.collections[0], ax=axes[1, 0], label='Vzdálenost od počátku')
axes[1, 0].grid(True, alpha=0.3)

# 4. Barva reprezentující hodnoty s bodem středu (divergentní paleta)
# Vytvoření nového pole s hodnotami centrovanými kolem středního bodu
data['centered_value'] = data['value'] - data['value'].mean()

scatter = sns.scatterplot(
    data=data,
    x='x', y='y',
    hue='centered_value',
    palette='coolwarm',  # Divergentní barevná mapa
    vmin=-3, vmax=3,     # Explicitně nastavit barevnou škálu
    ax=axes[1, 1]
)
axes[1, 1].set_title('Barva reprezentující divergentní hodnoty', fontsize=14)
plt.colorbar(scatter.collections[0], ax=axes[1, 1], label='Centrovaná hodnota')
axes[1, 1].grid(True, alpha=0.3)

plt.tight_layout()
plt.show()
```

Nejlepší postupy pro mapování dat na barvy:
- **Kategoriální proměnné**: Používejte odlišné barvy, které lze snadno rozlišit
- **Sekvenční data**: Používejte jednobarevné nebo vícebarevné sekvenční palety k zobrazení intenzity
- **Divergentní data**: Používejte divergentní palety s odlišnou barvou středového bodu
- **Zvýraznění důležitých dat**: Používejte jasné nebo syté barvy pro klíčové body
- **Přidání barevné legendy**: Vždy poskytněte legendu nebo barevnou lištu pro referenci

## 7. Sekvenční, divergentní a kategoriální barevná schémata

Různé typy dat vyžadují různá barevná schémata pro efektivní komunikaci informací:

```python
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import pandas as pd

# Vytvoření ukázkových dat
np.random.seed(42)
x = np.linspace(0, 10, 20)
y = np.linspace(0, 10, 20)
X, Y = np.meshgrid(x, y)

# Vytvoření obrázku s podgrafy
fig, axes = plt.subplots(3, 3, figsize=(18, 16))

# 1. SEKVENČNÍ BAREVNÁ SCHÉMATA (pro data postupující od nízkých k vysokým)
# Sekvenční data: teplotní gradient
Z1 = np.sqrt((X-5)**2 + (Y-5)**2)  # Vzdálenost od středu

# Jednobarevné sekvenční
im1 = axes[0, 0].imshow(Z1, cmap='Blues')
axes[0, 0].set_title('Jednobarevné sekvenční: Blues', fontsize=14)
plt.colorbar(im1, ax=axes[0, 0], orientation='horizontal')
axes[0, 0].set_xticks([])
axes[0, 0].set_yticks([])

# Vícebarevné sekvenční
im2 = axes[0, 1].imshow(Z1, cmap='viridis')
axes[0, 1].set_title('Vícebarevné sekvenční: Viridis', fontsize=14)
plt.colorbar(im2, ax=axes[0, 1], orientation='horizontal')
axes[0, 1].set_xticks([])
axes[0, 1].set_yticks([])

# Další příklad sekvenčního
im3 = axes[0, 2].imshow(Z1, cmap='YlGnBu')
axes[0, 2].set_title('Vícebarevné sekvenční: YlGnBu', fontsize=14)
plt.colorbar(im3, ax=axes[0, 2], orientation='horizontal')
axes[0, 2].set_xticks([])
axes[0, 2].set_yticks([])

# 2. DIVERGENTNÍ BAREVNÁ SCHÉMATA (pro data s významným středovým bodem)
# Divergentní data: hodnoty se rozcházejí od centrální hodnoty
Z2 = X - Y  # Rozdíl mezi souřadnicemi x a y

# Divergentní červeno-modrá
im4 = axes[1, 0].imshow(Z2, cmap='RdBu')
axes[1, 0].set_title('Divergentní: RdBu', fontsize=14)
plt.colorbar(im4, ax=axes[1, 0], orientation='horizontal')
axes[1, 0].set_xticks([])
axes[1, 0].set_yticks([])

# Divergentní purpurovo-zelená
im5 = axes[1, 1].imshow(Z2, cmap='PuOr')
axes[1, 1].set_title('Divergentní: PuOr', fontsize=14)
plt.colorbar(im5, ax=axes[1, 1], orientation='horizontal')
axes[1, 1].set_xticks([])
axes[1, 1].set_yticks([])

# Divergentní s coolwarm
im6 = axes[1, 2].imshow(Z2, cmap='coolwarm')
axes[1, 2].set_title('Divergentní: Coolwarm', fontsize=14)
plt.colorbar(im6, ax=axes[1, 2], orientation='horizontal')
axes[1, 2].set_xticks([])
axes[1, 2].set_yticks([])

# 3. KATEGORIÁLNÍ BAREVNÁ SCHÉMATA (pro odlišné, neuspořádané kategorie)
# Vytvoření ukázkových kategoriálních dat
categories = ['Kategorie A', 'Kategorie B', 'Kategorie C', 'Kategorie D', 'Kategorie E']
values = [23, 45, 56, 78, 32]

# Výchozí kategoriální
axes[2, 0].bar(categories, values, color=sns.color_palette("tab10", len(categories)))
```

- **Sekvenční barevná schémata**: Použijte pro data, která se pohybují od nízkých k vysokým hodnotám
  - Jednobarevná schémata: Použijte proměnnou sytost/jas jedné barvy
  - Vícebarevná schémata: Postupně přecházejte mezi odstíny
- **Divergentní barevná schémata**: Použijte pro data s významným středovým bodem
  - Použijte kontrastní barvy na extrémech s neutrální barvou uprostřed
  - Zajistěte, aby středový bod byl vizuálně odlišný
- **Kategoriální barevná schémata**: Použijte pro odlišné, neuspořádané kategorie

  - Zvažte použití palet přátelských k barvosleposti

```python
            n = n_colors_slider.value
            reverse = reverse_checkbox.value

    ui = widgets.VBox([
        widgets.HBox([palette_dropdown, n_colors_slider, reverse_checkbox]),
        output
    ])
    
    # Počáteční vykreslení
    update_plot()
    
    # Zobrazení UI
```

## 9. Nejlepší postupy pro výběr barev

Zde jsou klíčová doporučení pro výběr barev v datových vizualizacích:

```python
import matplotlib.pyplot as plt
import seaborn as sns
import numpy as np
import pandas as pd

# Vytvoření ukázkových dat
np.random.seed(42)
dates = pd.date_range('2023-01-01', periods=100)
data = pd.DataFrame({
    'date': dates,
    'value_a': np.cumsum(np.random.normal(0, 1, 100)),
    'value_b': np.cumsum(np.random.normal(0, 1, 100)) + 5,
    'value_c': np.cumsum(np.random.normal(0, 1, 100)) + 10,
    'category': np.random.choice(['Skupina 1', 'Skupina 2', 'Skupina 3'], 100)
})

# 1. DOBRÁ PRAXE: Konzistentní barevné schéma
plt.figure(figsize=(12, 6))

sns.lineplot(
    data=data, 
    x='date', 
    y='value_a', 
    label='Řada A',
    color='#1f77b4'  # Konzistentní modrá
)
sns.lineplot(
    data=data, 
    x='date', 
    y='value_b', 
    label='Řada B',
    color='#ff7f0e'  # Konzistentní oranžová
)
sns.lineplot(
    data=data, 
    x='date', 
    y='value_c', 
    label='Řada C',
    color='#2ca02c'  # Konzistentní zelená
)

plt.title('DOBRÉ: Konzistentní barevné schéma', fontsize=16)
plt.grid(True, alpha=0.3)
plt.legend(title='Datové řady')
plt.tight_layout()
plt.show()

# 2. ŠPATNÁ PRAXE: Příliš mnoho barev a špatný kontrast
plt.figure(figsize=(12, 6))

# Vytváření náhodných dat pro bodový graf
x = np.random.rand(100)
y = np.random.rand(100)
categories = np.random.choice(['A', 'B', 'C', 'D', 'E', 'F', 'G', 'H', 'I', 'J'], 100)

# Příliš mnoho barev se špatným kontrastem
rainbow_colors = plt.cm.rainbow(np.linspace(0, 1, len(np.unique(categories))))

for i, category in enumerate(np.unique(categories)):
    mask = categories == category
    plt.scatter(
        x[mask], y[mask], 
        label=category, 
        color=rainbow_colors[i], 
        s=100
    )

plt.title('ŠPATNÉ: Příliš mnoho barev se špatným kontrastem', fontsize=16)
plt.grid(True, alpha=0.3)
plt.legend(title='Kategorie', bbox_to_anchor=(1.05, 1), loc='upper left')
plt.tight_layout()
plt.show()

# 3. DOBRÁ PRAXE: Vhodné použití sekvenční palety pro číselná data
plt.figure(figsize=(10, 8))

# Vytvoření teplotní mapy pro korelační data

sns.heatmap(
    corr_data, 
    cmap='YlGnBu',  # Vhodná sekvenční paleta pro korelace
    annot=True, 
    fmt=".2f", 
    linewidths=0.5
)

plt.title('DOBRÉ: Vhodná sekvenční paleta pro korelace', fontsize=16)
plt.tight_layout()
plt.show()

# 4. ŠPATNÁ PRAXE: Použití duhové barevné mapy pro sekvenční data

# Stejná data, ale s duhovou barevnou mapou

plt.title('ŠPATNÉ: Duhová barevná mapa pro sekvenční data', fontsize=16)
plt.tight_layout()
plt.show()
```
