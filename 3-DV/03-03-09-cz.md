## Úvod

Komplexní sítě jsou systémy propojených komponent, které lze nalézt v mnoha oblastech včetně sociálních vztahů, biologických systémů, dopravních tras, počítačových sítí a citačních vzorů. Vizualizace těchto sítí nám pomáhá pochopit struktury, vzory a vztahy, které by jinak mohly zůstat skryty v datech.

V této příručce prozkoumáme techniky pro vizualizaci komplexních sítí v Pythonu, od základních grafových reprezentací až po interaktivní a rozsáhlé vizualizace sítí.

## Základy vizualizace sítí

### Co dělá vizualizaci sítí efektivní

Efektivní vizualizace sítě by měla:
- Jasně zobrazovat spojení mezi uzly
- Odhalovat důležité struktury (shluky, huby, mosty)
- Minimalizovat vizuální nepřehlednost
- Zvýrazňovat klíčové metriky nebo vlastnosti
- Být interpretovatelná pro zamýšlené publikum

### Běžné typy sítí

- **Neorientované sítě**: Spojení nemají směr (sítě přátelství)
- **Orientované sítě**: Spojení mají směr (vztahy sledování)
- **Vážené sítě**: Spojení mají různou sílu
- **Bipartitní sítě**: Uzly spadají do dvou odlišných skupin
- **Temporální sítě**: Spojení se mění v čase
- **Vícevrstvé sítě**: Existuje současně více typů vztahů

### Vytvoření jednoduché sítě

Začneme se základní vizualizací sítě pomocí NetworkX:

```python
# Vytvoření jednoduchého grafu
G = nx.Graph()

# Přidání hran
```

## Stylování uzlů a hran

Stylování uzlů a hran může zvýraznit důležité vzory a metriky:

```python
# Vytvoření komplexnějšího grafu
G_styled = nx.karate_club_graph()  # Známý dataset sociální sítě

# Výpočet centrality stupně uzlů
centrality = nx.degree_centrality(G_styled)

# Výpočet mezilehlosti (betweenness) hran

# Členství v klubu (skutečné komunity)

pos = nx.spring_layout(G_styled, seed=42)

# Vykreslení uzlů s různými velikostmi podle centrality
nx.draw_networkx_nodes(
    G_styled, pos, 
    node_size=[centrality[n] * 5000 for n in G_styled.nodes()],
    node_color=club_color, 
    alpha=0.8
)

# Vykreslení hran s různou šířkou podle mezilehlosti
edge_widths = [edge_betweenness[(u, v)] * 10 for u, v in G_styled.edges()]
nx.draw_networkx_edges(G_styled, pos, width=edge_widths, alpha=0.5, edge_color='gray')

# Přidání popisků
nx.draw_networkx_labels(G_styled, pos, font_size=12, font_weight='bold')

plt.title("Síť Karate klubu: Velikost uzlů = Centralita, Barva = Členství v klubu", fontsize=15)
plt.axis('off')
plt.show()

# Výpis informací o nejvíce centrálních uzlech a hranách
print("Top 3 uzly podle centrality:")
top_nodes = sorted(centrality.items(), key=lambda x: x[1], reverse=True)[:3]
for node, cent in top_nodes:
    print(f"Uzel {node}: {cent:.3f}")

print("\nTop 3 hrany podle mezilehlosti:")
top_edges = sorted(edge_betweenness.items(), key=lambda x: x[1], reverse=True)[:3]
for edge, bet in top_edges:
    print(f"Hrana {edge}: {bet:.3f}")
```

## Detekce a vizualizace komunit

Identifikace a vizualizace komunit v sítích může odhalit důležité strukturální vzorce:

```python
# Import algoritmu detekce komunit
from community import best_partition  # Z balíčku python-louvain

# Vytvoření středně velké sítě se strukturou komunit
G_comm = nx.random_partition_graph([15, 15, 20], 0.25, 0.01, seed=42)

partition = best_partition(G_comm)

# Převod rozdělení na seznam štítků komunit
communities = [partition[node] for node in G_comm.nodes()]

# Získání unikátních komunit
unique_communities = set(communities)
colors = plt.cm.rainbow(np.linspace(0, 1, len(unique_communities)))
community_colors = {com: colors[i] for i, com in enumerate(unique_communities)}

# Mapování barev komunit na uzly
node_colors = [community_colors[communities[i]] for i in range(len(communities))]

# Vizualizace sítě s komunitami
plt.figure(figsize=(12, 10))
pos = nx.spring_layout(G_comm, seed=42)

# Vykreslení sítě
nx.draw_networkx(
    G_comm,
    pos=pos,
    node_color=node_colors,
    node_size=150,
    with_labels=False,
    width=0.6,
    edge_color='lightgray'
)

# Přidání názvu a legendy
plt.title("Síť s komunitami detekovanými Louvainovým algoritmem", fontsize=15)

# Vytvoření vlastní legendy
from matplotlib.lines import Line2D
legend_elements = [Line2D([0], [0], marker='o', color='w', label=f'Komunita {com}',
                          markerfacecolor=community_colors[com], markersize=10)
                  for com in unique_communities]
plt.legend(handles=legend_elements, loc='upper right')

plt.axis('off')
plt.show()

# Výpis statistik o komunitách

    size = communities.count(com)
    community_sizes[com] = size
    print(f"Komunita {com}: {size} uzlů")
```

## Interaktivní vizualizace sítí

Interaktivní vizualizace umožňují průzkum komplexních sítí:

```python
# Použití Plotly pro interaktivní vizualizaci
def create_interactive_network(G, pos=None, title="Interaktivní síť"):
    if pos is None:
        pos = nx.spring_layout(G, seed=42)
    
    # Vytvoření tras hran
    edge_x = []
    edge_y = []
    for edge in G.edges():
        x0, y0 = pos[edge[0]]
        x1, y1 = pos[edge[1]]
        edge_x.extend([x0, x1, None])
        edge_y.extend([y0, y1, None])
    
    edge_trace = go.Scatter(
        x=edge_x, y=edge_y,
        line=dict(width=0.5, color='#888'),
        hoverinfo='none',
        mode='lines')
    
    # Vytvoření tras uzlů
    node_x = []
    node_y = []
    for node in G.nodes():
        x, y = pos[node]
        node_x.append(x)
        node_y.append(y)
    
    node_trace = go.Scatter(
        x=node_x, y=node_y,
        mode='markers',
        hoverinfo='text',
        marker=dict(
            showscale=True,
            colorscale='YlGnBu',
            reversescale=True,
            color=[],
            size=10,
            colorbar=dict(
                thickness=15,
                title='Spojení uzlů',
                xanchor='left',
                titleside='right'
            ),
            line_width=2))
    
    # Obarvení bodů uzlů podle počtu spojení
    node_adjacencies = []
    node_text = []
    for node in G.nodes():
        adjacencies = list(G.neighbors(node))
        node_adjacencies.append(len(adjacencies))
        node_text.append(f'Uzel: {node}<br>Spojení: {len(adjacencies)}')
    
    node_trace.marker.color = node_adjacencies
    node_trace.text = node_text
    
    # Vytvoření figury
    fig = go.Figure(data=[edge_trace, node_trace],
                 layout=go.Layout(
                    title=title,
                    titlefont_size=16,
                    showlegend=False,
                    hovermode='closest',
                    margin=dict(b=20,l=5,r=5,t=40),
                    xaxis=dict(showgrid=False, zeroline=False, showticklabels=False),
                    yaxis=dict(showgrid=False, zeroline=False, showticklabels=False))
                    )
    return fig

# Vytvoření středně velkého náhodného grafu pro demonstraci
G_interactive = nx.gnm_random_graph(30, 80, seed=42)

# Vygenerování interaktivní vizualizace

fig = create_interactive_network(G_interactive, pos_interactive, "Interaktivní vizualizace sítě")
fig.show()
```


## Vizualizace rozsáhlých sítí

Pro větší sítě potřebujeme speciální techniky, aby byly vizualizace interpretovatelné:

```python
# Vygenerování větší sítě
G_large = nx.barabasi_albert_graph(500, 3, seed=42)

# Výpočet stupňů uzlů
degrees = dict(nx.degree(G_large))

# Funkce pro filtrování na zobrazení pouze důležitých uzlů a spojení
def filter_graph_by_degree(G, min_degree=5):
    important_nodes = [node for node, degree in dict(G.degree()).items() if degree >= min_degree]
    subgraph = G.subgraph(important_nodes)
    return subgraph

# Vytvoření filtrovaného grafu
G_filtered = filter_graph_by_degree(G_large, min_degree=10)

print(f"Původní síť: {G_large.number_of_nodes()} uzlů, {G_large.number_of_edges()} hran")
print(f"Filtrovaná síť: {G_filtered.number_of_nodes()} uzlů, {G_filtered.number_of_edges()} hran")

# Vizuální porovnání úplného vs. filtrovaného grafu
fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(20, 10))

# Vykreslení úplného grafu s malými uzly
pos_large = nx.spring_layout(G_large, seed=42)
nx.draw(G_large, pos=pos_large, ax=ax1, node_size=10, node_color='blue', alpha=0.6, width=0.1)
ax1.set_title("Úplná síť (500 uzlů)", fontsize=15)

# Vykreslení filtrovaného grafu s velikostí uzlů úměrnou stupni
pos_filtered = nx.spring_layout(G_filtered, seed=42)
node_sizes = [degrees[node] * 20 for node in G_filtered.nodes()]
nx.draw(G_filtered, pos=pos_filtered, ax=ax2, node_size=node_sizes, 
        node_color='red', alpha=0.8, width=0.5, with_labels=True)
ax2.set_title("Filtrovaná síť (stupeň ≥ 10)", fontsize=15)

plt.tight_layout()
plt.show()
```

```python
def create_citation_network(num_papers=100, citations_per_paper=5):
    G = nx.DiGraph()  # Orientovaný graf pro citace
    
    # Přidání článků jako uzlů s metadaty
    topics = ["Strojové učení", "Sítě", "Vizualizace", "Statistika"]
    years = list(range(2010, 2023))
    
    for i in range(num_papers):
        G.add_node(i, 
                  title=f"Článek {i}",
                  topic=random.choice(topics),
                  year=random.choice(years),
                  citations=0)
    
    # Přidání citací (články mohou citovat pouze starší články)
    for i in range(num_papers):
        year_i = G.nodes[i]['year']
        # Nalezení potenciálních článků k citování (starší články)
        candidates = [j for j in range(num_papers) if G.nodes[j]['year'] < year_i]
        if candidates:
            # Přidání citací
            num_citations = min(len(candidates), random.randint(1, citations_per_paper))
            for cited_paper in random.sample(candidates, num_citations):
                G.add_edge(i, cited_paper)
                G.nodes[cited_paper]['citations'] += 1
    
    return G
```

```python
# Vytvoření simulované sociální sítě s komunitami
def create_social_network(num_users=100, num_communities=4, p_in=0.15, p_out=0.01):
    # Velikost oddílu pro každou komunitu
    partition_sizes = [num_users // num_communities] * num_communities
    for i in range(num_users % num_communities):
        partition_sizes[i] += 1
    
    # Vytvoření grafu se strukturou komunit
    G = nx.random_partition_graph(partition_sizes, p_in, p_out, seed=42)
    
    # Přidání atributů k uzlům
    age_groups = ["18-24", "25-34", "35-44", "45+"]
    interests = ["Sport", "Technologie", "Umění", "Cestování", "Jídlo", "Hudba"]
    activity_levels = ["Nízká", "Střední", "Vysoká"]
    
    for i, node in enumerate(G.nodes()):
        community_id = i // (num_users // num_communities)  # Přibližné přiřazení komunity
        G.nodes[node]['community'] = community_id
        G.nodes[node]['age_group'] = random.choice(age_groups)
        G.nodes[node]['interests'] = random.sample(interests, random.randint(1, 3))
        G.nodes[node]['activity'] = random.choice(activity_levels)
        G.nodes[node]['influence'] = random.uniform(0, 1)  # Náhodné skóre vlivu
    
    # Výpočet síťových metrik a přidání jako atributů
    degree_cent = nx.degree_centrality(G)
    betweenness_cent = nx.betweenness_centrality(G, k=min(20, num_users))  # Aproximace pro větší sítě
    eigenvector_cent = nx.eigenvector_centrality_numpy(G)
    
    for node in G.nodes():
        G.nodes[node]['degree_centrality'] = degree_cent[node]
        G.nodes[node]['betweenness_centrality'] = betweenness_cent[node]
        G.nodes[node]['eigenvector_centrality'] = eigenvector_cent[node]
    
    return G

# Vytvoření sociální sítě
G_social = create_social_network(num_users=100, num_communities=5, p_in=0.15, p_out=0.01)

# Získání přiřazení komunit

node_sizes = [1000 * eigenvector[node] + 20 for node in G_social.nodes()]

# Vykreslení uzlů
nx.draw_networkx_nodes(G_social, pos, 
                      node_color=node_colors, 
                      node_size=node_sizes, 
                      alpha=0.7)

# Vykreslení hran s průhledností
nx.draw_networkx_edges(G_social, pos, 
                      alpha=0.2, 
                      width=0.3)

# Přidání popisků pouze k nejvlivnějším uzlům

nx.draw_networkx_labels(G_social, pos, 
                       labels={node: str(node) for node in influential_nodes},
                       font_size=10, font_weight='bold')

# Přidání legendy pro komunity
from matplotlib.patches import Patch
legend_elements = [Patch(facecolor=community_colors[i], 
                        edgecolor='w',
                        label=f'Komunita {i}')

plt.title("Analýza sociální sítě: Struktura komunit a vliv", fontsize=20)
plt.axis('off')
plt.tight_layout()
plt.show()

# Analýza komunit
print("Analýza komunit:")
for community_id in unique_communities:
    community_nodes = [node for node, comm in communities.items() if comm == community_id]
    community_size = len(community_nodes)
    subgraph = G_social.subgraph(community_nodes)
    internal_edges = subgraph.number_of_edges()
    external_edges = sum(1 for u, v in G_social.edges() 
                       if (u in community_nodes and v not in community_nodes) or 
                          (v in community_nodes and u not in community_nodes))
    
    print(f"Komunita {community_id}: {community_size} členů, {internal_edges} interních spojení, "
          f"{external_edges} externích spojení")

# Nalezení klíčových propojovacích uzlů mezi komunitami (vysoká mezilehlost, spojení s více komunitami)
betweenness = nx.get_node_attributes(G_social, 'betweenness_centrality')
top_bridges = sorted(betweenness.items(), key=lambda x: x[1], reverse=True)[:5]

    node_comm = communities[node]
    neighbor_comms = set(communities[neigh] for neigh in G_social.neighbors(node))

          f"mezilehlost: {score:.4f}")
```

- Přizpůsobte vizualizace na základě specifické domény a cílů analýzy

Kombinací těchto technik můžete vytvářet informativní a názorné vizualizace, které pomáhají odhalit vzorce a vztahy v komplexních síťových datech.
