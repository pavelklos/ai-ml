## Úvod

Vizualizace dat v reálném čase je proces zobrazování dat v okamžiku jejich generování nebo přijetí, což umožňuje okamžitou analýzu a rozhodování. Tento přístup je zásadní v oblastech jako finanční obchodování, monitoring IoT zařízení, analýza sítí, vědecké experimenty a všude tam, kde jsou důležité včasné poznatky z kontinuálně se měnících dat.

Hlavní výhody vizualizace v reálném čase:
- Okamžitá detekce anomálií a trendů
- Rychlejší reakce na měnící se podmínky
- Lepší situační povědomí
- Zdokonalené rozhodovací schopnosti
- Poutavé a dynamické uživatelské zkušenosti

V tomto průvodci prozkoumáme různé knihovny a techniky v Pythonu pro vytváření efektivních vizualizací dat v reálném čase, od jednoduchých animovaných grafů až po komplexní dashboardy.

```python
import requests
import json

def poll_api(url, interval=1):
    """Dotazování API v pravidelných intervalech"""
    while True:
        try:
            response = requests.get(url)
            if response.status_code == 200:
                yield json.loads(response.text)
```

## Vizualizace v reálném čase s Matplotlib

Matplotlib nabízí animační schopnosti pro vytváření vizualizací v reálném čase.

### Základní animace

```python
import matplotlib.pyplot as plt
import matplotlib.animation as animation
from collections import deque
import numpy as np
import time

# Vytvoření obrázku a osy
fig, ax = plt.subplots(figsize=(10, 6))

# Nastavení parametrů grafu
ax.set_xlim(0, 100)
ax.set_ylim(-5, 5)
ax.set_title('Vizualizace dat v reálném čase')
ax.set_xlabel('Čas')
ax.set_ylabel('Hodnota')
ax.grid(True)

# Vytvoření objektu čáry
line, = ax.plot([], [], lw=2)

# Vytvoření front s pevnou délkou pro data x a y
max_points = 100
xdata = deque(maxlen=max_points)
ydata = deque(maxlen=max_points)

# Inicializační funkce
def init():
    line.set_data([], [])
    return line,

# Animační funkce volána periodicky
def update(frame):
    # Generování nebo získání nových dat
    new_y = np.sin(frame * 0.1) + np.random.normal(0, 0.3)
    
    # Přidání nových dat do front
    xdata.append(frame)
    ydata.append(new_y)
    
    # Aktualizace čáry s novými daty
    line.set_data(list(xdata), list(ydata))
    
    # Úprava limitů osy x podle potřeby
    if frame >= max_points:
        ax.set_xlim(frame - max_points + 1, frame)
    
    return line,

# Vytvoření animace
ani = animation.FuncAnimation(
    fig, update, frames=range(500),
    init_func=init, blit=True, interval=50)

plt.tight_layout()
plt.show()
```

```python
from matplotlib.animation import FuncAnimation

# Vytvoření obrázku a subplot
fig, (ax1, ax2) = plt.subplots(2, 1, figsize=(10, 8))
fig.suptitle('Více datových toků v reálném čase')

# Nastavení čar pro každý subplot
line1, = ax1.plot([], [], lw=2, color='blue')
line2, = ax2.plot([], [], lw=2, color='red')

# Inicializace datových kontejnerů
max_points = 100
xdata = list(range(max_points))
y1data = [0] * max_points
y2data = [0] * max_points

# Nastavení limitů grafu
ax1.set_ylim(-5, 5)
ax1.set_xlim(0, max_points)
ax1.set_ylabel('Senzor 1')
ax1.grid(True)

ax2.set_ylim(-5, 5)
ax2.set_xlim(0, max_points)
ax2.set_xlabel('Čas')
ax2.set_ylabel('Senzor 2')
ax2.grid(True)

# Funkce update pro animaci
def update(frame):
    # Simulace nových datových bodů pro dva senzory
    new_y1 = np.sin(frame * 0.1) + np.random.normal(0, 0.3)
    new_y2 = np.cos(frame * 0.1) + np.random.normal(0, 0.3)
    
    # Aktualizace dat posunutím doleva a přidáním nového bodu na konec
    y1data.pop(0)
    y1data.append(new_y1)
    y2data.pop(0)
    y2data.append(new_y2)
    
    # Aktualizace čar
    line1.set_data(xdata, y1data)
    line2.set_data(xdata, y2data)
    
    return line1, line2

# Vytvoření animace
ani = FuncAnimation(fig, update, frames=range(500),
                    blit=True, interval=50)

plt.tight_layout()
plt.show()
```

### Dynamická teplotní mapa

```python
import numpy as np
import matplotlib.pyplot as plt
import matplotlib.animation as animation

# Vytvoření obrázku a osy
fig, ax = plt.subplots(figsize=(8, 8))

# Inicializace 20x20 matice s nulami
data = np.zeros((20, 20))

# Vytvoření teplotní mapy
heatmap = ax.imshow(data, cmap='viridis', vmin=0, vmax=1, 
                    interpolation='nearest', aspect='auto')
plt.colorbar(heatmap)
ax.set_title('Teplotní mapa v reálném čase')

# Funkce pro aktualizaci teplotní mapy
def update(frame):
    # Generování náhodných změn v datech
    i, j = np.random.randint(0, 20, 2)
    data[i, j] = np.clip(data[i, j] + np.random.normal(0, 0.1), 0, 1)
    
    # Aplikace vyhlazení (volitelné)
    if frame % 10 == 0:
        data[:] = 0.95 * data + 0.05 * np.random.random((20, 20))
    
    # Aktualizace teplotní mapy
    heatmap.set_array(data)
    return [heatmap]

# Vytvoření animace
ani = animation.FuncAnimation(
    fig, update, frames=200,
    blit=True, interval=100)

plt.tight_layout()
plt.show()
```

## Interaktivní vizualizace v reálném čase s Plotly

Plotly poskytuje interaktivní vizualizace, které mohou být aktualizovány v reálném čase.

### Základní graf v reálném čase s Plotly

```python
import plotly.graph_objects as go
from IPython.display import display, clear_output
import numpy as np
import time

# Vytvoření obrázku
fig = go.Figure(
    data=[go.Scatter(x=[], y=[], mode='lines+markers', name='Signál')],
    layout=go.Layout(
        title="Monitorování signálu v reálném čase",
        xaxis=dict(title="Čas", range=[0, 30]),
        yaxis=dict(title="Hodnota", range=[-5, 5]),
        height=500,
    )
)

# Zobrazení počátečního prázdného obrázku
fig.show()

# Inicializace úložiště dat
x_data = []
y_data = []

# Simulace aktualizací v reálném čase (v prostředí notebooku)
for i in range(60):
    # Získání nových dat
    new_x = i
    new_y = np.sin(i * 0.2) + np.random.normal(0, 0.3)
    
    # Přidání do seznamů dat
    x_data.append(new_x)
    y_data.append(new_y)
    
    # Zachování pouze 30 nejnovějších bodů
    if len(x_data) > 30:
        x_data = x_data[-30:]
        y_data = y_data[-30:]
    
    # Aktualizace obrázku
    fig.data[0].x = x_data
    fig.data[0].y = y_data
    
    # Aktualizace rozsahu osy x pro zobrazení nejnovějších dat
    fig.update_layout(xaxis_range=[max(0, new_x-30), new_x])
    
    # Zobrazení aktualizovaného obrázku
    clear_output(wait=True)
    fig.show()
    
    # Pauza pro simulaci příchodu dat
    time.sleep(0.2)
```

## Dashboardy v reálném čase s Dash

Dash je mocný framework pro vytváření webových dashboardů v Pythonu.

### Základní dashboard v reálném čase

```python
from dash import Dash, dcc, html
from dash.dependencies import Input, Output
import plotly.graph_objs as go
import numpy as np
from collections import deque
import time

# Inicializace Dash aplikace
app = Dash(__name__)

# Maximální počet datových bodů k zobrazení
max_length = 50

# Inicializace datových vyrovnávacích pamětí
times = deque(maxlen=max_length)
values = deque(maxlen=max_length)

# Přidání počátečních dat
start_time = time.time()
for i in range(max_length):
    times.append(start_time - (max_length - i) * 0.2)
    values.append(0)

# Definice rozložení aplikace
app.layout = html.Div([
    html.H1("Dashboard pro monitorování dat v reálném čase"),
    
    dcc.Graph(id='live-graph', animate=True),
    
    dcc.Interval(
        id='graph-update',
        interval=200,  # v milisekundách
        n_intervals=0
    ),
])

# Definice callbacku pro aktualizaci grafu
@app.callback(
    Output('live-graph', 'figure'),
    [Input('graph-update', 'n_intervals')]
)
def update_graph(n):
    # Přidání nových dat
    times.append(time.time())
    values.append(np.sin(n * 0.1) + np.random.normal(0, 0.5))
    
    # Vytvoření grafu s novými daty
    fig = go.Figure(
        data=[go.Scatter(
            x=list(times),
            y=list(values),
            name='Data senzoru',
            mode='lines+markers',
            line=dict(color='#1E88E5', width=2)
        )],
        layout=go.Layout(
            title="Datový tok v reálném čase",
            xaxis=dict(title='Čas', range=[min(times), max(times)]),
            yaxis=dict(title='Hodnota', range=[-3, 3]),
            margin=dict(l=50, r=50, b=50, t=80),
            height=500,
        )
    )
    
    return fig

# Spuštění aplikace (odkomentujte pro spuštění)
# if __name__ == '__main__':
#     app.run_server(debug=True)

print("Pro spuštění tohoto dashboardu, uložte do souboru a spusťte pomocí 'python nazev_souboru.py'")
```

### Dashboard s více komponenty

```python
from dash import Dash, dcc, html
from dash.dependencies import Input, Output
import plotly.graph_objects as go
import plotly.express as px
from collections import deque
import pandas as pd
import numpy as np
import time

# Inicializace Dash aplikace
app = Dash(__name__)

# Maximální počet datových bodů k zobrazení
max_length = 100

# Inicializace datových vyrovnávacích pamětí pro více metrik
timestamps = deque(maxlen=max_length)
temperature = deque(maxlen=max_length)
humidity = deque(maxlen=max_length)
pressure = deque(maxlen=max_length)

# Přidání počátečních dat
start_time = time.time()
for i in range(20):  # Začínáme s 20 body
    timestamps.append(start_time - (20 - i) * 1.0)
    temperature.append(25 + np.random.normal(0, 1))
    humidity.append(50 + np.random.normal(0, 2))
    pressure.append(1013 + np.random.normal(0, 0.5))

# Inicializace stavových dat
status_data = {"system": "Online", "last_update": "N/A", "sensors": "3 aktivní"}

# Definice rozložení aplikace
app.layout = html.Div([
    html.H1("Dashboard pro monitorování IoT senzorů", style={"textAlign": "center"}),
    
    html.Div([
        html.Div([
            html.H3("Stav systému"),
            html.Div(id="status-indicators"),
        ], className="status-container", style={"width": "30%", "display": "inline-block"}),
        
        html.Div([
            html.H3("Aktuální hodnoty"),
            html.Div(id="current-values"),
        ], className="current-values-container", style={"width": "70%", "display": "inline-block"}),
    ], style={"display": "flex"}),
    
    html.Div([
        dcc.Graph(id='temperature-graph'),
    ], style={"width": "100%"}),
    
    html.Div([
        dcc.Graph(id='multi-graph'),
    ], style={"width": "100%"}),
    
    dcc.Interval(
        id='interval-component',
        interval=1000,  # aktualizace po 1 sekundě
        n_intervals=0
    )
])

# Callback pro aktualizaci stavových indikátorů
@app.callback(
    Output('status-indicators', 'children'),
    [Input('interval-component', 'n_intervals')]
)
def update_status(n):
    status_data["last_update"] = time.strftime("%Y-%m-%d %H:%M:%S")
    
    return html.Div([
        html.P(f"Stav: {status_data['system']}"),
        html.P(f"Poslední aktualizace: {status_data['last_update']}"),
        html.P(f"Senzory: {status_data['sensors']}")
    ])

# Callback pro aktualizaci aktuálních hodnot
@app.callback(
    Output('current-values', 'children'),
    [Input('interval-component', 'n_intervals')]
)
def update_current_values(n):
    return html.Div([
        html.Div([
            html.H4(f"{temperature[-1]:.1f} °C", style={"margin": "0"}),
            html.P("Teplota", style={"margin": "0"})
        ], style={"width": "33%", "display": "inline-block", "textAlign": "center"}),
        
        html.Div([
            html.H4(f"{humidity[-1]:.1f} %", style={"margin": "0"}),
            html.P("Vlhkost", style={"margin": "0"})
        ], style={"width": "33%", "display": "inline-block", "textAlign": "center"}),
        
        html.Div([
            html.H4(f"{pressure[-1]:.1f} hPa", style={"margin": "0"}),
            html.P("Tlak", style={"margin": "0"})
        ], style={"width": "33%", "display": "inline-block", "textAlign": "center"})
    ])

# Callback pro aktualizaci grafu teploty
@app.callback(
    Output('temperature-graph', 'figure'),
    [Input('interval-component', 'n_intervals')]
)
def update_temperature_graph(n):
    # Přidání nových dat
    timestamps.append(time.time())
    temperature.append(25 + np.sin(n * 0.05) + np.random.normal(0, 0.5))
    
    # Vytvoření DataFrame pro gauge chart
    df = pd.DataFrame({
        'Teplota': [temperature[-1]]
    })
    
    # Vytvoření gauge grafu
    fig = px.gauge(
        df,
        value='Teplota',
        range_color=[10, 40],
        color='Teplota',
        title='Aktuální teplota',
        labels={'Teplota': 'Teplota (°C)'},
        gauge={'axis': {'range': [15, 35]}, 'bar': {'color': "#1E88E5"}}
    )
    
    return fig

# Callback pro aktualizaci více grafů
@app.callback(
    Output('multi-graph', 'figure'),
    [Input('interval-component', 'n_intervals')]
)
def update_multi_graph(n):
    # Přidání nových dat
    humidity.append(50 + 5 * np.sin(n * 0.03) + np.random.normal(0, 1))
```

```python
import numpy as np

class CircularBuffer:
    """Efektivní kruhový buffer pro streamovaná data"""
    
    def __init__(self, size, dimensions=1):
        self.size = size
        self.dimensions = dimensions
        self.data = np.zeros((size, dimensions))
        self.index = 0
        self.count = 0
    
    def add(self, value):
        """Přidání nové hodnoty do bufferu"""
        self.data[self.index] = value
        self.index = (self.index + 1) % self.size
        self.count = min(self.count + 1, self.size)
    
    def get_all(self):
        """Získání všech dat v chronologickém pořadí"""
        if self.count < self.size:
            return self.data[:self.count]
        else:
            return np.concatenate((
                self.data[self.index:],
                self.data[:self.index]
            ))

# Příklad použití
buffer = CircularBuffer(size=1000, dimensions=3)

# Přidání dat
for i in range(1500):
    buffer.add([i, np.sin(i * 0.1), np.cos(i * 0.1)])

# Získání posledních 1000 bodů v chronologickém pořadí
recent_data = buffer.get_all()
print(f"Buffer obsahuje {len(recent_data)} bodů s tvarem {recent_data.shape}")
```

### Downsampling pro velké datasety

```python

# Funkce pro downsampling dat
def downsample(data, method='mean', bucket_size=100):
    """
    Downsampling dat pomocí různých metod
    
    Parametry:
    -----------
    data : DataFrame
        DataFrame s sloupci x a y
    method : str
        'mean', 'max', 'min', nebo 'lttb' (Largest Triangle Three Buckets)
    bucket_size : int
        Počet bodů k agregaci
        
    Vrací:
    --------
    DataFrame s downsamplovanými daty
    """
    if method in ['mean', 'max', 'min']:
        # Použití pandas groupby pro jednoduchou agregaci
        n_buckets = len(data) // bucket_size
        grouped = data.groupby(np.arange(len(data)) // bucket_size)
        
        if method == 'mean':
            return grouped.mean().reset_index(drop=True)
        elif method == 'max':
            return grouped.max().reset_index(drop=True)
        elif method == 'min':
            return grouped.min().reset_index(drop=True)
            
    else:
        # Výchozí jsou původní data
        return data

# Downsampling pomocí různých metod

df_max = downsample(df, method='max', bucket_size=100)
```

```python
class IoTSensor:
    def __init__(self, sensor_id, initial_value=0, drift=0, noise=1):
        self.sensor_id = sensor_id
        self.value = initial_value
        self.drift = drift
        self.noise = noise

            'value': [],
            'status': []

        timestamp = time.time()
        
        # Určení stavu na základě prahových hodnot

            status = 'Kritický'
        elif abs(self.value) > 10:
            status = 'Varování'

            'timestamp': [timestamp],
            'value': [self.value],
            'status': [status]
        })

        # Zachování pouze nedávné historie

            self.history = self.history.iloc[-100:]

            'sensor_id': self.sensor_id,
            'timestamp': timestamp,
            'value': self.value,
            'status': status
        }
```

```python
    values[missing_indices] = np.nan
    
    # Přidání odlehlých hodnot
    outlier_indices = np.random.choice(1000, 30, replace=False)
    values[outlier_indices] = values[outlier_indices] * 5
    
    # Vytvoření DataFrame
    df = pd.DataFrame({
        'timestamp': timestamps,
        'value': values
    })
    
    # Přidání některých duplicitních časových značek
    duplicate_rows = df.iloc[np.random.choice(1000, 20, replace=False)]
    df = pd.concat([df, duplicate_rows])
    
    # Přidání nerovnoměrného vzorkování
    df = df.sample(frac=1).reset_index(drop=True)
    
    return df

# Funkce pro předzpracování dat v reálném čase pro vizualizaci
def preprocess_for_visualization(df):
    """Předzpracování časových řad pro čistou vizualizaci"""
    
    # Seřazení podle časové značky
    df = df.sort_values('timestamp')
    
    # Odstranění duplicit
    df = df.drop_duplicates('timestamp')
    
    # Interpolace chybějících hodnot (NaN)
    df['value'] = df['value'].interpolate(method='linear')
    
    # Zpracování odlehlých hodnot pomocí klouzavého mediánu
    rolling_median = df['value'].rolling(window=5, center=True).median()
    df['rolling_median'] = rolling_median
    
    # Detekce odlehlých hodnot na základě vzdálenosti od klouzavého mediánu
    threshold = 3
    df['is_outlier'] = (
        np.abs(df['value'] - df['rolling_median']) > threshold * df['value'].std()
    )
    
    # Nahrazení odlehlých hodnot klouzavým mediánem
    df.loc[df['is_outlier'], 'value_cleaned'] = df.loc[df['is_outlier'], 'rolling_median']
    df.loc[~df['is_outlier'], 'value_cleaned'] = df.loc[~df['is_outlier'], 'value']
    
    # Převod na rovnoměrné časové vzorkování, pokud je potřeba
    uniform_index = pd.date_range(
        start=df['timestamp'].min(),
        end=df['timestamp'].max(),
        freq='1min'
    )
    
    uniform_df = df.set_index('timestamp').reindex(uniform_index).interpolate()
    uniform_df = uniform_df.reset_index().rename(columns={'index': 'timestamp'})
    
    return uniform_df
```
