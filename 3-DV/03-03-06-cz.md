## 3D vizualizace terénu

Vytváření 3D vizualizací terénu a výškových dat:

```python
# Toto je zástupné místo pro 3D vizualizaci terénu
# V reálné implementaci byste načítali DEM soubor
import numpy as np
from matplotlib import pyplot as plt
from mpl_toolkits.mplot3d import Axes3D

# Vygenerování ukázkových dat terénu
def generate_terrain(size=100, roughness=0.5):
    noise = np.random.normal(0, 1, (size, size))
    terrain = np.zeros((size, size))
    
    step = size
    while step > 1:
        for i in range(0, size, step):
            for j in range(0, size, step):
                square_avg = noise[i:i+step, j:j+step].mean()
                noise[i:i+step, j:j+step] += square_avg * roughness
        step = step // 2
        roughness /= 2
        
    # Normalizace na 0-1
    terrain = (noise - noise.min()) / (noise.max() - noise.min())
    return terrain

# Vytvoření povrchu terénu
terrain = generate_terrain(size=100, roughness=0.7)

# Vykreslení terénu ve 3D
fig = plt.figure(figsize=(12, 10))
ax = fig.add_subplot(111, projection='3d')

# Vytvoření mřížky pro vykreslení
x = np.linspace(0, 1, terrain.shape[0])
y = np.linspace(0, 1, terrain.shape[1])
x, y = np.meshgrid(x, y)

# Vykreslení povrchu
surf = ax.plot_surface(x, y, terrain, 
                       cmap='terrain',
                       linewidth=0,
                       antialiased=True,
                       shade=True)

# Přidání barevné škály
fig.colorbar(surf, ax=ax, shrink=0.5, aspect=5, label='Nadmořská výška')

# Nastavení popisků a titulku
ax.set_xlabel('X')
ax.set_ylabel('Y')
ax.set_zlabel('Nadmořská výška')
ax.set_title('3D vizualizace terénu', fontsize=14)

# Nastavení úhlu pohledu
ax.view_init(elev=30, azim=120)

plt.tight_layout()
plt.show()
```

# Toto je zástupné místo pro vytvoření animace povětrnostního systému
# V praxi byste použili skutečná meteorologická data
import matplotlib.pyplot as plt
import numpy as np
from matplotlib.animation import FuncAnimation
from IPython.display import HTML
import matplotlib.colors as mcolors

# Vygenerování syntetických meteorologických dat (změna teploty v průběhu času)
def generate_weather_data(width=50, height=25, frames=10):
    # Začneme základním rozložením teploty
    base = np.zeros((height, width))
    center_x, center_y = width // 2, height // 2
    
    # Vytvoření matice vzdáleností od centra
    y, x = np.ogrid[:height, :width]
    dist = np.sqrt((x - center_x)**2 + (y - center_y)**2)
    
    # Vytvoření základní teploty (teplejší uprostřed, chladnější na okrajích)
    max_dist = np.sqrt(center_x**2 + center_y**2)
    base = 30 * (1 - dist / max_dist)  # Teplota ve stupních Celsia
    
    # Generování snímků s pohybujícím se povětrnostním systémem
    frames_data = []
    
    for i in range(frames):
        # Vytvoření pohybující se povětrnostní fronty
        weather_front = np.zeros((height, width))
        front_center_x = int(width * 0.1) + int(width * 0.8 * i / frames)
        
        # Vzdálenost od povětrnostní fronty
        front_dist = np.sqrt((x - front_center_x)**2 + (y - center_y)**2)
        front_effect = 15 * np.exp(-front_dist**2 / (2 * (width/5)**2))
        
        # Snížení teploty vlivem povětrnostní fronty
        frame = base - front_effect
        frames_data.append(frame)
    
    return frames_data

# Vygenerování meteorologických dat
weather_frames = generate_weather_data(width=80, height=40, frames=20)

# Vytvoření vlastní barevné mapy pro teplotu
